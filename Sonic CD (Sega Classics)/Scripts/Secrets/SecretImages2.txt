// Aliases
private alias object.value0 : object.timer // Also used for fading

// States
private alias 0	: SECRETIMAGES2_SETUP
private alias 1	: SECRETIMAGES2_FADE_IN
private alias 2	: SECRETIMAGES2_CONTROL_DELAY
private alias 3	: SECRETIMAGES2_CONTROL_CHECK
private alias 4	: SECRETIMAGES2_EXIT

// Presentation Stage Aliases
private alias 3	: STAGE_P_LEVELSELECT

// Game Mode Aliases
private alias 0	: MODE_NOSAVE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case SECRETIMAGES2_SETUP
		if object.timer < 8
			object.timer++
		else
			object.timer = 288
			SetActivePalette(1, 0, screen.ysize)
			object.state++
		end if
		SetScreenFade(0, 0, 0, 255)
		break

	case SECRETIMAGES2_FADE_IN
		if object.timer > 0
			object.timer -= 16
		else
			PlayMusic(options.soundtrack)
			object.state++
		end if
		SetScreenFade(0, 0, 0, object.timer)
		break

	case SECRETIMAGES2_CONTROL_DELAY
		if object.timer < 60
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case SECRETIMAGES2_CONTROL_CHECK
		if keyPress[0].buttonA == true
			object.state = SECRETIMAGES2_EXIT
		end if

		if keyPress[0].buttonB == true
			object.state = SECRETIMAGES2_EXIT
		end if

		if keyPress[0].buttonC == true
			object.state = SECRETIMAGES2_EXIT
		end if

		if keyPress[0].start == true
			object.state = SECRETIMAGES2_EXIT
		end if

		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			object.state = SECRETIMAGES2_EXIT
		end if
		break

	case SECRETIMAGES2_EXIT
		if object.timer < 288
			object.timer += 8
			music.volume -= 5
		else
			SetActivePalette(0, 0, screen.ysize)
			StopMusic()
			options.gameMode = MODE_NOSAVE
			stage.activeList = PRESENTATION_STAGE
			stage.listPos    = STAGE_P_LEVELSELECT
			LoadStage()
		end if

		temp0 = object.timer
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break

	end switch
end event


event ObjectDraw
	DrawRect(0, 0, screen.xsize, screen.ysize, 224, 224, 224, 255)
	DrawSpriteScreenXY(0, screen.xcenter, screen.ycenter)
end event


event ObjectStartup
	LoadSpriteSheet("Secrets/DD_R2.gif")
	SpriteFrame(-160, -120, 320, 240, 0, 0)		// #0 - Desert Dazzle
	LoadPalette("DDZ_Pal.act", 1, 0, 0, 256)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Secrets/DD_R2.gif")
	SpriteFrame(-160, -120, 320, 240, 0, 0)		// #0 - Desert Dazzle

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

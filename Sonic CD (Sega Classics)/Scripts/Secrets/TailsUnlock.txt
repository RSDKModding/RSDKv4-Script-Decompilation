// Aliases
private alias object.value0 : object.timer

// States
private alias 0	: TAILSUNLOCK_SETUP
private alias 1	: TAILSUNLOCK_FADE_IN
private alias 2	: TAILSUNLOCK_CONTROL_DELAY
private alias 3	: TAILSUNLOCK_CONTROL_CHECK
private alias 4	: TAILSUNLOCK_EXIT

// Engine States
private alias 8	: ENGINE_RESETGAME

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case TAILSUNLOCK_SETUP
		if object.timer < 16
			// Yield on the black screen for a few moments
			object.timer++
		else
			if stage.debugMode == false
				// In the RSDKv3 version, we'd set the all stages clear achievement here.. usually :(
			end if

			object.timer = 288
			object.state++
		end if

		// Solid Black screen
		SetScreenFade(0, 0, 0, 255)
		break

	case TAILSUNLOCK_FADE_IN
		if object.timer > 0
			object.timer -= 8
		else
			// Fully faded in now, play the special SFX
			PlaySfx(SfxName[Event], false)
			object.state++
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break

	case TAILSUNLOCK_CONTROL_DELAY
		if object.timer < 60
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case TAILSUNLOCK_CONTROL_CHECK
		if keyPress[0].buttonA == true
			object.state = TAILSUNLOCK_EXIT
		end if

		if keyPress[0].buttonB == true
			object.state = TAILSUNLOCK_EXIT
		end if

		if keyPress[0].buttonC == true
			object.state = TAILSUNLOCK_EXIT
		end if

		if keyPress[0].start == true
			object.state = TAILSUNLOCK_EXIT
		end if

		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			object.state = TAILSUNLOCK_EXIT
		end if

		if object.timer < 300
			object.timer++
		else
			object.timer = 0
			object.state = TAILSUNLOCK_EXIT
		end if
		break

	case TAILSUNLOCK_EXIT
		if object.timer < 288
			// Fade the screen out
			object.timer += 8
		else
			engine.state = ENGINE_RESETGAME
		end if

		temp0 = object.timer
		if temp0 > 255
			temp0 = 255
		end if

		SetScreenFade(0, 0, 0, temp0)
		break

	end switch
end event


event ObjectDraw
	// The scene doesn't have a background, so draw a black rectangle one here
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)

	// And then draw the actual See You Next Game sprite
	DrawSpriteScreenXY(0, screen.xcenter, screen.ycenter)
end event


event ObjectStartup
	LoadSpriteSheet("Secrets/SecretImages.gif")
	SpriteFrame(-117, -36, 234, 64, 0, 900)		// #0 - See You Next Game
end event

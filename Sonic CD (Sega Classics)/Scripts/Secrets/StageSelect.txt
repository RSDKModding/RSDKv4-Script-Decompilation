// Aliases
private alias object.value0 : object.fade // Also used for timing
private alias object.value1 : object.level
private alias object.value2 : object.holdTimer
private alias object.value3 : object.touchButton

// States
private alias 0	: STAGESELECT_SETUP
private alias 1	: STAGESELECT_FADE_IN
private alias 2	: STAGESELECT_CONTROLS
private alias 3	: STAGESELECT_LOAD_LEVEL
private alias 4	: STAGESELECT_EXIT

// Presentation Stage Aliases
private alias 0	: STAGE_P_TITLESCREEN

// Game Mode Aliases
private alias 0	: MODE_NOSAVE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case STAGESELECT_SETUP
		if object.fade < 8
			object.fade++
		else
			object.fade = 254
			object.state++
		end if
		SetScreenFade(0, 0, 0, 255)
		break

	case STAGESELECT_FADE_IN
		if object.fade > 0
			SetScreenFade(0, 0, 0, object.fade)
			object.fade -= 8
		else
			object.state++
		end if
		break

	case STAGESELECT_CONTROLS
		if keyPress[0].up == true
			object.holdTimer = 0
			object.level--
			if object.level < 0
				object.level = 69
			end if
		else
			if keyPress[0].down == true
				object.holdTimer = 0
				object.level++
				if object.level > 69
					object.level = 0
				end if
			end if
		end if

		if keyDown[0].up == true
			object.holdTimer++
			if object.holdTimer == 8
				object.holdTimer = 0
				object.level--
				if object.level < 0
					object.level = 69
				end if
			end if
		else
			if keyDown[0].down == true
				object.holdTimer++
				if object.holdTimer == 8
					object.holdTimer = 0
					object.level++
					if object.level > 69
						object.level = 0
					end if
				end if
			end if
		end if

		if keyPress[0].start == true
			object.state = STAGESELECT_LOAD_LEVEL
		end if

		if keyPress[0].buttonB == true
			object.state = STAGESELECT_EXIT
		end if

		if engine.deviceType == MOBILE
			temp0 = screen.xcenter
			temp0 -= 138
			temp1 = screen.xcenter
			temp1 -= 76
			CheckTouchRect(temp0, 20, temp1, 72)
			if checkResult > -1
				keyDown[0].up = true
				object.holdTimer++
				if object.holdTimer == 8
					object.holdTimer = 0
					object.level--
					if object.level < 0
						object.level = 69
					end if
				end if
			end if

			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				keyDown[0].down = true
				object.holdTimer++
				if object.holdTimer == 8
					object.holdTimer = 0
					object.level++
					if object.level > 69
						object.level = 0
					end if
				end if
			end if

			temp0 = screen.xcenter
			temp0 += 28
			temp1 = screen.xcenter
			temp1 += 68
			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				object.touchButton = 1
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.touchButton == 1
						object.touchButton = 0
						PlaySfx(SfxName[Select], false)
						object.state = STAGESELECT_EXIT
					end if
				else
					if object.touchButton == 1
						object.touchButton = 0
					end if
				end if
			end if

			temp0 += 48
			temp1 += 48
			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				object.touchButton = 2
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.touchButton == 2
						object.touchButton = 0
						PlaySfx(SfxName[Select], false)
						object.state = STAGESELECT_LOAD_LEVEL
					end if
				else
					if object.touchButton == 2
						object.touchButton = 0
					end if
				end if
			end if
		end if
		break

	case STAGESELECT_LOAD_LEVEL
		if object.fade < 288
			object.fade += 8
		else
			options.gameMode = 0
			player.lives = 3
			stage.activeList = REGULAR_STAGE
			stage.listPos = object.level
			LoadStage()
		end if

		temp0 = object.fade
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break

	case STAGESELECT_EXIT
		if object.fade < 288
			object.fade += 8
		else
			options.gameMode = MODE_NOSAVE
			stage.activeList = PRESENTATION_STAGE
			stage.listPos    = STAGE_P_TITLESCREEN
			LoadStage()
		end if

		temp0 = object.fade
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break

	end switch
end event


event ObjectDraw
	temp0 = 0
	while temp0 < 240
		temp1 = 0
		while temp1 < screen.xsize
			DrawSpriteScreenXY(15, temp1, temp0)
			temp1 += 224
		loop
		temp0 += 48
	loop
	DrawSpriteScreenXY(16, screen.xcenter, 80)

	temp0 = screen.xcenter
	temp0 -= 48
	temp1 = object.level
	temp1 /= 10
	switch temp1
	case 0
		DrawSpriteScreenXY(1, temp0, 112)
		break
	case 1
		DrawSpriteScreenXY(3, temp0, 112)
		break
	case 2
		DrawSpriteScreenXY(4, temp0, 112)
		break
	case 3
		DrawSpriteScreenXY(5, temp0, 112)
		break
	case 4
		DrawSpriteScreenXY(6, temp0, 112)
		break
	case 5
		DrawSpriteScreenXY(7, temp0, 112)
		break
	case 6
		DrawSpriteScreenXY(8, temp0, 112)
		break
	end switch
	temp0 += 8
	DrawSpriteScreenXY(14, temp0, 112)

	temp0 += 8
	temp1 = object.level
	temp1 %= 10
	temp1 >>= 2
	temp1++
	DrawSpriteScreenXY(temp1, temp0, 112)

	temp0 += 8
	DrawSpriteScreenXY(14, temp0, 112)

	temp0 += 8
	temp1 = object.level
	temp1 %= 10
	switch temp1
	case 0
	case 4
		DrawSpriteScreenXY(10, temp0, 112)
		break

	case 1
	case 5
		DrawSpriteScreenXY(11, temp0, 112)
		break

	case 2
	case 6
	case 8
		DrawSpriteScreenXY(12, temp0, 112)
		break

	case 3
	case 7
	case 9
		DrawSpriteScreenXY(13, temp0, 112)
		break

	end switch

	if engine.deviceType == MOBILE
		temp0 -= 96
		if keyDown[0].up == true
			DrawSpriteScreenXY(18, temp0, 48)
		else
			DrawSpriteScreenXY(17, temp0, 48)
		end if
		if keyDown[0].down == true
			DrawSpriteScreenXY(20, temp0, 192)
		else
			DrawSpriteScreenXY(19, temp0, 192)
		end if
		temp0 += 160
		if object.touchButton == true
			DrawSpriteScreenXY(24, temp0, 192)
		else
			DrawSpriteScreenXY(23, temp0, 192)
		end if
		temp0 += 48
		if object.touchButton == 2
			DrawSpriteScreenXY(22, temp0, 192)
		else
			DrawSpriteScreenXY(21, temp0, 192)
		end if
	end if
end event

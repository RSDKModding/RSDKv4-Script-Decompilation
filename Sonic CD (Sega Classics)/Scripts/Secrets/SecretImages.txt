// Aliases
private alias object.value0 : object.timer

// States
private alias 0	: SECRETIMAGES_SETUP
private alias 1	: SECRETIMAGES_FADE_IN
private alias 2	: SECRETIMAGES_CONTROL_DELAY
private alias 3	: SECRETIMAGES_CONTROL_CHECK
private alias 4	: SECRETIMAGES_EXIT

// Presentation Stage Aliases
private alias 0	: STAGE_P_TITLESCREEN

// Game Mode Aliases
private alias 0	: MODE_NOSAVE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case SECRETIMAGES_SETUP
		if object.timer < 8
			// Yield on the black screen for a moment
			object.timer++
		else
			// Transition to the next state now!
			object.timer = 288
			object.state++
		end if

		SetScreenFade(0, 0, 0, 255)
		break

	case SECRETIMAGES_FADE_IN
		if object.timer > 0
			object.timer -= 16
		else
			PlayMusic(options.soundtrack)
			object.state++
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break

	case SECRETIMAGES_CONTROL_DELAY
		if object.timer < 60
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case SECRETIMAGES_CONTROL_CHECK
		if keyPress[0].buttonA == true
			object.state = SECRETIMAGES_EXIT
		end if

		if keyPress[0].buttonB == true
			object.state = SECRETIMAGES_EXIT
		end if

		if keyPress[0].buttonC == true
			object.state = SECRETIMAGES_EXIT
		end if

		if keyPress[0].start == true
			object.state = SECRETIMAGES_EXIT
		end if

		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			object.state = SECRETIMAGES_EXIT
		end if
		break

	case SECRETIMAGES_EXIT
		if object.timer < 288
			object.timer += 8
			music.volume -= 5
		else
			StopMusic()
			options.gameMode = MODE_NOSAVE
			stage.activeList = PRESENTATION_STAGE
			stage.listPos    = STAGE_P_TITLESCREEN
			LoadStage()
		end if

		temp0 = object.timer
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break
	end switch
end event


event ObjectDraw
	switch object.propertyValue
	case 0	// Batman Parody
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(0, screen.xcenter, screen.ycenter)
		break

	case 1	// Fun is Infinite
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(1, screen.xcenter, screen.ycenter)
		break

	case 2	// You Are Cool
		DrawRect(0, 0, screen.xsize, screen.ysize, 238, 238, 204, 255)
		DrawSpriteScreenXY(2, screen.xcenter, screen.ycenter)
		break

	case 3	// Fastest DJ Alive
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(3, screen.xcenter, screen.ycenter)
		break

	case 4	// See You Next Game
		DrawRect(0, 0, screen.xcenter, screen.ysize, 0, 34, 221, 255)
		DrawRect(screen.xcenter, 0, screen.xcenter, screen.ysize, 0, 34, 204, 255)
		DrawSpriteScreenXY(4, screen.xcenter, screen.ycenter)
		DrawSpriteScreenXY(5, screen.xcenter, screen.ycenter)
		break
	end switch
end event

	LoadSpriteSheet("Secrets/SecretImages.gif")

	SpriteFrame(-160, -112, 320, 224, 0, 0)		// #0 - Batman Parody
	SpriteFrame(-160, -112, 320, 224, 0, 225)	// #1 - Fun is Infinite
	SpriteFrame(-160, -112, 320, 224, 0, 450)	// #2 - You Are Cool
	SpriteFrame(-160, -112, 320, 224, 0, 675)	// #3 - Fastest DJ Alive
	SpriteFrame(-160, -112, 160, 224, 321, 0)	// #4 - See You Next Game first half
	SpriteFrame(0, -112, 160, 224, 321, 225)	// #5 - See You Next Game second half
end event

event RSDKDraw

	default

end event

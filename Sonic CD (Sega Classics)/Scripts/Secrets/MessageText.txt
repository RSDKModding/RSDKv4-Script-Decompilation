// Aliases
private alias object.value1 : object.index

// Text Info type
private alias 0	: TEXTINFO_TEXTDATA
private alias 1	: TEXTINFO_TEXTSIZE

// ========================
// Events
// ========================

event ObjectDraw
	arrayPos0 = 0

	// Get how long the target text line is
	GetTextInfo(temp1, MENU_1, TEXTINFO_TEXTSIZE, object.index, 0)

	// The text should be just about aligned to the center, so do some maths based on the character length to make that happen
	temp2 = temp1
	temp2 <<= 18
	FlipSign(temp2)
	temp2 += object.xpos

	// Loop through all the characters in the message
	while temp1 > 0
		// Get the actual character at the current char pos
		GetTextInfo(temp0, MENU_1, TEXTINFO_TEXTDATA, object.index, arrayPos0)
		object.frame = 0

		// Take uppercase letters and turn them into a matching Sprite Frame
		if temp0 > 64
			if temp0 < 91
				object.frame = temp0
				object.frame -= 64
			end if
		end if

		// Take lowercase letters and turn them into a matching Sprite Frame
		if temp0 > 96
			if temp0 < 123
				object.frame = temp0
				object.frame -= 96
			end if
		end if

		// Notably, the space character (32) is skipped over here, as it shouldn't draw anything

		if object.frame > 0
			// The the letter, but only if it's a valid one
			DrawSpriteXY(object.frame, temp2, object.ypos)
		end if

		// Move to the next letter
		arrayPos0++

		// And decrease the remaining letter count, as well
		temp1--

		// Each letter should be spaced 8 pixels apart
		temp2 += 0x80000
	loop
end event


event ObjectStartup
	LoadSpriteSheet("Secrets/SecretMenus.gif")

	SpriteFrame(-3, -9, 8, 16, 37, 35)		// #0 - Dummy Sprite

	// #1 -> #26 - Alphabet
	// (Using loops to automate Sprite Framing)
	temp1 = 49
	while temp1 < 100
		temp0 = 37
		while temp0 < 136
			SpriteFrame(-3, -6, 8, 16, temp0, temp1)
			temp0 += 9
		loop

		temp1 += 17
	loop
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	temp0 = 0
	while temp0 < 4
		DrawSprite(temp0)
		temp0++
	loop
end event

	SpriteFrame(9, -6, 8, 16, 73, 100)
	SpriteFrame(18, -6, 8, 16, 46, 134)
	SpriteFrame(27, -6, 8, 16, 109, 117)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

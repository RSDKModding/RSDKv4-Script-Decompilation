// Aliases
private alias object.value0 : object.timer

// ========================
// Events
// ========================

event ObjectDraw
	if object.timer > 144
		// Fade out the Heart
		temp0 = object.timer
		temp0 -= 144
		temp0 <<= 3
		if temp0 > 255
			temp0 = 255
		end if

		object.alpha = 255
		object.alpha -= temp0
		if object.alpha == 0
			object.type = TypeName[Blank Object]
		end if
	end if

	Sin(temp0, object.angle)
	temp0 <<= 9
	temp0 += object.xpos

	if object.timer > 19
		// Fading heart
		DrawSpriteFX(1, FX_INK, temp0, object.ypos)
	else
		// Normal heart
		DrawSpriteXY(0, temp0, object.ypos)
	end if

	// Hover up at a rate of 0.375 pixels per frame
	object.ypos -= 0x6000

	object.angle += 8
	object.angle &= 511

	object.timer++
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects2.gif")

	// Love Heart Frames
	SpriteFrame(-8, -8, 16, 16, 193, 134)
	SpriteFrame(-8, -8, 16, 16, 193, 150)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R7/Objects2.gif")
	SpriteFrame(-8, -8, 16, 16, 193, 150)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

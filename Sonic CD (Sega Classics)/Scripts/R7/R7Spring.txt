private alias object.value0 : object.timer

// ========================
// Events
// ========================

event ObjectDraw
	if object.timer == 0
		// Inactive spring, draw whatever the normal frame should be
		DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)
	else
		object.timer++

		temp0 = object.timer
		temp0 >>= 1

		switch temp0
		case 0
		case 1
			// Display the normal frame
			DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)
			break

		case 2
		case 6
			// Display the recoil frame
			temp1 = object.propertyValue
			temp1 += 8
			DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)
			break

		case 3
		case 4
		case 5
			// Display the launch frame
			temp1 = object.propertyValue
			temp1 += 16
			DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)
			break
			
		end switch

		// Only 12 frames of animation, go back to normal after that
		if object.timer > 12
			object.timer = 0
		end if
	end if
end event

event RSDKDraw
	temp0 = 0
	
	// If the last object is a Spring Cage, then try and match its direction
	GetObjectType(temp1, "Spring Cage")
	if object[-1].type == temp1
		switch object[-1].propertyValue
		case 0
			temp0 = 0;
			break

		case 1
			temp0 = 2
			break

		case 3
			temp0 = 4
			break

		case 2 // shouldn't be here, but just in case
		case 4
			temp0 = 0
			break

		case 5
			temp0 = 5
			break
		case 6

			temp0 = 1
			break
			
		end switch
	end if
	
	DrawSprite(temp0)
end event

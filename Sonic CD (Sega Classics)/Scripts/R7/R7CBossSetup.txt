public function R7Setup_GetLoopTileInfo
	// Gets info from out of bounds chunks, located past the signpost

	// Precondition:
	//   - temp0 is the chunk ID of the chunk currently being checked against

	temp3 = player[SLOT_PLAYER1].xpos
	temp3 >>= 16
	temp3 &= 255
	temp3 += 0x4000

	temp0 -= 16
	temp0 >>= 2
	temp0 <<= 8
	temp4 = player[SLOT_PLAYER1].ypos
	temp4 >>= 16
	temp4 &= 255
	temp4 += temp0

	Get16x16TileInfo(temp0, temp3, temp4, TILEINFO_INDEX)
end function

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		// "Layers" are managed here
		// (In reality, chunks are swapped on-the-fly in order to make it appear as if the player's moving behind/in front of them)

		// Get the position of the chunk the player's currently in
		temp1 = player[currentPlayer].xpos
		temp1 >>= 23
		temp2 = player[currentPlayer].ypos
		temp2 >>= 23

		// And find what chunk ID it holds
		GetTileLayerEntry(temp0, 0, temp1, temp2)

		// If applicable, jump to the loop chunk set it corresponds to
		switch temp0
		case 16
		case 17
		case 18
		case 19
			// Loop chunks, open from left side
			CallFunction(R7Setup_GetLoopTileInfo)
			if player[currentPlayer].collisionMode == CMODE_FLOOR
				if temp0 == 1001
					temp1 &= 1022
					temp2 &= 1022
					SetTileLayerEntry(20, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(21, 0, temp1, temp2)
					temp1--
					temp2++
					SetTileLayerEntry(22, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(23, 0, temp1, temp2)
				end if
			else
				if player[currentPlayer].speed > 0
					if temp0 == 1003
						temp1 &= 1022
						temp2 &= 1022
						SetTileLayerEntry(20, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(21, 0, temp1, temp2)
						temp1--
						temp2++
						SetTileLayerEntry(22, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(23, 0, temp1, temp2)
					end if
				end if
			end if
			break

		case 20
		case 21
		case 22
		case 23
			// Loop chunks, open from right side
			CallFunction(R7Setup_GetLoopTileInfo)
			if player[currentPlayer].collisionMode == CMODE_FLOOR
				if temp0 == 1000
					temp1 &= 1022
					temp2 &= 1022
					SetTileLayerEntry(16, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(17, 0, temp1, temp2)
					temp1--
					temp2++
					SetTileLayerEntry(18, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(19, 0, temp1, temp2)
				end if
			else
				if player[currentPlayer].speed < 0
					if temp0 == 1002
						temp1 &= 1022
						temp2 &= 1022
						SetTileLayerEntry(16, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(17, 0, temp1, temp2)
						temp1--
						temp2++
						SetTileLayerEntry(18, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(19, 0, temp1, temp2)
					end if
				end if
			end if
			break

		case 24
		case 25
		case 26
		case 27
			// Wavy chunks, open from left side
			// Note - These don't actually exist in Act 3! So the function is comparing against blank chunks...
			CallFunction(R7Setup_GetLoopTileInfo)
			if temp0 == 1001
				temp1 &= 1022
				temp2 &= 1022
				SetTileLayerEntry(28, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(29, 0, temp1, temp2)
				temp1--
				temp2++
				SetTileLayerEntry(30, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(31, 0, temp1, temp2)
			end if
			break

		case 28
		case 29
		case 30
		case 31
			// Wavy chunks, open from right side, but they don't actually exist
			// See note above
			CallFunction(R7Setup_GetLoopTileInfo)
			if temp0 == 1000
				temp1 &= 1022
				temp2 &= 1022
				SetTileLayerEntry(24, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(25, 0, temp1, temp2)
				temp1--
				temp2++
				SetTileLayerEntry(26, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(27, 0, temp1, temp2)
			end if
			break
		end switch
	next
end event


event ObjectDraw
	// Find the XPos to draw it at, keeping the screen width in mind
	object.xpos = screen.xoffset
	object.xpos /= 420
	object.xpos *= 420
	object.xpos += 320
	object.xpos *= 3
	object.xpos >>= 1
	temp0 = screen.xoffset
	temp0 *= 3
	temp0 >>= 1
	object.xpos -= temp0

	object.ypos = 520
	temp0 = screen.yoffset
	temp0 *= 23
	temp0 >>= 4
	object.ypos -= temp0

	// Draw the street lamp
	DrawSpriteScreenXY(0, object.xpos, object.ypos)
	DrawSpriteScreenXY(1, object.xpos, object.ypos)
	DrawSpriteScreenXY(2, object.xpos, object.ypos)
	DrawSpriteScreenXY(3, object.xpos, object.ypos)
end event

	object[21].priority = PRIORITY_ACTIVE
	object[21].drawOrder = 5

	// Street Lamps
	SpriteFrame(-24, -24, 48, 24, 207, 1)

	// Street Lamp Post Frames
	SpriteFrame(-8, 0, 16, 224, 239, 26)
	SpriteFrame(-8, 224, 16, 224, 239, 26)
	SpriteFrame(-8, 448, 16, 224, 239, 26)

	// Set initial stage bounds
	stage.curXBoundary2 -= 288
end event

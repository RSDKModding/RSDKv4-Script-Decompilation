// ========================
// Aliases
// ========================

// Player Aliases
private alias object.angle         : player.angle
private alias object.xpos          : player.xpos
private alias object.collisionLeft : player.collisionLeft

// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if player[currentPlayer].angle == 0
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -14, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				// Push the player out
				player[currentPlayer].xpos = player[currentPlayer].collisionLeft
				player[currentPlayer].xpos <<= 16
				player[currentPlayer].xpos += object.xpos
				player[currentPlayer].xpos -= 0x100000 // 10 pixels left
			end if
		end if
	next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)

	// Draw the object's switch's hitbox
	temp0 = object.ixpos
	temp0 -= 16
	temp1 = object.iypos
	temp1 -= 14
	DrawRectOutline(temp0, temp1, 32, 30, 255, 255, 255, 255)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -14, 32, 30, 1, 145) // "Script" Icon - it's a tad bit large for the actual hitbox but I can't think of a better fit...

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

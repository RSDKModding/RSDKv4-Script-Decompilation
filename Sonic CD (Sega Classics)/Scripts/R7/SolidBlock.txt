event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = player[currentPlayer].ypos
		if player[currentPlayer].prevGravity == GRAVITY_GROUND
			temp1 = -10
		else
			temp1 = 0x10000
		end if

		switch object.propertyValue
		case BLOCK_SINGLE
		case BLOCK_INVISIBLE
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, temp1, C_BOX, C_BOX)
			break

		case BLOCK_TWO_H
			BoxCollisionTest(C_SOLID, object.entityPos, -32, -16, 32, 16, currentPlayer, C_BOX, temp1, C_BOX, C_BOX)
			break

		case BLOCK_THREE_H
			BoxCollisionTest(C_SOLID, object.entityPos, -48, -16, 48, 16, currentPlayer, C_BOX, temp1, C_BOX, C_BOX)
			break

		case BLOCK_FOUR_H
			BoxCollisionTest(C_SOLID, object.entityPos, -64, -16, 64, 16, currentPlayer, C_BOX, temp1, C_BOX, C_BOX)
			break
			
		case BLOCK_TWO_V
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			break

		case BLOCK_THREE_V
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			break

		case BLOCK_FOUR_V
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -64, 16, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			break

		end switch

		if checkResult == COL_LEFT
			player.rcollisionFlag = true // Used in the Metal Sonic race sequence, but set by the Block regardless of current level
		end if

		if player[currentPlayer].prevGravity == GRAVITY_GROUND
			if checkResult == COL_BOTTOM
				player[currentPlayer].ypos = temp0 // Move the player back ontop of the block
			end if
		end if
	next
end event

event ObjectStartup
	LoadSpriteSheet("R7/Objects.gif")

	// Load the appropriate version of the sprite based on the current time period
	switch HUD.currentTimePeriod
	case TIME_PRESENT
	case TIME_PAST
	case TIME_BAD_FUTURE
		// Normal version of the sprite
		SpriteFrame(-16, -16, 32, 32, 1, 1)
		break

	case TIME_GOOD_FUTURE
		// Good future version of the sprite
		SpriteFrame(-16, -16, 32, 32, 117, 219)
		break

	end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Type
			checkResult = object.propertyValue
			checkResult &= 7
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // Type
			object.propertyValue = editor.variableValue
			object.propertyValue &= 7
			break

		end switch
	end if
end event

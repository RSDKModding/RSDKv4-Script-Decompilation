// ========================
// Aliases
// ========================

// Aliases
private alias 0 : BOUNDADJUST_TYPE_FROMRIGHT
private alias 1 : BOUNDADJUST_TYPE_FROMLEFT

// Player Aliases
private alias object.xpos : player.xpos

// ========================
// Events
// ========================

event ObjectUpdate
	// hardcoded Y positions are used here :(
	if object.propertyValue == BOUNDADJUST_TYPE_FROMRIGHT
		if player[SLOT_PLAYER1].xpos > object.xpos
			stage.newYBoundary2 = 520
		else
			stage.newYBoundary2 = 768
		end if
	else
		if player[SLOT_PLAYER1].xpos < object.xpos
			stage.newYBoundary2 = 520
		else
			stage.newYBoundary2 = 768
		end if
	end if
end event

// ========================
// Editor Events
// ========================

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // Type
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // Type
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event


event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-8, -8, 16, 16, 173, 67) // #0 - Trigger icon
	
	AddEditorVariable("type")
	SetActiveVariable("type")
	AddEnumVariable("Right", BOUNDADJUST_TYPE_FROMRIGHT)
	AddEnumVariable("Left",  BOUNDADJUST_TYPE_FROMLEFT)
end event

public function Player_State_Hugged
    // No hugs if you're in the air
    if player.gravity == GRAVITY_AIR
        player.state = Player_State_Air

        player.animation = ANI_WALKING
        player.timer     = 0
        CallFunction(Player_HandleAirMovement)
    else
        CallFunction(Player_HandleOnGround)

        // Yes, the only thing the function does is make you wait
        if player.timer < 240
            player.animation = ANI_STOPPED
            player.timer++
        else
            player.animation = ANI_WAITING
        end if

        // Warp cancel check
        if warp.destination > WARPDEST_NONE
            if warp.timer > 99
                if warp.timer < 204
                    warp.destination = WARPDEST_NONE
                end if
            end if
            warp.timer = 0
        end if

        // Get free from Amy
        if player.jumpPress == true
            CallFunction(Player_Action_Jump)
        end if
    end if
end function

event ObjectUpdate
	// Always face Sonic
    if player[SLOT_PLAYER1].xpos < object.xpos
        object.direction = FACING_LEFT
    else
        object.direction = FACING_RIGHT
    end if

    if object.fallBackTimer < 600
        object.fallBackTimer++
    else
        if object.fallBackTimer == 600
			// Can't run away forever!

			// It's been 10 seconds of chasing Sonic already so end the level regardless if he wants it or not
            object.fallBackTimer = 601

			// Lock Sonic's controls
            player[SLOT_PLAYER1].left  = false
            player[SLOT_PLAYER1].right = false
            player[SLOT_PLAYER1].up    = false
            player[SLOT_PLAYER1].down  = false
            player[SLOT_PLAYER1].controlMode = CONTROLMODE_NONE

            CreateTempObject(TypeName[Fade Music], FADEMUSIC_SSZAMY, object.xpos, object.ypos)
        end if
    end if

    switch object.state
    case AMY2_CHASEPLAYER
        object.timer++
        switch object.frame
        case 4
        case 6
            if object.timer == 4
                object.timer = 0
                object.frame++
            end if
            break

        case 5
        case 7
            if object.timer == 3
                object.timer = 0
                object.frame++
            end if
            break

        end switch

        if object.frame > 7
            object.frame = 4
        end if

        if object.direction == FACING_RIGHT
			// We're chasing Sonic, going right of course
            if object.xvel < 0x20000
                object.xvel += 0x2000
            end if

            BoxCollisionTest(C_TOUCH, object.entityPos, -32, -360, 4, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[SLOT_PLAYER1].gravity == GRAVITY_GROUND
                    if player[SLOT_PLAYER1].direction == object.direction
                        BoxCollisionTest(C_TOUCH, object.entityPos, -32, -20, 4, 20, 0, C_BOX, C_BOX, C_BOX, C_BOX)
                        if checkResult == true
                            if player[SLOT_PLAYER1].animation == ANI_SPINDASH
								// Sonic's doing a spindash, throw Amy back (!)
                                object.state = AMY2_RECOIL
                                object.frame = 9
                                object.yvel = -0x40000
                                object.inAir = true
                                object.xvel = -0x10000
                            else
                                if object.fallBackTimer < 600
                                    object.fallBackTimer = 601

                                    player[SLOT_PLAYER1].left  = false
                                    player[SLOT_PLAYER1].right = false
                                    player[SLOT_PLAYER1].up    = false
                                    player[SLOT_PLAYER1].down  = false
                                    player[SLOT_PLAYER1].controlMode = CONTROLMODE_NONE

                                    CreateTempObject(TypeName[Fade Music], FADEMUSIC_SSZAMY, object.xpos, object.ypos)
                                end if

                                object.state = AMY2_HUGGING
                                object.frame = 8
                                object.timer = 0
                                player[SLOT_PLAYER1].state = Player_State_Hugged
                                PlaySfx(SfxName[Giggle], false)
                            end if
                        else
                            object.state = AMY2_TRACKPLAYER
                            object.frame = 0
                            object.timer = 0
                            object.xvel  = 0
                        end if
                    else
						// Presumably some code should be here, given that the condition exists and all, and yet there isn't...
                    end if
                end if
            end if

            if object.xpos < object.bounds.right
                object.xpos += object.xvel
            else
				// Sonic has somehow escaped out of bounds to the left, stand still and patiently await his return
                object.xvel = 0
                object.xpos = object.bounds.right
                object.state = AMY2_WAITNG_R
                object.frame = 0
                object.timer = 0
            end if
        else
            if object.xvel > -0x20000
                object.xvel -= 0x2000
            end if

            BoxCollisionTest(C_TOUCH, object.entityPos, -4, -360, 32, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[SLOT_PLAYER1].gravity == GRAVITY_GROUND
                    if player[SLOT_PLAYER1].direction == object.direction
                        BoxCollisionTest(C_TOUCH, object.entityPos, -32, -20, 4, 20, 0, C_BOX, C_BOX, C_BOX, C_BOX)
                        if checkResult == true
                            if player[SLOT_PLAYER1].animation == ANI_SPINDASH
								// Sonic's doing a spindash, hit Amy back (!)
                                object.state = AMY2_RECOIL
                                object.frame = 9
                                object.yvel  = -0x40000
                                object.inAir = true
                                object.xvel  = 0x10000
                            else
                                if object.fallBackTimer < 600
                                    object.fallBackTimer = 601

                                    player[SLOT_PLAYER1].left  = false
                                    player[SLOT_PLAYER1].right = false
                                    player[SLOT_PLAYER1].up    = false
                                    player[SLOT_PLAYER1].down  = false
                                    player[SLOT_PLAYER1].controlMode = CONTROLMODE_NONE

                                    CreateTempObject(TypeName[Fade Music], FADEMUSIC_SSZAMY, object.xpos, object.ypos)
                                end if

                                object.state = AMY2_HUGGING
                                object.frame = 8
                                object.timer = 0
                                player[SLOT_PLAYER1].state = Player_State_Hugged
                                PlaySfx(SfxName[Giggle], false)
                            end if
                        else
                            object.state = AMY2_TRACKPLAYER
                            object.frame = 0
                            object.timer = 0
                            object.xvel  = 0
                        end if
                    else
						// This condition exists and yet there's no code inside of it...
                    end if
                end if
            end if

            if object.xpos > object.bounds.left
                object.xpos += object.xvel
            else
				// Sonic somehow escaped out of bounds to the right, look at him and wait
                object.xvel  = 0
                object.xpos  = object.bounds.left
                object.state = AMY2_WAITNG_L
                object.frame = 0
                object.timer = 0
            end if
        end if
        break

    case AMY2_TRACKPLAYER
        object.timer++
        if object.timer == 12
            object.timer = 0
            object.frame++
        end if

        if object.frame > 3
            object.frame = 0
        end if

        BoxCollisionTest(C_TOUCH, object.entityPos, -4, -360, 4, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == false
            if player[SLOT_PLAYER1].gravity == GRAVITY_GROUND
                object.state = AMY2_CHASEPLAYER
                object.frame = 4
                object.timer = 0
            end if
        else
            if player[SLOT_PLAYER1].gravity == GRAVITY_AIR
                object.state = AMY2_CHASEPLAYER
                object.frame = 4
                object.timer = 0
            end if
        end if
        break

    case AMY2_HUGGING
        player[SLOT_PLAYER1].speed = 0
        player[SLOT_PLAYER1].xvel = 0
        player[SLOT_PLAYER1].direction = object.direction
        break

    case AMY2_RECOIL
        if object.yvel > 0
            object.frame = 10
        end if

        if object.xvel > 0
            object.xvel -= 0x800
        end if

        if object.xvel < 0
            object.xvel += 0x800
        end if

        object.xpos += object.xvel
        if object.inAir == false
            if object.timer == 16
                object.state = AMY2_CHASEPLAYER
                object.frame = 4
                object.timer = 0
            else
                object.timer++
            end if
        end if
        break

    case AMY2_WAITNG_R
		// Staring at Sonic to the right
        object.timer++
        if object.timer == 12
            object.timer = 0
            object.frame++
        end if

        if object.frame > 3
            object.frame = 0
        end if

        if object.direction == FACING_LEFT
            object.state = AMY2_CHASEPLAYER
            object.frame = 4
            object.timer = 0
        end if
        break

    case AMY2_WAITNG_L
		// Staring at Sonic to the Left
        object.timer++
        if object.timer == 12
            object.timer = 0
            object.frame++
        end if

        if object.frame > 3
            object.frame = 0
        end if

        if object.direction == FACING_RIGHT
            object.state = AMY2_CHASEPLAYER
            object.frame = 4
            object.timer = 0
        end if
        break

    end switch

    object.heartsTimer++

	// Wait what? State 7? Where'd that come from? This object's states end at 5!
	// But I digress, spawn hearts at a faster rate while in state 7
	// (Perhaps it was the hugging state? It is for PPZ's Amy, which is what this object seems to be based on)
    if object.state == AMY2_UNUSED
        if object.heartsTimer > 15
            CreateTempObject(TypeName[Love Heart], 0, object.xpos, object.ypos)
            object[tempObjectPos].iypos -= 6

            object[tempObjectPos].inkEffect = INK_ALPHA
            object[tempObjectPos].alpha = 255

            object.heartsTimer = 0
        end if
    else
        if object.heartsTimer > 41
            CreateTempObject(TypeName[Love Heart], 0, object.xpos, object.ypos)
            object[tempObjectPos].iypos -= 6

            object[tempObjectPos].inkEffect = INK_ALPHA
            object[tempObjectPos].alpha = 255

            object.heartsTimer = 0
        end if
    end if

    if object.inAir == false
		// Track the ground
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)
    else
		// Got launched into the air by Sonic...

		// Apply air movements & gravity
        object.yvel += 0x4000
        object.ypos += object.yvel

		// See if Amy's landed yet
        ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
        if checkResult == true
            object.yvel = 0
            object.inAir = false
        end if
    end if
end event

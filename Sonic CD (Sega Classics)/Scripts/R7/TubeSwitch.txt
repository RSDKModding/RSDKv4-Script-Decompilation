event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			switch object.propertyValue
			case TUBESW_ENTER_H_L
				if object.inTube == false
					if player[currentPlayer].state != Player_State_TubeAirRoll_CD
						player[currentPlayer].state = Player_State_TubeAirRoll_CD
						player[currentPlayer].gravity     = GRAVITY_GROUND
						player[currentPlayer].prevGravity = GRAVITY_GROUND
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].minRollSpeed = 0xD0000
					else
						player[currentPlayer].state = Player_State_Roll
						player[currentPlayer].animation = ANI_JUMPING
					end if
				end if
				break

			case TUBESW_ENTER_V
				player[currentPlayer].state = Player_State_Air
				player[currentPlayer].animation = ANI_JUMPING

				if player[currentPlayer].gravity == GRAVITY_AIR
					if player[currentPlayer].yvel < 0
						player[currentPlayer].yvel = -0xD0000
					else
						player[currentPlayer].yvel = 0xD0000
					end if
				else
					switch player[currentPlayer].collisionMode
					case CMODE_LWALL
						if player[currentPlayer].speed < 0
							player[currentPlayer].yvel = 0xD0000
						else
							player[currentPlayer].yvel = -0xD0000
						end if
						break

					case CMODE_RWALL
						if player[currentPlayer].speed < 0
							player[currentPlayer].yvel = -0xD0000
						else
							player[currentPlayer].yvel = 0xD0000
						end if
						break

					end switch
				end if

				player[currentPlayer].gravity = GRAVITY_AIR
				player[currentPlayer].xpos = object.xpos
				player[currentPlayer].xvel = 0
				player[currentPlayer].speed = 0
				break

			case TUBESW_ENTER_V_TUBE
				if object.inTube == false
					if player[currentPlayer].state != Player_State_TubeAirRoll_CD
						player[currentPlayer].xpos = object.xpos
						player[currentPlayer].state = Player_State_TubeAirRoll_CD
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].minRollSpeed = 0xD0000
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].prevGravity = GRAVITY_GROUND
						player[currentPlayer].xvel = 0

						if player[currentPlayer].yvel < 0
							player[currentPlayer].speed = -0xD0000
						else
							player[currentPlayer].speed = 0xD0000
						end if

						player[currentPlayer].yvel = 0
						player[currentPlayer].right = false
						player[currentPlayer].left = false
					else
						player[currentPlayer].state = Player_State_Air
						player[currentPlayer].animation = ANI_JUMPING

						switch player[currentPlayer].collisionMode
						case CMODE_LWALL
							if player[currentPlayer].speed < 0
								player[currentPlayer].yvel = 0xD0000
							else
								player[currentPlayer].yvel = -0xD0000
							end if
							break

						case CMODE_RWALL
							if player[currentPlayer].speed < 0
								player[currentPlayer].yvel = -0xD0000
							else
								player[currentPlayer].yvel = 0xD0000
							end if
							break

						end switch

						player[currentPlayer].gravity = GRAVITY_AIR
						player[currentPlayer].xpos = object.xpos
						player[currentPlayer].xvel = 0
						player[currentPlayer].speed = 0
					end if
				end if
				break

			case TUBESW_H_LABYRINTH
				if object.inTube == false
					if player[currentPlayer].collisionMode == CMODE_ROOF
						player[currentPlayer].collisionMode = CMODE_FLOOR
						player[currentPlayer].angle = 0
						FlipSign(player[currentPlayer].speed)
						FlipSign(player[currentPlayer].xvel)
					end if
				end if
				break

			case TUBESW_ENTER_H_R
				if object.inTube == false
					player[currentPlayer].state = Player_State_Roll
					player[currentPlayer].animation = ANI_JUMPING

					if player[currentPlayer].xpos < object.xpos
						if player[currentPlayer].speed < 0xA0000
							player[currentPlayer].speed = 0xA0000
						end if
						player[currentPlayer].direction = FACING_RIGHT
					else
						if player[currentPlayer].speed > -0xA0000
							player[currentPlayer].speed = -0xA0000
						end if
						player[currentPlayer].direction = FACING_LEFT
					end if
				end if
				break

			case TUBESW_V_LABYRINTH
				if player[currentPlayer].yvel < 0
					if player[currentPlayer].state != Player_State_TubeAirRoll_CD
						if player[currentPlayer].left == true
							player[currentPlayer].state = Player_State_TubeAirRoll_CD
							player[currentPlayer].gravity = GRAVITY_GROUND
							player[currentPlayer].prevGravity = GRAVITY_GROUND
							player[currentPlayer].collisionMode = CMODE_RWALL
							player[currentPlayer].angle = 64
							player[currentPlayer].animation = ANI_JUMPING
							player[currentPlayer].minRollSpeed = 0xD0000
							player[currentPlayer].speed = -0xD0000
							player[currentPlayer].xvel = 0
							player[currentPlayer].yvel = 0
							player[currentPlayer].xpos -= 0x80000
						end if

						if player[currentPlayer].right == true
							player[currentPlayer].state = Player_State_TubeAirRoll_CD
							player[currentPlayer].gravity = GRAVITY_GROUND
							player[currentPlayer].prevGravity = GRAVITY_GROUND
							player[currentPlayer].collisionMode = CMODE_LWALL
							player[currentPlayer].angle = 192
							player[currentPlayer].animation = ANI_JUMPING
							player[currentPlayer].minRollSpeed = 0xD0000
							player[currentPlayer].speed = 0xD0000
							player[currentPlayer].xvel = 0
							player[currentPlayer].yvel = 0
							player[currentPlayer].xpos += 0x80000
						end if
					end if
				end if
				break

			case TUBESW_LABYRINTH_SPRING
				if object.inTube == false
					if player[currentPlayer].state == Player_State_Air
						player[currentPlayer].xpos = object.xpos
						player[currentPlayer].xvel = 0
						player[currentPlayer].speed = 0
					end if
				end if
				break

			case TUBESW_LABYRINTH_EXIT_UPWARDS
				if object.inTube == false
					if player[currentPlayer].collisionMode == CMODE_ROOF
						player[currentPlayer].collisionMode = CMODE_FLOOR
						player[currentPlayer].angle += 128
						FlipSign(player[currentPlayer].speed)
						FlipSign(player[currentPlayer].xvel)
					end if
				end if
				break

			end switch

			object.inTube = true
		else
			object.inTube = false
		end if
	next
end event

event RSDKDraw
	// This object doesn't have a debug view, so this is custom
	// It draws a square out of tubeswitch icons
	temp0 = object.xpos
	temp0 -= 0x80000
	temp1 = object.ypos
	temp1 -= 0x80000
	DrawSpriteXY(0, temp0, temp1)

	temp0 += 0x100000
	DrawSpriteXY(0, temp0, temp1)

	temp1 += 0x100000
	DrawSpriteXY(0, temp0, temp1)

	temp0 -= 0x100000
	DrawSpriteXY(0, temp0, temp1)

	if editor.showGizmos == true
		// Draw the tube switch's hitbox, purposefully ignoring Editor.DrawingOverlay
		temp0 = object.ixpos
		temp0 -= 16
		temp1 = object.iypos
		temp1 -= 16
		DrawRectOutline(temp0, temp1, 32, 32, 255, 255, 255, 255)
	end if
end event

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

public function SpinningTop_PlayerJump
	temp1 = 0
	if temp1 == 0
		player[currentPlayer].controlLock = 0
		player[currentPlayer].gravity = 1
		temp1 = player[currentPlayer].jumpStrength
		temp1 += player[currentPlayer].gravityStrength  // temp1 is just thrown into the void here...
		player[currentPlayer].yvel = player[currentPlayer].jumpStrength
		FlipSign(player[currentPlayer].yvel)
		GetTableValue(player[currentPlayer].speed, temp7, SpinningTop_PlayerSpeeds)
		player[currentPlayer].scrollTracking = 1
		player[currentPlayer].animation = ANI_JUMPING
		player[currentPlayer].angle = 0
		player[currentPlayer].collisionMode = 0
		player[currentPlayer].timer = 1
		player[currentPlayer].state = Player_State_Air
		player[currentPlayer].collisionDisabled = true
		player[currentPlayer].jumpAbilityState = 1
	end if
end function

event ObjectUpdate
	temp7 = 0
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		GetBit(temp6, object.activePlayers, temp7)
		SetBit(object.activePlayers, temp7, 0)
		temp3 = temp7
		temp3 <<= 3
		temp2 = 255
		temp2 <<= temp3
		temp2 ^= -1
		temp4 = object.playerRadius
		temp4 >>= temp3
		temp4 &= 255
		object.playerRadius &= temp2
		temp5 = object.playerAngles
		temp5 >>= temp3
		temp5 &= 255
		object.playerAngles &= temp2
		BoxCollisionTest(C_SOLID, object.entityPos, -16, -8, 16, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			SetBit(object.activePlayers, temp7, 1)
			if player[currentPlayer].state != Player_State_SpinningTop
				player[currentPlayer].state = Player_State_SpinningTop
				temp0 = player[currentPlayer].xpos
				temp0 -= object.xpos
				temp0 >>= 16
				if temp0 < 0
					FlipSign(temp0)
					temp5 = 128
				else
					temp5 = 0
				end if
				temp4 = temp0
				SetTableValue(player[currentPlayer].speed, temp7, SpinningTop_PlayerSpeeds)
				player[currentPlayer].animation = ANI_SPINNING_TOP
			end if
			player[currentPlayer].xvel = 0
			player[currentPlayer].speed = 0
			temp5 += 8
			Cos256(temp0, temp5)
			temp0 *= temp4
			temp0 <<= 8
			temp0 += object.xpos
			player[currentPlayer].xpos = temp0
			temp0 = temp5
			temp0 >>= 4
			GetTableValue(player[0].frame, temp0, SpinningTop_PlayerFrames)
			temp0 = temp5
			temp0 &= 63
			if temp0 == 0
				temp4++
			end if
			if player[currentPlayer].xpos < object.xpos
				if player[currentPlayer].left == 1
					temp4++
				end if
				if player[currentPlayer].right == 1
					temp4--
					if object.playerRadius == -1
						temp4 = 0
					end if
				end if
			else
				if player[currentPlayer].left == 1
					temp4--
					if temp4 == -1
						temp4 = 0
					end if
				end if
				if player[currentPlayer].right == 1
					temp4++
				end if
			end if
			if player[currentPlayer].jumpPress == 1
				CallFunction(SpinningTop_PlayerJump)
				if player[currentPlayer].state == Player_State_Air
					PlaySfx(SfxName[Shot], 0)
					SetBit(object.activePlayers, temp7, 0)
				end if
			end if
		else
			if temp6 == 1
				player[currentPlayer].state = Player_State_Air
				player[currentPlayer].animation = ANI_WALKING
				GetTableValue(player[currentPlayer].speed, temp7, SpinningTop_PlayerSpeeds)
			end if
		end if
		temp4 &= 255
		temp4 <<= temp3
		object.playerRadius += temp4
		temp5 &= 255
		temp5 <<= temp3
		object.playerAngles += temp5
		temp7++
	next
	temp0 = oscillation
	temp0 &= 1
	if temp0 == 0
		object.frame++
		if object.frame > 2
			object.frame = 0
		end if
	end if
end event

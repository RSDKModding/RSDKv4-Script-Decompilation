event ObjectUpdate
    switch object.state
    case CLEDGE3_DELAY_FROM_RIGHT
        if object.timer > 0
            object.timer--
        else
            PlaySfx(SfxName[Crumble], false)
            object.state = CLEDGE3_COLLAPSE_RIGHT
        end if
        break

    case CLEDGE3_DELAY_FROM_LEFT
        if object.timer > 0
            object.timer--
        else
            PlaySfx(SfxName[Crumble], false)
            object.state = CLEDGE3_COLLAPSE_LEFT
        end if
        break

    case CLEDGE3_COLLAPSE_RIGHT
        object.timer++
        if object.timer == 12
            object.timer = 0
            if object.hitboxLeft < object.hitboxRight
                object.length--
                object.hitboxLeft += 16

                temp0 = object.hitboxLeft
                temp0 -= 8
                temp0 <<= 16
                temp0 += object.xpos

                CreateTempObject(TypeName[Crumble Ledge 3], 0, temp0, object.ypos)
                object[tempObjectPos].state = CLEDGE3_DEBRIS
                object[tempObjectPos].hitboxTop = object.hitboxTop

                if object.frame < 2
                    object[tempObjectPos].frame = object.frame
                else
                    temp0 = object.propertyValue
                    temp0--
                    if object.length == temp0
                        object[tempObjectPos].frame = 4
                    else
                        if object.length == 0
                            object[tempObjectPos].frame = 6
                        else
                            object[tempObjectPos].frame = 5
                        end if
                    end if
                end if
            else
                ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            end if
        end if
        break

    case CLEDGE3_COLLAPSE_LEFT
        object.timer++
        if object.timer == 12
            object.timer = 0
            if object.hitboxRight > object.hitboxLeft
                object.length--
                object.hitboxRight -= 16

                temp0 = object.hitboxRight
                temp0 += 8
                temp0 <<= 16
                temp0 += object.xpos

                CreateTempObject(TypeName[Crumble Ledge 3], 0, temp0, object.ypos)
                object[tempObjectPos].state = CLEDGE3_DEBRIS
                object[tempObjectPos].hitboxTop = object.hitboxTop

                if object.frame < 2
                    object[tempObjectPos].frame = object.frame
                else
                    temp0 = object.propertyValue
                    temp0--
                    if object.length == temp0
                        object[tempObjectPos].frame = 6
                    else
                        if object.length == 0
                            object[tempObjectPos].frame = 4
                        else
                            object[tempObjectPos].frame = 5
                        end if
                    end if
                end if
            else
                ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            end if
        end if
        break

    case CLEDGE3_RESTORE
        if object.outOfBounds == true
            object.state = CLEDGE3_IDLE
            object.length = object.propertyValue
            object.hitboxLeft = object.propertyValue
            object.hitboxLeft <<= 3
            object.hitboxRight = object.hitboxLeft
            FlipSign(object.hitboxLeft)
            object.priority = PRIORITY_BOUNDS
            object.timer = 0
        end if
        break

    case CLEDGE3_DEBRIS
        object.yvel += 0x4000
        if object.yvel > 0
            object.ypos += object.yvel
        end if

        if object.outOfBounds == true
            object.type = TypeName[Blank Object]
        end if
        break
    end switch

    if object.state == CLEDGE3_IDLE
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, object.hitboxTop, object.hitboxRight, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.timer = 1
                object.priority = PRIORITY_ACTIVE
                if player[currentPlayer].xpos > object.xpos
                    object.state = CLEDGE3_DELAY_FROM_RIGHT
                else
                    object.state = CLEDGE3_DELAY_FROM_LEFT
                end if
            end if
        next
    else
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, object.hitboxTop, object.hitboxRight, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        next
    end if
end event

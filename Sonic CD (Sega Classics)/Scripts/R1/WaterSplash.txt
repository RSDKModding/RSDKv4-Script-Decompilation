// Aliases
private alias object.value0 : object.timer

// States
private alias 0 : WATERSPLASH_IDLE
private alias 1 : WATERSPLASH_DELAY
private alias 2 : WATERSPLASH_DRAW

// ========================
// Events
// ========================

event ObjectUpdate
    if object.state == WATERSPLASH_IDLE
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_TOUCH, object.entityPos, -16, 0, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.timer = 4
                object.state = WATERSPLASH_DELAY
                object.drawOrder = 4
                object.frame = 0
            end if
        next
    end if
end event


event ObjectDraw
    switch object.state
    case WATERSPLASH_DELAY
        if object.timer > 0
            object.timer--
        else
            object.state = WATERSPLASH_DRAW
            PlaySfx(SfxName[Water Splash], false)
        end if
        break

    case WATERSPLASH_DRAW
        DrawSprite(object.frame)
        object.timer++
        if object.timer > 4
            object.timer = 0
            object.frame++
            if object.frame > 4
                object.state = WATERSPLASH_IDLE
            end if
        end if
        break
    end switch
end event

// Amy Rose's favorite function
public function Player_State_Hugged
    // No hugs if you're in the air
    if player.gravity == GRAVITY_AIR
        player.state = Player_State_Air

        player.animation = ANI_WALKING
        player.timer     = 0
        CallFunction(Player_HandleAirMovement)
    else
        CallFunction(Player_HandleOnGround)

        // Yes, the only thing the function does is make you wait
        if player.timer < 240
            player.animation = ANI_STOPPED
            player.timer++
        else
            player.animation = ANI_WAITING
        end if

        // Warp cancel check
        if warp.destination > WARPDEST_NONE
            if warp.timer > 99
                if warp.timer < 204
                    warp.destination = WARPDEST_NONE
                end if
            end if
            warp.timer = 0
        end if

        // Get free from Amy
        if player.jumpPress == true
            CallFunction(Player_Action_Jump)
        end if
    end if
end function

public function AmyRose_DebugSpawn
	CreateTempObject(TypeName[Amy Rose], 4, object.xpos, object.ypos)

	object[tempObjectPos].leftLimit = object[arrayPos0].xpos
	object[tempObjectPos].leftLimit -= 0x1400000

	object[tempObjectPos].rightLimit = object[arrayPos0].xpos
	object[tempObjectPos].rightLimit += 0x400000

	object[tempObjectPos].drawOrder = 4

	object[tempObjectPos].gravity = GRAVITY_AIR
end function

event ObjectUpdate
	object.priority = PRIORITY_ACTIVE
	if player[SLOT_PLAYER1].xpos < object.xpos
		object.direction = FACING_LEFT
	else
		object.direction = FACING_RIGHT
	end if

	switch object.state
	case AMY_IDLE
		temp0 = object.frameTimer
		temp0 /= 10
		temp0 += 11
		object.frame = temp0
		object.frameTimer++

		if object.frameTimer > 139
			object.frameTimer = 0
		end if

		if player[SLOT_PLAYER1].type != TypeName[Debug Mode]
			// Check to see if Sonic's anywhere near, and if so, start giving chase!
			BoxCollisionTest(C_TOUCH, object.entityPos, -96, -360, 96, 360, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.xvel = 0
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			end if
		end if
		break

	case AMY_CHASE
		object.frameTimer++

		switch object.frame
		case 4
		case 6
			if object.frameTimer == 4
				object.frameTimer = 0
				object.frame++
			end if
			break

		case 5
		case 7
			if object.frameTimer == 3
				object.frameTimer = 0
				object.frame++
			end if
			break
		end switch

		if object.frame > 7
			object.frame = 4
		end if

		if object.direction == FACING_RIGHT
			if object.xvel < 0x20000
				object.xvel += 0x1000
			end if

			BoxCollisionTest(C_TOUCH, object.entityPos, -32, -360, 4, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				if player[SLOT_PLAYER1].gravity == GRAVITY_GROUND
					if player[SLOT_PLAYER1].direction == object.direction
						BoxCollisionTest(C_TOUCH, object.entityPos, -32, -20, 4, 20, 0, C_BOX, C_BOX, C_BOX, C_BOX)
						if checkResult == true
							CheckEqual(player[SLOT_PLAYER1].animation, ANI_SPINDASH)
							temp0 = checkResult

							// Before commencing the hugging, make sure Sonic's not already being hugged by someone else first
							// -> This check was only added in via an update, so it's not present in the Steam version of the game
							CheckEqual(object[1].type, TypeName[Blue Shield])
							temp0 |= checkResult

							if temp0 == true
								object.state = AMY_REJECTED
								object.frame = 9
								object.yvel = -0x40000
								object.gravity = GRAVITY_AIR
								object.xvel = -0x10000
							else
								object.sonicHugged = true
								object.state = AMY_HUG
								object.frame = 8
								object.frameTimer = 0
								player[SLOT_PLAYER1].state = Player_State_Hugged
								PlaySfx(SfxName[Giggle], false)

								// Just one hug is enough unlock criteria
								if stage.debugMode == false
									// (unused)
								end if
							end if
						else
							object.state = AMY_HEART_STARE
							object.frame = 0
							object.frameTimer = 0
							object.xvel = 0
						end if
					else
						object.state = AMY_HEART_STARE
						object.frame = 0
						object.frameTimer = 0
						object.xvel = 0
					end if
				end if
			end if

			if object.xpos < object.rightLimit
				object.xpos += object.xvel
			else
				object.xvel = 0
				object.xpos = object.rightLimit

				if object.sonicHugged == true
					object.state = AMY_CHEER_RIGHT
				else
					object.state = AMY_AWAIT_RIGHT
					object.frame = 0
					object.frameTimer = 0
				end if

			end if
		else
			if object.xvel > -0x20000
				object.xvel -= 0x1000
			end if

			BoxCollisionTest(C_TOUCH, object.entityPos, -4, -360, 32, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				if player[SLOT_PLAYER1].gravity == GRAVITY_AIR
					if player[SLOT_PLAYER1].direction == object.direction
						BoxCollisionTest(C_TOUCH, object.entityPos, -32, -20, 4, 20, 0, C_BOX, C_BOX, C_BOX, C_BOX)
						if checkResult == true
							CheckEqual(player[SLOT_PLAYER1].animation, ANI_SPINDASH)
							temp0 = checkResult

							// Make sure Sonic isn't wearing a shield before having Amy grab on
							// -> This change was only introduced in an update, the Steam version doesn't have this check
							CheckEqual(object[1].type, TypeName[Blue Shield])
							temp0 |= checkResult

							if temp0 == true
								object.state = AMY_REJECTED
								object.frame = 9
								object.yvel = -0x40000
								object.gravity = GRAVITY_AIR
								object.xvel = 0x10000
							else
								object.sonicHugged = true
								object.state = AMY_HUG
								object.frame = 8
								object.frameTimer = 0
								player[SLOT_PLAYER1].state = Player_State_Hugged
								PlaySfx(SfxName[Giggle], false)

								// Just one hug is enough unlock criteria
								if stage.debugMode == false
									// (unused)
								end if
							end if
						else
							object.state = AMY_HEART_STARE
							object.frame = 0
							object.frameTimer = 0
							object.xvel = 0
						end if
					else
						object.state = AMY_HEART_STARE
						object.frame = 0
						object.frameTimer = 0
						object.xvel = 0
					end if
				end if
			end if

			if object.xpos > object.leftLimit
				object.xpos += object.xvel
			else
				object.xvel = 0
				object.xpos = object.leftLimit
				if object.sonicHugged == true
					object.state = AMY_AWAIT_LEFT
				else
					object.state = AMY_CHEER_LEFT
					object.frame = 0
					object.frameTimer = 0
				end if
			end if
		end if
		break

	case AMY_AWAIT_RIGHT
		object.frameTimer++
		if object.frameTimer == 12
			object.frameTimer = 0
			object.frame++
		end if

		if object.frame > 3
			object.frame = 0
		end if

		if object.direction == FACING_LEFT
			object.state = AMY_CHASE
			object.frame = 4
			object.frameTimer = 0
		end if
		break

	case AMY_CHEER_LEFT
		object.frameTimer++
		if object.frameTimer == 12
			object.frameTimer = 0
			object.frame++
		end if

		if object.frame > 3
			object.frame = 0
		end if

		if object.direction == FACING_RIGHT
			object.state = AMY_CHASE
			object.frame = 4
			object.frameTimer = 0
		end if
		break

	case AMY_CHEER_RIGHT
		if object.frameTimer < 8
			object.frame = 0
		else
			if object.frameTimer > 47
				if object.gravity == GRAVITY_GROUND
					if object.frameTimer < 120
						object.yvel = -0x40000
						object.gravity = GRAVITY_AIR
						object.frame = 9
					else
						object.frameTimer = 0
					end if
				else
					if object.yvel > 0
						object.frame = 0
					end if
				end if
			else
				temp0 = object.frameTimer
				temp0 -= 8
				temp0 %= 20
				temp0 /= 10
				object.frame = temp0
				object.frame += 9
			end if
		end if

		if object.gravity == GRAVITY_GROUND
			if object.direction == FACING_LEFT
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			end if
		end if

		object.frameTimer++
		break

	case AMY_AWAIT_LEFT
		if object.frameTimer < 8
			object.frame = 0
		else
			if object.frameTimer > 47
				if object.gravity == GRAVITY_GROUND
					if object.frameTimer < 120
						object.yvel = -0x40000
						object.gravity = GRAVITY_AIR
						object.frame = 9
					else
						object.frameTimer = 0
					end if
				else
					if object.yvel > 0
						object.frame = 0
					end if
				end if
			else
				temp0 = object.frameTimer
				temp0 -= 8
				temp0 %= 20
				temp0 /= 10
				object.frame = temp0
				object.frame += 9
			end if
		end if

		if object.gravity == GRAVITY_GROUND
			if object.direction == FACING_RIGHT
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			end if
		end if

		object.frameTimer++
		break

	case AMY_HEART_STARE
		object.frameTimer++
		if object.frameTimer == 12
			object.frameTimer = 0
			object.frame++
		end if

		if object.frame > 3
			object.frame = 0
		end if

		BoxCollisionTest(C_TOUCH, object.entityPos, -4, -360, 4, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == false
			if player[SLOT_PLAYER1].gravity == GRAVITY_GROUND
				// If you're not exactly above her, she'll start chasing you again
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			end if
		else
			if player[SLOT_PLAYER1].gravity == GRAVITY_AIR
				// If you jump while she waits for you, she'll start chasing you again
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			end if
		end if
		break

	case AMY_HUG
		player[SLOT_PLAYER1].speed = 0
		player[SLOT_PLAYER1].xvel = 0
		player[SLOT_PLAYER1].direction = object.direction

		if player[SLOT_PLAYER1].animation == ANI_JUMPING_A // not the global, odd?
			object.state = AMY_REJECTED
			object.frame = 9
			object.yvel = -0x40000
			object.gravity = GRAVITY_AIR
			if object.direction == FACING_RIGHT
				object.xvel = -0x10000
			else
				object.xvel = 0x10000
			end if
		end if
		break

	case AMY_REJECTED
		if object.yvel > 0
			object.frame = 10
		end if

		if object.xvel > 0
			object.xvel -= 0x800
		end if

		if object.xvel < 0
			object.xvel += 0x800
		end if

		object.xpos += object.xvel

		if object.gravity == GRAVITY_GROUND
			if object.frameTimer == 16
				object.state = AMY_CHASE
				object.frame = 4
				object.frameTimer = 0
			else
				object.frameTimer++
			end if
		end if

		if object.xpos > object.rightLimit
			object.xpos = object.rightLimit
		end if
		break

	end switch

	object.heartTimer++

	if object.state == AMY_HUG
		if object.heartTimer > 15
			CreateTempObject(TypeName[Love Heart], 0, object.xpos, object.ypos)
			object[tempObjectPos].iypos -= 6
			object[tempObjectPos].inkEffect = INK_ALPHA
			object[tempObjectPos].alpha = 255
			object.heartTimer = 0
		end if
	else
		if object.heartTimer > 41
			CreateTempObject(TypeName[Love Heart], 0, object.xpos, object.ypos)
			object[tempObjectPos].iypos -= 6
			object[tempObjectPos].inkEffect = INK_ALPHA
			object[tempObjectPos].alpha = 255
			object.heartTimer = 0
		end if
	end if

	if object.gravity == GRAVITY_GROUND
		ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)
	else
		object.yvel += 0x4000
		object.ypos += object.yvel
		ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
		if checkResult == true
			object.yvel = 0
			object.gravity = GRAVITY_GROUND
		end if
	end if

	if object.outOfBounds == true
		object.priority = PRIORITY_BOUNDS
		object.state = AMY_IDLE
		object.sonicHugged = false
	end if
end event

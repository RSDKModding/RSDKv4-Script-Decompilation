event ObjectUpdate
    object.timer++
    if object.timer == 8
        object.timer = 0
        object.currentPalette++
        object.currentPalette %= 3
        RotatePalette(0, 177, 179, true)
        RotatePalette(0, 172, 174, false)
    end if
end event

// Aliases
private alias object.xvel   : object.offset.x
private alias object.yvel   : object.offset.y
private alias object.value0 : object.faceEntity

// Face Aliases
private alias object.xpos : face.xpos
private alias object.ypos : face.ypos

// States
private alias 0 : BOSSSHOULDER_SETUP
private alias 1 : BOSSSHOULDER_IDLE
private alias 2 : BOSSSHOULDER_DESTROYED

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case BOSSSHOULDER_SETUP
		object.state = BOSSSHOULDER_IDLE
		// [Fallthrough]

	case BOSSSHOULDER_IDLE
		arrayPos0 = object.faceEntity

		object.xpos = face[arrayPos0].xpos
		object.xpos += 0x180000

		object.ypos = face[arrayPos0].ypos
		object.ypos -= 0xC0000
		break

	case BOSSSHOULDER_DESTROYED
		object.offset.x -= 0x600
		object.offset.y += 0x1800

		object.xpos += object.offset.x
		object.ypos += object.offset.y

		if object.outOfBounds == true
			object.type = TypeName[Blank Object]
			object.priority = PRIORITY_BOUNDS
		end if
		break
	end switch
end event


event ObjectDraw
	DrawSprite(0)
end event

// ========================

// Aliases
private alias object.value0 : object.originPos.x // Set by Ramp3D

// Property Values
private alias 0 : FOREGROUND_BUSH // This one here is still bugged, just like in the original 2011 remaster.
private alias 1 : SMALL_BUSH
private alias 2 : BACKGROUND_BUSH

// ========================
// Events
// ========================

event ObjectDraw
    if object.propertyValue == FOREGROUND_BUSH
        DrawSpriteXY(0, object.originPos.x, object.ypos)
    else
        DrawSprite(object.propertyValue)
    end if
end event


event ObjectStartup
    foreach (TypeName[Star Bush], arrayPos0, ALL_ENTITIES)
        if object[arrayPos0].propertyValue == FOREGROUND_BUSH
            object[arrayPos0].drawOrder = 4 // This should be 5.
        end if
        if object[arrayPos0].propertyValue == BACKGROUND_BUSH
            object[arrayPos0].drawOrder = 4
        end if
    next

    LoadSpriteSheet("R1/Objects2.gif")
    
    SpriteFrame(-24, -20, 48, 40, 115, 131)
    SpriteFrame(-12, -12, 24, 24, 1, 229)
    SpriteFrame(-24, -20, 48, 40, 115, 131)
end event


// ========================
// Editor Events
// ========================

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // property value
            checkResult = object.propertyValue
            checkResult &= 2
            break
            
        case 0 // direction
            checkResult = object.propertyValue
            checkResult &= 2
            break       
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // property value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 2
            break
            
        case 0 // direction
            object.propertyValue = editor.variableValue
            object.propertyValue &= 2
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event


event RSDKLoad
    LoadSpriteSheet("R1/Objects2.gif")

    SpriteFrame(-24, -20, 48, 40, 115, 131)     // #0 - Big StarBush
    SpriteFrame(-12, -12, 24, 24, 1, 229)       // #1 - Small StarBush
    SpriteFrame(-24, -20, 48, 40, 115, 131)     // #2 - Big StarBush (Background)

    AddEditorVariable("type")
    SetActiveVariable("type")
    AddEnumVariable("Big Bush (3D Ramp)", 0)
    AddEnumVariable("Small Bush", 1)
    AddEnumVariable("Big Bush (Background)", 2)
end event

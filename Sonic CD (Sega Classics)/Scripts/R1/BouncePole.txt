event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        // Check:
        // Collision with object
        // Player is in the air
        // The vertical momentum goes downwards
        BoxCollisionTest(C_TOUCH, object.entityPos, -16, -8, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            if player[currentPlayer].gravity == GRAVITY_AIR
                if player[currentPlayer].yvel > 0
                    player[currentPlayer].state = Player_State_Air
                    player[currentPlayer].timer = 0
                    player[currentPlayer].animation = ANI_JUMPING

                    FlipSign(player[currentPlayer].yvel)
                    player[currentPlayer].yvel -= 0x20000

                    // Vertical velocity cap
                    if player[currentPlayer].yvel < -0xA0000
                        player[currentPlayer].yvel = -0xA0000
                    end if

                    // Start bounce pole animation
                    object.frame = 1
                end if
            end if
        end if

        if object.frame > 0
            object.frameTimer++
            if object.frameTimer == 3
                object.frameTimer = 0

                object.frame++
                if object.frame > 20
                    object.frame = 0
                end if
            end if
        end if
    next
end event


event ObjectDraw

    DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
end event

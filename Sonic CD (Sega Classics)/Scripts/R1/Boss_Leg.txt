// ========================
// Function Declarations

reserve function BossLeg_SaveLegPosition
reserve function BossLeg_HandleMovement

// ========================
// Function Definitions
// ========================

public function BossLeg_SaveLegPosition
    GetBit(temp0, object.legSide, 5)
    if temp0 == true
        arrayPos0 = object.legJointEntity
        legJoint[arrayPos0].xpos += temp1
        legJoint[arrayPos0].ypos += temp2

        arrayPos1 = legJoint[arrayPos0].bodyEntity
        body[arrayPos1].xpos += temp1
        body[arrayPos1].ypos += temp2

        arrayPos0 = body[arrayPos1].faceEntity
        face[arrayPos0].xpos += temp1
        face[arrayPos0].ypos += temp2
    end if
end function


public function BossLeg_HandleMovement
    arrayPos0 = object.legJointEntity
    object.xpos = legJoint[arrayPos0].xpos
    object.xpos += 0x40000
    object.xpos += object.velocity.x

    object.ypos = legJoint[arrayPos0].ypos
    object.ypos += 0x100000
    object.ypos += object.velocity.y
end function


// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case BOSSLEG_SETUP
        object.offset.x = 0x8000
        object.offset.y = 0x4000
        object.state = BOSSLEG_LANDING
        break

    case BOSSLEG_LANDING
        arrayPos0 = object.legJointEntity

        object.xpos = legJoint[arrayPos0].xpos
        object.xpos += 0x40000

        object.ypos = legJoint[arrayPos0].ypos
        object.ypos += 0x100000
        break

    case BOSSLEG_L_MIMIC_R
        // No matter the leg, it will end up picking the right leg here
        arrayPos0 = object.legJointEntity
        arrayPos1 = legJoint[arrayPos0].bodyEntity
        arrayPos0 = body[arrayPos1].legJointREntity
        arrayPos1 = legJointR[arrayPos0].legREntity

        object.xpos = legR[arrayPos1].xpos
        object.xpos -= 0xA0000
        object.ypos = legR[arrayPos1].ypos
        break

    case BOSSLEG_MOVE // whatever leg this is, it will move
        arrayPos0 = object.footEntity
        if object.velocity.x < 0x80000
            temp1 = object.offset.x
            object.velocity.x += temp1

            GetBit(temp0, object.legSide, 4)
            if temp0 == true
                FlipSign(temp1)
                temp2 = false
                CallFunction(BossLeg_SaveLegPosition)
            else
                CallFunction(BossLeg_HandleMovement)
            end if
        else
            CallFunction(BossLeg_HandleMovement)
        end if
        break

    case BOSSLEG_MOVE_OPPOSITE // whatever leg this is, it will wait for the opposite to move
        arrayPos0 = object.footEntity
        if object.velocity.x > -0x80000
            temp1 = object.offset.x
            object.velocity.x -= temp1

            GetBit(temp0, object.legSide, 4)
            if temp0 == true
                temp2 = false
                CallFunction(BossLeg_SaveLegPosition)
            else
                CallFunction(BossLeg_HandleMovement)
            end if
        else
            CallFunction(BossLeg_HandleMovement)
        end if
        break

    case BOSSLEG_MOVE_C // whatever leg this is, it will move
        if object.velocity.x > -0x80000
            temp1 = object.offset.x
            object.velocity.x -= temp1
        else
            temp1 = false
        end if

        if object.velocity.y > -0x40000
            temp2 = object.offset.y
            object.velocity.y -= temp2
        else
            temp2 = false
        end if

        GetBit(temp0, object.legSide, 4)
        if temp0 == true
            CallFunction(BossLeg_SaveLegPosition)
        else
            CallFunction(BossLeg_HandleMovement)
        end if
        break

    case BOSSLEG_MOVE_OPPOSITE_C // whatever leg this is, it will wait for it's opposite to move
        if object.velocity.x > -0x80000
            temp1 = object.offset.x
            object.velocity.x -= temp1
        else
            temp1 = false
        end if

        if object.velocity.y < 0x40000
            temp2 = object.offset.y
            object.velocity.y += temp2
        else
            temp2 = false
        end if

        GetBit(temp0, object.legSide, 4)
        if temp0 == true
            FlipSign(temp2)
            CallFunction(BossLeg_SaveLegPosition)
        else
            CallFunction(BossLeg_HandleMovement)
        end if
        break

    case BOSSLEG_DESTROYED
        GetBit(temp0, object.legSide, 4)
        if temp0 == false
            object.offset.x -= 0x600
        else
            object.offset.x += 0x600
        end if

        object.offset.y += 0x1A60

        object.xpos += object.offset.x
        object.ypos += object.offset.y

        if object.outOfBounds == true
            object.type = TypeName[Blank Object]
            object.priority = PRIORITY_BOUNDS
        end if
        break
    end switch

    if object.state < BOSSLEG_DESTROYED
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            if player[currentPlayer].invincibleTimer == 0
                BoxCollisionTest(C_TOUCH, object.entityPos, -12, -20, 12, -20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    arrayPos0 = object.legJointEntity
                    arrayPos1 = legJoint[arrayPos0].bodyEntity
                    arrayPos0 = body[arrayPos1].faceEntity

                    face[arrayPos0].animation = EGGANI_LAUGH
                    face[arrayPos0].faceTimer = 180
                    CallFunction(Player_Hit)
                end if
            end if
        next
    end if
end event

// Fake Table for Oscillation of the boss height
public function R1_Eggman_Oscillation
    switch object.oscillation
    case 0
        temp0 = 1
        break
    case 1
        temp0 = 0
        break
    case 2
        temp0 = 2
        break
    case 3
        temp0 = 0
        break
    case 4
        temp0 = 3
        break
    case 5
        temp0 = 0
        break
    case 6
        temp0 = 4
        break
    case 7
        temp0 = 5
        break
    case 8
        temp0 = 5
        break
    case 9
        temp0 = 6
        break
    case 10
        temp0 = 6
        break
    case 11
        temp0 = 10
        break
    case 12
        temp0 = 4
        break
    case 13
        temp0 = 10
        break
    case 14
        temp0 = 8
        break
    case 15
        temp0 = 50
        break
    case 16
        temp0 = 5
        break
    case 17
        temp0 = 10
        break
    case 18
    case 19
        temp0 = -1
        break
    end switch
end function

event ObjectUpdate
    switch object.state
    case R1_EGGMAN_SETUP
        object.animation = EGGANI_NONE
        object.priority = PRIORITY_ACTIVE

        object.bodyEntity = object.entityPos
        object.bodyEntity += 1

        object.shoulderEntity = object.entityPos
        object.shoulderEntity += 8

        object.newXBoundary1Value = bossBody.ixpos

        object.newXBoundary2Value = object.newXBoundary1Value
        object.newXBoundary2Value += 320

        temp0 = screen.xcenter
        temp0 -= 160
        temp0 >>= 1
        object.newXBoundary2Value += temp0

        object.savedXBoundary2 = stage.curXBoundary2
        object.savedYBoundary2 = stage.curYBoundary2

        stage.newXBoundary2 = object.newXBoundary2Value

        object.state = R1_EGGMAN_WAIT_PLAYER
        break

    case R1_EGGMAN_WAIT_PLAYER
        temp0 = object.newXBoundary1Value
        if screen.xcenter > 160
            temp0 += 180
        else
            temp0 += 160
        end if

        temp0 <<= 16

        if player[SLOT_PLAYER1].xpos > temp0
            temp0 = object.ypos
            temp0 += 0xC00000

            if player[SLOT_PLAYER1].ypos > temp0
                stage.newXBoundary1 = object.newXBoundary1Value
                stage.curYBoundary2 = player[SLOT_PLAYER1].ypos
                stage.curYBoundary2 >>= 16
                stage.curYBoundary2 += 160

                if stage.curYBoundary2 < 256
                    stage.curYBoundary2 = 256
                end if

                object.state = R1_EGGMAN_SET_BOUNDARIES
            end if
        end if
        break

    case R1_EGGMAN_BODY_CONTROLLER
        if object.timer > 0
            object.timer--
        end if

        if bossBody.state > BOSSBODY_LANDING
            if object.faceTimer > 0
                object.faceTimer--
            else
                object.animation = EGGANI_IDLE
            end if
        end if

        arrayPos0 = object.bodyEntity
        GetBit(temp0, body[arrayPos0].bitHolder, 0)
        if temp0 == KEY_EVENT_TRUE
            if object[arrayPos0].state == BOSSBODY_PREPARE_CHARGING
                // Object +15 is a blank object used to hold values
                object.ypos          = object[+15].value0
                bossBody.ypos        = object[+15].value1
                bossLegJointR.ypos   = object[+15].value2
                bossLegR.velocity.y  = object[+15].value3
                bossLegR.ypos        = object[+15].ypos
                bossFootR.velocity.y = object[+15].value4
                bossLegJointL.ypos   = object[+15].value5
                bossLegL.velocity.y  = object[+15].value6
                bossFootL.velocity.y = object[+15].value7

                GetBit(temp0, body[arrayPos0].bitHolder, 6)
                if temp0 == LEFT_LEG_ACTIVE
                    SetBit(body[arrayPos0].bitHolder, 6, RIGHT_LEG_ACTIVE)
                else
                    SetBit(body[arrayPos0].bitHolder, 6, LEFT_LEG_ACTIVE)
                end if
            end if

            SetBit(body[arrayPos0].bitHolder, 0, KEY_EVENT_FALSE)
            if object.oscillation == 2
                // This may be a typo... it's supposed to be changing the [Boss Body]'s value instead I imagine
                SetBit(object.bitHolder, 3, 1)
            end if

            SetBit(body[arrayPos0].bitHolder, 1, MOVE_FORWARD)
            object.oscillation += 2
            CallFunction(R1_Eggman_Oscillation)
            if temp0 < 0
                object.oscillation = 6
            end if

            CallFunction(R1_Eggman_Oscillation)
            if temp0 == 6 // Go back
                // Object +15 is a blank object used to hold values
                object[+15].value0 = object.ypos
                object[+15].value1 = bossBody.ypos
                object[+15].value2 = bossLegJointR.ypos
                object[+15].value3 = bossLegR.velocity.y
                object[+15].ypos   = bossLegR.ypos
                object[+15].value4 = bossFootR.velocity.y
                object[+15].value5 = bossLegJointL.ypos
                object[+15].value6 = bossLegL.velocity.y
                object[+15].value7 = bossFootL.velocity.y
            end if

            if body[arrayPos0].state == BOSSBODY_MOVE_BACKWARDS
                if temp0 == 4
                    GetBit(temp1, body[arrayPos0].bitHolder, 6)
                    if temp1 == LEFT_LEG_ACTIVE
                        SetBit(body[arrayPos0].bitHolder, 6, RIGHT_LEG_ACTIVE)
                    else
                        SetBit(body[arrayPos0].bitHolder, 6, LEFT_LEG_ACTIVE)
                    end if
                end if
            end if

            body[arrayPos0].state = temp0
            object.oscillation++
            CallFunction(R1_Eggman_Oscillation)
            object.oscillation--
            body[arrayPos0].chargeStart = temp0
        end if
        break

    case R1_EGGMAN_DEFEATED
        if object.timer < 160
            object.timer++
            temp0 = object.timer
            temp0 &= 3
            if temp0 == 3
                Rand(temp0, 128)
                temp0 -= 64
                temp0 <<= 16
                temp0 += object.xpos

                Rand(temp1, 64)
                temp1 -= 32
                temp1 <<= 16
                temp1 += object.ypos

                CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
                PlaySfx(SfxName[Explosion], false)
            end if

            if object.timer == 94
                temp0 = object.entityPos
                temp0++

                player.score += 1000
                ResetObjectEntity(temp0, TypeName[Eggman Jetpack], 0, object.xpos, object.ypos)

                object.animation = EGGANI_DESTROYED
            end if
        end if
        break

    case R1_EGGMAN_BLANK
        break

    case R1_EGGMAN_RESTORE_BOUNDS
        if object.timer < 120
            music.volume--

            stage.curYBoundary2 += 4
            stage.curXBoundary2 += 6

            object.timer++
        else
            stage.curYBoundary2 = object.savedYBoundary2
            stage.curXBoundary2 = object.savedXBoundary2
            PlayMusic(0)
            object.type = TypeName[Blank Object]
        end if
        break

    case R1_EGGMAN_SET_BOUNDARIES
        if stage.curYBoundary2 > bossBody.iypos
            stage.curYBoundary2 -= 2
            stage.curYBoundary1 += 2
        else
            stage.curYBoundary2 = bossBody.iypos

            temp0 = bossBody.iypos
            temp0 -= screen.ysize
            if temp0 < 0
                temp0 = 0
            end if

            stage.newYBoundary1 = temp0

            object.timer = 0
            object.state = R1_EGGMAN_SPAWN_HVC
        end if
        break

    case R1_EGGMAN_SPAWN_HVC
        if object.timer < 120
            if object.timer < 100
                music.volume--
            else
                if object.timer == 100
                    PlayMusic(4)
                end if
            end if

            object.timer++
        else
            object.timer = 0
            object.state = R1_EGGMAN_BODY_CONTROLLER
            object.animation = EGGANI_LAUGH

            arrayPos0 = object.entityPos
            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Body], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            body[arrayPos0].faceEntity = object.entityPos

            body[arrayPos0].legJointREntity = object.entityPos
            body[arrayPos0].legJointREntity += 2

            body[arrayPos0].legJointLEntity = object.entityPos
            body[arrayPos0].legJointLEntity += 5

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Leg Joint], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Boss Body
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 1

            // Leg R
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 3

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Leg], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Leg Joint R
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 2

            // Foot R
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 4

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Foot], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Leg R
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 3

            // Boss Body
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 1

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Leg Joint], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Boss Body
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 1

            // Leg L
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 6

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Leg], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Leg Joint L
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 5

            // Foot L
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 7

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Foot], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Leg L
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 6

            // Boss Body
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 1
            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Shoulder], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 4

            shoulder[arrayPos0].faceEntity = object.entityPos

            // Arm Joint L
            shoulder[arrayPos0].armJointLEntity = object.entityPos
            shoulder[arrayPos0].armJointLEntity += 9

            // Arm Joint R
            shoulder[arrayPos0].armJointREntity = object.entityPos
            shoulder[arrayPos0].armJointREntity += 12

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Arm Joint], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Shoulder
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 8

            // Arm L
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 10

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Arm], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Arm Joint L
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 9

            // Bumper L
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 11

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Bumper], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE

            // Arm L
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 10

            // Boss Body
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 1

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Arm Joint], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Shoulder
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 8

            // Arm R
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 13

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Arm], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Arm Joint R
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 12

            // Bumper R
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 14

            arrayPos0++

            ResetObjectEntity(arrayPos0, TypeName[Boss Bumper], 0, object.xpos, object.ypos)
            bossPart[arrayPos0].priority = PRIORITY_ACTIVE
            bossPart[arrayPos0].drawOrder = 2

            // Arm R
            bossPart[arrayPos0].prevEntity = object.entityPos
            bossPart[arrayPos0].prevEntity += 13

            // Boss Body
            bossPart[arrayPos0].nextEntity = object.entityPos
            bossPart[arrayPos0].nextEntity += 1

            arrayPos0++
        end if
        break
    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = player[currentPlayer].collisionRight
        temp0 <<= 16
        temp0 += player[currentPlayer].xpos

        temp1 = stage.curXBoundary2
        temp1 <<= 16
        if temp0 > temp1
            player[currentPlayer].xvel = 0
            player[currentPlayer].speed = 0
            player[currentPlayer].xpos = temp1
            temp0 = player[currentPlayer].collisionRight
            temp0 <<= 16
            player[currentPlayer].xpos -= temp0
        end if

        if object.state > R1_EGGMAN_WAIT_PLAYER
            if player[currentPlayer].animation != ANI_HURT_A
                if object.hit < 3
                    BoxCollisionTest(C_TOUCH, object.entityPos, -28, -28, 28, 28, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                    if checkResult == true
                        if player[currentPlayer].animation == ANI_JUMPING
                            player[currentPlayer].xvel = player[currentPlayer].xpos
                            player[currentPlayer].xvel -= object.xpos
                            player[currentPlayer].xvel >>= 3

                            player[currentPlayer].speed = player[currentPlayer].xvel

                            player[currentPlayer].yvel = player[currentPlayer].ypos
                            player[currentPlayer].yvel -= object.ypos
                            player[currentPlayer].yvel >>= 3

                            if object.timer == 0
                                if bossBody.state != BOSSBODY_DESCEND
                                    PlaySfx(SfxName[Explosion], false)
                                    switch object.hit
                                    case 0
                                        object.hit++
                                        object.timer = 30
                                        object.animation = 3
                                        object.animationTimer = 120

                                        arrayPos0 = object.shoulderEntity
                                        arrayPos1 = shoulder[arrayPos0].armJointREntity
                                        debris[arrayPos1].state = BOSSARMJOINT_DESTROYED
                                        debris[arrayPos1].offset.x = 0x10000
                                        debris[arrayPos1].offset.y = 0x20000

                                        // Arm Joint R to Arm R
                                        arrayPos0 = bossPart[arrayPos1].nextEntity
                                        debris[arrayPos0].state = BOSSARM_DESTROYED
                                        debris[arrayPos0].offset.x = 0
                                        debris[arrayPos0].offset.y = 0x10000

                                        // Arm R to Bumper R
                                        arrayPos1 = bossPart[arrayPos0].nextEntity
                                        debris[arrayPos1].state = BOSSBUMPER_DESTROYED
                                        debris[arrayPos1].offset.x = 0
                                        debris[arrayPos1].offset.y = -0x18000
                                        break

                                    case 1
                                        object.hit++
                                        object.timer = 30
                                        object.animation = EGGANI_HURT
                                        object.animationTimer = 120

                                        arrayPos0 = object.shoulderEntity
                                        arrayPos1 = shoulder[arrayPos0].armJointLEntity
                                        debris[arrayPos1].state = BOSSARMJOINT_DESTROYED
                                        debris[arrayPos1].offset.x = 0x10000
                                        debris[arrayPos1].offset.y = 0x20000

                                        // Arm Joint L to Arm L
                                        arrayPos0 = bossPart[arrayPos1].nextEntity
                                        debris[arrayPos0].state = BOSSARM_DESTROYED
                                        debris[arrayPos0].offset.x = 0
                                        debris[arrayPos0].offset.y = 0x10000

                                        // ArmL to Bumper L
                                        arrayPos1 = bossPart[arrayPos0].nextEntity
                                        debris[arrayPos1].state = BOSSBUMPER_DESTROYED
                                        debris[arrayPos1].offset.x = 0
                                        debris[arrayPos1].offset.y = -0x18000
                                        break

                                    case 2
                                        object.hit++
                                        object.state = R1_EGGMAN_DEFEATED
                                        object.animation = EGGANI_HURT

                                        arrayPos0 = object.bodyEntity
                                        body[arrayPos0].state = BOSSBODY_DESTROYED

                                        arrayPos0 = object.shoulderEntity
                                        debris[arrayPos0].state = BOSSSHOULDER_DESTROYED
                                        debris[arrayPos0].offset.x = -0x10000
                                        debris[arrayPos0].offset.y = -0x20000

                                        arrayPos0 = object.bodyEntity
                                        arrayPos1 = body[arrayPos0].legJointREntity
                                        debris[arrayPos1].state = BOSSLEGJOINT_DESTROYED
                                        debris[arrayPos1].offset.x = 0
                                        debris[arrayPos1].offset.y = -0x28000

                                        // Leg Joint R to Leg R
                                        arrayPos0 = bossPart[arrayPos1].nextEntity
                                        debris[arrayPos0].state = BOSSLEG_DESTROYED
                                        debris[arrayPos0].offset.x = 0
                                        debris[arrayPos0].offset.y = -0x20000

                                        // Leg R to Foot R
                                        arrayPos1 = bossPart[arrayPos0].nextEntity
                                        debris[arrayPos1].state = BOSSFOOT_DESTROYED
                                        debris[arrayPos1].offset.x = 0
                                        debris[arrayPos1].offset.y = -0x23000

                                        arrayPos0 = object.bodyEntity
                                        arrayPos1 = body[arrayPos0].legJointLEntity
                                        debris[arrayPos1].state = BOSSLEGJOINT_DESTROYED
                                        debris[arrayPos1].offset.x = 0

                                        debris[arrayPos0].state = BOSSLEG_DESTROYED
                                        debris[arrayPos0].offset.x = 0
                                        debris[arrayPos0].offset.y = -0x20000

                                        // Leg L to Foot L
                                        arrayPos1 = bossPart[arrayPos0].nextEntity
                                        debris[arrayPos1].state = BOSSFOOT_DESTROYED
                                        debris[arrayPos1].offset.x = 0
                                        debris[arrayPos1].offset.y = -0x23000
                                        break
                                    end switch
                                end if
                            end if
                        else
                            if player[currentPlayer].invincibleTimer == 0
                                object.animation = EGGANI_LAUGH
                                object.faceTimer = 180
                                player[currentPlayer].state = Player_State_GotHit
                                if player[currentPlayer].xpos > object.xpos
                                    player[currentPlayer].speed = 0x20000
                                else
                                    player[currentPlayer].speed = -0x20000
                                end if
                            end if
                        end if
                    end if
                end if
            end if
        end if
    next
end event


event ObjectDraw
    switch object.animation
    case EGGANI_IDLE
        DrawSprite(0)
        break

    case EGGANI_LAUGH
        if object.frameTimer < 8
            object.frameTimer = 8
        else
            if object.frameTimer > 23
                object.frameTimer = 8
            end if
        end if

        object.frame = object.frameTimer
        object.frame >>= 3
        object.frameTimer++
        DrawSprite(object.frame)
        break

    case EGGANI_HURT
        if object.frameTimer < 12
            object.frameTimer = 12
        else
            if object.frameTimer > 27
                object.frameTimer = 12
            end if
        end if

        object.frame = object.frameTimer
        object.frame >>= 2
        object.frameTimer++
        DrawSprite(object.frame)
        break
    end switch
end event

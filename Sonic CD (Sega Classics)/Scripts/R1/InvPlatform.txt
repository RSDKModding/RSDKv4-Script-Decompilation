event ObjectUpdate
    if object.state < INVPLATFORM_RESTORE
        if object.state == INVPLATFORM_IDLE
            if object.stood == true
                if object.frame < 24
                    object.frame++
                end if
            else
                if object.frame > 0
                    object.frame--
                end if
            end if
        end if

        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_PLATFORM, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            object.stood = checkResult
        next
    end if

    if HUD.currentTimePeriod > TIME_PAST // Inv Platforms in the future fall upon contact
        switch object.state
        case INVPLATFORM_IDLE
            if object.stood == true
                object.state = INVPLATFORM_FALL
                object.priority = PRIORITY_ACTIVE
                player[SLOT_PLAYER1].ypos += 0x20000
            end if
            break

        case INVPLATFORM_FALL
            if object.stood == true
                player[SLOT_PLAYER1].ypos += 0x20000
            end if

            object.ypos += 0x20000

            if object.outOfBounds == true
                object.state = INVPLATFORM_RESTORE
                object.ypos = object.originPos.y
            end if
            break

        case INVPLATFORM_RESTORE
            if object.outOfBounds == true
                object.state = INVPLATFORM_IDLE
                object.priority = PRIORITY_BOUNDS
            end if
            break
        end switch
    end if
end event


event ObjectDraw
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        DrawSprite(object.frame)
        break

    case TIME_PAST
        DrawSprite(5)
        break

    case TIME_GOOD_FUTURE
    case TIME_BAD_FUTURE
        if object.state < INVPLATFORM_RESTORE
            DrawSprite(5)
        end if
        break
    end switch
end event

private alias object.value0 : object.originPos.x // Set by Ramp3D
private alias object.value1 : object.timer

private alias 1 : BOOSTSPINNER_SPIN

// ========================
// Events
// ========================

event ObjectUpdate
    if object.state == BOOSTSPINNER_IDLE
        if object.frame == 3
            foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -24, 128, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state = BOOSTSPINNER_SPIN
                    object.spinTimer = 4
                end if
            next
        end if
    end if
end event


event ObjectDraw
    temp0 = object.originPos.x
    temp0 -= object.xpos
    temp0 >>= 20
    object.frame = temp0

    if object.frame > 3
        object.frame = 3
    end if

    if object.state == BOOSTSPINNER_SPIN
        DrawSpriteXY(object.spinTimer, object.originPos.x, object.ypos)
        object.timer++
        if object.timer == 3
            object.timer = 0

            object.spinTimer++
            if object.spinTimer == 16
                object.state = BOOSTSPINNER_IDLE
            end if
        end if
    else
        DrawSpriteXY(object.frame, object.originPos.x, object.ypos)
    end if
end event

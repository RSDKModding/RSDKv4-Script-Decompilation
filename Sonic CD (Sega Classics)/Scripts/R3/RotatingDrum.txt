event ObjectUpdate
	switch object.state
	case ROTATINGDRUM_FRAME_0
		object.frame = 0
		object.direction = FLIP_NONE
		object.drawOrder = 3
		object.collisionOffset.y = 0x10000
		break

	case ROTATINGDRUM_FRAME_1
		object.frame = 1
		object.direction = FLIP_NONE
		object.drawOrder = 3
		object.collisionOffset.y = 0x50000
		break

		break

	case ROTATINGDRUM_FRAME_3
		object.frame = 3
		object.direction = FLIP_NONE
		object.drawOrder = 3
		object.collisionOffset.y = 0x80000
		break

	case ROTATINGDRUM_FRAME_4
		object.frame = 4
		object.direction = FLIP_NONE
		object.drawOrder = 3
		object.collisionOffset.y = 0x80000
		break

	case ROTATINGDRUM_FRAME_5
		object.frame = 5
		object.direction = FLIP_Y
		object.drawOrder = 3
		object.collisionOffset.y = 0x80000
		break

	case ROTATINGDRUM_FRAME_6

	case ROTATINGDRUM_FRAME_13
		object.frame = 9
		object.direction = FLIP_Y
		object.drawOrder = 2
		object.collisionOffset.y = -0x80000
		break

	case ROTATINGDRUM_FRAME_14
		object.frame = 9
		object.direction = FLIP_NONE
		object.drawOrder = 2
		object.collisionOffset.y = -0x80000
		break

	case ROTATINGDRUM_FRAME_15
		object.frame = 9
		object.direction = FLIP_NONE
		object.drawOrder = 2
		object.collisionOffset.y = -0x60000
		break

	case ROTATINGDRUM_FRAME_16
		object.frame = 9
		object.direction = FLIP_NONE
		object.drawOrder = 2
		object.collisionOffset.y = -0x50000
		break

	case ROTATINGDRUM_FRAME_17
		object.frame = 9
		object.direction = FLIP_NONE
		object.drawOrder = 2
		object.collisionOffset.y = -0x10000
		break

	end switch

	temp0 = R3Setup_drumTimer
	temp0 %= 10

	if temp0 == 0
		object.timer++
		if object.timer > 3
			object.timer = 0

			object.state++
			object.state %= 18
		end if
	else
		object.collisionOffset.y = 0
	end if

	object.priority = PRIORITY_ACTIVE

	temp0 = object.ixpos
	temp0 -= screen.xoffset
	temp0 -= screen.xsize

	if temp0 > 128
		object.ypos = object.originPos.y
		object.state = object.propertyValue
		object.timer = 0
		object.collisionOffset.y = 0
		object.priority = PRIORITY_XBOUNDS
	end if

	temp0 = object.ixpos
	temp0 -= screen.xoffset

	if temp0 < -128
		object.ypos = object.originPos.y
		object.state = object.propertyValue
		object.timer = 0
		object.collisionOffset.y = 0
		object.priority = PRIORITY_XBOUNDS
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		switch object.state
		case ROTATINGDRUM_FRAME_2
			BoxCollisionTest(C_PLATFORM, object.entityPos, -24, -8, 24, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
			break

		case ROTATINGDRUM_FRAME_3
		case ROTATINGDRUM_FRAME_4
		case ROTATINGDRUM_FRAME_5
		case ROTATINGDRUM_FRAME_6
			BoxCollisionTest(C_PLATFORM, object.entityPos, -24, -4, 24, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
			break

		end switch
	next

	object.ypos += object.collisionOffset.y
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			checkResult %= 18
			break

		case 0 // startFrame
			checkResult = object.propertyValue
			checkResult %= 18
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			object.propertyValue %= 18
			break

		case 0 // startFrame
			object.propertyValue = editor.variableValue
			object.propertyValue %= 18
			break

		end switch
	end if
end event

public function BossBody_LaunchBomb
    temp0 = object.ypos
    temp0 -= object.ypos
    if temp0 < 0x2400000
        CreateTempObject(TypeName[Pinball Bomb], 0, object.xpos, object.ypos)

        Rand(object[tempObjectPos].xvel, 0x400)
        object[tempObjectPos].xvel <<= 8

        if object.attackTimer < 128
            FlipSign(object[tempObjectPos].xvel)
            object[tempObjectPos].xpos -= 0x80000
        else
            object[tempObjectPos].xpos += 0x80000
        end if

        object[tempObjectPos].ypos += 0x180000
        object[tempObjectPos].yvel = 0x30000
        object[tempObjectPos].drawOrder = 2
    end if
end function

event ObjectUpdate
    switch object.state
    case BOSSBODY_DESCEND
        object.ypos = player[SLOT_PLAYER1].ypos
        object.ypos &= -0x10000 // Truncate the value

		// Height Limit
        if object.ypos < object.yOriginPos
            object.ypos = object.yOriginPos
            object.frame = 0
        end if

        if object.ypos > object.yvel
            object.ypos = object.yvel
            object.frame = 0
            object.state = BOSSBODY_DROP_BOMBS
            object.priority = PRIORITY_ACTIVE
        end if

        if object.ypos != object.yGearPos
            object.yGearPos = object.ypos
            object.frame++
            object.frame %= 9
        end if

        eggman.ypos = object.ypos

        object.attackTimer++
        object.attackTimer &= 255

        if object.attackTimer == 0
            CallFunction(BossBody_LaunchBomb)
        end if

        if object.attackTimer == 128
            CallFunction(BossBody_LaunchBomb)
        end if
        break

    case BOSSBODY_DROP_BOMBS
        eggman.ypos = object.ypos

        if EPlatform.length < 8 // if the platform didn't close, keep shooting
            object.attackTimer++
            object.attackTimer &= 255
        else
            object.attackTimer = 32
        end if

        if object.attackTimer == 0
            CallFunction(BossBody_LaunchBomb)
        end if

        if object.attackTimer == 128
            CallFunction(BossBody_LaunchBomb)
        end if
        break

    case BOSSBODY_HIT_1
        if object.ypos > object.yvel
            object.ypos -= 0x28000
            object.frame++
            object.frame %= 9
        else
            object.ypos = object.yvel
            object.frame = 0
            object.state = BOSSBODY_DRAG_EGGMAN_1
        end if

        eggman.ypos = object.ypos

        if object.damageTimer == 2
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos -= 32
            object[tempObjectPos].iypos += 10
        end if

        if object.damageTimer == 3
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos += 32
            object[tempObjectPos].iypos += 6
        end if

        if object.damageTimer == 4
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].iypos += 4
        end if
        break

    case BOSSBODY_DRAG_EGGMAN_1
        eggman.ypos = object.ypos
        break

    case BOSSBODY_HIT_2
        if object.ypos > object.yvel
            object.ypos -= 0x28000
            object.frame++
            object.frame %= 9
        else
            object.ypos = object.yvel
            object.frame = 0
            object.state = BOSSBODY_DRAG_EGGMAN_2
        end if

        eggman.ypos = object.ypos

        if object.damageTimer == 2
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos -= 32
            object[tempObjectPos].iypos -= 48
        end if

        if object.damageTimer == 3
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos += 32
            object[tempObjectPos].iypos -= 52
        end if

        if object.damageTimer == 4
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].iypos -= 56
        end if
        break

    case BOSSBODY_DRAG_EGGMAN_2
        eggman.ypos = object.ypos
        break

    case BOSSBODY_HIT_3
        if object.ypos > object.yvel
            object.ypos -= 0x28000
            object.frame++
            object.frame %= 9
            music.volume--
        else
            object.ypos = object.yvel
            object.frame = 0
            object.state = BOSSBODY_DESTROYED
            object.timer = 0
            object.priority = PRIORITY_ACTIVE
        end if

        if object.damageTimer > 0
            eggman.ypos = object.damageTimer
            eggman.ypos <<= 17
            FlipSign(eggman.ypos)
            eggman.ypos += object.ypos
        else
            if eggman.state == EGGANI_IDLE
                eggman.state = EGGANI_FLEE
                eggman.priority = PRIORITY_ACTIVE
                eggman.frameTimer &= 7
            end if

            if eggman.state == EGGANI_NERVOUS
                eggman.state = EGGANI_FLEE
                eggman.priority = PRIORITY_ACTIVE
                eggman.frameTimer &= 7
            end if
        end if

        if object.damageTimer == 2
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos -= 16
            object[tempObjectPos].iypos -= 52
        end if

        if object.damageTimer == 3
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            object[tempObjectPos].ixpos += 16
            object[tempObjectPos].iypos -= 56
        end if
        break

    case BOSSBODY_DESTROYED
        temp0 = object.timer
        temp0 &= 3
        if temp0 == 0
            Rand(temp0, 128)
            temp0 -= 64
            temp0 <<= 16
            temp0 += object.xpos

            Rand(temp1, 48)
            temp1 -= 24
            temp1 <<= 16
            temp1 += object.ypos

            CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
            PlaySfx(SfxName[Explosion], false)
        end if

        object.timer++
        if object.timer == 80
            player.score += 1000
            ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            PlayMusic(0)
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.state
        case BOSSBODY_DROP_BOMBS
        case BOSSBODY_DRAG_EGGMAN_1
            BoxCollisionTest(C_TOUCH, object.entityPos, -64, -16, 64, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                player[currentPlayer].yvel = 0
                object.yvel -= 0x480000

                object.damageTimer = 1
                object.state++
				
                eggman.state = EGGANI_HURT
                PlaySfx(SfxName[Explosion], false)
            end if
            break

        case BOSSBODY_DRAG_EGGMAN_2
            BoxCollisionTest(C_TOUCH, object.entityPos, -64, -16, 64, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                player[currentPlayer].yvel = 0
                object.yvel -= 0x800000

                object.damageTimer = 1
                object.state++

                PlaySfx(SfxName[Explosion], false)
            end if
            break

        end switch
    next
end event


event ObjectDraw
    switch object.state
    case BOSSBODY_DESCEND
    case BOSSBODY_DROP_BOMBS
        DrawSprite(4)
        DrawSprite(5)

        if object.attackTimer < 128
            if object.attackTimer < 8
                temp0 = object.attackTimer
                temp0 >>= 1
                temp0 += 7
            else
                temp0 = 8
            end if

            DrawSprite(temp0)
            DrawSprite(12)
        else
            if object.attackTimer < 136
                temp0 = object.attackTimer
                temp0 -= 128
                temp0 >>= 1
                temp0 += 11
            else
                temp0 = 12
            end if

            DrawSprite(8)
            DrawSprite(temp0)
        end if

        DrawSprite(6)
        DrawSprite(3)

        temp0 = object.frame
        temp0 /= 3
        DrawSprite(temp0)
        break

    case BOSSBODY_HIT_1
    case BOSSBODY_DRAG_EGGMAN_1
    case BOSSBODY_HIT_2
        temp0 = object.damageTimer
        temp0 &= 1
        if temp0 == 0
            DrawSprite(4)
            DrawSprite(5)
            DrawSprite(3)
        else
            DrawSprite(16)
            DrawSprite(17)
            DrawSprite(15)
        end if

        if object.damageTimer > 0
            object.damageTimer++
            object.damageTimer &= 15
        end if

        temp0 = object.frame
        temp0 /= 3
        DrawSprite(temp0)
        break

    case BOSSBODY_DRAG_EGGMAN_2
        temp0 = object.damageTimer
        temp0 &= 1

        temp1 = object.ypos
        temp1 += object.yControllerPos
        if temp0 == 0
            DrawSprite(4)
            DrawSprite(5)
            DrawSpriteXY(18, object.xpos, temp1)
            DrawSprite(3)
        else
            DrawSprite(16)
            DrawSprite(17)
            DrawSpriteXY(19, object.xpos, temp1)
            DrawSprite(15)
        end if

        if eggman.state == EGGANI_NERVOUS
            if object.yControllerPos < 0x100000
                object.yControllerPos += 0x8000
            end if
        end if

        if object.damageTimer > 0
            object.damageTimer++
            object.damageTimer &= 15
        end if

        temp0 = object.frame
        temp0 /= 3
        DrawSprite(temp0)
        break

    case BOSSBODY_HIT_3
        temp0 = object.ypos
        temp0 -= 0x200000

        if temp0 < object.yvel
            DrawSprite(16)
            DrawSprite(17)
            DrawSprite(3)
        else
            temp0 = object.damageTimer
            temp0 &= 1

            temp1 = object.ypos
            temp1 += object.yControllerPos

            if temp0 == 0
                DrawSprite(4)
                DrawSprite(5)
                DrawSprite(3)
            else
                DrawSprite(16)
                DrawSprite(17)
                DrawSprite(15)
            end if

            if object.yControllerPos < 0x100000
                object.yControllerPos += 0x8000
            end if
        end if

        if object.damageTimer > 0
            object.damageTimer++
            object.damageTimer &= 15
        end if

        temp0 = object.frame
        temp0 /= 3
        DrawSprite(temp0)
        break

    case BOSSBODY_DESTROYED
        DrawSprite(3)
        DrawSprite(0)
        break

    end switch
end event

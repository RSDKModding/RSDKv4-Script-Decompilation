	DrawSprite(0)
end function


public function FPlatform_DebugSpawn
	CreateTempObject(TypeName[F Platform], 0, object.xpos, object.ypos)
	object[tempObjectPos].originPos.y = object.ypos
end function


// ========================
// Events
// ========================

event ObjectUpdate
	if object.stood == true
		if object.timer < 16
			object.timer++

			temp0 = object.timer
			temp0 &= 3

			if temp0 == 3
				object.collisionOffset.y = 0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
		end if
	else
		if object.timer > 0
			object.timer--

			temp0 = object.timer
			temp0 &= 3

			if temp0 == 3
				object.collisionOffset.y = -0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
			object.ypos = object.originPos.y
		end if
	end if

	object.stood = false

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -17, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object.stood = true
			player[currentPlayer].ypos += object.collisionOffset.y
			player[currentPlayer].ypos += 0x10000
		end if
	next

	object.ypos += object.collisionOffset.y
end event


event ObjectDraw
	DrawSprite(0)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-32, -16, 64, 32, 100, 1) // #0 - Floating Platform

	foreach (TypeName[F Platform], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].originPos.y = object[arrayPos0].ypos
	next

	SetTableValue(TypeName[F Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(FPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(FPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

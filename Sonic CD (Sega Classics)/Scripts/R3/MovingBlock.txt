// ========================
// Aliases

// Aliases
private alias object.value0 : object.oscillation
private alias object.value1 : object.originPos.x

// Spring Aliases
private alias object[+1].xpos : spring.xpos

// ========================
// Events
// ========================

event ObjectUpdate
	object.oscillation += 2
	object.oscillation &= 511
	Sin(object.xpos, object.oscillation)

	object.xpos <<= 13
	object.xpos += object.originPos.x
	spring.xpos = object.xpos

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID2, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
	next
end event


event ObjectDraw
	DrawSprite(0)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 1)

	foreach (TypeName[Moving Block], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].originPos.x = object[arrayPos0].xpos
	next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

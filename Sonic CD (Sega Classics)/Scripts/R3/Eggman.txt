// Aliases
private alias object.value0 : object.frameTimer
private alias object.value1 : object.timer

// Boss Body Aliases
private alias object[+1].value5 : bossBody.damageTimer
private alias object[+1].state  : bossBody.state

// States
private alias 0 : EGGMAN_IDLE
private alias 1 : EGGMAN_HURT
private alias 2 : EGGMAN_NERVOUS
private alias 3 : EGGMAN_FLEE

// BossBody Aliases
private alias 5 : BOSSBODY_DRAG_EGGMAN_2

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case EGGMAN_IDLE
        object.frameTimer++
        object.frameTimer %= 40
        break

    case EGGMAN_HURT
        if object.timer < 41
            object.timer++
        else
            object.timer = 0
            object.frameTimer = 0

            if bossBody.state < BOSSBODY_DRAG_EGGMAN_2
                object.state = EGGMAN_IDLE
            else
                object.state = EGGMAN_NERVOUS
            end if
        end if
        break

    case EGGMAN_NERVOUS
        object.frameTimer++
        object.frameTimer &= 31
        break

    case EGGMAN_FLEE
        object.frameTimer++
        object.frameTimer &= 7
        break

    end switch
end event


event ObjectDraw
    switch object.state
    case EGGMAN_IDLE
        DrawSprite(4)

        temp0 = object.frameTimer
        temp0 /= 10
        DrawSprite(temp0)
        break

    case EGGMAN_HURT
        temp1 = object.timer
        temp1 <<= 8
        temp1 /= 40

        Sin(temp0, temp1)
        temp0 <<= 11
        FlipSign(temp0)
        temp0 += object.ypos

        temp1 = object.timer
        temp1 &= 15
        temp1 >>= 3
        temp1 += 7
        DrawSpriteXY(temp1, object.xpos, temp0)

        temp0 = bossBody.damageTimer
        temp0 &= 1
        temp0 += 4
        DrawSprite(temp0)

        DrawSprite(6)
        break

    case EGGMAN_NERVOUS
        temp0 = object.frameTimer
        temp0 >>= 4
        temp0 += 9
        DrawSprite(temp0)
        break

    case EGGMAN_FLEE
        temp0 = object.frameTimer
        temp0 >>= 2
        temp0 += 11
        DrawSprite(temp0)

        object.xpos += 0x40000

        if object.outOfBounds == true
            ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
        end if
        break
    end switch
end event

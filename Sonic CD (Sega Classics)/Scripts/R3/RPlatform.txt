event ObjectUpdate
	if object.outOfBounds == true
		object.ypos = object.originPos.y
		object.state = RISEPLATFORM_RESET
		object.timer = 0
	end if

	switch object.state
	case RISEPLATFORM_IDLE
		object.collisionOffset.y = 0
		if object.stood == true
			object.state = RISEPLATFORM_DELAY

			if object.priority != PRIORITY_XBOUNDS_DESTROY
				object.priority = PRIORITY_ACTIVE
			end if

			object.yvel = 0
			object.riseTimer = 0
		end if
		break

	case RISEPLATFORM_DELAY
		if object.riseTimer < 8
			object.riseTimer++
		else
			object.riseTimer = 0
			object.state++
		end if
		break

	case RISEPLATFORM_ASCEND
		if object.yvel > -0x40000
			object.yvel -= 0x1000
		else
			object.state++
		end if
		break

	case RISEPLATFORM_DELAY_2
		if object.riseTimer < 36
			object.riseTimer++
		else
			object.riseTimer = 0
			object.state++
		end if
		break

	case RISEPLATFORM_SLOW
		if object.yvel < 0
			object.yvel += 0x1000
		else
			object.yvel = 0
			object.state++
		end if
		break

	case RISEPLATFORM_FLOAT
		if object.stood == true
			if object.timer < 16
				object.timer++

				temp0 = object.timer
				temp0 &= 3

				if temp0 == 3
					object.yvel = 0x10000
				else
					object.yvel = 0
				end if
			else
				object.yvel = 0
			end if
		else
			if object.timer > 0
				object.timer--

				temp0 = object.timer
				temp0 &= 3

				if temp0 == 3
					object.yvel = -0x10000
				else
					object.yvel = 0
				end if
			else
				object.yvel = 0
			end if
		end if
		break

	case RISEPLATFORM_RESET
		if object.outOfBounds == true
			object.state = RISEPLATFORM_IDLE
			if object.priority != PRIORITY_XBOUNDS_DESTROY
				object.priority = PRIORITY_BOUNDS
			end if
		end if
		break

	end switch

	object.stood = false

	temp0 = object.ypos
	temp0 += object.yvel

	object.ypos &= -0x10000
	object.collisionOffset.y = temp0
	object.collisionOffset.y &= -0x10000
	object.collisionOffset.y -= object.ypos

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object.stood = true
			player[currentPlayer].ypos += object.collisionOffset.y
		end if
	next

	object.ypos = temp0
end event

		object[arrayPos0].originPos.y = object[arrayPos0].ypos
	next

	SetTableValue(TypeName[R Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(RPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(RPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

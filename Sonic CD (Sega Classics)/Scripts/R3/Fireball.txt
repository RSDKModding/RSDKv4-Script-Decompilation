// Aliases
private alias object.value0 : object.fade

// ========================
// Events
// ========================

event ObjectUpdate
    object.xpos += object.xvel
    if object.xvel > 0
        object.xvel += 0x1000
    else
        object.xvel -= 0x1000
    end if

    if object.outOfBounds == true
        object.type = TypeName[Blank Object]
    else
        temp0 = screen.yoffset
        temp0 -= 8
        temp0 <<= 16
        if object.ypos < temp0
            object.type = TypeName[Blank Object]
        end if
    end if

    if object.fade > 0
        object.fade--
        if object.fade < 32
            object.alpha = object.fade
            object.alpha <<= 3
        end if
    else
        object.type = TypeName[Blank Object]
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -8, -6, 8, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            CallFunction(Player_FireHit)
        end if
    next
end event


event ObjectDraw
    DrawSpriteFX(object.frame, FX_INK, object.xpos, object.ypos)
end event


event ObjectStartup
    LoadSpriteSheet("R3/Objects.gif")

    SpriteFrame(-8, -8, 16, 16, 230, 1)     // #0 - Fireball facing left
    SpriteFrame(-8, -8, 16, 16, 230, 18)    // #1 - Fireball facing right
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(object.propertyValue)
end event


event RSDKLoad
    LoadSpriteSheet("R3/Objects.gif")

    SpriteFrame(-8, -8, 16, 16, 230, 1)     // #0 - Fireball facing left
    SpriteFrame(-8, -8, 16, 16, 230, 18)    // #1 - Fireball facing right

    SetVariableAlias(ALIAS_VAR_PROPVAL, "direction")
end event

event ObjectUpdate
    switch object.propertyValue
    case BUMPER_RIGHT_TO_LEFT_G
        object.xpos = object.originPos.x
        object.xpos += R3Setup.bumperOscillation
        break

    case BUMPER_LEFT_TO_RIGHT_G
        object.xpos = object.originPos.x
        object.xpos -= R3Setup.bumperOscillation
        break

    case BUMPER_TOP_TO_BOTTOM
        if object.turnDirection == false
            if object.oscillation < 0x380000
                object.oscillation += 0x10000
            else
                object.turnDirection = true
            end if
        else
            if object.oscillation > -0x380000
                object.oscillation -= 0x10000
            else
                object.turnDirection = false
            end if
        end if

        object.ypos = object.originPos.y
        object.ypos += object.oscillation
        break

    case BUMPER_BOTTOM_TO_TOP
        if object.turnDirection == false
            if object.oscillation < 0x380000
                object.oscillation += 0x10000
            else
                object.turnDirection = true
            end if
        else
            if object.oscillation > -0x380000
                object.oscillation -= 0x10000
            else
                object.turnDirection = false
            end if
        end if

        object.ypos = object.originPos.y
        object.ypos -= object.oscillation
        break

    case BUMPER_TOP_TO_BOTTOM_G
        object.ypos = object.originPos.y
        object.ypos += R3Setup.bumperOscillation
        break

    case BUMPER_BOTTOM_TO_TOP_G
        object.ypos = object.originPos.y
        object.ypos -= R3Setup.bumperOscillation
        break

    end switch

    if object.state > BUMPER_IDLE
        object.frame = object.frameTimer
        object.frame /= 5
        object.frame++

        object.frameTimer++
        if object.frameTimer > 22
            object.frameTimer = 0
            object.state = BUMPER_IDLE
            object.frame = 0
        end if
    end if

    if object.outOfBounds == true
        if object.priority != PRIORITY_XBOUNDS_DESTROY
            object.bonusScore = 0
            object.priority = PRIORITY_BOUNDS
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if player[currentPlayer].animation != ANI_HURT
            BoxCollisionTest(C_TOUCH, object.entityPos, -14, -14, 14, 14, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if object.state == BUMPER_IDLE
                    PlaySfx(SfxName[Bumper], false)

                    if object.bonusScore < 5
                        CreateTempObject(TypeName[Object Score], 0, object.xpos, object.ypos)
                        player.score += 100
                        object.bonusScore++
                    end if
                end if

                if object.frameTimer > 5
                    PlaySfx(SfxName[Bumper], false)
                end if

                object.state = BUMPER_ACTIVE
                object.priority = PRIORITY_ACTIVE

                temp0 = player[currentPlayer].xpos
                temp0 -= object.xpos

                temp1 = player[currentPlayer].ypos
                temp1 -= object.ypos

                ATan2(temp2, temp0, temp1)
                Cos256(temp0, temp2)
                Sin256(temp1, temp2)

                if player[currentPlayer].state == Player_State_Fly
                    temp0 <<= 10
                    temp1 <<= 10
                else
                    temp0 *= 0x700
                    temp1 *= 0x700
                    player[currentPlayer].timer = 0
                end if

                player[currentPlayer].xvel = temp0
                player[currentPlayer].yvel = temp1

                player[currentPlayer].speed = player[currentPlayer].xvel
                player[currentPlayer].gravity = GRAVITY_AIR
            end if
        end if
    next
end event


event ObjectDraw
    DrawSprite(object.frame)
end event


event ObjectStartup
    LoadSpriteSheet("R3/Objects.gif")

    foreach (TypeName[Bumper], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].originPos.x = object[arrayPos0].xpos
        object[arrayPos0].originPos.y = object[arrayPos0].ypos
    next

    SpriteFrame(-16, -16, 32, 32, 67, 167)      // #0 - Bumper
    SpriteFrame(-12, -12, 24, 24, 206, 170)     // #1 - Bumper Hit Frame 0
    SpriteFrame(-20, -20, 40, 40, 215, 203)     // #2 - Bumper Hit Frame 1
    SpriteFrame(-12, -12, 24, 24, 206, 170)     // #3 - Bumper Hit Frame 2
    SpriteFrame(-20, -20, 40, 40, 215, 203)     // #4 - Bumper Hit Frame 3
    SpriteFrame(-20, -20, 40, 40, 215, 203)     // #5 - Bumper Hit Frame 4

    SetTableValue(TypeName[Bumper], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(Bumper_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(Bumper_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
    DebugMode_ObjCount++
end event

private alias object.yvel           : player.yvel

private alias object.gravity        : player.gravity

private alias 7  : TUBESWITCH_V_FORCE_L_DW
private alias 8  : TUBESWITCH_FORCE_ROLL
private alias 9  : TUBESWITCH_V_GROUNDED_DW

private alias 11 : TUBESWITCH_EDGE_BOUNCE
private alias 12 : TUBESWITCH_ROLL_MASK

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			switch object.propertyValue
			case TUBESWITCH_H_L
				if player[currentPlayer].speed > 0
					player[currentPlayer].direction = FACING_RIGHT
					player[currentPlayer].state = Player_State_TubeRoll
					player[currentPlayer].animation = ANI_JUMPING
					player[currentPlayer].minRollSpeed = 0xA0000
					if player[currentPlayer].speed < 0xA0000
						player[currentPlayer].speed = 0xA0000
					end if
				else
					if player[currentPlayer].gravity == GRAVITY_GROUND
						if player[currentPlayer].speed > -0x20000
							player[currentPlayer].speed = -0x20000
						end if
					end if

					player[currentPlayer].state = Player_State_Roll
					player[currentPlayer].animation = ANI_JUMPING
				end if
				break

			case TUBESWITCH_H_R
				if player[currentPlayer].speed < 0
					player[currentPlayer].state = Player_State_TubeRoll
					player[currentPlayer].animation = ANI_JUMPING
					player[currentPlayer].minRollSpeed = 0xA0000
					if player[currentPlayer].speed > -0xA0000
						player[currentPlayer].speed = -0xA0000
					end if
				else
					if player[currentPlayer].gravity == GRAVITY_GROUND
						if player[currentPlayer].speed < 0x20000
							player[currentPlayer].speed = 0x20000
						end if
					end if

					player[currentPlayer].state = Player_State_Roll
					player[currentPlayer].animation = ANI_JUMPING
				end if
				break

			case TUBESWITCH_V_T
				if object.inTube == false
					if player[currentPlayer].gravity == GRAVITY_AIR
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].speed = 0x100000
						player[currentPlayer].minRollSpeed = 0x100000
					else
						if player[currentPlayer].speed > 0
							player[currentPlayer].gravity = GRAVITY_GROUND
							player[currentPlayer].state = Player_State_TubeRoll
							player[currentPlayer].animation = ANI_JUMPING
							player[currentPlayer].collisionMode = CMODE_RWALL
							player[currentPlayer].angle = 64
							player[currentPlayer].speed = 0x100000
							player[currentPlayer].minRollSpeed = 0x100000
						else
							player[currentPlayer].gravity = GRAVITY_AIR
							player[currentPlayer].yvel = player[currentPlayer].speed
							player[currentPlayer].speed = 0
							player[currentPlayer].xvel = 0
						end if
					end if
				end if
				break

			case TUBESWITCH_V_B
				if object.inTube == false
					if player[currentPlayer].gravity == GRAVITY_AIR
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].speed = -0xA0000
						player[currentPlayer].minRollSpeed = 0xA0000
					end if

					if player[currentPlayer].collisionMode == CMODE_RWALL
						if player[currentPlayer].speed < 0
							player[currentPlayer].gravity = GRAVITY_GROUND
							player[currentPlayer].state = Player_State_TubeRoll
							player[currentPlayer].animation = ANI_JUMPING
							player[currentPlayer].collisionMode = CMODE_RWALL
							player[currentPlayer].angle = 64
							player[currentPlayer].speed = -0xA0000
							player[currentPlayer].minRollSpeed = 0xA0000
						else
							player[currentPlayer].gravity = GRAVITY_AIR
							player[currentPlayer].yvel = player[currentPlayer].speed
							player[currentPlayer].speed = 0
							player[currentPlayer].xvel = 0
						end if
					end if
				end if
				break

			case TUBESWITCH_ROLL_ALIGN
				if player[currentPlayer].gravity == GRAVITY_GROUND
					player[currentPlayer].state = Player_State_Roll
				else
					player[currentPlayer].xpos = object.xpos
				end if
				if player[currentPlayer].animation != ANI_JUMPING
					player[currentPlayer].animation = ANI_JUMPING
					player[currentPlayer].rollAnimationSpeed = 240
				end if
				break

			case TUBESWITCH_V_GROUNDED
				if object.inTube == false
					if player[currentPlayer].yvel > 0
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].speed = 0x100000
						player[currentPlayer].minRollSpeed = 0x100000
					end if
				else
					if player[currentPlayer].state == Player_State_Air
						player[currentPlayer].animation = ANI_JUMPING
					end if
				end if
				break

			case TUBESWITCH_V_FORCE_DW
				if object.inTube == false
					if player[currentPlayer].gravity == GRAVITY_AIR
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].speed = 0x100000
						player[currentPlayer].minRollSpeed = 0x100000
					else
						player[currentPlayer].gravity = GRAVITY_AIR
						player[currentPlayer].yvel = player[currentPlayer].speed
						FlipSign(player[currentPlayer].yvel)
						player[currentPlayer].speed = 0
						player[currentPlayer].xvel = 0
					end if
				end if
				break

			case TUBESWITCH_V_FORCE_L_DW
				if object.inTube == false
					if player[currentPlayer].yvel > 0
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_LWALL
						player[currentPlayer].angle = 192
						player[currentPlayer].speed = -0x100000
						player[currentPlayer].minRollSpeed = 0x100000
					end if
				else
					player[currentPlayer].animation = ANI_JUMPING
				end if
				break

			case TUBESWITCH_FORCE_ROLL
				if player[currentPlayer].state != Player_State_Fly
					if player[currentPlayer].gravity == GRAVITY_GROUND
						player[currentPlayer].state = Player_State_Roll
					end if
					player[currentPlayer].animation = ANI_JUMPING
				end if
				break

			case TUBESWITCH_V_GROUNDED_DW
				if object.inTube == false
					if player[currentPlayer].yvel > 0
						player[currentPlayer].gravity = GRAVITY_GROUND
						player[currentPlayer].state = Player_State_TubeRoll
						player[currentPlayer].animation = ANI_JUMPING
						player[currentPlayer].collisionMode = CMODE_RWALL
						player[currentPlayer].angle = 64
						player[currentPlayer].speed = 0x80000
						player[currentPlayer].minRollSpeed = 0x80000
					end if
				else
					player[currentPlayer].animation = ANI_JUMPING
				end if
				break

			case TUBESWITCH_GROUNDED_CANCEL
				if object.inTube == false
					if player[currentPlayer].gravity == GRAVITY_GROUND
						player[currentPlayer].gravity = GRAVITY_AIR
						player[currentPlayer].yvel = player[currentPlayer].speed
						player[currentPlayer].speed = 0
						player[currentPlayer].xvel  = 0
					end if
				end if
				break

			case TUBESWITCH_EDGE_BOUNCE
				if player[currentPlayer].yvel > 0
					if player[currentPlayer].xpos > object.xpos
						temp0 = player[currentPlayer].xpos
						temp0 -= object.xpos
						if temp0 < 0x60000
							player[currentPlayer].gravity = GRAVITY_AIR
							player[currentPlayer].yvel = -0x30000
						end if
					else
						temp0 = object.xpos
						temp0 -= player[currentPlayer].xpos
						if temp0 < 0x60000
							player[currentPlayer].gravity = GRAVITY_AIR
							player[currentPlayer].yvel = -0x30000
						end if
					end if
				end if
				break

			case TUBESWITCH_ROLL_MASK
				if player[currentPlayer].state != Player_State_Fly
					if player[currentPlayer].xvel > 0x40000
						if player[currentPlayer].gravity == GRAVITY_GROUND
							player[currentPlayer].state = Player_State_Roll
						end if
						player[currentPlayer].animation = ANI_JUMPING
					end if
				end if
				break

			end switch

			object.inTube = true
		else
			object.inTube = false
		end if
	next
end event

event RSDKDraw
	temp0 = object.xpos
	temp0 -= 0x80000
	temp1 = object.ypos
	temp1 -= 0x80000
	DrawSpriteXY(0, temp0, temp1)

	temp0 += 0x100000
	DrawSpriteXY(0, temp0, temp1)

	temp1 += 0x100000
	DrawSpriteXY(0, temp0, temp1)

	temp0 -= 0x100000
	DrawSpriteXY(0, temp0, temp1)
end event

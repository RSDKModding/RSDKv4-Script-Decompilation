// ========================
// Aliases
// ========================

// Aliases
private alias object.value0 : object.offset.y
private alias object.value1 : object.drawPos.y

// Attachment Aliases
private alias object[+1].ypos : attachment.ypos

// Player Aliases
private alias object.xpos : player.xpos

// Property Values
private alias 0 : FLIPOUTBLOCK_FLOOR
private alias 1 : FLIPOUTBLOCK_FLOOR_NO_ATTACH
private alias 2 : FLIPOUTBLOCK_ROOF
private alias 3 : FLIPOUTBLOCK_ROOF_NO_ATTACH

// ========================
// Events
// ========================

event ObjectUpdate
    if object.propertyValue > FLIPOUTBLOCK_FLOOR_NO_ATTACH
        object.ypos += object.offset.y
    else
        object.ypos -= object.offset.y
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID2, object.entityPos, -15, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
    next

    if object.propertyValue > FLIPOUTBLOCK_FLOOR_NO_ATTACH
        object.ypos -= object.offset.y
    else
        object.ypos += object.offset.y
    end if

    temp0 = object.xpos
    temp0 += 0x480000
    if player[SLOT_PLAYER1].xpos > temp0
        if object.offset.y < 0x200000
            object.offset.y += 0x80000
        end if
    end if

    temp0 = object.xpos
    temp0 -= 0x100000
    if player[SLOT_PLAYER1].xpos < temp0
        if object.offset.y > 0
            object.offset.y -= 0x80000
        end if
    end if

    temp1 = object.propertyValue
    temp1 &= 1
    object.drawPos.y = object.ypos
    if object.propertyValue > FLIPOUTBLOCK_FLOOR_NO_ATTACH
        object.drawPos.y += object.offset.y
        if temp1 == 0
            attachment.ypos = object.ypos
            attachment.ypos += object.offset.y
        end if
    else
        object.drawPos.y -= object.offset.y
        if temp1 == 0
            attachment.ypos = object.ypos
            attachment.ypos -= object.offset.y
        end if
    end if
end event


event ObjectDraw
    DrawSpriteXY(object.propertyValue, object.xpos, object.drawPos.y)
end event


event ObjectStartup
    LoadSpriteSheet("R3/Objects.gif")
    
    SpriteFrame(-16, -16, 32, 32, 34, 1) // #0 - Flipout Block
    SpriteFrame(-16, -16, 32, 32, 34, 1) // #1 - Flipout Block
    SpriteFrame(-16, -16, 32, 32, 67, 1) // #2 - Flipout Block Upside-Down
    SpriteFrame(-16, -16, 32, 32, 67, 1) // #3 - Flipout Block Upside-Down
end event


// ========================
// Editor Events
// ========================

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
        case 0 // Type
            checkResult = object.propertyValue
            checkResult &= 3
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
        case 0 // Type
            object.propertyValue = editor.variableValue
            object.propertyValue &= 3
            break

        end switch
    end if
end event


event RSDKDraw
	DrawSprite(object.propertyValue)

	if object.propertyValue < FLIPOUTBLOCK_ROOF
		temp0 = object.ypos
		temp0 -= 0x200000
	else
		temp0 = object.ypos
		temp0 += 0x200000
	end if

	temp1 = object.xpos
	temp1 += 0x100000

	temp2 = object.xpos
	temp2 -= 0x100000

	temp3 = object.ypos
	temp3 -= 0x100000

	if editor.showGizmos == true
		switch object.propertyValue
		case FLIPOUTBLOCK_FLOOR
		case FLIPOUTBLOCK_ROOF
			DrawArrow(temp1, object.ypos, temp1, temp0, 255, 255, 255, 0)
			DrawRectOutline(temp2, temp3, 0x40, 0x20, 255, 255, 255, 255)
			break

		case FLIPOUTBLOCK_FLOOR_NO_ATTACH
		case FLIPOUTBLOCK_ROOF_NO_ATTACH
			DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 255, 255, 0)
			DrawRectOutline(temp2, temp3, 0x20, 0x20, 255, 255, 255, 255)
			break

		end switch
	end if
end event


event RSDKLoad
    LoadSpriteSheet("R3/Objects.gif")

    SpriteFrame(-16, -16, 32, 32, 34, 1) // #0 - Flipout Block
    SpriteFrame(-16, -16, 32, 32, 34, 1) // #1 - Flipout Block
    SpriteFrame(-16, -16, 32, 32, 67, 1) // #2 - Flipout Block Upside-Down
    SpriteFrame(-16, -16, 32, 32, 67, 1) // #3 - Flipout Block Upside-Down

    AddEditorVariable("type")
    SetActiveVariable("type")
    AddEnumVariable("Normal", FLIPOUTBLOCK_FLOOR)
    AddEnumVariable("Without Attachment", FLIPOUTBLOCK_FLOOR_NO_ATTACH)
    AddEnumVariable("Upside-Down", FLIPOUTBLOCK_ROOF)
    AddEnumVariable("Upside-Down w/o Attachment", FLIPOUTBLOCK_ROOF_NO_ATTACH)
end event

event ObjectUpdate
    object.priority = PRIORITY_ACTIVE

    if player[SLOT_PLAYER1].xpos < object.xpos
        object.direction = FACING_LEFT
    else
        object.direction = FACING_RIGHT
    end if

    switch object.state
    case AMY_IDLE
        temp0 = object.frameTimer
        temp0 /= 10
        temp0 += 11
        object.frame = temp0

        object.frameTimer++
        if object.frameTimer > 139
            object.frameTimer = 0
        end if

        if player[SLOT_PLAYER1].type != TypeName[Debug Mode]
            BoxCollisionTest(C_TOUCH, object.entityPos, -96, -360, 96, 360, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.xvel = 0
                object.state = AMY_CHASE
                object.frame = 4
                object.frameTimer = 0
            end if
        end if
        break

    case AMY_CHASE
        object.frameTimer++

        switch object.frame
        case 4
        case 6
            if object.frameTimer == 4
                object.frameTimer = 0
                object.frame++
            end if
            break

        case 5
        case 7
            if object.frameTimer == 3
                object.frameTimer = 0
                object.frame++
            end if
            break

        end switch

        if object.frame > 7
            object.frame = 4
        end if

        if object.direction == FACING_RIGHT
            if object.xvel < 0x20000
                object.xvel += 0x2000
            end if

            if object.xpos < object.rightLimit
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.xpos = object.rightLimit
                object.state = AMY_AWAIT_RIGHT
                object.frame = 0
                object.frameTimer = 0
            end if
        else
            if object.xvel > -0x20000
                object.xvel -= 0x2000
            end if

            if object.xpos > object.leftLimit
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.xpos = object.leftLimit
                object.state = AMY_AWAIT_LEFT
                object.frame = 0
                object.frameTimer = 0
            end if
        end if
        break

    case AMY_AWAIT_RIGHT
        object.frameTimer++
        if object.frameTimer == 12
            object.frameTimer = 0
            object.frame++
        end if

        if object.frame > 3
            object.frame = 0
        end if

        if metal.state == METALSONIC_WAIT
            if metal.timer < 60
                metal.timer++
            else
                metal.timer = 0
                metal.state = METALSONIC_APPROACH
                metal.priority = PRIORITY_ACTIVE
            end if
        end if
        break

    case AMY_AWAIT_LEFT
        object.frameTimer++
        if object.frameTimer == 12
            object.frameTimer = 0
            object.frame++
        end if

        if object.frame > 3
            object.frame = 0
        end if

        if object.direction == FACING_RIGHT
            object.state = AMY_CHASE
            object.frame = 4
            object.frameTimer = 0
        end if
        break

    case AMY_KIDNAP
        object.frameTimer++
        if object.frameTimer > 13
            object.frameTimer = 0
            object.frame++
            if object.frame > 9
                object.frame = 8
            end if
        end if
        break

    end switch

    object.heartTimer++
    if object.state < AMY_KIDNAP
        if object.heartTimer > 41
            CreateTempObject(TypeName[Love Heart], 0, object.xpos, object.ypos)
            object[tempObjectPos].iypos -= 6
            object[tempObjectPos].inkEffect = INK_ALPHA
            object[tempObjectPos].alpha = 255
            object.heartTimer = 0
        end if
    end if

    if object.gravity == GRAVITY_GROUND
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)
    else
        object.yvel += 0x4000
        object.ypos += object.yvel
        ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
        if checkResult == true
            object.yvel = 0
            object.gravity = GRAVITY_GROUND
        end if
    end if
end event


event ObjectDraw
    DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
end event

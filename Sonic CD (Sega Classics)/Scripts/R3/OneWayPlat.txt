// ========================
// Aliases
// ========================

// Aliases
private alias object.value0 : object.yvel

// Player Aliases
private alias object.ypos : player.ypos

// ========================
// Function Declarations
// ========================

reserve function OneWayPlatform_DebugDraw
reserve function OneWayPlatform_DebugSpawn

// ========================
// Function Definitions
// ========================

public function OneWayPlatform_DebugDraw
	DrawSprite(0)
end function


public function OneWayPlatform_DebugSpawn
	CreateTempObject(TypeName[One Way Platform], 0, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -8, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
	next

	if player[SLOT_PLAYER1].ypos < object.ypos
		if object.yvel < 0x400000
			object.yvel += 0x100000
		end if
	else
		if object.yvel > 0
			object.yvel -= 0x100000
		end if
	end if
end event


event ObjectDraw
	temp0 = object.xpos
	temp0 += object.yvel
	DrawSpriteXY(0, temp0, object.ypos)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-96, -8, 64, 16, 165, 1)

	SetTableValue(TypeName[One Way Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(OneWayPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(OneWayPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)

	if editor.showGizmos == true
		temp0 = object.ixpos
		temp0 -= 32
		temp0 <<= 16

		DrawArrow(temp0, object.ypos, object.xpos, object.ypos, 255, 0, 0, 0)
	end if
end event


event RSDKLoad
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-96, -8, 64, 16, 165, 1)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

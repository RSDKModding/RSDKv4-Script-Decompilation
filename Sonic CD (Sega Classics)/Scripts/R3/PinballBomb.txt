// Aliases
private alias object.value0 : object.frameTimer
private alias object.value3 : object.bounces

// Player Aliases
private alias object.xpos    : player.xpos
private alias object.ypos    : player.ypos
private alias object.xvel    : player.xvel
private alias object.yvel    : player.yvel
private alias object.speed   : player.speed
private alias object.gravity : player.gravity
private alias object.value1  : player.timer

// ========================
// Events
// ========================

event ObjectUpdate
    object.yvel += 0x2000
    if object.yvel > 0x40000
        object.drawOrder = 3 // If the bomb goes fast enough, get to the background
    end if

    object.xpos += object.xvel
    object.ypos += object.yvel

    object.frameTimer++
    object.frameTimer &= 3

    if object.xvel > 0
        ObjectTileCollision(CSIDE_LWALL, 8, 0, 0)
    else
        ObjectTileCollision(CSIDE_RWALL, -8, 0, 0)
    end if

    if checkResult == true
        FlipSign(object.xvel)
    end if

    ObjectTileCollision(CSIDE_FLOOR, 0, 8, 0)
    if checkResult == true
        FlipSign(object.yvel)
        object.yvel >>= 1

        if object.bounces < 1
            object.bounces++
        else
            PlaySfx(SfxName[Explosion], false)
            ResetObjectEntity(object.entityPos, TypeName[Explosion], 0, object.xpos, object.ypos)
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            PlaySfx(SfxName[Explosion], false)

            ResetObjectEntity(object.entityPos, TypeName[Explosion], 0, object.xpos, object.ypos)
            object.priority = PRIORITY_ACTIVE

            temp0 = player[currentPlayer].xpos
            temp0 -= object.xpos

            temp1 = player[currentPlayer].ypos
            temp1 -= object.ypos
            ATan2(temp2, temp0, temp1)

            Cos256(temp0, temp2)
            Sin256(temp1, temp2)
            temp0 *= 0x700
            temp1 *= 0x700

            player[currentPlayer].xvel = temp0
            player[currentPlayer].yvel = temp1
            player[currentPlayer].speed = player[currentPlayer].xvel
            player[currentPlayer].gravity = GRAVITY_AIR
            player[currentPlayer].timer = 0
        end if
    next
end event


event ObjectDraw
    if object.frameTimer == 0
        DrawSprite(0)
    else
        DrawSprite(1)
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R3/Objects2.gif")

    SpriteFrame(-12, -12, 24, 24, 232, 0)   // #0 - Pinball Bomb Frame 0
    SpriteFrame(-8, -8, 16, 16, 130, 67)    // #1 - Pinball Bomb Frame 1
end event

end event

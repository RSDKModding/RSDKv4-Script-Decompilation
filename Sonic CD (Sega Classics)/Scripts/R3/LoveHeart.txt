// Aliases
private alias object.value0 : object.timer

// ========================
// Events
// ========================

event ObjectDraw
    if object.timer > 144
        temp0 = object.timer
        temp0 -= 144
        temp0 <<= 3
        if temp0 > 255
            temp0 = 255
        end if

        object.alpha = 255
        object.alpha -= temp0

        if object.alpha == 0
            object.type = TypeName[Blank Object]
        end if
    end if

    Sin(temp0, object.angle)
    temp0 <<= 9
    temp0 += object.xpos

    if object.timer > 19
        DrawSpriteFX(1, FX_INK, temp0, object.ypos)
    else
        DrawSpriteXY(0, temp0, object.ypos)
    end if

    object.ypos -= 0x6000

    object.angle += 8
    object.angle &= 511

    object.timer++
end event


event ObjectStartup
    LoadSpriteSheet("R3/Objects3.gif")
    
    SpriteFrame(-8, -8, 16, 16, 193, 134)   // #0 - Small Heart
    SpriteFrame(-8, -8, 16, 16, 193, 150)   // #1 - Big Heart
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R3/Objects3.gif")
    SpriteFrame(-8, -8, 16, 16, 193, 134)   // #0 - Small Heart
    
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

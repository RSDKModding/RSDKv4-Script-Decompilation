    CreateTempObject(TypeName[Score Chute], 0, object.xpos, object.ypos)
    object[tempObjectPos].drawOrder = 4
end function

// Events
// ========================

event ObjectUpdate
    switch object.state
    case SCORECHUTE_CATCH_PLAYER
        if object.timer < 8
            object.timer++

            player[SLOT_PLAYER1].xpos = object.xpos
            player[SLOT_PLAYER1].ypos = object.ypos
            player[SLOT_PLAYER1].ypos -= 0x80000
            player[SLOT_PLAYER1].yvel = 0
            player[SLOT_PLAYER1].xvel = 0
            player[SLOT_PLAYER1].speed = 0
        else
            object.state = SCORECHUTE_INSERT_1
            object.timer = 0

            player[SLOT_PLAYER1].yvel = 0
            player[SLOT_PLAYER1].xvel = 0

            player[SLOT_PLAYER1].visible = false

            if player[SLOT_PLAYER1].blinkTimer > 0
                player[SLOT_PLAYER1].invincibleTimer = 0
                player[SLOT_PLAYER1].blinkTimer = 0
            end if

            player[SLOT_PLAYER1].state = Player_State_Static
        end if
        break

    case SCORECHUTE_INSERT_1
        if object.timer < 8
            object.timer++

            player[SLOT_PLAYER1].xpos = object.xpos
            player[SLOT_PLAYER1].ypos = object.ypos
            player[SLOT_PLAYER1].ypos -= 0x80000
        else
            object.state = SCORECHUTE_INSERT_2
            object.timer = 0
        end if
        break

    case SCORECHUTE_INSERT_2
        if object.timer < 4
            object.timer++
        else
            object.state = SCORECHUTE_GIVE_POINTS
            object.timer = 0
        end if

		// incase of time over
        if player[SLOT_PLAYER1].state == Player_State_Death
            object.state = SCORECHUTE_COOLDOWN
            object.timer = 0

            player[SLOT_PLAYER1].visible = true
        end if
        break

    case SCORECHUTE_GIVE_POINTS
        if object.scoreBonus < 4
            if object.timer == 32
                CreateTempObject(TypeName[Object Score], 0, object.xpos, object.ypos)
                player.score += 100
            end if
        end if

        if object.scoreBonus < 3
            if object.timer == 64
                CreateTempObject(TypeName[Object Score], 0, object.xpos, object.ypos)
                player.score += 100
            end if
        end if

        if object.timer < 80
            object.timer++
        else
            object.state = 5
            object.timer = 0
        end if

		// incase of time over
        if player[SLOT_PLAYER1].state == Player_State_Death
            object.state = SCORECHUTE_COOLDOWN
            object.timer = 0

            player[SLOT_PLAYER1].visible = true
        end if
        break

    case SCORECHUTE_PULL_PLAYER
        if object.timer < 4
            object.timer++
        else
            object.state = SCORECHUTE_EJECT_PLAYER
            object.timer = 0
        end if

		// incase of time over
        if player[SLOT_PLAYER1].state == Player_State_Death
            object.state = SCORECHUTE_COOLDOWN
            object.timer = 0

            player[SLOT_PLAYER1].visible = true
        end if
        break

    case SCORECHUTE_EJECT_PLAYER
        if object.timer < 8
            object.timer++
        else
            object.state = SCORECHUTE_COOLDOWN
            object.timer = 0

            player[SLOT_PLAYER1].visible = true
            player[SLOT_PLAYER1].state = Player_State_Air
            player[SLOT_PLAYER1].animation = ANI_JUMPING
            player[SLOT_PLAYER1].interaction = true
            player[SLOT_PLAYER1].yvel = -0x74000
            player[SLOT_PLAYER1].timer = 0

            PlaySfx(SfxName[Chute Exit], false)
        end if
        break

    case SCORECHUTE_COOLDOWN
        if object.timer < 120
            object.timer++
        else
            object.state = 0
            object.timer = 0
            object.scoreBonus++
        end if
        break

    end switch

    if object.outOfBounds == true
        object.scoreBonus = 0
        object.priority = PRIORITY_BOUNDS
    end if

    if object.state == SCORECHUTE_IDLE
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            if player[currentPlayer].yvel >= 0
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -8, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state = SCORECHUTE_CATCH_PLAYER
                    object.priority = PRIORITY_ACTIVE

                    player[currentPlayer].animation = ANI_JUMPING
                    player[currentPlayer].interaction = false

                    PlaySfx(SfxName[Score Chute], false)
                end if
            end if
        next
    end if
end event


event ObjectDraw
    switch object.state
    case SCORECHUTE_CATCH_PLAYER
        DrawSprite(0)
        break

    case SCORECHUTE_INSERT_1
        DrawSprite(0)
        DrawSprite(1)
        break

    case SCORECHUTE_INSERT_2
        DrawSprite(0)
        DrawSprite(2)
        break

    case SCORECHUTE_GIVE_POINTS
        temp0 = object.timer
        temp0 &= 7
        temp0 >>= 2
        if temp0 > 0
            DrawSprite(0)
        end if
        break

    case SCORECHUTE_PULL_PLAYER
        DrawSprite(0)
        DrawSprite(2)
        break

    case SCORECHUTE_EJECT_PLAYER
        DrawSprite(0)
        DrawSprite(1)
        break

    end switch
end event

    LoadSpriteSheet("R3/Objects.gif")

    foreach (TypeName[Score Chute], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].drawOrder = 4
    next

    SpriteFrame(-24, -8, 48, 16, 117, 233)       // #0 - Score Chute Active

    if stage.playerListPos == PLAYER_SONIC_A
        SpriteFrame(-16, -24, 32, 16, 2, 240)    // #1 - Jump Insert Frame 0
        SpriteFrame(-8, -16, 16, 8, 35, 241)     // #2 - Jump Insert Frame 1
    else
        SpriteFrame(-12, -20, 24, 12, 228, 244)  // #1 - Jump Insert Frame 0
        SpriteFrame(-8, -16, 16, 8, 211, 247)    // #2 - Jump Insert Frame 1
    end if

    SetTableValue(TypeName[Score Chute], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(ScoreChute_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(ScoreChute_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
    DebugMode_ObjCount++
end event

// Aliases
private alias object.value0 : object.timer

// States
private alias 0 : FLOWER_SEED_SPAWN
private alias 1 : FLOWER_SEED_FALL

private alias 2 : FLOWER_SEED_PLANTED_1
private alias 3 : FLOWER_SEED_PLANTED_2
private alias 4 : FLOWER_SEED_PLANTED_3

private alias 5 : FLOWER_ROOT_GROWTH_1
private alias 6 : FLOWER_ROOT_GROWTH_2

private alias 7 : FLOWER_ANIM_1
private alias 8 : FLOWER_ANIM_2
private alias 9 : FLOWER_ANIM_3

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case FLOWER_SEED_SPAWN
        object.frame = 8
        object.timer++
        if object.timer == 20
            object.timer = 0
            object.state++
        end if
        break

    case FLOWER_SEED_FALL
        object.frame = object.timer
        object.frame >>= 2

        object.timer++
        object.timer &= 7

        object.ypos += 0x20000
        ObjectTileCollision(CSIDE_FLOOR, 0, 8, 0) // Check if it touched the ground
        if checkResult == true
            object.ypos += 0x80000
            object.state++
            object.timer = 0
            object.frame = 2
        end if
        break

    case FLOWER_SEED_PLANTED_1
        object.timer++
        if object.timer == 4
            object.timer = 0
            object.state++
            object.frame = 3
        end if
        break

    case FLOWER_SEED_PLANTED_2
        object.timer++
        if object.timer == 4
            object.timer = 0
            object.state++
            object.frame = 2
        end if
        break

    case FLOWER_SEED_PLANTED_3
        object.timer++
        if object.timer == 4
            object.timer = 0
            object.state++
            object.frame = 3
        end if
        break

    case FLOWER_ROOT_GROWTH_1
        object.timer++
        if object.timer == 5
            object.timer = 0
            object.state++
            object.frame = 4
        end if
        break

    case FLOWER_ROOT_GROWTH_2
        object.timer++
        if object.timer == 4
            object.timer = 0
            object.state++
            object.frame = 5
        end if
        break

    case FLOWER_ANIM_1
        object.timer++
        if object.timer == 2
            object.timer = 0
            object.state++
            object.frame = 6
        end if
        break

    case FLOWER_ANIM_2
        object.timer++
        if object.timer == 20
            object.timer = 0
            object.state++
            object.frame = 7
        end if
        break

    case FLOWER_ANIM_3
        object.timer++
        if object.timer == 20
            object.timer = 0
            object.state--
            object.frame = 6
        end if
        break

    end switch
end event


event ObjectDraw
    DrawSprite(object.frame)
end event

    SpriteFrame(-12, -48, 24, 48, 1, 1)     // #6 - Flower Frame 0

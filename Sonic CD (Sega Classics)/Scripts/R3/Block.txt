// ========================
// Function Declarations
// ========================

reserve function Block_DebugDraw
reserve function Block_DebugSpawn

// ========================
// Function Definitions
// ========================

public function Block_DebugDraw
	DrawSprite(0)
end function


public function Block_DebugSpawn
	CreateTempObject(TypeName[Block], 0, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
	next
end event


event ObjectDraw
	DrawSprite(0)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 1)

	SetTableValue(TypeName[Block], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(Block_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(Block_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 1)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

reserve function Piston_DebugDraw

// ========================
// Function Definitions

public function Piston_DebugDraw
    DrawSprite(1)
    DrawSprite(0)
end function


public function Piston_DebugSpawn
    CreateTempObject(TypeName[Piston], 0, object.xpos, object.ypos)
    object[tempObjectPos].frame = 1
end function


// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case PISTON_DETECT_PLAYER
        BoxCollisionTest(C_TOUCH, object.entityPos, -112, -256, 112, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            if object.timer < 80
                object.timer++
                object.frame = 1
            else
                object.timer = 6
                object.state = PISTON_PREPARE_FIRE
            end if
        end if
        break

    case PISTON_PREPARE_FIRE
        if object.timer < 33
            object.timer++
            object.frame = object.timer
            object.frame /= 3
        else
            object.frame = 1
            object.timer = 0
            object.state = PISTON_EMERGE
        end if
        break

    case PISTON_EMERGE
        if object.offset.y < 0x200000
            object.offset.y += 0x80000
        else
            object.state = PISTON_FIRE
        end if
        break

    case PISTON_FIRE
        if object.timer < 8
            if object.timer == 0
                switch HUD.currentTimePeriod
                case TIME_PRESENT
                case TIME_BAD_FUTURE
                    CreateTempObject(TypeName[Fireball], 0, object.xpos, object.ypos)
                    object[tempObjectPos].xpos -= 0x180000
                    object[tempObjectPos].ypos -= 0x140000
                    object[tempObjectPos].fade = 64
                    object[tempObjectPos].xvel = -0x10000
                    object[tempObjectPos].inkEffect = INK_ALPHA
                    object[tempObjectPos].alpha = 255
                    object[tempObjectPos].frame = 0

                    CreateTempObject(TypeName[Fireball], 0, object.xpos, object.ypos)
                    object[tempObjectPos].xpos += 0x180000
                    object[tempObjectPos].ypos -= 0x140000
                    object[tempObjectPos].fade = 64
                    object[tempObjectPos].xvel = 0x10000
                    object[tempObjectPos].inkEffect = INK_ALPHA
                    object[tempObjectPos].alpha = 255
                    object[tempObjectPos].frame = 1
                end switch
            end if

            object.timer++
        else
            object.timer = 0
            object.state = PISTON_HIDE
        end if
        break

    case PISTON_HIDE
        if object.offset.y > 0
            object.offset.y -= 0x40000
        else
            object.state = PISTON_DETECT_PLAYER
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = -0xC0000
        temp0 -= object.offset.y
        temp0 >>= 16
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -12, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        BoxCollisionTest(C_PLATFORM, object.entityPos, -16, temp0, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

		// if the player is on top when the piston emerges, launch it to the sky
		if checkResult == true
            if object.offset.y == 0x200000
                if object.state == PISTON_EMERGE
                    player[currentPlayer].yvel = -0x70000
                    player[currentPlayer].timer = 0
                    player[currentPlayer].gravity = GRAVITY_AIR
                    player[currentPlayer].animation = ANI_BOUNCING
                    player[currentPlayer].state = Player_State_Air
                end if
            end if
        end if
    next
end event


event ObjectDraw
    temp0 = object.ypos
    temp0 -= object.offset.y
    DrawSpriteXY(object.frame, object.xpos, temp0)
    DrawSprite(0)
end event


event ObjectStartup

    foreach (TypeName[Piston], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].frame = 1
    next

    SpriteFrame(-16, -4, 32, 8, 214, 34)       // #0 - Piston Head
    SpriteFrame(-16, -12, 32, 40, 1, 34)       // #1 - Piston

        SpriteFrame(-16, -12, 32, 40, 1, 34)   // #7  - Piston Emerge Frame 5
        SpriteFrame(-16, -12, 32, 40, 34, 34)  // #8  - Piston Emerge Frame 6
        SpriteFrame(-16, -12, 32, 40, 1, 34)   // #9  - Piston Emerge Frame 7
        SpriteFrame(-16, -12, 32, 40, 34, 34)  // #10 - Piston Emerge Frame 8

    case TIME_PAST

        SpriteFrame(-16, -12, 32, 40, 1, 34)  // #9  - Piston Emerge Frame 7
        SpriteFrame(-16, -12, 32, 40, 1, 34)  // #10 - Piston Emerge Frame 8
        SpriteFrame(-16, -12, 32, 40, 1, 34)  // #11 - Piston Emerge Frame 9
        break

    end switch

    SetTableValue(TypeName[Piston], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(Piston_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(Piston_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)

	if editor.showGizmos == true

	end if
end event

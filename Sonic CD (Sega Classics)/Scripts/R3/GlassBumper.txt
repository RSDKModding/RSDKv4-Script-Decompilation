// Aliases
private alias object.value0 : object.timer

// Player Aliases
private alias object.xpos      : player.xpos
private alias object.ypos      : player.ypos
private alias object.xvel      : player.xvel
private alias object.yvel      : player.yvel
private alias object.speed     : player.speed
private alias object.animation : player.animation
private alias object.gravity   : player.gravity
private alias object.value1    : player.timer

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0 : GLASSBUMPER_IDLE
private alias 1 : GLASSBUMPER_DESTROYED

// Time Period Aliases
private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

// Tile Info
private alias 0 : TILEINFO_INDEX

// ========================
// Events
// ========================

event ObjectUpdate
	if object.state == GLASSBUMPER_DESTROYED
		object.timer++
		if object.timer > 15
			object.type = TypeName[Blank Object]
		end if
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if player[currentPlayer].animation != ANI_HURT
			if object.state == GLASSBUMPER_IDLE
				BoxCollisionTest(C_TOUCH, object.entityPos, -14, -14, 14, 14, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					PlaySfx(SfxName[Glass Smash], false)

					CreateTempObject(TypeName[Object Score], 0, object.xpos, object.ypos)
					player.score += 100

					object.state = GLASSBUMPER_DESTROYED
					object.priority = PRIORITY_ACTIVE

					temp0 = player[currentPlayer].xpos
					temp0 -= object.xpos

					temp1 = player[currentPlayer].ypos
					temp1 -= object.ypos

					ATan2(temp2, temp0, temp1)
					Cos256(temp0, temp2)
					Sin256(temp1, temp2)

					temp0 *= 0x700
					temp1 *= 0x700

					player[currentPlayer].xvel = temp0
					player[currentPlayer].yvel = temp1
					player[currentPlayer].speed = player[currentPlayer].xvel

					player[currentPlayer].gravity = GRAVITY_AIR
					player[currentPlayer].timer = 0

					temp0 = object.ixpos
					temp1 = object.iypos

					// Replace the bumper tiles with blank tiles
					temp0 -= 8
					temp1 -= 8
					Set16x16TileInfo(0, temp0, temp1, TILEINFO_INDEX)

					temp0 += 16
					Set16x16TileInfo(0, temp0, temp1, TILEINFO_INDEX)

					temp0 -= 16
					temp1 += 16
					Set16x16TileInfo(0, temp0, temp1, TILEINFO_INDEX)

					temp0 += 16
					Set16x16TileInfo(0, temp0, temp1, TILEINFO_INDEX)
				end if
			end if
		end if
	next
end event


event ObjectDraw
	if object.state == GLASSBUMPER_DESTROYED
		temp0 = object.timer
		temp0 >>= 3
		DrawSprite(temp0)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects3.gif")

	switch HUD.currentTimePeriod
	case TIME_PRESENT
		SpriteFrame(-16, -16, 32, 32, 66, 1)
		SpriteFrame(-16, -16, 32, 32, 99, 1)
		break

	case TIME_PAST
		SpriteFrame(-16, -16, 32, 32, 66, 34)
		SpriteFrame(-16, -16, 32, 32, 99, 34)
		break

	case TIME_GOOD_FUTURE
		SpriteFrame(-16, -16, 32, 32, 66, 67)
		SpriteFrame(-16, -16, 32, 32, 99, 67)
		break

	case TIME_BAD_FUTURE
		SpriteFrame(-16, -16, 32, 32, 66, 100)
		SpriteFrame(-16, -16, 32, 32, 99, 100)
		break

	end switch
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	object.inkEffect = INK_BLEND
	DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
end event


event RSDKLoad
	LoadSpriteSheet("R3/Objects3.gif")
	CallFunction(EditorHelpers_FindTimePeriod)

	switch checkResult
	case TIME_PRESENT
		SpriteFrame(-16, -16, 32, 32, 66, 1)
		SpriteFrame(-16, -16, 32, 32, 99, 1)
		break

	case TIME_PAST
		SpriteFrame(-16, -16, 32, 32, 66, 34)
		SpriteFrame(-16, -16, 32, 32, 99, 34)
		break

	case TIME_GOOD_FUTURE
		SpriteFrame(-16, -16, 32, 32, 66, 67)
		SpriteFrame(-16, -16, 32, 32, 99, 67)
		break

	case TIME_BAD_FUTURE
		SpriteFrame(-16, -16, 32, 32, 66, 100)
		SpriteFrame(-16, -16, 32, 32, 99, 100)
		break

	end switch
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

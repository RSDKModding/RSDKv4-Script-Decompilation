public function SRecords_DrawRecords
    object.minutes = saveRAM[arrayPos0]
    object.minutes /= 6000

    object.seconds = saveRAM[arrayPos0]
    object.seconds /= 100
    object.seconds %= 60

    object.milliSeconds = saveRAM[arrayPos0]
    object.milliSeconds %= 100

    temp0 = object.xpos
    temp0 >>= 16
    temp0 -= 16
    DrawNumbers(0, temp0, temp1, object.minutes, 2, 8, 1)

    temp0 += 24
    DrawNumbers(0, temp0, temp1, object.seconds, 2, 8, 1)

    temp0 += 24
    DrawNumbers(0, temp0, temp1, object.milliSeconds, 2, 8, 1)
end function

event ObjectUpdate
    object.ypos = object.record
    object.ypos *= 192
    object.ypos -= menuControl.smYPos
    if object.ypos < -384
        object.ypos += 576
    end if
    object.ypos += 224
    object.ypos <<= 16
end event


event ObjectDraw
    DrawSprite(18)
    DrawSprite(10)

    switch object.record
    case FIRST_RECORD
        temp0 = sLargeIcon[-1].portrait
        temp0 += 11
        DrawSprite(temp0)
        break

    case SECOND_RECORD
        temp0 = sLargeIcon[-2].portrait
        temp0 += 11
        DrawSprite(temp0)
        break

    case THIRD_RECORD
        temp0 = sLargeIcon[-3].portrait
        temp0 += 11
        DrawSprite(temp0)
        break

    end switch

    arrayPos0 = temp0
    arrayPos0 -= 11
    arrayPos0 *= 18

    temp0 = object.record
    temp0 *= 6

    arrayPos0 += temp0
    arrayPos0 += 49

    temp1 = object.ypos
    temp1 >>= 16
    temp1 += 97
    CallFunction(SRecords_DrawRecords)

    temp1 += 24
    arrayPos0 += 2
    CallFunction(SRecords_DrawRecords)

    temp1 += 24
    arrayPos0 += 2
    CallFunction(SRecords_DrawRecords)
    DrawSprite(21)

    switch menuControl.state
    case MENUCONTROL_SM_CONTROLS
        switch menuControl.smCurrentButton
        case 0
            DrawSprite(19)
            break

        case 2
            DrawSprite(20)
            break

        end switch
        break

    case MENUCONTROL_SM_CONFIRMED
        temp0 = menuControl.timer
        temp0 &= 3
        temp0 >>= 1
        if temp0 == 0
            DrawSprite(20)
        end if
        break

    case MENUCONTROL_SM_LOAD_LEVEL
        DrawSprite(20)
        break

    case MENUCONTROL_SM_COMFIRMED_LEAVE
        temp0 = menuControl.timer
        temp0 &= 3
        temp0 >>= 1
        if temp0 == 0
            DrawSprite(19)
        end if
        break

    end switch
end event

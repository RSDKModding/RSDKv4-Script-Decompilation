// saveRAM[39] is how many zones are unlocked for Time Attack

public function MenuControl_IconTransitions
	arrayPos0 = 32

	// Top half
	temp2 = screen.xcenter
	temp2 <<= 16

	temp0 = -210
	temp1 = -46
	while arrayPos0 < 36
		// Horizontal movement
		object[arrayPos0].xpos = temp0
		object[arrayPos0].xpos *= object.offset
		object[arrayPos0].xpos >>= 1

		// Vertical movement
		object[arrayPos0].xpos += temp2
		object[arrayPos0].ypos = temp1
		object[arrayPos0].ypos *= object.offset
		object[arrayPos0].ypos += 0x640000

		temp0 += 140
		arrayPos0++
	loop

	// Bottom half
	temp0 = -210
	temp1 = 46
	while arrayPos0 < 40
		// Horizontal movement
		object[arrayPos0].xpos = temp0
		object[arrayPos0].xpos *= object.offset
		object[arrayPos0].xpos >>= 1
		object[arrayPos0].xpos += temp2

		// Vertical movement
		object[arrayPos0].ypos = temp1
		object[arrayPos0].ypos *= object.offset
		object[arrayPos0].ypos += 0x640000

		temp0 += 140
		arrayPos0++
	loop
end function


public function MenuControl_DefaultRecord
	arrayPos0 = 48
	temp0 = 0
	temp1 = 0

	// Checks in all Time Attack levels if there is a record saved
	while temp0 < 21
		temp1 += saveRAM[arrayPos0]
		arrayPos0 += 6
		temp0++
	loop

	// If there are no records, set the default times (5 Minutes per round)
	if temp1 == 0
		arrayPos0 = 48
		temp0 = 0
		while temp0 < 21
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			saveRAM[arrayPos0] = 30000
			arrayPos0++
			temp0++
		loop
	end if
end function


public function MenuControl_ZoneUnlockCheck
	temp0 = -1

	// Save 0 stage.listPos
	arrayPos0 = 3
	while arrayPos0 < 40
		temp1 = saveRAM[arrayPos0]

		// In this order, check if your save is post PPZ 3, post MMZ 2 or if it's on a special stage
		if temp1 > 10
			if temp1 > 67
				if temp1 < 80
					temp1 = 7
				else
					temp1 -= 80
					temp1 /= 10
				end if
			else
				temp1 /= 10
			end if
		else
			temp1 = 0
		end if

		// Sets the amount of unlocked zones
		if temp1 > temp0
			temp0 = temp1
		end if

		// Jump to next save
		arrayPos0 += 8
	loop

	if temp0 > saveRAM[39]
		saveRAM[39] = temp0 // Set the new value if there are new unlocked zones
	else
		temp0 = saveRAM[39] // Otherwise just use the saved value
	end if

	// Unlocks the playable zones
	arrayPos0 = 32
	while temp0 > 0
		object[arrayPos0].zoneLocked = false
		arrayPos0++ // Skips the locking of this zone
		temp0--
	loop

	// Locks the rest of zones
	while arrayPos0 < 39
		object[arrayPos0].zoneLocked = true
		arrayPos0++
	loop
end function


public function MenuControl_TotalRegularRecord
	// Get the times of all zones
	arrayPos0 = 32
	arrayPos1 = 48
	temp1 = 0
	while arrayPos0 < 39
		temp0 = saveRAM[arrayPos1]
		arrayPos1 += 6

		temp0 += saveRAM[arrayPos1]
		arrayPos1 += 6

		temp0 += saveRAM[arrayPos1]
		arrayPos1 += 6

		temp1 += temp0

		object[arrayPos0].minutes = temp0
		object[arrayPos0].minutes /= 6000

		object[arrayPos0].seconds = temp0
		object[arrayPos0].seconds /= 100
		object[arrayPos0].seconds %= 60

		object[arrayPos0].milliseconds = temp0
		object[arrayPos0].milliseconds %= 100
		arrayPos0++
	loop

	// Calculate the total time
	object[arrayPos0].minutes = temp1
	object[arrayPos0].minutes /= 6000

	if object[arrayPos0].minutes < 25
		// No achievement unlock
	end if

	object[arrayPos0].seconds = temp1
	object[arrayPos0].seconds /= 100
	object[arrayPos0].seconds %= 60

	object[arrayPos0].milliseconds = temp1
	object[arrayPos0].milliseconds %= 100

	if object[arrayPos0].minutes < 30
		object.ssUnlocked = true
	end if
end function


public function MenuControl_TotalSpecialRecord
	// Get the times of all zones
	arrayPos0 = 32
	arrayPos1 = 49
	temp1 = 0
	while arrayPos0 < 39
		temp0 = saveRAM[arrayPos1]
		arrayPos1 += 18

		temp1 += temp0

		object[arrayPos0].minutes = temp0
		object[arrayPos0].minutes /= 6000

		object[arrayPos0].seconds = temp0
		object[arrayPos0].seconds /= 100
		object[arrayPos0].seconds %= 60

		object[arrayPos0].milliseconds = temp0
		object[arrayPos0].milliseconds %= 100

		arrayPos0++
	loop

	// Calculate the total time
	object[arrayPos0].minutes = temp1
	object[arrayPos0].minutes /= 6000

	object[arrayPos0].seconds = temp1
	object[arrayPos0].seconds /= 100
	object[arrayPos0].seconds %= 60

	object[arrayPos0].milliseconds = temp1
	object[arrayPos0].milliseconds %= 100

	object.ssUnlocked = true
end function

event ObjectUpdate
	switch object.state
	case MENUCONTROL_SETUP
		// Makes sure nothing gets carried away from the saves
		stage.debugMode     = false
		stage.playerListPos = PLAYER_SONIC_A

		lampPostID = 0

		goodFuture.enabled = false
		goodFuture.count   = 0
		goodFuture.list    = 0

		specialStage.timeStones = 0
		ReadSaveRAM()

		object.state  = MENUCONTROL_FADEIN
		object.timer  = 384
		object.offset = 0x40000
		SetScreenFade(0, 0, 0, 255)

		if engine.deviceType == MOBILE
			object.savedRound      = -1
			object.selectedButton  = -1
			object.smCurrentButton = -1
		end if

		CallFunction(MenuControl_ZoneUnlockCheck)

		// Check if you come from playing time attack
		if timeAttack.round > -1
			object.state = MENUCONTROL_SM_LEVEL_RETURN

			largeIcon.portrait = timeAttack.round
			object.savedRound  = timeAttack.round

			largeIcon.xpos = screen.xcenter
			largeIcon.xpos -= 72
			largeIcon.xpos <<= 16

			object.smYPos = timeAttack.zone
			object.smYPos *= 192
			object.smYPos += 192

			if timeAttack.result > 0
				arrayPos0 = timeAttack.round
				arrayPos0 *= 18

				temp0 = timeAttack.zone
				temp0 *= 6

				arrayPos0 += temp0
				arrayPos0 += 48
				arrayPos0 += timeAttack.menuSection

				if timeAttack.result < saveRAM[arrayPos0]
					// 1st Record Beated
					arrayPos1 = arrayPos0
					arrayPos1 += 4
					arrayPos0 += 2
					saveRAM[arrayPos1] = saveRAM[arrayPos0]

					arrayPos1 -= 2
					arrayPos0 -= 2
					saveRAM[arrayPos1] = saveRAM[arrayPos0]
					saveRAM[arrayPos0] = timeAttack.result

					object.beatedRecord = 1
				else
					arrayPos0 += 2
					if timeAttack.result < saveRAM[arrayPos0]
						// 2st Record Beated
						arrayPos1 = arrayPos0
						arrayPos1 += 2
						saveRAM[arrayPos1] = saveRAM[arrayPos0]
						saveRAM[arrayPos0] = timeAttack.result

						object.beatedRecord = 2
					else
						arrayPos0 += 2
						if timeAttack.result < saveRAM[arrayPos0]
							// 3st Record Beated
							saveRAM[arrayPos0]  = timeAttack.result
							object.beatedRecord = 3
						end if
					end if
				end if
			end if

			object.offset = 0x50000
		else
			timeAttack.menuSection = 0
		end if

		CallFunction(MenuControl_DefaultRecord)

		if timeAttack.menuSection == 0
			CallFunction(MenuControl_TotalRegularRecord)
		else
			CallFunction(MenuControl_TotalSpecialRecord)

			object[40].type = TypeName[S Large Icon]
			object[41].type = TypeName[S Records]

			arrayPos0 = 32
			while arrayPos0 < 40
				object[arrayPos0].type = TypeName[Special Icon]
				arrayPos0++
			loop
		end if

		WriteSaveRAM()
		CallFunction(MenuControl_IconTransitions)
		PlayMusic(0)
		break

	case MENUCONTROL_FADEIN
		if object.timer > 0
			object.timer -= 8
			object.offset -= 0x1000
		else
			object.state = MENUCONTROL_CONTROLS
		end if

		SetScreenFade(0, 0, 0, object.timer)
		CallFunction(MenuControl_IconTransitions)
		break

	case MENUCONTROL_CONTROLS
		if options.physicalControls == true
			arrayPos0 = 32
			arrayPos0 += object.savedRound

			checkResult = false

			if object[arrayPos0].state != ROUNDICON_SELECTED_ZONE
				if keyPress[0].left == true
					object.savedRound--
					if object.savedRound < 0
						object.savedRound = 7
					end if

					PlaySfx(SfxName[Menu Button], false)
				end if

				if keyPress[0].right == true
					object.savedRound++
					if object.savedRound > 7
						object.savedRound = 0
					end if

					PlaySfx(SfxName[Menu Button], false)
				end if

				if keyPress[0].up == true
					object.savedRound -= 4
					if object.savedRound < 0
						object.savedRound += 8
						if object.ssUnlocked == true
							object.state = MENUCONTROL_SCROLL_UP
						end if
					end if

					PlaySfx(SfxName[Menu Button], false)
				end if

				if keyPress[0].down == true
					object.savedRound += 4
					if object.savedRound > 7
						object.savedRound -= 8
						if object.ssUnlocked == true
							object.state = MENUCONTROL_SCROLL_DOWN
						end if
					end if

					PlaySfx(SfxName[Menu Button], false)
				end if

				if keyPress[0].start == true
					checkResult = true
				end if

				if keyPress[0].buttonA == true
					checkResult = true
				end if
			end if

			if checkResult == true
				arrayPos0 = 32
				arrayPos0 += object.savedRound
				if object[arrayPos0].zoneLocked == false
					object[arrayPos0].state = ROUNDICON_SELECTED_ZONE
					object.smCurrentButton  = 0

					switch object.savedRound
					case 0
						PlaySfx(SfxName[Select], false)
						break

					case 1
						PlaySfx(SfxName[Select], false)
						break

					case 2
						PlaySfx(SfxName[Select], false)
						break

					case 3
						PlaySfx(SfxName[Select], false)
						break

					case 4
						PlaySfx(SfxName[Select], false)
						break

					case 5
						PlaySfx(SfxName[Select], false)
						break

					case 6
						PlaySfx(SfxName[Select], false)
						break

					case 7
						PlaySfx(SfxName[Select], false)
						object.state++
						StopMusic()
						break

					end switch
				end if
			end if
		else
			if object.ssUnlocked == true
				temp0 = screen.xsize
				temp0 -= 72
				CheckTouchRect(temp0, 192, screen.xsize, screen.ysize)
				if checkResult > -1
					object.selectedButton = 8
				else
					CheckTouchRect(0, 0, screen.xsize, screen.ysize)
					if checkResult < 0
						if object.selectedButton == 8
							PlaySfx(SfxName[Menu Button], false)
							object.state = MENUCONTROL_SCROLL_DOWN
							object.selectedButton = -1
						end if
					else
						if object.selectedButton == 8
							object.selectedButton = -1
						end if
					end if
				end if
			end if

			if object.selectedButton > -1
				object.savedRound = object.selectedButton
			else
				object.savedRound = -1
			end if

			if keyPress[0].left == true
				object.savedRound     = 7
				object.selectedButton = -1

				options.physicalControls = true
			end if

			if keyPress[0].right == true
				object.savedRound     = 0
				object.selectedButton = -1

				options.physicalControls = true
			end if

			if keyPress[0].up == true
				object.savedRound     = 7
				object.selectedButton = -1

				options.physicalControls = true
			end if

			if keyPress[0].down == true
				object.savedRound     = 0
				object.selectedButton = -1

				options.physicalControls = true
			end if
		end if

		if keyPress[0].buttonB == true
			object.savedRound = 7

			arrayPos0 = 32
			arrayPos0 += object.savedRound

			// Tell RoundIcon that we want to leave
			object[arrayPos0].state = ROUNDICON_SELECTED_ZONE
			object.state++

			StopMusic()
			PlaySfx(SfxName[Select], false)
		end if
		break

	case MENUCONTROL_BLANK
		// chillin
		break

	case MENUCONTROL_RETURN_TO_MENU
		if object.timer < 384
			object.timer += 8
			object.offset += 0x1000
		else
			engine.state = ENGINE_RESETGAME
		end if

		SetScreenFade(0, 0, 0, object.timer)
		CallFunction(MenuControl_IconTransitions)
		break

	case MENUCONTROL_ENTER_SM
		if object.timer < 256
			object.timer += 8
			object.offset += 0x2000
		else
			if timeAttack.menuSection == 0
				object[40].type = TypeName[Large Icon]
				object[41].type = TypeName[Records]
			else
				object[40].type = TypeName[S Large Icon]
				object[41].type = TypeName[S Records]
			end if

			largeIcon.portrait = object.savedRound
			largeIcon.state    = LARGEICON_ENTER_SCREEN

			object[41].drawOrder = 2
			object[42].drawOrder = 2
			object[43].drawOrder = 2

			object.timer  = 0
			object.smYPos = 0

			object.state++
		end if

		CallFunction(MenuControl_IconTransitions)
		break

	case MENUCONTROL_CALL_RECORD_MENU
		if object.smYPos < 192
			object.smYPos += 8
		else
			object.state++
		end if

		if options.physicalControls == true
			object.smCurrentButton = 0
		else
			object.smCurrentButton = -1
		end if
		break

	case MENUCONTROL_SM_CONTROLS
		if options.physicalControls == true
			if timeAttack.menuSection == 0
				if keyPress[0].up == true
					object.state = MENUCONTROL_SM_SCROLL_UP
				end if

				if keyPress[0].down == true
					object.state = MENUCONTROL_SM_SCROLL_DOWN
				end if

				if keyPress[0].left == true
					object.smCurrentButton--
					if object.smCurrentButton < 0
						object.smCurrentButton = 2
					end if
				end if

				if keyPress[0].right == true
					object.smCurrentButton++
					if object.smCurrentButton > 2
						object.smCurrentButton = 0
					end if
				end if
			else
				if keyPress[0].left == true
					object.smCurrentButton -= 2
					if object.smCurrentButton < 0
						object.smCurrentButton = 2
					end if
				end if

				if keyPress[0].right == true
					object.smCurrentButton += 2
					if object.smCurrentButton > 2
						object.smCurrentButton = 0
					end if
				end if
			end if

			checkResult = false
			if keyPress[0].start == true
				checkResult = true
			end if

			if keyPress[0].buttonA == true
				checkResult = true
			end if

			if keyPress[0].buttonB == true
				PlaySfx(SfxName[Back Button], false)

				largeIcon.state = LARGEICON_LEAVE_SCREEN
				object.state    = MENUCONTROL_SM_COMFIRMED_LEAVE

				object[41].drawOrder = 8
				object[42].drawOrder = 8
				object[43].drawOrder = 8

				arrayPos0 = object.smYPos
				arrayPos0 /= 192
				arrayPos0 += 40
				object[arrayPos0].drawOrder = 2
			end if

			if checkResult == true
				switch object.smCurrentButton
				case 0
					PlaySfx(SfxName[Back Button], false)

					largeIcon.state = LARGEICON_LEAVE_SCREEN
					object.state    = MENUCONTROL_SM_COMFIRMED_LEAVE

					object[41].drawOrder = 8
					object[42].drawOrder = 8
					object[43].drawOrder = 8

					arrayPos0 = object.smYPos
					arrayPos0 /= 192
					arrayPos0 += 40
					object[arrayPos0].drawOrder = 2
					break

				case 1
					PlaySfx(SfxName[Menu Button], false)
					object.state = MENUCONTROL_SM_SCROLL_UP
					break

				case 2
					PlaySfx(SfxName[Select], false)
					object.state = MENUCONTROL_SM_CONFIRMED
					StopMusic()
					break

				end switch
			end if

			CheckTouchRect(0, 0, screen.xsize, screen.ysize)
			if checkResult > -1
				options.physicalControls = false
				object.smCurrentButton = -1
			end if
		else
			temp0 = screen.xcenter
			temp0 -= 140
			temp1 = temp0
			temp1 += 66
			CheckTouchRect(temp0, 0, temp1, 40)
			if checkResult > -1
				object.smCurrentButton = 0
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.smCurrentButton == 0
						PlaySfx(SfxName[Back Button], false)

						largeIcon.state = LARGEICON_LEAVE_SCREEN
						object.state    = MENUCONTROL_SM_COMFIRMED_LEAVE

						object[41].drawOrder = 8
						object[42].drawOrder = 8
						object[43].drawOrder = 8

						arrayPos0 = object.smYPos
						arrayPos0 /= 192
						arrayPos0 += 40
						object[arrayPos0].drawOrder = 2
					end if
				else
					if object.smCurrentButton == 0
						object.smCurrentButton = -1
					end if
				end if
			end if

			if keyPress[0].buttonB == true
				PlaySfx(SfxName[Back Button], false)

				largeIcon.state = LARGEICON_LEAVE_SCREEN
				object.state    = MENUCONTROL_SM_COMFIRMED_LEAVE

				object[41].drawOrder = 8
				object[42].drawOrder = 8
				object[43].drawOrder = 8

				arrayPos0 = object.smYPos
				arrayPos0 /= 192
				arrayPos0 += 40
				object[arrayPos0].drawOrder = 2
			end if

			temp0 += 159
			temp1 += 159
			if timeAttack.menuSection == 0
				CheckTouchRect(temp0, 0, temp1, 40)
				if checkResult > -1
					object.smCurrentButton = 1
				else
					CheckTouchRect(0, 0, screen.xsize, screen.ysize)
					if checkResult < 0
						if object.smCurrentButton == 1
							PlaySfx(SfxName[Menu Button], false)
							object.state = MENUCONTROL_SM_SCROLL_DOWN
						end if
					else
						if object.smCurrentButton == 1
							object.smCurrentButton = -1
						end if
					end if
				end if
			end if

			temp0 += 68
			temp1 += 68
			CheckTouchRect(temp0, 0, temp1, 40)
			if checkResult > -1
				object.smCurrentButton = 2
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.smCurrentButton == 2
						PlaySfx(SfxName[Select], false)
						object.state = MENUCONTROL_SM_CONFIRMED
						StopMusic()
					end if
				else
					if object.smCurrentButton == 2
						object.smCurrentButton = -1
					end if
				end if
			end if

			if keyPress[0].left == true
				object.smCurrentButton   = 2
				options.physicalControls = true
			end if

			if keyPress[0].right == true
				object.smCurrentButton   = 0
				options.physicalControls = true
			end if
		end if
		break

	case MENUCONTROL_SM_SCROLL_DOWN
		if object.timer < 192
			object.timer += 8
			object.smYPos += 8
			if object.smYPos > 767
				object.smYPos -= 576
			end if
		else
			object.timer = 0
			object.state = MENUCONTROL_SM_CONTROLS
			if options.physicalControls == false
				object.smCurrentButton = -1
			end if
		end if
		break

	case MENUCONTROL_SM_SCROLL_UP
		if object.timer < 192
			object.timer += 8
			object.smYPos -= 8
			if object.smYPos < 192
				object.smYPos += 576
			end if
		else
			object.timer = 0
			object.state = MENUCONTROL_SM_CONTROLS
			if options.physicalControls == false
				object.smCurrentButton = -1
			end if
		end if
		break

	case MENUCONTROL_SM_CONFIRMED
		if object.timer < 30
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case MENUCONTROL_SM_LOAD_LEVEL
		if object.timer < 320
			object.timer += 8
		else
			if timeAttack.menuSection == 0
				stage.activeList = REGULAR_STAGE
				stage.listPos    = largeIcon.portrait
				timeAttack.round = largeIcon.portrait
				stage.listPos *= 10

				temp0 = object.smYPos
				temp0 -= 192
				temp0 /= 192

				timeAttack.zone = temp0
				temp0 <<= 2
				stage.listPos += temp0
				if temp0 == 8
					stage.listPos++
				end if
			else
				stage.activeList = BONUS_STAGE
				stage.listPos    = largeIcon.portrait
				timeAttack.round = largeIcon.portrait
				timeAttack.zone  = 0
			end if

			player.lives     = 1
			player.score     = 0
			options.gameMode = MODE_TIMEATTACK
			LoadStage()
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break

	case MENUCONTROL_SM_LEVEL_RETURN
		if object.timer > 0
			object.timer -= 8
		else
			timeAttack.round = -1
			if object.beatedRecord > 0
				PlaySfx(SfxName[Achievement], false)
				object.state = MENUCONTROL_SM_RECORD_BEATED
				object[0].drawOrder = 4
			else
				object.state = MENUCONTROL_SM_CONTROLS
			end if
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break

	case MENUCONTROL_SM_COMFIRMED_LEAVE
		if object.timer < 30
			object.timer++
		else
			object.offset = 0x50000
			object.timer = 0
			object.state++
		end if

		if object.smYPos > 0
			object.smYPos -= 8
		end if
		break

	case MENUCONTROL_SM_EXIT
		if object.timer < 256
			object.timer += 8
			object.offset -= 0x2000
		else
			object.timer = 0
			object.state = MENUCONTROL_CONTROLS
		end if

		CallFunction(MenuControl_IconTransitions)
		break

	case MENUCONTROL_SM_RECORD_BEATED
		if object.timer < 120
			object.timer++
		else
			object.timer = 0
			object.state = MENUCONTROL_SM_CONTROLS
			object.beatedRecord = false
			object[0].drawOrder = 1
		end if
		break

	case MENUCONTROL_SCROLL_DOWN
		if object.timer < 20
			object.timer++

			arrayPos0 = 32
			while arrayPos0 < 40
				object[arrayPos0].ypos -= 0xC0000
				arrayPos0++
			loop
		else
			object.timer = 0

			arrayPos0 = 32
			if timeAttack.menuSection == 0
				while arrayPos0 < 40
					object[arrayPos0].ypos += 0x1E00000
					object[arrayPos0].type = TypeName[Special Icon]
					arrayPos0++
				loop

				timeAttack.menuSection = 1
				CallFunction(MenuControl_TotalSpecialRecord)
			else
				while arrayPos0 < 40
					object[arrayPos0].ypos += 0x1E00000
					object[arrayPos0].type = TypeName[Round Icon]
					arrayPos0++
				loop

				timeAttack.menuSection = 0
				CallFunction(MenuControl_TotalRegularRecord)
			end if

			object.state++
		end if
		break

	case MENUCONTROL_SCROLLING_1 // Follow-up to MENUCONTROL_SCROLL_DOWN
		if object.timer < 20
			object.timer++
			arrayPos0 = 32
			while arrayPos0 < 40
				object[arrayPos0].ypos -= 0xC0000
				arrayPos0++
			loop
		else
			object.timer = 0
			object.state = MENUCONTROL_CONTROLS
		end if
		break

	case MENUCONTROL_SCROLL_UP
		if object.timer < 20
			object.timer++

			arrayPos0 = 32
			while arrayPos0 < 40
				object[arrayPos0].ypos += 0xC0000
				arrayPos0++
			loop
		else
			object.timer = 0

			arrayPos0 = 32
			if timeAttack.menuSection == 0
				while arrayPos0 < 40
					object[arrayPos0].ypos -= 0x1E00000
					object[arrayPos0].type = TypeName[Special Icon]
					arrayPos0++
				loop

				timeAttack.menuSection = 1
				CallFunction(MenuControl_TotalSpecialRecord)
			else
				while arrayPos0 < 40
					object[arrayPos0].ypos -= 0x1E00000
					object[arrayPos0].type = TypeName[Round Icon]
					arrayPos0++
				loop

				timeAttack.menuSection = 0
				CallFunction(MenuControl_TotalRegularRecord)
			end if

			object.state++
		end if
		break

	case MENUCONTROL_SCROLLING_2 // Follow-up to MENUCONTROL_SCROLL_UP
		if object.timer < 20
			object.timer++
			arrayPos0 = 32
			while arrayPos0 < 40
				object[arrayPos0].ypos += 0xC0000
				arrayPos0++
			loop
		else
			object.timer = 0
			object.state = MENUCONTROL_CONTROLS
		end if
		break

	end switch
end event


event ObjectDraw
	switch object.state
	case MENUCONTROL_SM_COMFIRMED_LEAVE
		temp0 = screen.xcenter
		temp0 += 16
		DrawRect(temp0, 31, screen.xcenter, 9, 0, 0, 0, 255)
		DrawRect(temp0, 40, screen.xcenter, 135, 224, 0, 0, 255)
		DrawRect(temp0, 175, screen.xcenter, 38, 0, 0, 96, 255)
		break

	case MENUCONTROL_SM_RECORD_BEATED
		temp0 = object.timer
		temp0 >>= 2
		temp0 &= 1
		if temp0 == 0
			temp0 = screen.xcenter
			temp0 += 16
			temp1 = object.beatedRecord
			temp1 *= 24
			temp1 += 105
			DrawRect(temp0, temp1, 34, 24, 8, 0, 104, 255)
		end if
		break

	end switch
end event


event ObjectStartup
	if options.soundtrack == OST_JP
		SetMusicTrack("JP/timeAttack.ogg", 0, 100512)
	else
		SetMusicTrack("US/DAGarden.ogg", 0, 117382)
	end if

	object[0].type = TypeName[Menu Control]
	object[0].drawOrder = 1

	options.physicalControls = true

	if engine.deviceType == MOBILE
		options.physicalControls = true
	end if
end event

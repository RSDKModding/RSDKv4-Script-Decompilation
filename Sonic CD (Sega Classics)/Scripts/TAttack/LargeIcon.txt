event ObjectUpdate
    switch object.state
    case LARGEICON_ENTER_SCREEN
        temp0 = screen.xcenter
        temp0 -= 72
        temp0 <<= 16
        if object.xpos < temp0
            object.xpos += 0x100000
            if object.xpos > temp0
                object.xpos = temp0
            end if
        else
            object.state = LARGEICON_BLANK
        end if
        break

    case LARGEICON_LEAVE_SCREEN
        if object.xpos > -0x480000
            object.xpos -= 0x100000
            if object.xpos < -0x480000
                object.xpos = -0x480000
            end if
        else
            object.state = LARGEICON_BLANK
        end if
        break

    end switch
end event


event ObjectDraw
    DrawSprite(15)
    DrawSprite(object.portrait)

	// Update both temp0 and temp1, but don't use them...
	// These are used by the touch controls for the mobile version of the game
    temp0 = screen.xcenter
    temp0 -= 72
    temp0 <<= 16
    temp0 -= object.xpos
    temp0 >>= 16
    temp0 += screen.xcenter
    temp0 += 20
    temp1 = temp0
    temp1 += 68
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // frame
            checkResult = object.propertyValue
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // frame
            object.propertyValue = editor.variableValue
            break

        end switch
    end if
end event

    object.seconds = saveRAM[arrayPos0]
    object.seconds /= 100
    object.seconds %= 60

    object.milliSeconds = saveRAM[arrayPos0]

    temp0 = object.xpos
    temp0 >>= 16
    temp0 -= 16
    DrawNumbers(26, temp0, temp1, object.minutes, 2, 8, 1)

    temp0 += 24

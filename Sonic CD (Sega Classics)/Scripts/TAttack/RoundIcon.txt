event ObjectUpdate
    if options.physicalControls == false
        if menuControl.state == MENUCONTROL_CONTROLS
            if object.lockedZone == false
                temp0 = object.ixpos
                temp0 -= 34
                temp1 = object.ixpos
                temp1 += 34
                temp2 = object.iypos
                temp2 -= 45
                temp3 = object.iypos
                temp3 += 45
                CheckTouchRect(temp0, temp2, temp1, temp3)
                if checkResult > -1
                    menuControl.selectedButton = object.portrait
                else
                    CheckTouchRect(0, 0, screen.xsize, screen.ysize)
                    if checkResult < 0
                        if menuControl.selectedButton == object.portrait
                            PlaySfx(SfxName[Select], false)

                            menuControl.state = MENUCONTROL_BLANK
                            object.state      = ROUNDICON_SELECTED_ZONE

                            if object.portrait == EXIT_ICON // Exit button pressed?
                                StopMusic()
                            end if
                        end if
                    else
                        if menuControl.selectedButton == object.portrait
                            menuControl.selectedButton = -1
                        end if
                    end if
                end if
            end if
        end if
    end if
end event


event ObjectDraw
    DrawSprite(10)

    // Icon Movement
    object.rotation++
    object.rotation &= 511

    Cos(temp0, object.rotation)
    temp0 >>= 4

    Sin(temp1, object.rotation)
    temp1 >>= 4

    switch object.portrait
    case PALMTREE_PANIC_ICON
        temp0 += 64
        temp1 += 76
        break

    case COLLISION_CHAOS_ICON
        temp0 += 320
        temp1 += 76
        break

    case TIDAL_TEMPEST_ICON
        temp0 += 448
        temp1 += 76
        break

    case QUARTZ_QUADRANT_ICON
        temp0 += 64
        temp1 += 220
        break

    case WACKY_WORKBENCH_ICON
        temp0 += 192
        temp1 += 220
        break

    case STARDUST_SPEEDWAY_ICON
        temp0 += 320
        temp1 += 220
        break

    case METALLIC_MADNESS_ICON
        temp0 += 448
        temp1 += 220
        break

    end switch

    if object.portrait < EXIT_ICON // Draw for all main stages
        vertexBuffer[0].x = object.xpos
        vertexBuffer[0].x >>= 16
        vertexBuffer[0].x -= 32
        vertexBuffer[0].u = temp0
        vertexBuffer[0].u -= 32
        vertexBuffer[0].y = object.ypos
        vertexBuffer[0].y >>= 16
        vertexBuffer[0].y -= 43
        vertexBuffer[0].v = temp1
        vertexBuffer[0].v -= 43

        vertexBuffer[1].x = object.xpos
        vertexBuffer[1].x >>= 16
        vertexBuffer[1].x += 31
        vertexBuffer[1].u = temp0
        vertexBuffer[1].u += 31
        vertexBuffer[1].y = vertexBuffer[0].y
        vertexBuffer[1].v = vertexBuffer[0].v

        vertexBuffer[2].x = vertexBuffer[0].x
        vertexBuffer[2].u = vertexBuffer[0].u
        vertexBuffer[2].y = object.ypos
        vertexBuffer[2].y >>= 16
        vertexBuffer[2].y += 24
        vertexBuffer[2].v = temp1
        vertexBuffer[2].v += 24

        vertexBuffer[3].x = vertexBuffer[1].x
        vertexBuffer[3].u = vertexBuffer[1].u
        vertexBuffer[3].y = vertexBuffer[2].y
        vertexBuffer[3].v = vertexBuffer[2].v

        Draw3DScene()
        DrawSprite(19)

        temp0 = object.xpos
        temp0 >>= 16
        temp0 -= 24

        temp1 = object.ypos
        temp1 >>= 16
        temp1 += 34

        // Zone specific time
        DrawNumbers(0, temp0, temp1, object.minutes, 2, 6, 1)

        temp0 += 16
        DrawNumbers(0, temp0, temp1, object.seconds, 2, 6, 1)

        temp0 += 19
        DrawNumbers(0, temp0, temp1, object.milliseconds, 2, 6, 1)
    else
        // Draw the Exit
        DrawSprite(20)

        temp0 = object.xpos
        temp0 >>= 16
        temp0 -= 18

        temp1 = object.ypos
        temp1 >>= 16
        temp1 += 34

        // Total Time
        DrawNumbers(0, temp0, temp1, object.minutes, 3, 6, 1)

        temp0 += 16
        DrawNumbers(0, temp0, temp1, object.seconds, 2, 6, 1)

        temp0 += 19
        DrawNumbers(0, temp0, temp1, object.milliseconds, 2, 6, 1)

        DrawSprite(22)
    end if

    DrawSprite(object.frame)

    if object.state == ROUNDICON_SELECTED_ZONE
        temp0 = object.timer
        temp0 &= 3
        temp0 >>= 1
        if temp0 == 0
            if menuControl.savedRound == object.portrait
                DrawSprite(21)
            end if
        end if

        object.timer++
        if object.timer == 30
            object.timer = 0
            object.state = ROUNDICON_BLANK
            if object.portrait == EXIT_ICON
                menuControl.state = MENUCONTROL_RETURN_TO_MENU
            else
                menuControl.state = MENUCONTROL_ENTER_SM
                menuControl.selectedButton = -1
            end if
        end if
    else
        if menuControl.state == MENUCONTROL_CONTROLS
            if menuControl.savedRound == object.portrait
                DrawSprite(21)
            end if
        end if
    end if

    if object.lockedZone == true
        DrawSprite(23)
    end if
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.portrait
            checkResult &= 7
            break

        case 0 // roundID
            checkResult = object.portrait
            checkResult &= 7
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.portrait = editor.variableValue
            object.portrait &= 7
            break

        case 0 // roundID
            object.portrait = editor.variableValue
            object.portrait &= 7
            break

        end switch
    end if
end event


event RSDKDraw
    object.frame = object.portrait
    object.frame += 11

    DrawSprite(10)

    if object.portrait < EXIT_ICON
        // Draw for all main stages
        DrawSprite(19)
    else
        // Draw the Exit
        DrawSprite(20)
        DrawSprite(22)
    end if

    DrawSprite(object.frame)
end event

public function MSProjector_PlayerInteraction
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if player[currentPlayer].animation == ANI_JUMPING
            BoxCollisionTest(C_TOUCH, object.entityPos, -13, -13, 13, 13, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.timer         = 0
                metalSonic.destroyed = true
                object.priority      = PRIORITY_ACTIVE
                object.state         = MSPROJECTOR_DESTROYED
                metalSonic.list++

                // Saviour of the Planet Unlock Criteria
                if stage.playerListPos == PLAYER_SONIC_A
                    if stage.debugMode == 0
                        if goodFuture.list > 16382 // 0b11111111111110
                            if metalSonic.list > 11
                                // (unused)
                            end if
                        end if
                    end if
                end if
            end if

            if player[currentPlayer].gravity == GRAVITY_AIR
                BoxCollisionTest(C_SOLID, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            end if
        else
            BoxCollisionTest(C_SOLID, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        end if
    next
end function

event ObjectUpdate
    switch object.state
    case MSPROJECTOR_ANIM_1
        object.timer++
        if object.timer == 16
            object.timer = 0
            object.state = MSPROJECTOR_ANIM_2
        end if

        object.flicker = object.timer
        object.flicker &= 3

        object.frame++
        if object.frame == 56
            object.frame = 0
        end if

        CallFunction(MSProjector_PlayerInteraction)
        break

    case MSPROJECTOR_ANIM_2
        object.timer++
        if object.timer == 12
            object.timer = 0
            object.state = MSPROJECTOR_ANIM_1
        end if

        object.flicker = object.timer
        object.flicker %= 3

        object.frame++
        if object.frame == 56
            object.frame = 0
        end if

        CallFunction(MSProjector_PlayerInteraction)
        break

    case MSPROJECTOR_DESTROYED
        temp0 = object.timer
        temp0 &= 3
        if temp0 == 0
            Rand(temp0, 32)
            temp0 -= 16
            temp0 <<= 16
            temp0 += object.xpos

            Rand(temp1, 32)
            temp1 -= 16
            temp1 <<= 16
            temp1 += object.ypos

            CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
            object[tempObjectPos].drawOrder = 4
            PlaySfx(SfxName[Explosion], false)
        end if

        object.timer++
        if object.timer == 60
            ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, object.xpos, object.ypos)
        end if
        break
    end switch
end event


event ObjectDraw
    object.spriteSheet = MSProjector_projectorSheet
    DrawSprite(0)

    if object.state < MSPROJECTOR_DESTROYED
        object.angle += MSProjector_speed
        object.angle &= 511

        object.currentFrame++
        object.currentFrame &= 31
        temp2 = object.currentFrame
        temp2 >>= 4
        temp2 += 7
        Cos(temp0, object.angle)
        Sin(temp1, object.angle)

        if temp1 < 0
            object.direction = FACING_RIGHT
        else
            object.direction = FACING_LEFT
        end if

        temp0 <<= 12
        temp1 <<= 11

        temp0 += object.xpos
        temp1 += object.ypos

        temp0 -= 0x580000
        temp1 -= 0x180000

        // Projector Particles
        temp3 = object.timer
        temp3 &= 1
        if temp3 == 0
            temp3 = object.timer
            temp3 &= 3
            temp3 >>= 1
            temp3++
            DrawSprite(temp3)
        end if

        if object.flicker < 2
            if MSProjector_speed == 8 // This is true for every animal except the TTZ Fishes
                // Metal and stomped animal
                if object.frame < 28
                    object.spriteSheet = MSProjector_animalSheet
                    DrawSprite(5)
                    object.spriteSheet = MSProjector_projectorSheet
                    DrawSprite(3)
                else
                    object.spriteSheet = MSProjector_animalSheet
                    DrawSprite(6)
                    object.spriteSheet = MSProjector_projectorSheet
                    DrawSprite(4)
                end if

                // Free animal
                object.spriteSheet = MSProjector_animalSheet
                DrawSpriteFX(temp2, FX_FLIP, temp0, temp1)
            else
                // Metal
                object.spriteSheet = MSProjector_projectorSheet
                if object.frame < 28
                    DrawSprite(3)
                else
                    DrawSprite(4)
                end if

                // Fishes
                object.spriteSheet = MSProjector_animalSheet
                DrawSpriteFX(temp2, FX_FLIP, temp0, temp1)
                temp4 = object.angle
                temp4 += 320
                temp4 &= 511
                Cos(temp0, temp4)
                Sin(temp1, temp4)

                if temp1 < 0
                    object.direction = FACING_RIGHT
                else
                    object.direction = FACING_LEFT
                end if

                temp0 <<= 12
                temp1 <<= 11

                temp0 += object.xpos
                temp1 += object.ypos

                temp0 -= 0x600000
                temp1 -= 0x180000

                temp2 -= 2
                DrawSpriteFX(temp2, FX_FLIP, temp0, temp1)
            end if
        end if
    end if
end event

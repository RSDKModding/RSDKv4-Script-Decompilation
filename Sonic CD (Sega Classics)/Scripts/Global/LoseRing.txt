// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.targetPlayer

// Player Aliases
private alias object.state  : player.state
private alias object.value0 : player.rings

// Path ID Aliases
private alias 0 : PATH_A

// Pan
private alias 0 : PAN_RIGHT
private alias 1 : PAN_LEFT

// ========================
// Events
// ========================

event ObjectUpdate
	object.yvel += 0x1800
	object.xpos += object.xvel
	object.ypos += object.yvel

	if object.yvel >= 0
		ObjectTileCollision(CSIDE_FLOOR, 0, 8, PATH_A)
		if checkResult == true
			temp0 = object.yvel
			temp0 >>= 2
			object.yvel -= temp0
			FlipSign(object.yvel)
			if object.yvel > -0x10000
				object.yvel = -0x10000
			end if
		end if
	end if

	object.timer++
	if object.timer == 0x100
		object.type = TypeName[Blank Object]
		object.xvel = 0
		object.yvel = 0
		object.timer = 0
	else
		object.animationTimer += object.animationSpeed
		if object.animationTimer >= 0x100
			object.animationTimer -= 0x100
			object.frame++
			if object.frame == 8
				object.frame = 0
				if object.animationSpeed > 16
					object.animationSpeed -= 16
				end if
			end if
		end if

		if object.timer >= 240
			object.alpha -= 16
		end if
	end if

	if object.timer > 63
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Check if the player should be able to collect rings
			if player[currentPlayer].state != Player_State_Hurt
				BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					// Make the ring disappear
					object.type = TypeName[Ring Sparkle]

					if object.propertyValue == 0
						object.drawOrder = 4
					else
						object.drawOrder = 2
					end if

					object.timer        = 0
					object.targetPlayer = SLOT_PLAYER1
					object.timer        = 0
					object.frame        = 0

					// Add to the player's ring count
					player[SLOT_PLAYER1].rings++

					// Max 999 rings
					if player[SLOT_PLAYER1].rings > 999
						// Bug Details:
						// This sets the current object instance's value0 to 999 instead of object[0].value0,
						// which should be aliased as player[SLOT_PLAYER1].rings.
						player.rings = 999
					end if

					// Give the player an extra life if needed
					if player[SLOT_PLAYER1].rings >= ringExtraLife
						if options.gameMode != MODE_TIMEATTACK
							player.lives++
							PlaySfx(SfxName[1 Up], false)

							ringExtraLife += 100
							if ringExtraLife > 300
								ringExtraLife = 1000
							end if
						end if

						if stage.debugMode == false
							if player[SLOT_PLAYER1].rings >= 200
							end if
						end if
					end if

					// Play the ring chime
					if ringPan == PAN_RIGHT
						PlaySfx(SfxName[Ring L], false)
						SetSfxAttributes(SfxName[Ring L], -1, -100)
						ringPan = PAN_LEFT
					else
						PlaySfx(SfxName[Ring R], false)
						SetSfxAttributes(SfxName[Ring R], -1, 100)
						ringPan = PAN_RIGHT
					end if
				end if
			end if
		next
	end if
end event


event ObjectDraw
	DrawSpriteFX(object.frame, FX_INK, object.xpos, object.ypos)
end event

	// Ring Frames, they're the same as the standard Ring object sprites

	SpriteFrame(-8, -8, 16, 16, 1, 35)

	SpriteFrame(-8, -8, 16, 16, 1, 69)
	SpriteFrame(-8, -8, 16, 16, 1, 86)
	SpriteFrame(-8, -8, 16, 16, 1, 103)
	SpriteFrame(-8, -8, 16, 16, 1, 120)
end event

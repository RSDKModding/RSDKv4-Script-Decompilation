event ObjectDraw
    switch object.state
    case TIMEWARP_SETUP
        if object.timer < 320
            if object.timer == 0
                warp.xpos          = player[SLOT_PLAYER1].xpos
                warp.ypos          = player[SLOT_PLAYER1].ypos
                warp.xvel          = player[SLOT_PLAYER1].xvel
                warp.yvel          = player[SLOT_PLAYER1].yvel
                warp.speed         = player[SLOT_PLAYER1].speed
                warp.collisionMode = player[SLOT_PLAYER1].collisionMode
                warp.minRollSpeed  = player[SLOT_PLAYER1].minRollSpeed
                warp.gravity       = player[SLOT_PLAYER1].gravity

                if player[SLOT_PLAYER1].state == Player_State_Static
                    warp.state = Player_State_Air
                else
                    if player[SLOT_PLAYER1].state == Player_State_SpinningTop
                        warp.state = Player_State_Air
                    else
                        if player[SLOT_PLAYER1].state == Player_State_WaterCurrent
                            warp.state = Player_State_Air
                        else
                            warp.state = player[SLOT_PLAYER1].state
                        end if
                    end if
                end if

                warp.animation   = player[SLOT_PLAYER1].animation
                warp.angle       = player[SLOT_PLAYER1].angle
                warp.playerFrame = player[SLOT_PLAYER1].frame

                player[SLOT_PLAYER1].invincibleTimer = 1000
                player[SLOT_PLAYER1].controlMode     = CONTROLMODE_NONE
                player[SLOT_PLAYER1].visible         = false
            end if

            music.volume -= 8
            SetScreenFade(208, 255, 224, object.timer)
            object.timer += 10
        else
            // 88 Miles per Hour unlock criteria
            if stage.playerListPos == PLAYER_SONIC_A
                if stage.debugMode == false
                    // (unused)
                end if
            end if

            StopSfx(SfxName[Flying])
            StopSfx(SfxName[Tired])

            stage.state       = STAGE_RUNNING
            stage.timeEnabled = false

            // Saves everything
            CallFunction(StageSetup_SaveStageState)

            recMilliSeconds = stage.milliSeconds
            recSeconds      = stage.seconds
            recMinutes      = stage.minutes

            warp.rings = player[SLOT_PLAYER1].rings
            temp2      = HUD.currentTimePeriod

            // Sets every object as a blank object in the current period
            // (likely to avoid the player's death during the cutscene)
            temp0 = 0
            while temp0 < 1184
                ResetObjectEntity(temp0, TypeName[Blank Object], 0, 0, 0)
                temp0++
            loop

            // Blanks the level layers
            screen.xoffset = 0
            screen.yoffset = 0
            stage[0].activeLayer = 9
            stage[1].activeLayer = 9
            stage[2].activeLayer = 9
            stage[3].activeLayer = 9
            arrayPos0 = 0

            // Sets the fade in
            object[arrayPos0].type          = TypeName[Time Warp]
            object[arrayPos0].state         = TIMEWARP_FADE_IN
            object[arrayPos0].timer         = 320
            object[arrayPos0].currentPeriod = temp2
            arrayPos0++

            // Creates the background
            temp0 = -32
            while temp0 < 256
                temp1 = 0
                while temp1 < screen.xsize
                    object[arrayPos0].type  = TypeName[Time Warp]
                    object[arrayPos0].state = TIMEWARP_BACKGROUND
                    Rand(object[arrayPos0].frame, 27)
                    object[arrayPos0].frame >>= 2
                    object[arrayPos0].frame <<= 2
                    object[arrayPos0].ixpos = temp1
                    object[arrayPos0].iypos = temp0
                    object[arrayPos0].ixpos += 16
                    object[arrayPos0].iypos += 16
                    Rand(object[arrayPos0].direction, 15)
                    object[arrayPos0].direction >>= 2
                    arrayPos0++
                    temp1 += 32
                loop
                temp0 += 32
            loop

            // And of course, the star of the show appears
            object[arrayPos0].type  = TypeName[Warp Sonic]
            object[arrayPos0].ixpos = screen.xcenter
            object[arrayPos0].iypos = 480

            SetScreenFade(208, 255, 224, 255)

            PlaySfx(SfxName[Time Warp], false)
        end if
        break

    case TIMEWARP_FADE_IN
        if object.timer > 0
            SetScreenFade(208, 255, 224, object.timer)
            object.timer -= 10
        else
            object.state = TIMEWARP_BLANK
        end if
        break

    case TIMEWARP_BACKGROUND
        temp0 = object.frame
        temp0 >>= 1
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

        object.frame++
        if object.frame == 28
            object.frame = 0
        end if

        object.ypos += 0x40000
        if object.ypos >= 0x1000000
            object.ypos -= 0x1200000
        end if
        break

    case TIMEWARP_BLANK
        break

    case TIMEWARP_LEAVING
        object.timer++
        if object.timer == 40
            object.timer = 0
            object.state = TIMEWARP_TRAVEL
        end if
        break

    case TIMEWARP_TRAVEL
        if object.timer < 1800
            SetScreenFade(208, 255, 224, object.timer)
            object.timer += 10
        else
            SetScreenFade(208, 255, 224, 255)
            switch object.currentPeriod
            case TIME_PRESENT
                if warp.destination == WARPDEST_PAST
                    stage.listPos++
                else
                    if generator.destroyed == true
                        stage.listPos += 2
                    else
                        stage.listPos += 3
                    end if
                end if
                break

            case TIME_PAST
                stage.listPos--
                break

            case TIME_GOOD_FUTURE
                stage.listPos -= 2
                break

            case TIME_BAD_FUTURE
                stage.listPos -= 3
                break

            end switch

            fadeColor = 208
            fadeColor <<= 16
            temp0 = 255
            temp0 <<= 8
            fadeColor += temp0
            fadeColor += 224

            LoadStage()
        end if
        break
    end switch
end event

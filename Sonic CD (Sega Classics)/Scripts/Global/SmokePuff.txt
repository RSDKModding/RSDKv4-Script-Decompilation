// ========================

// Aliases
private alias object.value0 : object.timer

// ========================
// Events
// ========================

event ObjectUpdate
	object.timer++
	if object.timer > 3
		object.timer = 0
		object.frame++
		if object.frame > 4
			object.type = TypeName[Blank Object]
			object.frame = 0
		end if
	end if
end event


event ObjectDraw
	DrawSprite(object.frame)
end event


event ObjectStartup
	if stage.playerListPos == PLAYER_SONIC_A
		LoadSpriteSheet("Global/Items2.gif")
	else
		LoadSpriteSheet("Global/Items2_t.gif")
	end if

	SpriteFrame(-16, -8, 32, 16, 1, 1)		// #0 - Smoke Puff Frame 0
	SpriteFrame(-16, -16, 32, 32, 1, 18)	// #1 - Smoke Puff Frame 1
	SpriteFrame(-16, -16, 32, 32, 1, 51)	// #2 - Smoke Puff Frame 2
	SpriteFrame(-24, -24, 48, 48, 34, 1)	// #3 - Smoke Puff Frame 3
	SpriteFrame(-24, -24, 48, 48, 34, 50)	// #4 - Smoke Puff Frame 4
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Items2.gif")
	SpriteFrame(-16, -8, 32, 16, 1, 1)		// #0 - Smoke Puff Frame 0

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

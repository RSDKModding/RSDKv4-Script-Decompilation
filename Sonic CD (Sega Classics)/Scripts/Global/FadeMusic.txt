// ========================

// Aliases
private alias object.value0 : object.timer

// Property Values
private alias 0 : FADEMUSIC_TO_BOSS
private alias 1 : FADEMUSIC_TO_LEVEL
private alias 2 : FADEMUSIC_SSZEGGMAN
private alias 3 : FADEMUSIC_SSZAMY

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.propertyValue
	case FADEMUSIC_TO_BOSS
		if object.timer < 100
			object.timer++
			music.volume--
		else
			PlayMusic(4)
			object.type = TypeName[Blank Object]
		end if
		break

	case FADEMUSIC_TO_LEVEL
		if object.timer < 100
			object.timer++
			music.volume--
		else
			PlayMusic(0)
			object.type = TypeName[Blank Object]
		end if
		break

	case FADEMUSIC_SSZEGGMAN
		if object.timer < 50
			object.timer++
			music.volume -= 2
		else
			PlayMusic(0)
			object.type = TypeName[Blank Object]
		end if
		break

	case FADEMUSIC_SSZAMY
		if object.timer < 100
			object.timer++
			music.volume--
		else
			PlayMusic(1)

			object.type = TypeName[Blank Object]

			object[30].type      = TypeName[Act Finish]
			object[30].drawOrder = 6
			object[30].priority  = PRIORITY_ACTIVE
		end if
		break

	end switch
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)		// #0 - "Script" Icon

	// Although used by the object, it's not to be set from the editor
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

private alias 1 : TRACK_ACTFINISH

// ========================

public function SignPost_DebugDraw
    DrawSprite(0)
    DrawSprite(1)
end function


public function SignPost_DebugSpawn
    CreateTempObject(TypeName[Sign Post], 0, object.xpos, object.ypos)
    object[tempObjectPos].frame     = 1
    object[tempObjectPos].spinCount = 8
end function


// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case SIGNPOST_IDLE
        temp0 = object.ypos
        temp0 += 0x400000
        if player[SLOT_PLAYER1].ypos < temp0 // Makes sure the player triggers the signpost no matter the height
            stage.newXBoundary1 = object.xpos
            stage.newXBoundary1 >>= 16
            stage.newXBoundary1 -= screen.xcenter
            stage.newXBoundary1 -= screen.xcenter

            stage.newXBoundary2 = object.xpos
            stage.newXBoundary2 >>= 16
            stage.newXBoundary2 += screen.xcenter

            if player[SLOT_PLAYER1].xpos > object.xpos // Check that the player crossed the sign post
                stage.newXBoundary1 = object.xpos
                stage.newXBoundary1 >>= 16
                stage.newXBoundary1 -= screen.xcenter

                stage.timeEnabled = false

                object.state = SIGNPOST_FLIP
                PlaySfx(SfxName[Sign Post], false)
            end if
        end if
        break

    case SIGNPOST_FLIP
        music.volume--
        object.spinTimer++
        if object.spinTimer == 2
            object.spinTimer = 0
            object.frame++
            if object.frame == 5
                Rand(temp0, 48)
                temp0 <<= 16
                temp0 += object.xpos
                temp0 -= 0x180000

                Rand(temp1, 32)
                temp1 <<= 16
                temp1 += object.ypos
                temp1 -= 0x1C0000

                CreateTempObject(TypeName[Ring Sparkle], 0, temp0, temp1)

                object.spinCount--
                if object.spinCount == 0
                    SetBit(object.activePlayers, 0, true)
                    object.state = SIGNPOST_MOVE_PLAYER
                end if
            end if

            if object.frame == 9
                Rand(temp0, 48)
                temp0 <<= 16
                temp0 += object.xpos
                temp0 -= 0x180000

                Rand(temp1, 32)
                temp1 <<= 16
                temp1 += object.ypos
                temp1 -= 0x1C0000

                CreateTempObject(TypeName[Ring Sparkle], 0, temp0, temp1)
                object.frame = 1
            end if
        end if
        break

    case SIGNPOST_MOVE_PLAYER
        temp0 = screen.xcenter
        temp0 <<= 16
        temp0 += object.xpos

        if player[SLOT_PLAYER1].xpos > temp0
            PlayMusic(TRACK_ACTFINISH)

            object.state = SIGNPOST_BLANK

            object[SLOT_ACTFINISH].type      = TypeName[Act Finish]
            object[SLOT_ACTFINISH].drawOrder = 6
            object[SLOT_ACTFINISH].priority  = PRIORITY_ACTIVE
        else
            if player[SLOT_PLAYER1].type == TypeName[Blank Object]
                PlayMusic(TRACK_ACTFINISH)

                object.state = SIGNPOST_BLANK

                object[SLOT_ACTFINISH].type      = TypeName[Act Finish]
                object[SLOT_ACTFINISH].drawOrder = 6
                object[SLOT_ACTFINISH].priority  = PRIORITY_ACTIVE
            end if
        end if
        break

    case SIGNPOST_BLANK
        break

    end switch

    // Invisible off-screen barrier
    temp0 = 40
    temp0 += 212
    temp0 <<= 16
    temp0 += object.xpos

    currentPlayer = SLOT_PLAYER1
    while currentPlayer < playerCount
        GetBit(temp1, object.activePlayers, currentPlayer)
        if temp1 == true
            player[currentPlayer].interaction = false
            player[currentPlayer].controlMode = CONTROLMODE_NONE
            player[currentPlayer].up          = false
            player[currentPlayer].down        = false
            player[currentPlayer].left        = false
            player[currentPlayer].right       = true
            player[currentPlayer].jumpPress   = false
            player[currentPlayer].jumpHold    = false
        end if

        if player[currentPlayer].xpos > temp0
            player[currentPlayer].xpos  = temp0
            player[currentPlayer].speed = 0
            player[currentPlayer].xvel  = 0
        end if

        if player[currentPlayer].type == TypeName[Debug Mode]
            SetBit(object.activePlayers, currentPlayer, false)
        end if

        if player[currentPlayer].type == TypeName[Blank Object]
            SetBit(object.activePlayers, currentPlayer, false)
        end if

        currentPlayer++
    loop
end event


event ObjectDraw
    DrawSprite(0)
    DrawSprite(object.frame)
end event


event ObjectStartup
    if stage.playerListPos == PLAYER_SONIC_A
        LoadSpriteSheet("Global/Items2.gif")
    else
        LoadSpriteSheet("Global/Items2_t.gif")
    end if

    SpriteFrame(-12, -12, 24, 24, 0, 150)		// #0 - Sign Post
    SpriteFrame(-24, -44, 48, 32, 34, 99)		// #1 - Sign Post Head (Eggman)
    SpriteFrame(-16, -44, 32, 32, 1, 183)		// #2 - Sign Post Head (Flip 1)
    SpriteFrame(-4, -44, 8, 32, 25, 150)		// #3 - Sign Post Head (Flip 2)
    SpriteFrame(-16, -44, 32, 32, 1, 216)		// #4 - Sign Post Head (Flip 3)
    SpriteFrame(-24, -44, 48, 32, 34, 132)		// #5 - Sign Post Head (Player)
    SpriteFrame(-16, -44, 32, 32, 1, 183)		// #6 - Sign Post Head (Flip 1)
    SpriteFrame(-4, -44, 8, 32, 25, 150)		// #7 - Sign Post Head (Flip 2)
    SpriteFrame(-16, -44, 32, 32, 1, 216)		// #8 - Sign Post Head (Flip 3)

    foreach (TypeName[Sign Post], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].frame     = 1
        object[arrayPos0].spinCount = 8
        object[arrayPos0].priority  = PRIORITY_XBOUNDS
    next

    SetTableValue(TypeName[Sign Post], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(SignPost_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(SignPost_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
    DebugMode_ObjCount++
end event

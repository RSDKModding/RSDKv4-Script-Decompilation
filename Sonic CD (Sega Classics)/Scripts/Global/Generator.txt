public function Generator_DebugDraw
    temp0 = object.direction
    object.direction = FLIP_Y

public function Generator_DebugSpawn
    CreateTempObject(TypeName[Generator], 0, object.xpos, object.ypos)
    object[tempObjectPos].direction = FLIP_Y
    object[tempObjectPos].yOriginPos  = object[tempObjectPos].ypos
    object[tempObjectPos].oscillation = -0x2000
end function


public function Generator_PlayerInteraction
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if player[currentPlayer].animation == ANI_JUMPING
            BoxCollisionTest(C_TOUCH, object.entityPos, -38, -32, 38, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.state = GENERATOR_EXPLODE
                object.ypos  = object.yOriginPos

                generator.destroyed = true
                goodFuture.enabled  = true

                object.priority = PRIORITY_ACTIVE

                player.score += 1500
                if stage.actNum < 3
                    temp0 = stage.listPos
                    temp0 /= 10
                    temp0 <<= 1
                    temp0 += stage.actNum
                    temp0--
                    SetBit(goodFuture.list, temp0, true)
                end if

                // Saviour of the Planet unlock criteria
                if stage.playerListPos == PLAYER_SONIC_A
                    if stage.debugMode == false
                        if goodFuture.list > 16382 // 0b11111111111110
                            if metalSonic.list > 11
                                // no achievement
                            end if
                        end if
                    end if
                end if
            end if
        else
            BoxCollisionTest(C_SOLID, object.entityPos, -38, -33, 38, 36, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                player[currentPlayer].ypos += 0x10000
            end if
        end if
    next
end function

event ObjectUpdate
    if object.propertyValue == OPERATIVE
        switch object.state
        case GENERATOR_HOVER_DOWN
            object.timer++
            if object.timer == 32
                object.oscillation = 0x2000
                object.timer = 0
                object.state = GENERATOR_HOVER_UP
            end if

            object.ypos += object.oscillation
            CallFunction(Generator_PlayerInteraction)
            break

        case GENERATOR_HOVER_UP
            object.timer++
            if object.timer == 32
                object.oscillation = -0x2000
                object.timer = 0
                object.state = GENERATOR_HOVER_DOWN
            end if

            object.ypos += object.oscillation
            CallFunction(Generator_PlayerInteraction)
            break

        case GENERATOR_EXPLODE
            // Explosion code
            Rand(temp0, 80)
            temp0 -= 40
            temp0 <<= 16
            temp0 += object.xpos

            Rand(temp1, 64)
            temp1 -= 32
            temp1 <<= 16
            temp1 += object.ypos

            CreateTempObject(TypeName[Explosion], 0, temp0, temp1)

            temp0 = object.timer
            temp0 &= 3
            if temp0 == 0
                PlaySfx(SfxName[Explosion], false)
            end if

            object.timer++
            if object.timer == 60
                object.priority      = PRIORITY_BOUNDS
                object.timer         = 0
                object.state         = GENERATOR_HOVER_DOWN
                object.propertyValue = DESTROYED

                object.iypos += 16
                PlaySfx(SfxName[Achievement], false)
            end if
            break
        end switch
    end if
end event


event ObjectDraw
    switch object.propertyValue
    case OPERATIVE
        if object.state < GENERATOR_EXPLODE
            DrawSprite(object.frame)
            DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)

            // Electricity
            temp0 = object.frame
            temp0 += 4
            DrawSprite(temp0)
            temp0 += 4
            DrawSprite(temp0)
            temp0 += 4
            DrawSprite(temp0)

            object.frameTimer++
            if object.frameTimer > 3
                object.frameTimer = 0
                object.frame++
                if object.frame > 3
                    object.frame = 0
                end if
            end if
        end if
        break

    case OLD
        DrawSprite(17)
        DrawSpriteFX(17, FX_FLIP, object.xpos, object.ypos)
        break

    case DESTROYED
        DrawSprite(16)
        break

    case OLD_DESTROYED
        DrawSprite(18)
        break
    end switch
end event

    LoadSpriteSheet("Global/Items3.gif")

    foreach (TypeName[Generator], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].direction = FLIP_Y

        if generator.destroyed == true
            switch object[arrayPos0].propertyValue
            case OPERATIVE
                object[arrayPos0].propertyValue = DESTROYED
                break

            case OLD
                object[arrayPos0].propertyValue = OLD_DESTROYED
                break
            end switch
        else
            if object[arrayPos0].propertyValue == OPERATIVE
                object[arrayPos0].iypos -= 16
            end if

            object[arrayPos0].yOriginPos  = object[arrayPos0].ypos

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // inactive
			checkResult = object.propertyValue
			break
		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // inactive
			object.propertyValue = editor.variableValue
			break
		end switch
	end if
end event

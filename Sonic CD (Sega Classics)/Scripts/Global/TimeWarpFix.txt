// Aliases
private alias object.value0 : object.timer

// Player Aliases
private alias object.xpos    : player.xpos
private alias object.ypos    : player.ypos
private alias object.gravity : player.gravity

// Property Values
private alias 0 : TIMEWARPFIX_ERASE_TOUCH    // Touch this while trying to warp, no more time travel for you
private alias 1 : TIMEWARPFIX_ERASE_AREA     // Get past this, no more warp for you
private alias 2 : TIMEWARPFIX_SAVE_POINT     // Sets your position to the next object after time-travelling
private alias 3 : TIMEWARPFIX_SAVE_POINT_AIR // Sets your position to the next object after time-travelling, but in the air

// Warp Destination
private alias 0 : WARP_NONE


// ========================
// Events
// ========================

event ObjectUpdate
	switch object.propertyValue
	case TIMEWARPFIX_ERASE_TOUCH
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if warp.destination > WARP_NONE
				if warp.timer > 99
					if warp.timer < 220 // They forgot to fix the warp timer here
						warp.destination = WARP_NONE
					end if
				end if

				warp.timer = 0
			end if
		end if
		break

	case TIMEWARPFIX_ERASE_AREA
		if player[SLOT_PLAYER1].xpos < object.xpos
			warp.destination = WARP_NONE
			warp.timer = 0
		end if
		break

	case TIMEWARPFIX_SAVE_POINT
		BoxCollisionTest(C_TOUCH, object.entityPos, -48, -48, 48, 48, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object.type = TypeName[Blank Object]
			player[SLOT_PLAYER1].xpos = object[+1].xpos
			player[SLOT_PLAYER1].ypos = object[+1].ypos
		end if

		object.timer++
		if object.timer == 2
			object.type = TypeName[Blank Object]
		end if
		break

	case TIMEWARPFIX_SAVE_POINT_AIR
		BoxCollisionTest(C_TOUCH, object.entityPos, -96, -96, 96, 96, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object.type = TypeName[Blank Object]
			player[SLOT_PLAYER1].xpos = object[+1].xpos
			player[SLOT_PLAYER1].ypos = object[+1].ypos
			player[SLOT_PLAYER1].gravity = GRAVITY_AIR
		end if

		object.timer++
		if object.timer == 2
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
end event


// ========================
// Editor Events
// ========================

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Type
			checkResult = object.propertyValue
			checkResult &= 3
			break

		end switch
	else
		switch editor.VariableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // type
			object.propertyValue = editor.variableValue
			object.propertyValue &= 3
			break

		end switch
	end if
end event


event RSDKDraw
	DrawSprite(0)

	if editor.showGizmos == true
 		editor.drawingOverlay = true

		temp0 = -1

		switch object.PropertyValue
		case TIMEWARPFIX_ERASE_TOUCH
		case TIMEWARPFIX_FORCE_ERASE_TOUCH
			temp0 = 16; temp1 = 16; temp2 = 16; temp3 = 16;
			break

		case TIMEWARPFIX_ERASE_AREA
			// Draw a rightward arrow, should convey what this object does well enough
			temp1 = object.xpos
			temp1 += 0x380000
			DrawArrow(object.xpos, object.ypos, temp1, object.ypos, 0, 0, 255)
			break

		case TIMEWARPFIX_SAVE_POINT
			temp0 = 48; temp1 = 48; temp2 = 48; temp3 = 48;

			// Draw an arrow to where the player will be sent
			DrawArrow(object.xpos, object.ypos, object[+1].xpos, object[+1].ypos, 255, 255, 0)
			break

		case TIMEWARPFIX_SAVE_POINT_AIR
			temp0 = 96; temp1 = 96; temp2 = 96; temp3 = 96;

			// Draw an arrow to where the player will be sent
			DrawArrow(object.xpos, object.ypos, object[+1].xpos, object[+1].ypos, 255, 255, 0)
			break

		end switch

		if temp0 != -1
			CallFunction(EditorHelpers_DrawHitbox)
		end if

		editor.drawingOverlay = false
	end if
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-8, -8, 16, 16, 173, 67)

	AddEditorVariable("type")
	SetActiveVariable("type")
	AddEnumVariable("Erase Time Warp on Touch", 0)
	AddEnumVariable("Erase Time Warp on Pass", 1)
	AddEnumVariable("Set Time Warp Pos (Ground)", 2)
	AddEnumVariable("Set Time Warp Pos (Air)", 3)
	AddEnumVariable("Erase Time Warp on Touch", 4)
end event

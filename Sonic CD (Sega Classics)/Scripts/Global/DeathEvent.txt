// Aliases
private alias object.value1 : object.text1Pos
private alias object.value2 : object.text2Pos
private alias object.value3 : object.timer

// States
private alias 0 : DEATHEVENT_GAMEOVER
private alias 1 : DEATHEVENT_TIMEOVER
private alias 2 : DEATHEVENT_FADEOUT
private alias 3 : DEATHEVENT_TIMEATTACK

// Presentation Stages
private alias 0 : STAGE_P_TITLE
private alias 2 : STAGE_P_TATTACK

// Engine States
private alias 8 : ENGINE_RESETGAME

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case DEATHEVENT_GAMEOVER
	case DEATHEVENT_TIMEOVER
		temp0 = screen.xcenter
		temp0 -= 72
		if object.text1Pos < temp0
			object.text1Pos += 16
			if object.text1Pos > temp0
				object.text1Pos = temp0
			end if
		end if

		temp0 += 80
		if object.text2Pos > temp0
			object.text2Pos -= 16
			if object.text2Pos < temp0
				object.text2Pos = temp0
			end if
		end if

		if object.timer < 288
			object.timer += 4
		else
			if object.state == DEATHEVENT_GAMEOVER
				stage.activeList = PRESENTATION_STAGE
				stage.listPos    = STAGE_P_TITLE
				lampPostID       = 0

				generator.destroyed  = false
				metalSonic.destroyed = false

				if options.gameMode == MODE_SAVEGAME
					arrayPos1 = options.saveSlot
					arrayPos1 <<= 3
					arrayPos1++
					if saveRAM[arrayPos1] < 3
						saveRAM[arrayPos1] = 3
					end if

					WriteSaveRAM()
				end if
			end if

			if object.state == DEATHEVENT_TIMEOVER
				recMilliSeconds = 0
				recSeconds      = 0
				recMinutes      = 0
			end if

			engine.state = ENGINE_RESETGAME
		end if

		if object.timer > 0
			music.volume -= 2
			SetScreenFade(0, 0, 0, object.timer)
		else
			if keyPress[0].buttonA == true
				object.timer = 0
			end if

			if keyPress[0].buttonB == true
				object.timer = 0
			end if

			if keyPress[0].buttonC == true
				object.timer = 0
			end if
		end if
		break

	case DEATHEVENT_FADEOUT
		if object.timer < 288
			music.volume -= 2
			object.timer += 4
		else
			LoadStage()
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break

	case DEATHEVENT_TIMEATTACK
		if object.timer < 288
			music.volume -= 2
			object.timer += 4
		else
			timeAttack.result = 0
			lampPostID        = 0
			stage.activeList  = PRESENTATION_STAGE
			stage.listPos     = STAGE_P_TATTACK
			LoadStage()
		end if

		SetScreenFade(0, 0, 0, object.timer)
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case DEATHEVENT_GAMEOVER
		DrawSpriteScreenXY(0, object.text1Pos, 96)
		DrawSpriteScreenXY(2, object.text2Pos, 96)
		break

	case DEATHEVENT_TIMEOVER
		DrawSpriteScreenXY(1, object.text1Pos, 96)
		DrawSpriteScreenXY(2, object.text2Pos, 96)
		break
	end switch
end event


event ObjectStartup
	if stage.playerListPos == PLAYER_SONIC_A
		LoadSpriteSheet("Global/Display.gif")
	else
		LoadSpriteSheet("Global/Display_t.gif")
	end if

	SpriteFrame(0, 0, 64, 16, 0, 189)   // 0 - "Game"
	SpriteFrame(0, 0, 56, 16, 65, 189)  // 1 - "Time"
	SpriteFrame(0, 0, 64, 16, 122, 189) // 2 - "Over"
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143) // #0 - "Script" Icon

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

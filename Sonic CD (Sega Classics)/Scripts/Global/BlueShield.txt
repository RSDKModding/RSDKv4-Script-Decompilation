// ========================
// Aliases
// ========================

// Aliases
private alias object.value0 : object.timer

// Player Aliases
private alias object.state     : player.state
private alias object.xpos      : player.xpos
private alias object.ypos      : player.ypos
private alias object.visible   : player.visible
private alias object.drawOrder : player.drawOrder

// ========================
// Events
// ========================

event ObjectUpdate
	object.drawOrder = player[-playerCount].drawOrder

	// Dead player shouldn't have any shield
	if player[-playerCount].state == Player_State_Death
		object.type = TypeName[Blank Object]
	end if

	// If they drowned, they shouldn't have them either
	if player[-playerCount].state == Player_State_Drown
		object.type = TypeName[Blank Object]
	end if

	// Update the object's animation
	object.timer++
	if object.timer == 2
		object.timer = 0
		object.frame++
		if object.frame > 5
			object.frame = 0
		end if
	end if
end event


event ObjectDraw
	// Don't draw the shield if the player is warping
	// The shield isn't just erased incase the player stops
	if warp.timer == 0
		if player[-playerCount].visible == true
			DrawSpriteFX(object.frame, FX_INK, player[-playerCount].xpos, player[-playerCount].ypos)
		end if
	end if
end event


event ObjectStartup
	if stage.playerListPos == PLAYER_SONIC_A
		LoadSpriteSheet("Global/Items2.gif")
	else
		LoadSpriteSheet("Global/Items2_t.gif")
	end if

	// Shield frames
	SpriteFrame(-24, -24, 48, 48, 109, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
	SpriteFrame(-24, -24, 48, 48, 158, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
	SpriteFrame(-24, -24, 48, 48, 207, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Items2.gif")
	SpriteFrame(-24, -24, 48, 48, 109, 196)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

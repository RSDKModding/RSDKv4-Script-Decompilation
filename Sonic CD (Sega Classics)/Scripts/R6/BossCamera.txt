// Aliases
private alias object.value0 : object.cameraHeight
private alias object.value1 : object.boundaryHeight

// Player Aliases
private alias object.ypos : player.ypos

// ========================
// Events
// ========================

event ObjectUpdate
    temp0 = object.ypos
    temp0 -= player[SLOT_PLAYER1].ypos
    temp0 >>= 25

    if temp0 != object.cameraHeight
        if object.cameraHeight < temp0
            stage.newYBoundary2 = temp0
            stage.newYBoundary2 *= -512
            stage.newYBoundary2 += object.iypos
            object.boundaryHeight = stage.newYBoundary2
        else
            stage.curYBoundary2 = temp0
            stage.curYBoundary2 *= -512
            stage.curYBoundary2 += object.iypos
            object.boundaryHeight = stage.curYBoundary2
        end if

        object.cameraHeight = temp0
    else
        if object.boundaryHeight < stage.curYBoundary2
            stage.curYBoundary2--

            if object.boundaryHeight < stage.curYBoundary2
                stage.curYBoundary2--
            end if

            if object.boundaryHeight < stage.curYBoundary2
                stage.curYBoundary2--
            end if

            if object.boundaryHeight < stage.curYBoundary2
                stage.curYBoundary2--
            end if
        end if
    end if
end event

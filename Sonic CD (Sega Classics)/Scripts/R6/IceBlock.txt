event ObjectUpdate
    switch object.propertyValue
    case CHARACTER_ICEBLOCK
        if object.state == ICEBLOCK_FALLING
            object.ypos += object.yvel
            object.yvel += player[SLOT_PLAYER1].gravityStrength
            if object.yvel > 0x140000
                object.yvel = 0x140000
            end if

            ObjectTileCollision(CSIDE_FLOOR, 0, 24, 0)
            if checkResult == true
                object.state = ICEBLOCK_BREAKICE
                object.yvel = 0
                PlaySfx(SfxName[Glass Smash], false)
            end if

            player[SLOT_PLAYER1].xpos = object.xpos
            player[SLOT_PLAYER1].ypos = object.ypos
        else
            if object.timer < 16
                object.timer++
                if player[SLOT_PLAYER1].type == TypeName[Player Object]
                    if player[SLOT_PLAYER1].jumpPress == true
                        CallFunction(IceBlock_CreateIceChunks)
                        object.type = TypeName[Blank Object]

                        arrayPos1 = stage.entityPos
                        stage.entityPos = SLOT_PLAYER1
                        CallFunction(Player_Action_Jump)
                        stage.entityPos = arrayPos1

                        player[SLOT_PLAYER1].visible = true
                        player[SLOT_PLAYER1].interaction = true
                    end if
                end if
            else
                CallFunction(IceBlock_CreateIceChunks)
                object.type = TypeName[Blank Object]

                player[SLOT_PLAYER1].state = Player_State_Air

                CallFunction(Player_Hit)

                player[SLOT_PLAYER1].visible = true
                player[SLOT_PLAYER1].interaction = true
            end if
        end if
        break

    case BIG_CHUNK_ICEBLOCK
        if object.timer < 10
            object.timer++
            object.xpos += object.xvel
            object.ypos += object.yvel
        else
            object.type = TypeName[Blank Object]

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = 0
            object[tempObjectPos].yvel = -0x10000

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = -0x10000
            object[tempObjectPos].yvel = 0

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = 0x10000
            object[tempObjectPos].yvel = 0

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = 0
            object[tempObjectPos].yvel = 0x10000
        end if
        break

    case MEDIUM_CHUNK_ICEBLOCK
        if object.timer < 16
            object.timer++
            object.xpos += object.xvel
            object.ypos += object.yvel
        else
            object.type = TypeName[Blank Object]

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = 0
            object[tempObjectPos].yvel = -0x10000

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = -0x10000
            object[tempObjectPos].yvel = 0x10000

            CreateTempObject(TypeName[Ice Block], SMALL_CHUNK_ICEBLOCK, object.xpos, object.ypos)
            object[tempObjectPos].drawOrder = 4
            object[tempObjectPos].xvel = 0x10000
            object[tempObjectPos].yvel = 0x10000
        end if
        break

    case CENTRAL_CHUNK_ICEBLOCK
        if object.timer < 1
            object.timer++
        else
            object.type = TypeName[Blank Object]
        end if
        break

    case SMALL_CHUNK_ICEBLOCK
        if object.timer < 12
            object.timer++
            object.xpos += object.xvel
            object.ypos += object.yvel
        else
            object.type = TypeName[Blank Object]
        end if
        break
    end switch
end event

    DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)

event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.propertyValue
        case CATWALK_W_TUBESWITCH
            if object.state == TILESWAPPER_OUTSIDE_CHUNK
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -64, 64, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state    = TILESWAPPER_INSIDE_CHUNK
                    object.priority = PRIORITY_ACTIVE

                    temp0 = object.xpos
                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23
                    SetTileLayerEntry(386, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(387, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(390, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(391, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(394, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(395, 0, temp0, temp1)
                end if
            else
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -64, 692, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == false
                    object.state    = TILESWAPPER_OUTSIDE_CHUNK
                    object.priority = PRIORITY_BOUNDS

                    temp0 = object.xpos
                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23
                    SetTileLayerEntry(86, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(87, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(122, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(123, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(126, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(127, 0, temp0, temp1)
                end if
            end if
            break

        case CATWALK_3_CHUNKS
            if object.state == TILESWAPPER_OUTSIDE_CHUNK
                BoxCollisionTest(C_TOUCH, object.entityPos, -48, -64, -8, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state    = TILESWAPPER_INSIDE_CHUNK
                    object.priority = PRIORITY_ACTIVE

                    temp0 = object.xpos
                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23
                    SetTileLayerEntry(390, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(391, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(394, 0, temp0, temp1)
                end if
            else
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -64, 384, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == false
                    object.state    = TILESWAPPER_OUTSIDE_CHUNK
                    object.priority = PRIORITY_BOUNDS

                    temp0 = object.xpos
                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23
                    SetTileLayerEntry(122, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(123, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(126, 0, temp0, temp1)
                end if
            end if
            break

        case CATWALK_4_CHUNKS
            if object.state == TILESWAPPER_OUTSIDE_CHUNK
                BoxCollisionTest(C_TOUCH, object.entityPos, -48, -64, -8, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state    = TILESWAPPER_INSIDE_CHUNK

                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23

                    temp0++
                    SetTileLayerEntry(390, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(391, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(394, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(395, 0, temp0, temp1)
                end if
            else
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -64, 544, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == false
                    object.state    = TILESWAPPER_OUTSIDE_CHUNK
                    object.priority = PRIORITY_BOUNDS

                    temp0 = object.xpos
                    temp0 >>= 23
                    temp1 = object.ypos
                    temp1 >>= 23
                    SetTileLayerEntry(203, 0, temp0, temp1)

                    temp0++
                    SetTileLayerEntry(123, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(126, 0, temp0, temp1)
                    temp0++
                    SetTileLayerEntry(127, 0, temp0, temp1)
                end if
            end if
            break

        case TUNNEL
            if object.state == TILESWAPPER_OUTSIDE_CHUNK

                    SetTileLayerEntry(86, 0, temp0, temp1)

                    SetTileLayerEntry(87, 0, temp0, temp1)
                end if
            end if
            break
        end switch
    next
end event

event RSDKDraw
    temp0 = object.xpos
    temp0 -= 0x80000
    temp1 = object.ypos
    temp1 -= 0x80000
    DrawSpriteXY(0, temp0, temp1)

    temp0 += 0x100000
    DrawSpriteXY(0, temp0, temp1)

    temp1 += 0x100000
    DrawSpriteXY(0, temp0, temp1)

    temp0 -= 0x100000
    DrawSpriteXY(0, temp0, temp1)
end event

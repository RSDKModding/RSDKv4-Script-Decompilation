event ObjectUpdate
    switch object.state
    case EGGMANSTATUE_DESTROYED
        temp0 = object.timer
        temp0 &= 3
        if temp0 == 0
            Rand(temp0, 32)
            temp0 -= 16
            temp0 <<= 16
            temp0 += object.xpos

            Rand(temp1, 48)
            temp1 -= 24
            temp1 <<= 16
            temp1 += object.ypos

            CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
            PlaySfx(SfxName[Explosion], false)
        end if

        object.timer++
        if object.timer == 40
            object.timer = 0
            object.state++
        end if
        break

    case EGGMANSTATUE_BOMB_DROP
        object.timer++
        if object.timer == 60
            object.timer = 0
            CallFunction(EggmanStatue_HandleBombDrop)
            object.type = TypeName[Blank Object]
        end if
        break
    end switch

    if object.state == EGGMANSTATUE_INACTIVE
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            if player[currentPlayer].animation == ANI_JUMPING
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -24, 16, 24, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
                if checkResult == true
                    object.state = EGGMANSTATUE_DESTROYED
                end if
            end if
        next
	// just casually missing an end if here yeah that's fine
end event


event ObjectDraw
    if object.state < EGGMANSTATUE_BOMB_DROP
        DrawSprite(0)
    end if
end event

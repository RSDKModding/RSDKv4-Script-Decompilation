reserve function MovingBlocks_HandleMovement

// ========================
// Function Definitions
// ========================

public function MovingBlocks_HandleMovement
    switch object.currentBlock
    case 0
        object.block1.ypos -= 0x8000
        if object.animationTimer == 64
            object.block2.ypos = object.block1.ypos
        end if
        break

    case 1
        object.block1.ypos -= 0x8000
        break

    case 2
        object.block1.ypos += 0x8000
        object.block3.ypos += 0x8000
        if object.animationTimer == 64
            object.block1.ypos = object.block3.ypos
        end if
        break

    case 3
        object.block1.ypos += 0x8000
        object.block2.ypos += 0x8000
        break

    case 4
        object.block1.ypos -= 0x8000
        object.block2.ypos -= 0x8000
        if object.animationTimer == 64
            object.block1.ypos = object.block2.ypos
        end if
        break

    case 5
        object.block1.ypos -= 0x8000
        object.block3.ypos -= 0x8000
        break
    end switch
end function


// ========================

event ObjectUpdate
    object.rotation -= 2
    if object.rotation < 0
        object.rotation += 512
    end if
    object.animationTimer++
    CallFunction(MovingBlocks_HandleMovement)
    if object.animationTimer == 64
        object.animationTimer = 0
        object.currentBlock++
        if object.currentBlock > 5
            object.currentBlock = 2
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = object.xpos
        temp1 = object.ypos
        switch object.currentBlock
        case 0
        case 1
            object.xpos = object.block2.xpos
            object.ypos = object.block2.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if
            break // ?

            object.xpos = object.block1.xpos
            object.ypos = object.block1.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if
            break

        case 2
            object.xpos = object.block2.xpos
            object.ypos = object.block2.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if

            object.xpos = object.block3.xpos
            object.ypos = object.block3.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_BOTTOM
                if player[currentPlayer].gravity == GRAVITY_GROUND
                    PlaySfx(SfxName[Hurt], false)

                    player[currentPlayer].drawOrder = 5

                    player[currentPlayer].speed = 0
                    player[currentPlayer].xvel = 0
                    player[currentPlayer].yvel = -0x68000

                    player[currentPlayer].state = Player_State_Death
                    player[currentPlayer].animation = ANI_DYING

                    player[currentPlayer].tileCollisions = false
                    player[currentPlayer].interaction = false
                    screen[0].cameraEnabled = false
                end if
            end if
            break

        case 3
        case 4
            object.xpos = object.block2.xpos
            object.ypos = object.block2.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if

            object.xpos = object.block3.xpos
            object.ypos = object.block3.ypos
            BoxCollisionTest(C_TOUCH, object.entityPos, -12, -16, 12, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

                object.xpos = object.block1.xpos
                object.ypos = object.block1.ypos

                BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == COL_BOTTOM
                    if player[currentPlayer].gravity == GRAVITY_GROUND
                        PlaySfx(SfxName[Hurt], false)

                        player[currentPlayer].drawOrder = 5

                        player[currentPlayer].speed = 0
                        player[currentPlayer].xvel = 0
                        player[currentPlayer].yvel = -0x68000

                        player[currentPlayer].state = Player_State_Death
                        player[currentPlayer].animation = ANI_DYING

                        player[currentPlayer].tileCollisions = false
                        player[currentPlayer].interaction = false
                        screen[0].cameraEnabled = false
                    end if
                end if
            else
                BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 32, 0, C_BOX, C_BOX, C_BOX, C_BOX)
            end if
            break

        case 5
            object.xpos = object.block2.xpos
            object.ypos = object.block2.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if

            object.xpos = object.block1.xpos
            object.ypos = object.block1.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += 0x10000
            end if
            break // ?

            object.xpos = object.block3.xpos
            object.ypos = object.block3.ypos
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break
        end switch

        object.xpos = temp0
        object.ypos = temp1
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
    next
end event


event ObjectDraw
    DrawSpriteXY(0, object.block1.xpos, object.block1.ypos)
    DrawSpriteXY(0, object.block2.xpos, object.block2.ypos)
    DrawSpriteXY(0, object.block3.xpos, object.block3.ypos)
    DrawSprite(0)
end event


event ObjectStartup
    LoadSpriteSheet("R6/Objects.gif")
    SpriteFrame(-16, -16, 32, 32, 173, 1) // #0 - Block

    foreach (TypeName[Moving Blocks], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].block1.xpos = object[arrayPos0].xpos
        object[arrayPos0].block1.ypos = object[arrayPos0].ypos
        object[arrayPos0].block2.xpos = object[arrayPos0].xpos
        object[arrayPos0].block2.ypos = object[arrayPos0].ypos
        object[arrayPos0].block3.xpos = object[arrayPos0].xpos
        object[arrayPos0].block3.ypos = object[arrayPos0].ypos
        object[arrayPos0].rotation = 480
    next
end event

event RSDKDraw
    DrawSprite(0)

    temp0 = object.ypos
    temp0 -= 0x100000

    temp1 = object.xpos
    temp1 -= 0x100000

    if editor.showGizmos == true
        editor.drawingOverlay = true

        temp0 -= 0x200000
        DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
        DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 0, 0)

        temp0 -= 0x200000
        DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)

        editor.drawingOverlay = false
    end if
end event

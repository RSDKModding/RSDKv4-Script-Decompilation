// Aliases
private alias object.value0 : object.timer

// States
private alias 0 : BOSSSPIKE_FALLING
private alias 1 : BOSSSPIKE_DESPAWN

// ========================
// Events
// ========================

event ObjectUpdate
    if object.state == BOSSSPIKE_FALLING
        object.ypos += 0x38000
        ObjectTileCollision(CSIDE_FLOOR, 0, 12, 0)
        if checkResult == true
            object.state = BOSSSPIKE_DESPAWN
        end if
    else
        object.timer++
        if object.timer == 64
            object.type = TypeName[Blank Object]
        end if
    end if

    if object.state == BOSSSPIKE_FALLING
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_TOUCH, object.entityPos, -4, -12, 4, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                CallFunction(Player_Hit)
            end if
        next
    end if
end event


event ObjectDraw
    temp0 = object.timer
    temp0 &= 7
    if temp0 < 4
        DrawSprite(0)
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R6/Objects2.gif")
    SpriteFrame(-4, -12, 8, 24, 228, 1) // #0 - Boss Spike
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R6/Objects2.gif")
    SpriteFrame(-4, -12, 8, 24, 228, 1) // #0 - Boss Spike

    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

public function R6Setup_PresentPastStageUpdate
    switch object.state
    case R6_SETUP_PREPARE_CHARGE
        if object.timer < 480
            object.timer++
        else
            PlaySfx(SfxName[Electricity], false)
            object.timer = 0
            object.state = R6_SETUP_WEB_CHARGE
        end if
        break

    case R6_SETUP_WEB_CHARGE
        if object.timer < 90
            object.timer++
            object.frame++
            object.frame &= 1
        else
            object.timer = 0
            object.frame = 2
            object.state = R6_SETUP_WEB_SHOCK
        end if
        break

    case R6_SETUP_WEB_SHOCK
        if object.timer < 120
            object.timer++
            object.frame++
            if object.frame == 9
                object.frame = 2
            end if
        else
            object.timer = 0
            object.frame = 0
            object.state = R6_SETUP_PREPARE_CHARGE
            object.webDirection++
            object.webDirection %= 3
        end if
        break
    end switch

    switch object.alarmTile
    case 0
        if object.alarmTimer < 15
            object.alarmTimer++
        else
            Copy16x16Tile(578, 801)
            Copy16x16Tile(677, 801)
            object.alarmTimer = 0
            object.alarmTile++
        end if
        break

    case 1
        if object.alarmTimer < 4
            object.alarmTimer++
        else
            Copy16x16Tile(578, 802)
            Copy16x16Tile(677, 802)
            object.alarmTimer = 0
            object.alarmTile++
        end if
        break

    case 2
        if object.alarmTimer < 9
            object.alarmTimer++
        else
            Copy16x16Tile(578, 803)
            Copy16x16Tile(677, 803)
            object.alarmTimer = 0
            object.alarmTile++
        end if
        break

    case 3
        if object.alarmTimer < 4
            object.alarmTimer++
        else
            Copy16x16Tile(578, 800)
            Copy16x16Tile(677, 800)
            object.alarmTimer = 0
            object.alarmTile = 0
        end if
        break
    end switch

    temp0 = object.lightTile
    temp0 &= 1
    if temp0 == 0
        temp0 = object.lightTile
        temp0 >>= 1
        temp0 += 804
        Copy16x16Tile(579, temp0)
    end if

    object.lightTile++
    object.lightTile %= 24

    temp0 = object.electricTile
    temp0 %= 5
    if temp0 == 0
        temp0 = object.electricTile
        temp0 /= 5
        temp0 += 816
        Copy16x16Tile(580, temp0)
        temp0 += 3
        Copy16x16Tile(581, temp0)
    end if

    object.electricTile++
    object.electricTile %= 15

    if object.bounceFloorAnim < 32
        object.bounceFloorAnim++
        object.bounceFloorFlag = false
    else
        if object.bounceFloorAnim < 212
            temp0 = object.bounceFloorAnim
            temp0 &= 1
            if temp0 == 0
                temp0 = object.bounceFloorTile
                temp0 += 822
                object.bounceFloorTile++
                object.bounceFloorTile %= 6
                Copy16x16Tile(543, temp0)
            end if

            object.bounceFloorAnim++
            object.bounceFloorFlag = true
        else
            object.bounceFloorAnim = 0
            object.bounceFloorFlag = false
            Copy16x16Tile(543, 822)
        end if
    end if
end function

    end if

    object.lightTile++
    object.lightTile %= 24

    object.timer++
    object.timer %= 120
    if object.timer < 31
        temp0 = object.electricTile
        temp0 %= 5
        if temp0 == 0
            temp0 = object.electricTile
            temp0 /= 5
            temp0 += 820
            Copy16x16Tile(580, temp0)
            temp0 += 3
            Copy16x16Tile(581, temp0)
        end if

        object.electricTile++
        object.electricTile %= 15
    else
        Copy16x16Tile(580, 832)

    end if

    temp0 = object.bounceFloorAnim

    end if

    object.bounceFloorAnim++
    object.bounceFloorAnim &= 511
    object.bounceFloorFlag = true
end function

    temp0 = oscillation

        RotatePalette(0, 161, 163, false)
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp2 = player[currentPlayer].xpos
        temp2 >>= 16
        temp2 += player[currentPlayer].collisionRight

        temp3 = player[currentPlayer].ypos
        temp3 >>= 16
        temp3 += player[currentPlayer].collisionBottom
        temp3 += 2

        Get16x16TileInfo(temp0, temp2, temp3, TILEINFO_ANGLEB)

        temp2 = player[currentPlayer].xpos
        temp2 >>= 16
        temp2 += player[currentPlayer].collisionLeft

        Get16x16TileInfo(temp1, temp2, temp3, TILEINFO_ANGLEB)

        temp0 |= temp1
        if temp0 == 1
            if object.bounceFloorFlag == true
                if player[currentPlayer].yvel > -1
                    player[currentPlayer].animation = ANI_JUMPING
                    player[currentPlayer].gravity = GRAVITY_AIR
                    player[currentPlayer].timer = 0
                    player[currentPlayer].yvel = -0x160000
                    player[currentPlayer].state = Player_State_Air
                    PlaySfx(SfxName[Bounce Floor], false)
                end if
            end if
        end if

        temp2 -= player[currentPlayer].collisionLeft
        temp3 -= player[currentPlayer].collisionBottom
        temp3 -= 2

        Get16x16TileInfo(temp0, temp2, temp3, TILEINFO_ANGLEB)

        switch temp0
        case 2
            object.xpos = player[currentPlayer].xpos

            if player[currentPlayer].direction == FACING_LEFT
                object.xpos += 0x80000
            else
                object.xpos -= 0x80000
            end if

            CallFunction(Player_Hit)
            break

event ObjectStartup
    object[SLOT_R6SETUP].type = TypeName[R6 Setup]
    object[SLOT_R6SETUP].priority = PRIORITY_ACTIVE
    object[SLOT_R6SETUP].drawOrder = 2

    switch HUD.currentTimePeriod
    case TIME_PRESENT
    case TIME_PAST
        object[SLOT_R6SETUP].stateStageUpdate = R6Setup_PresentPastStageUpdate
        break

    case TIME_GOOD_FUTURE
        object[SLOT_R6SETUP].stateStageUpdate = R6Setup_GoodFutureStageUpdate
        break

    case TIME_BAD_FUTURE
        if stage.actNum == 3
            object[SLOT_R6SETUP].stateStageUpdate = R6Setup_BadFutureStageUpdateBoss
        else
            object[SLOT_R6SETUP].stateStageUpdate = R6Setup_BadFutureStageUpdate
        end if
        break
    end switch
end event

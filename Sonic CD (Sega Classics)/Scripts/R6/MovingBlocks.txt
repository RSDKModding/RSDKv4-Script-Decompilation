event ObjectUpdate
    if object.state == MOVINGBLOCKS_MOVE
        if object.animationTimer < 64
            object.animationTimer++
            CallFunction(object.movePattern)
        else
            object.animationTimer = 0
            object.state = MOVINGBLOCKS_CHANGE_PATTERN
            object.blockState++
            if object.blockState > 14
                object.blockState = 3
            end if
        end if
    else
        if object.animationTimer < 30
            object.animationTimer++
        else
            object.animationTimer = 0
            object.state = MOVINGBLOCKS_MOVE
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = object.xpos
        temp1 = object.ypos

        object.xpos = object.block1.xpos
        object.ypos = object.block1.ypos
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        object.xpos = object.block2.xpos
        object.ypos = object.block2.ypos
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        object.xpos = object.block3.xpos
        object.ypos = object.block3.ypos
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        object.xpos = object.block4.xpos
        object.ypos = object.block4.ypos
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -17, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        object.xpos = temp0
        object.ypos = temp1
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
    next
end event

event ObjectStartup
    LoadSpriteSheet("R6/Objects.gif")
    SpriteFrame(-16, -16, 32, 32, 173, 1) // #0 - Block

    foreach (TypeName[Moving Blocks], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].block1.xpos = object[arrayPos0].xpos
        object[arrayPos0].block1.ypos = object[arrayPos0].ypos
        object[arrayPos0].block2.xpos = object[arrayPos0].xpos
        object[arrayPos0].block2.ypos = object[arrayPos0].ypos
        object[arrayPos0].block3.xpos = object[arrayPos0].xpos
        object[arrayPos0].block3.ypos = object[arrayPos0].ypos
        object[arrayPos0].block4.xpos = object[arrayPos0].xpos
        object[arrayPos0].block4.ypos = object[arrayPos0].ypos

        switch object[arrayPos0].propertyValue
        case FORMATION_L_AND_LINES
            object[arrayPos0].movePattern = MovingBlocks_MoveType_1
            break

        case FORMATION_TETROMINOS_LEFT
            object[arrayPos0].movePattern = MovingBlocks_MoveType_4
            break

        case FORMATION_L_AND_LINES_CCW
            object[arrayPos0].movePattern = MovingBlocks_MoveType_2
            break

        case FORMATION_TETROMINOS_RIGHT
            object[arrayPos0].movePattern = MovingBlocks_MoveType_3
            break
        end switch
    next
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // movementPattern
            checkResult = object.propertyValue
            checkResult &= 3
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // movementPattern
            object.propertyValue = editor.variableValue
            object.propertyValue &= 3
            break
        end switch
    end if
end event


event RSDKDraw
    temp0 = object.ypos
    temp0 -= 0x100000

    temp1 = object.xpos
    temp1 -= 0x100000

    switch object.propertyValue
    case FORMATION_L_AND_LINES
    case FORMATION_L_AND_LINES_CCW
        if editor.showGizmos == true
            editor.drawingOverlay = true

            temp0 -= 0x200000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 0, 0)

            temp0 -= 0x200000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            
            temp0 -= 0x200000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            
            if object.propertyValue == FORMATION_L_AND_LINES
                temp1 += 0x200000
            else
                temp1 -= 0x200000
            end if

            temp0 = object.ypos
            temp0 -= 0x100000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            
            editor.drawingOverlay = false
        end if
        break

    case FORMATION_TETROMINOS_LEFT
        if editor.showGizmos == true
            editor.drawingOverlay = true

            temp1 += 0x200000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            
            temp1 += 0x200000
            DrawArrow(object.xpos, object.ypos, temp1, object.ypos, 255, 0, 0)
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)

            temp0 -= 0x200000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)

            temp0 = object.ypos
            temp0 -= 0x100000
            DrawRectOutline(temp1, temp0, 0x20, 0x20, 255, 255, 0, 255)
            
            editor.drawingOverlay = false
        end if
        break

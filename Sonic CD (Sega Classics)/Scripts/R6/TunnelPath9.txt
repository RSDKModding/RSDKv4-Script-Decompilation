event RSDKDraw
    object.inkEffect = INK_BLEND
    DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)

    if editor.showGizmos == true
        editor.drawingOverlay = true

        CallFunction(EditorHelpers_FindTimePeriod)

        switch checkResult
        case TIME_PRESENT
        case TIME_BAD_FUTURE
            temp6 = 255
            temp7 = 0
            break

        case TIME_PAST
            temp6 = 0
            temp7 = 128
            break

        case TIME_GOOD_FUTURE
            temp6 = 128
            temp7 = 0
            break
        end switch

        temp0 = 0
        while temp0 < 4
            temp1 = object.xpos
            temp2 = object.ypos
            temp3 = object.xpos
            temp4 = object.ypos
            temp5 = false
            switch temp0
            case 0
                temp1 += 0x3800
                temp3 += 0x3800
                temp2 += 0x7D00
                temp4 += 0x2A87D00
                temp5 = true
                break

            case 1
                temp1 += 0x3800
                temp3 += 0x3C03800
                temp2 += 0x2A87D00
                temp4 += 0x2A87D00
                break
            case 2
                temp1 += 0x3C03800
                temp3 += 0x3C03800
                temp2 += 0x2A87D00
                temp4 += 0x1A87D00
                temp5 = true
                break

            case 3
                temp1 += 0x3C03800
                temp3 += 0x1903800
                temp2 += 0x1A87D00
                temp4 += 0x1A87D00
                break
            endswitch

            DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)

            // Make the arrow thicker
            if temp5 == true
                temp1 += 0x10000
                temp3 += 0x10000
                DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)
            else
                temp2 -= 0x10000
                temp4 -= 0x10000
                DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)
            end if

            temp0++
        loop

        editor.drawingOverlay = false
    end if
end event

// Aliases
private alias object.value0 : object.fade
private alias object.value1 : object.eggMobileEntityPos

// Player Aliases
private alias object.xpos : player.xpos

// States
private alias 0	: FADESCREEN_FADE_IN
private alias 1	: FADESCREEN_FADE_OUT

// Fade Music Property Values
private alias 1 : FADEMUSIC_TO_LEVEL

// ========================
// Events
// ========================

event ObjectUpdate
	if object.state == FADESCREEN_FADE_IN
		if object.fade < 320
			object.fade += 4
		else
			object.state = FADESCREEN_FADE_OUT

			arrayPos0 = object.eggMobileEntityPos
			object[arrayPos0].type = TypeName[Destroyed Mobile]
			object[arrayPos0].priority = PRIORITY_BOUNDS

			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				player[currentPlayer].xpos = object[arrayPos0].xpos
			next

			stage.curXBoundary2 = 0xF00

			arrayPos0++
			ResetObjectEntity(arrayPos0, TypeName[Eggman Escape], 0, object[arrayPos0].xpos, object[arrayPos0].ypos)
		end if
	else
		if object.fade > 0
			object.fade -= 4
		else
			object.type = TypeName[Blank Object]
			CreateTempObject(TypeName[Fade Music], FADEMUSIC_TO_LEVEL, object.xpos, object.ypos)
		end if
	end if

	SetScreenFade(224, 224, 224, object.fade)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143) // #0 - "Script" Icon
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

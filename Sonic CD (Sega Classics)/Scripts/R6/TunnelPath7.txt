event ObjectUpdate
    if object.state == TUNNELPATH_7_INACTIVE
        if object.propertyValue == 0
            foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
                if player[currentPlayer].state != Player_State_Static
                    BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                    if checkResult == true
                        player[currentPlayer].state = Player_State_Static
                        player[currentPlayer].interaction = false

                        if player[currentPlayer].blinkTimer > 0
                            player[currentPlayer].invincibleTimer = 0
                            player[currentPlayer].blinkTimer = 0
                        end if

                        player[currentPlayer].visible = false

                        object.state        = TUNNELPATH_7_ACTIVE
                        object.moveSteps    = -1
                        object.priority     = PRIORITY_ACTIVE
                        object.activePlayer = currentPlayer

                        PlaySfx(SfxName[Release], false)
                    end if
                end if
            next
        end if

event RSDKDraw
    object.inkEffect = INK_BLEND
    DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)

    if editor.showGizmos == true
        editor.drawingOverlay = true

        CallFunction(EditorHelpers_FindTimePeriod)

        switch checkResult
        case TIME_PRESENT
        case TIME_BAD_FUTURE
            temp6 = 255
            temp7 = 0
            break

        case TIME_PAST
            temp6 = 0
            temp7 = 128
            break

        case TIME_GOOD_FUTURE
            temp6 = 128
            temp7 = 0
            break
        end switch

        temp1 = object.xpos
        temp2 = object.ypos
        temp3 = object.xpos
        temp4 = object.ypos
        temp1 += 0x2B00
        temp3 += 0x1302B00
        temp2 += 0x2000
        temp4 += 0x2000
        DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)

        // Make the arrow thicker
        temp2 += 0x10000
        temp4 += 0x10000
        DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)

        editor.drawingOverlay = false
    end if
end event

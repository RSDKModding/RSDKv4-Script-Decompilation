// ========================

// Player Aliases
private alias object.state        : player.state
private alias object.xpos         : player.xpos
private alias object.ypos         : player.ypos
private alias object.yvel         : player.yvel
private alias object.animation    : player.animation
private alias object.collisionTop : player.collisionTop

// ========================
// Events
// ========================

event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -8, -2, 8, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if player[currentPlayer].yvel > -1
            if player[currentPlayer].xpos > object.xpos
                if checkResult == true
                    if player[currentPlayer].animation != ANI_HANGING
                        player[currentPlayer].state = Player_State_HangBar
                        player[currentPlayer].yvel = 0
                        player[currentPlayer].animation = ANI_HANGING

                        player[currentPlayer].ypos = player[currentPlayer].collisionTop
                        player[currentPlayer].ypos *= -0x10000
                        player[currentPlayer].ypos += object.ypos

                        PlaySfx(SfxName[Catch], false)

                        screen[0].adjustCameraY = 0
                    end if
                end if
            end if
        end if
    next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    object.inkEffect = INK_BLEND
    DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
end event


event RSDKLoad
    LoadSpriteSheet("Players/Sonic3.gif")
    SpriteFrame(0, 0, 32, 48, 1, 132) // #0 - Hanging Sonic

    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

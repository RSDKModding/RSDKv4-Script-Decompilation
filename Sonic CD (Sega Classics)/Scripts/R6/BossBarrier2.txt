private alias object.value0 : object.barrierHeight

// States
private alias 0 : BOSSBARRIER2_IDLE

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case BOSSBARRIER2_IDLE
        break

    case BOSSBARRIER2_UNUSED
        if object.barrierHeight < 0x600000
            object.barrierHeight += 0x20000
            object.ypos -= 0x20000
        end if

        temp0 = object.xpos
        temp0 -= object.xpos
        break

    case BOSSBARRIER2_OPEN
        if object.barrierHeight < 0x600000
            object.barrierHeight += 0x20000
            object.ypos -= 0x20000
        else
            object.state = BOSSBARRIER2_IDLE
        end if
        break
    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.barrierHeight < 1
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        else
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        end if
    next
end event

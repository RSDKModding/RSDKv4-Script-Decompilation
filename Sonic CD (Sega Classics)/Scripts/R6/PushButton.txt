// ========================
// Aliases

// Aliases
private alias object.propertyValue : object.pressed

// Player Aliases
private alias object.ypos            : player.ypos
private alias object.yvel            : player.yvel
private alias object.gravity         : player.gravity
private alias object.collisionBottom : player.collisionBottom

// ========================
// Function Declarations
// ========================

reserve function PushButton_DebugDraw
reserve function PushButton_DebugSpawn

// ========================
// Function Definitions
// ========================

public function PushButton_DebugDraw
	DrawSprite(0)
end function


public function PushButton_DebugSpawn
	CreateTempObject(TypeName[Push Button], false, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.pressed == false
			if player[currentPlayer].yvel >= 0
				BoxCollisionTest(C_SOLID, object.entityPos, -16, -8, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == COL_TOP
					object.pressed = true
					player[currentPlayer].ypos += 0x80000
					PlaySfx(SfxName[Switch], false)
				else
					object.pressed = false
				end if
			else
				object.pressed = false
			end if
		else
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -2, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				if player[currentPlayer].yvel > -1
					object.pressed = true

					player[currentPlayer].gravity = GRAVITY_GROUND
					player[currentPlayer].yvel = 0

					BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -2, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					player[currentPlayer].ypos = player[currentPlayer].collisionBottom
					FlipSign(player[currentPlayer].ypos)
					player[currentPlayer].ypos <<= 16
					player[currentPlayer].ypos += object.ypos
				end if
			else
				object.pressed = false
				if player[currentPlayer].yvel > -1
					player[currentPlayer].gravity = GRAVITY_GROUND
					player[currentPlayer].yvel = 0

					player[currentPlayer].ypos = player[currentPlayer].collisionBottom
					FlipSign(player[currentPlayer].ypos)
					player[currentPlayer].ypos -= 8
					player[currentPlayer].ypos <<= 16
					player[currentPlayer].ypos += object.ypos
				end if
			end if
		end if
	next
end event


event ObjectDraw
	DrawSprite(object.pressed)
end event


event ObjectStartup
	LoadSpriteSheet("R6/Objects.gif")

	SpriteFrame(-16, -8, 32, 16, 140, 1)  // #0 - Push Button
	SpriteFrame(-16, 0, 32, 8, 140, 18)   // #1 - Push Button pressed

	SetTableValue(TypeName[Push Button], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(PushButton_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(PushButton_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R6/Objects.gif")
	SpriteFrame(-16, -8, 32, 16, 140, 1)	// #0 - Push Button
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

event ObjectDraw
    switch object.propertyValue
    case 0
        DrawSprite(R6Setup.frame)

        temp0 = object.xpos
        temp1 = object.ypos
        temp1 -= 0x800000
        DrawSpriteXY(R6Setup.frame, temp0, temp1)

        temp1 += 0x1000000
        DrawSpriteXY(R6Setup.frame, temp0, temp1)

        if R6Setup.state > R6_SETUP_PREPARE_CHARGE
            switch R6Setup.direction
            case 0
                object.hWebFrame = 9
                object.hWebFrame += R6Setup.frame
                DrawSprite(object.hWebFrame)

                object.direction = FLIP_Y
                DrawSpriteXY(object.hWebFrame, object.xpos, temp1)
                break

            case 1
                object.vWebFrame = 18
                object.vWebFrame += R6Setup.frame
                DrawSprite(object.vWebFrame)

                object.direction = FLIP_X
                temp0 += 0x800000
                DrawSpriteXY(object.vWebFrame, temp0, object.ypos)
                break

            case 2
                temp0 += 0x800000

                object.dWebFrame = 27
                object.dWebFrame += R6Setup.frame
                object.direction = FLIP_X

                temp2 = object.xpos
                temp2 -= 0x800000
                DrawSprite(object.dWebFrame)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, object.xpos, object.ypos)
                DrawSpriteXY(object.dWebFrame, temp0, temp1)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, temp2, temp1)
                
				object.dWebFrame += 9
                DrawSprite(object.dWebFrame)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, object.xpos, object.ypos)
                DrawSpriteXY(object.dWebFrame, temp0, temp1)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, temp2, temp1)
                
				object.dWebFrame += 9
                DrawSprite(object.dWebFrame)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, object.xpos, object.ypos)
                DrawSpriteXY(object.dWebFrame, temp0, temp1)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, temp2, temp1)
                
				object.dWebFrame += 9
                DrawSprite(object.dWebFrame)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, object.xpos, object.ypos)
                DrawSpriteXY(object.dWebFrame, temp0, temp1)
                DrawSpriteFX(object.dWebFrame, FX_FLIP, temp2, temp1)
                break
            end switch
        end if
        break

    case 1
        DrawSprite(R6Setup.frame)
        break

    case 2
        if R6Setup.state == R6_SETUP_PREPARE_CHARGE
            R6Setup.timer = 0
        end if
    end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Type
			checkResult = object.propertyValue
			checkResult %= 3
			break
		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // Type
			object.propertyValue = editor.variableValue
			object.propertyValue %= 3
			break
		end switch
	end if
end event


event RSDKDraw
	LoadSpriteSheet("R6/Objects3.gif")

	switch object.propertyValue
	case WEB_PARENT_OBJECT
		DrawSprite(0)

		temp0 = object.xpos
		temp1 = object.ypos
		temp1 -= 0x800000
		DrawSpriteXY(0, temp0, temp1)

		temp1 += 0x1000000
		DrawSpriteXY(0, temp0, temp1)
		DrawSprite(1)

		object.direction = FLIP_Y
		DrawSpriteXY(1, object.xpos, temp1)
		break

	case WEB_CHILD_OBJECT
		DrawSprite(0)
		break

	case WEB_DELAYER
		object.inkEffect = INK_BLEND
		DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)

		RemoveSpriteSheet("R6/Objects3.gif")
		LoadSpriteSheet("Global/Items.gif")
		DrawSpriteFX(2, FX_INK, object.xpos, object.ypos)
		break
	end switch
end event

event ObjectUpdate
    if object.state == CLIMBER_ACTIVE
        if object.bottomStep == RIGHT_STEP
            if object.frame > 0
                object.ypos -= 0xC0000
                object.frame--
            else
                if object.timer < 120
                    object.timer++
                    if object.timer == 60
                        object.flameAnimation = 1
                    end if
                else
                    object.timer = 0
                    object.state = CLIMBER_DESCENDING
                    object.flameAnimation = 2
                end if
            end if
        else
            if object.frame < 4
                object.frame++
                object.ypos -= 0xC0000
            else
                if object.timer < 120
                    object.timer++
                    if object.timer == 60
                        object.flameAnimation = 1
                    end if
                else
                    object.timer = 0
                    object.state = CLIMBER_DESCENDING
                    object.flameAnimation = 2
                end if
            end if
        end if
    else
        object.ypos += 0x10000
        if object.bottomStep == LEFT_STEP
            ObjectTileCollision(CSIDE_FLOOR, -25, 32, 0)
        else
            ObjectTileCollision(CSIDE_FLOOR, 25, 32, 0)
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.frame
        case 0
            BoxCollisionTest(C_PLATFORM, object.entityPos, -56, -33, -24, -16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

                if object.bottomStep == RIGHT_STEP
                    object.bottomStep = LEFT_STEP
                    object.state = CLIMBER_ACTIVE
                    object.flameAnimation = 0
                    object.timer = 0
                end if
            end if

            BoxCollisionTest(C_PLATFORM, object.entityPos, 24, 15, 56, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                player[currentPlayer].ypos += 0x10000
                if object.bottomStep == LEFT_STEP
                    object.bottomStep = RIGHT_STEP
                    object.state = CLIMBER_ACTIVE
                    object.flameAnimation = 0
                    object.timer = 0
                end if
            end if
            break

            if checkResult == true

event ObjectDraw
    switch object.frame
    case 0
        DrawSprite(0)
        DrawSprite(3)
        DrawSpriteFX(7, FX_FLIP, object.xpos, object.ypos)
        break

    case 1
        DrawSprite(1)
        DrawSprite(4)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)
        break

    case 2
        DrawSprite(2)
        DrawSprite(5)
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        break

    case 3
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        DrawSprite(6)
        DrawSpriteFX(4, FX_FLIP, object.xpos, object.ypos)
        break

    case 4
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSprite(7)
        DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
        break
    end switch

    switch object.flameAnimation
    case 0
        temp0 = object.flameFrame
        temp0 /= 3
        temp0 += 8
        object.flameFrame++
        object.flameFrame %= 6
        break

    case 1 // leaving this animation sometimes creates a bug where the frame gets out of index, resulting in garbage sprites
        temp0 = object.flameFrame
        temp0 /= 3
        temp0 += 8
        object.flameFrame++
        object.flameFrame %= 12
        break

    case 2
        temp0 = object.flameFrame
        temp0 /= 3
        temp0 += 10
        object.flameFrame++
        object.flameFrame %= 6
        break
    end switch

    DrawSprite(temp0)
end event

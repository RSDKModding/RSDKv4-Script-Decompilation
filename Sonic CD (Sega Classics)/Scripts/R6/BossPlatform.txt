event ObjectUpdate
    object.originPos.y = object.ypos
    if object.fallSpeed > 0
        object.yvel += object.fallSpeed
        object.ypos += object.yvel

        ObjectTileCollision(CSIDE_FLOOR, -16, 16, 0)
        temp0 = checkResult

        ObjectTileCollision(CSIDE_FLOOR, 16, 16, 0)
        temp0 |= checkResult

        if temp0 == true
            PlaySfx(SfxName[Impact1], false)
            object.fallSpeed = 0
            object.yvel = 0
        end if
    else
        arrayPos0 = object.fcEntityPos
        if floorControl[arrayPos0].state == FLOORCONTROL_TILE_SWAP
            if floorControl[arrayPos0].timer > 29
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -512, 16, 512, 0, C_BOX, C_BOX, C_BOX, C_BOX)

				if checkResult == true
                    object.fallSpeed = 0xE00
                else
                    object.fallSpeed = 0xB00
                end if

                object.yvel = -0x60000
            end if
        end if
    end if

    object.collisionOffset.y = object.ypos
    object.collisionOffset.y -= object.originPos.y

    if object.intangibleTime > 0
        object.intangibleTime--
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.intangibleTime == 0
            temp0 = object.ypos
            object.ypos = object.originPos.y

            BoxCollisionTest(C_SOLID, object.entityPos, -16, -18, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_TOP
                player[currentPlayer].ypos += object.collisionOffset.y
                player[currentPlayer].ypos += 0x20000
            else
                if object.fallSpeed > 0
                    if checkResult == COL_BOTTOM
                        CallFunction(Player_Hit)
                        if player[currentPlayer].state == Player_State_GotHit
                            object.intangibleTime = 20
                        end if
                    end if
                end if

                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -32, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    if player[currentPlayer].yvel < -0x40000
                        if player[currentPlayer].jumpPress == true
                            temp1 = object.yvel
                            temp1 >>= 1
                            player[currentPlayer].yvel += temp1
                        end if
                    end if
                end if
            end if

            object.ypos = temp0
        end if
    next
end event

    end if
end event

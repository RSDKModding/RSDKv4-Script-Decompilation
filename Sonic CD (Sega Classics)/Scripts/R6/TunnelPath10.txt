event RSDKDraw
    object.inkEffect = INK_BLEND
    DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)

    if editor.showGizmos == true
        editor.drawingOverlay = true

        CallFunction(EditorHelpers_FindTimePeriod)

        switch checkResult
        case TIME_PRESENT
        case TIME_BAD_FUTURE
            temp6 = 255
            temp7 = 0
            break

        case TIME_PAST
            temp6 = 0
            temp7 = 128
            break

        case TIME_GOOD_FUTURE
            temp6 = 128
            temp7 = 0
            break
        end switch

        temp0 = 0
        while temp0 <= 10
            temp1 = object.xpos
            temp2 = object.ypos
            temp3 = object.xpos
            temp4 = object.ypos
            temp5 = false
            switch temp0
            case 0
                temp1 += 0x5200
                temp3 += 0x5200
                temp2 += 0x3500
                temp4 += 0x2B83500
                temp5 = true
                break

            case 1
                temp1 += 0x5200
                temp3 -= 0x77AE00
                temp2 += 0x2B83500
                temp4 += 0x2B83500
                break

            case 2
                temp1 -= 0x77AE00
                temp3 -= 0x77AE00
                temp2 += 0x2B83500
                temp4 += 0x3683500
                temp5 = true
                break

            case 3
                temp1 -= 0x77AE00
                temp3 -= 0x17AE00
                temp2 += 0x3683500
                temp4 += 0x3683500
                break

            case 4
                temp1 -= 0x17AE00
                temp3 -= 0xA7A600
                temp2 += 0x3683500
                temp4 += 0x3C03500
                break

            case 5
                temp1 -= 0xA7A600
                temp3 -= 0x47A600
                temp2 += 0x3B83500
                temp4 += 0x3B83500
                break
            case 6
                temp1 -= 0x47A600
                temp3 -= 0x47A600
                temp2 += 0x3B83500
                temp4 += 0x3F83500
                temp5 = true
                break

            case 7
                temp1 -= 0x47A600

                temp2 += 0x3F83500
                temp4 += 0x3F83500
                break

            case 8
                temp1 += 0x185A00

                temp2 += 0x3F83500
                temp4 += 0x4583500
                temp5 = true
                break

            case 9

                temp1 -= 0x37A600
                temp3 -= 0x37A600
                temp2 += 0x4583500
                temp4 += 0x4E83500
                temp5 = true
                break
            end switch

            DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)

            // Make the arrow thicker
            if temp5 == true

                temp3 += 0x10000

                temp2 -= 0x10000

                DrawArrow(temp1, temp2, temp3, temp4, 255, temp6, temp7)
            end if

            temp0++
        loop

        editor.drawingOverlay = false
    end if
end event

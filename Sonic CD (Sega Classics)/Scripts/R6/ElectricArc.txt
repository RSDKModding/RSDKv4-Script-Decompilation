event ObjectUpdate
    switch object.state
    case ELECTRICARC_POWERING
        object.timer++
        if object.timer == 30
            object.timer = 0
            object.state++
        end if
        break

    case ELECTRICARC_EXTEND_1
    case ELECTRICARC_EXTEND_2
    case ELECTRICARC_EXTEND_3
        object.timer++
        if object.timer == 3
            object.timer = 0
            object.state++
        end if
        break

    case ELECTRICARC_SWAP_TO_TILES
        object.timer++
        if object.timer == 3
            object.timer = 0
            object.type = TypeName[Blank Object]
            temp1 = object.xpos
            temp1 >>= 23
            temp1--
            temp2 = object.ypos
            temp2 >>= 23
            GetTileLayerEntry(temp0, 0, temp1, temp2)
            temp0 += 3
            SetTileLayerEntry(temp0, 0, temp1, temp2)
            temp1++
            GetTileLayerEntry(temp0, 0, temp1, temp2)
            temp0 += 3
            SetTileLayerEntry(temp0, 0, temp1, temp2)
            temp1++
            GetTileLayerEntry(temp0, 0, temp1, temp2)
            temp0 += 3
            SetTileLayerEntry(temp0, 0, temp1, temp2)
        end if
        break
    end switch

    object.frame++
    object.frame %= 15
end event


event ObjectDraw
    switch object.state
    case ELECTRICARC_POWERING
        temp0 = object.frame
        temp0 /= 5
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    case ELECTRICARC_EXTEND_1
        temp0 = object.frame
        temp0 /= 5
        temp0 += 3
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    case ELECTRICARC_EXTEND_2
        temp0 = object.frame
        temp0 /= 5
        temp0 += 6
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    case ELECTRICARC_EXTEND_3
        temp0 = object.frame
        temp0 /= 5
        temp0 += 9
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    case ELECTRICARC_SWAP_TO_TILES
        temp0 = object.frame
        temp0 /= 5
        temp0 += 9
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        temp0 += 3
        DrawSprite(temp0)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break
    end switch
end event

// Aliases
private alias object.value1 : object.paletteTimer
private alias object.value2 : object.whiteFlickerTimer
private alias object.value3 : object.whiteFlickerIndex1
private alias object.value4 : object.whiteFlickerIndex2

// Player Aliases
private alias object.xpos            : player.xpos
private alias object.ypos            : player.ypos
private alias object.collisionLeft   : player.collisionLeft
private alias object.collisionTop    : player.collisionTop
private alias object.collisionRight  : player.collisionRight
private alias object.collisionBottom : player.collisionBottom

// Reserved Object Slot Aliases
private alias 21 : SLOT_R8SETUP

// Tile Info
private alias 1 : TILEINFO_DIRECTION
private alias 8 : TILEINFO_ANGLEB

// ========================
// Tables
// ========================

public table R8Setup_WhiteFlickerPal
	0xF0F0F0, 0x000000, 0xF0F0F0, 0x000000, 0xF0F0F0, 0x000000, 0x000000, 0x000000
	0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000
	0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000, 0x000000
end table

// ========================
// Events
// ========================

event ObjectUpdate
	object.rotation++
	object.rotation %= 144

	object.paletteTimer++
	if object.paletteTimer == 5
		object.paletteTimer = 0
		RotatePalette(0, 166, 168, true)
	end if

	object.whiteFlickerTimer++
	if object.whiteFlickerTimer == 3
		object.whiteFlickerTimer = 0

		GetTableValue(temp0, object.whiteFlickerIndex1, R8Setup_WhiteFlickerPal)
		SetPaletteEntry(0, 177, temp0)
		object.whiteFlickerIndex1++
		object.whiteFlickerIndex1 %= 24

		GetTableValue(temp0, object.whiteFlickerIndex2, R8Setup_WhiteFlickerPal)
		SetPaletteEntry(0, 178, temp0)
		object.whiteFlickerIndex2++
		object.whiteFlickerIndex2 %= 24
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp1 = player[currentPlayer].xpos
		temp1 >>= 16
		temp1 += player[currentPlayer].collisionRight

		temp2 = player[currentPlayer].ypos
		temp2 >>= 16
		temp2 += player[currentPlayer].collisionBottom
		temp2 += 2
		Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_ANGLEB)
		if temp0 == 1
			Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_DIRECTION)
			if temp0 == 0
				player[currentPlayer].xpos += 0x10000
			else
				player[currentPlayer].xpos -= 0x10000
			end if
		else
			temp1 = player[currentPlayer].xpos
			temp1 >>= 16
			temp1 += player[currentPlayer].collisionLeft
			Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_ANGLEB)
			if temp0 == 1
				Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_DIRECTION)
				if temp0 == 0
					player[currentPlayer].xpos += 0x10000
				else
					player[currentPlayer].xpos -= 0x10000
				end if
			end if
		end if
	next
end event


event ObjectStartup
	object[SLOT_R8SETUP].type = TypeName[R8 Setup]
	object[SLOT_R8SETUP].priority = PRIORITY_ACTIVE

	object[SLOT_R8SETUP].whiteFlickerIndex1 = 5
	object[SLOT_R8SETUP].whiteFlickerIndex2 = 8
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

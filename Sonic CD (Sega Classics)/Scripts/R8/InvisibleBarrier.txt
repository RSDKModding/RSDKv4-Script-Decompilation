// ========================
// Aliases
// ========================

// Player Aliases
private alias object.xpos           : player.xpos
private alias object.collisionLeft  : player.collisionLeft
private alias object.collisionRight : player.collisionRight

// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -14, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if player[currentPlayer].xpos < object.xpos
				player[currentPlayer].xpos = player[currentPlayer].collisionLeft
				player[currentPlayer].xpos <<= 16
				player[currentPlayer].xpos += object.xpos
				player[currentPlayer].xpos -= 0x100000
			else
				player[currentPlayer].xpos = player[currentPlayer].collisionRight
				player[currentPlayer].xpos <<= 16
				player[currentPlayer].xpos += object.xpos
				player[currentPlayer].xpos += 0x100000
			end if
		end if
	next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	temp0 = object.xpos
	temp0 -= 0x100000
	temp1 = object.ypos
	temp1 -= 0xE0000
	DrawRectOutline(temp0, temp1, 0x20, 0x1C, 255, 255, 0, 255)

	object.inkEffect = INK_ALPHA
	object.alpha = 1
	DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

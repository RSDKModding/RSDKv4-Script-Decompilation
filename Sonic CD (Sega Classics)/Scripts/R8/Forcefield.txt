private alias object.propertyValue : object.scrollDistance

// States
private alias 0 : FORCEFIELD_MOVE_TO_SCREEN
private alias 1 : FORCEFIELD_IDLE
private alias 2 : FORCEFIELD_VANISH

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case FORCEFIELD_MOVE_TO_SCREEN
        if object.scrollDistance > 168
            object.scrollDistance--
        else
            object.state++
        end if

        object.frame = object.animationTimer
        object.frame >>= 2

        object.animationTimer++
        object.animationTimer &= 15

        temp1 = oscillation
        temp1 <<= 3
        Sin(temp0, temp1)
        temp0 >>= 4
        temp0 += 190

        object.alpha = temp0
        object.inkEffect = INK_ALPHA
        object.drawOrder = 4
        break

    case FORCEFIELD_IDLE
        object.frame = object.animationTimer
        object.frame >>= 2

        object.animationTimer++
        object.animationTimer &= 15

        temp1 = oscillation
        temp1 <<= 3
        Sin(temp0, temp1)
        temp0 >>= 4
        temp0 += 190

        object.alpha = temp0
        break

    case FORCEFIELD_VANISH
        object.frame = 0
        if object.alpha > 0
            object.alpha -= 2
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.alpha > 0
            temp0 = object.scrollDistance
            temp0 -= 8
            BoxCollisionTest(C_SOLID, object.entityPos, temp0, -96, 240, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            FlipSign(temp0)
            BoxCollisionTest(C_SOLID, object.entityPos, -240, -96, temp0, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        end if
    next
end event


event ObjectDraw
    temp0 = object.scrollDistance
    temp0 *= -0x10000
    temp0 += object.xpos
    DrawSpriteFX(4, FX_INK, temp0, object.ypos)
    DrawSpriteFX(5, FX_INK, temp0, object.ypos)

    object.direction = FLIP_NONE
    DrawSpriteFX(object.frame, FX_FLIP, temp0, object.ypos)
    object.direction = FLIP_Y
    DrawSpriteFX(object.frame, FX_FLIP, temp0, object.ypos)

    temp0 = object.scrollDistance
    temp0 <<= 16
    temp0 += object.xpos
    DrawSpriteFX(4, FX_INK, temp0, object.ypos)
    DrawSpriteFX(5, FX_INK, temp0, object.ypos)

    object.direction = FLIP_NONE
    DrawSpriteFX(object.frame, FX_FLIP, temp0, object.ypos)
    object.direction = FLIP_Y
    DrawSpriteFX(object.frame, FX_FLIP, temp0, object.ypos)
end event

// Player Aliases
private alias object.speed   : player.speed
private alias object.gravity : player.gravity

// ========================
// Events
// ========================

event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -64, -64, 64, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            if player[currentPlayer].gravity == GRAVITY_GROUND
                if player[currentPlayer].speed < 0x50000
                    player[currentPlayer].speed = 0x50000
                else
                    if player[currentPlayer].speed > 0x100000
                        player[currentPlayer].speed = 0x100000
                    end if
                end if
            end if
        end if
    next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	if editor.showGizmos == true
		editor.drawingOverlay = true
		temp0 = object.ixpos
		temp0 -= 64
		temp0 <<= 16
		temp1 = object.iypos
		temp1 -= 64
		temp1 <<= 16
		DrawRectOutline(temp0, temp1, 128, 128, 255, 255, 0, 255)
		editor.drawingOverlay = false
	end if

	DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("Global/Display.gif")
    SpriteFrame(-16, -16, 32, 32, 1, 143)       // #0 - "Script" Icon
    
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

public function EggMobile_StopMovement
    object.timer++
    if object.timer == helper2.stopTimer
        object.timer = 0
        helper1.controllerDisabled = false
    end if
end function


public function EggMobile_MoveToLeft
    object.xpos -= object.xvel
    if object.xpos <= helper1.leftBound
        helper1.controllerDisabled = false
    end if
end function


public function EggMobile_MoveToRight
    object.xpos += object.xvel
    if object.xpos >= helper1.rightBound
        helper1.controllerDisabled = false
    end if
end function


public function EggMobile_GroundSpinAscend
    object.ypos -= object.yvel
    if object.ypos <= helper1.targetPos
        object.ypos = helper1.targetPos
        helper1.controllerDisabled = false
    end if
end function


public function EggMobile_GroundSpinDescend
    object.ypos += object.yvel
    if object.ypos >= helper1.targetPos
        object.ypos = helper1.targetPos
        helper1.controllerDisabled = false
    end if
    if object.oscillationValue != 0
        if helper1.bossHealth < 4
            temp0 = object.oscillation
            temp0 >>= 6
            if temp0 == helper2.maxOscillation
                object.oscillationValue = 0
                object.oscillation = temp0
                object.oscillation <<= 6
            end if
        else
            temp0 = object.oscillation
            temp0 %= 0x500
            if temp0 == 0
                object.oscillationValue = 0
            end if
        end if
    end if
end function

public function EggMobile_GroundSpinLeft
    object.xpos -= 0x14CCC
    if object.xpos <= helper1.targetPos
        object.xpos = helper1.targetPos
        object.oscillationValue = 0
        helper1.controllerDisabled = false
    end if
    temp0 = object.rotation
    temp0 <<= 1
    temp0 &= 511
    Sin(temp1, temp0)
    if temp1 < 0
        FlipSign(temp1)
    end if
    temp1 >>= 7
    temp1 += 64
    ObjectTileGrip(CSIDE_FLOOR, 0, temp1, 0)
end function

public function EggMobile_HandleWingThrowATK
    switch helper2.secondaryState
    case HELPER2_PREPARE_DETACH
        object.timer++
        if object.timer == 20
            object.timer = 0
            helper2.secondaryState++
        end if
        break

    case HELPER2_CENTER_WINGS
        if object.oscillationValue != 0
            temp0 = object.oscillation
            temp0 >>= 7
            if temp0 == 0
                temp0 <<= 7
                object.oscillation = temp0
                object.oscillationValue = 0
                helper2.secondaryState++
            end if
        end if
        break

    case HELPER2_DETACH_WINGS
        object.timer++
        object.wingOffset += 128

        if object.timer == 22
            bossWing[-4].frame = 1
            bossWing[-3].frame = 1
            bossWing[-2].frame = 1
            bossWing[-1].frame = 1
        end if

        if object.timer == 26
            bossWing[-4].frame = 2
            bossWing[-3].frame = 2
            bossWing[-2].frame = 2
            bossWing[-1].frame = 2
        end if

        if object.timer == 36
            object.timer = 0
            helper2.secondaryState++
        end if
        break

    case HELPER2_PREPARE_THROW
        object.timer++
        if object.timer == 10
            object.timer = 0
            helper2.secondaryState++
            object.oscillationValue = -64
            helper1.targetPos = object[+3].ypos
            helper1.targetPos -= 0x440000
        end if
        break

    case HELPER2_THROW_WINGS
        temp0 = 0
        arrayPos0 = object.entityPos
        arrayPos0 -= 4
        while temp0 < 4
            if bossWing[arrayPos0].attached == true
                temp0 = bossWing[arrayPos0].rotation
                temp0 >>= 4
                if temp0 == 0
                    bossWing[arrayPos0].rotation = 0
                    bossWing[arrayPos0].attached = 0
                    bossWing[arrayPos0].state = BOSSWING_DETACH
                    bossWing[arrayPos0].yvel = 0x28000
                end if
            end if
            arrayPos0++
            temp0++
        loop

        if object.ypos <= helper1.targetPos
            object.ypos = helper1.targetPos
            object.timer++
            if object.timer == 60
                object.timer = 0
                object.oscillationValue = 0
                object.oscillation = 0
                helper1.controllerDisabled = false
            end if
        else
            object.ypos -= 0x14000
        end if
        break
    end switch
end function


public function EggMobile_WingThrowVertical
    if helper2.animSpeedState < helper1.bossHealth
        if object.timer == 0
            arrayPos0 = object.entityPos
            arrayPos0 -= 4
            arrayPos0 += helper2.animSpeedState

            bossWing[arrayPos0].type = TypeName[Boss Wing]
            bossWing[arrayPos0].state = BOSSWING_FALL_VERTICAL
            bossWing[arrayPos0].rotation = 0
            bossWing[arrayPos0].timer = 0

            if helper1.bossHealth > 2
                bossWing[arrayPos0].frame = 4
            else
                bossWing[arrayPos0].frame = 3
            end if

            bossWing[arrayPos0].ypos = object[+3].ypos
            bossWing[arrayPos0].ypos -= 0x800000
            bossWing[arrayPos0].xvel = 0
            bossWing[arrayPos0].yvel = 0x38000

            switch helper2.wingSpawnXPos
            case 0
                bossWing[arrayPos0].xpos = -0x400000
                break

            case 1
                bossWing[arrayPos0].xpos = 0x400000
                break

            case 2
                bossWing[arrayPos0].xpos = 0x800000
                break

            case 3
                bossWing[arrayPos0].xpos = -0x800000
                break

            case 4
                bossWing[arrayPos0].xpos = -0x800000
                break

            case 5
                bossWing[arrayPos0].xpos = -0x400000
                break

            case 6
                bossWing[arrayPos0].xpos = 0
                break

            case 7
                bossWing[arrayPos0].xpos = 0x400000
                break

            case 8
                bossWing[arrayPos0].xpos = 0x800000
                break

            case 9
                bossWing[arrayPos0].xpos = 0x400000
                break

            case 10
                bossWing[arrayPos0].xpos = 0
                break

            case 11
                bossWing[arrayPos0].xpos = -0x800000
                break

            case 12
                bossWing[arrayPos0].xpos = -0x500000
                break

            case 13
                bossWing[arrayPos0].xpos = 0x500000
                break

            case 14
                bossWing[arrayPos0].xpos = -0x300000
                break

            case 15
                bossWing[arrayPos0].xpos = 0xA0000
                break

            end switch

            bossWing[arrayPos0].xpos += object[+3].xpos
            object.timer++
        else
            if object.timer < 30
                object.timer++
            else
                object.timer = 0
                helper2.animSpeedState++
                helper2.wingSpawnXPos++
            end if
        end if
    else
        if bossWing[-4].type == TypeName[Blank Object]
            if bossWing[-3].type == TypeName[Blank Object]
                if bossWing[-2].type == TypeName[Blank Object]
                    if bossWing[-1].type == TypeName[Blank Object]
                        helper1.controllerDisabled = false
                    end if
                end if
            end if
        end if
    end if
end function


public function EggMobile_WingThrowDiagonal
    if helper2.animSpeedState < 1
        bossWing[-1].type = TypeName[Boss Wing]
        bossWing[-1].state = BOSSWING_FALL_VERTICAL
        bossWing[-1].timer = 0
        bossWing[-1].frame = 3

        if player[SLOT_PLAYER1].xpos < object[+3].xpos
            bossWing[-1].ypos = object[+3].ixpos
            bossWing[-1].ypos += 248

            temp0 = player[SLOT_PLAYER1].xpos
            temp0 >>= 16
            bossWing[-1].ypos -= temp0
            bossWing[-1].ypos *= -0x6A0A
            bossWing[-1].ypos >>= 16
            bossWing[-1].ypos &= 0xFFFF
            bossWing[-1].ypos <<= 16
            bossWing[-1].ypos += object[+3].ypos
            bossWing[-1].ypos += 0x600000
            bossWing[-1].xpos = object[+3].xpos
            bossWing[-1].xpos += 0xC80000

            bossWing[-1].rotation = 96
            bossWing[-1].xvel = -0x33BBA
            bossWing[-1].yvel = 0x156CF
        else
            object[-1].ypos = player[SLOT_PLAYER1].xpos
            object[-1].ypos >>= 16

            temp0 = object[+3].ixpos
            temp0 -= 248
            object[-1].ypos -= temp0
            object[-1].ypos *= -0x6A0A
            object[-1].ypos >>= 16
            object[-1].ypos &= 0xFFFF
            object[-1].ypos <<= 16
            object[-1].ypos += object[+3].ypos
            object[-1].ypos += 0x600000
            object[-1].xpos = object[+3].xpos
            object[-1].xpos -= 0xC80000

            object[-1].rotation = 416
            object[-1].xvel = 0x33BBA
            object[-1].yvel = 0x156CF
        end if

        helper2.animSpeedState++
    else
        if bossWing[-4].type == TypeName[Blank Object]
            if bossWing[-3].type == TypeName[Blank Object]
                if bossWing[-2].type == TypeName[Blank Object]
                    if bossWing[-1].type == TypeName[Blank Object]
                        helper1.controllerDisabled = false
                    end if
                end if
            end if
        end if
    end if
end function

public function EggMobile_SetTargetDestiny
    object.xvel = player[SLOT_PLAYER1].xpos
    object.xvel -= object.xpos
    object.xvel >>= 16

    temp0 = object.xvel
    if temp0 < 0
        FlipSign(temp0)
    end if

    if temp0 < 64
        object.xvel <<= 4
        object.xvel /= 90
        object.xvel <<= 12
        object.yvel = object[+3].ypos
        object.yvel -= 0x40000

        helper1.targetPos = object.yvel

        object.yvel -= object.ypos
        object.yvel >>= 16
        object.yvel <<= 4
        object.yvel /= 90
        object.yvel <<= 12

        if helper1.bossHealth == 2
            helper2.maxOscillation = 0x1180
        else
            helper2.maxOscillation = 0xF00
        end if
    else
        if object.xvel >= 0
            object.xvel -= 16
        else
            object.xvel += 16
        end if

        object.xvel <<= 4
        object.xvel /= 90
        object.xvel <<= 12
        object.yvel = object[+3].ypos
        object.yvel += 0xA0000

        helper1.targetPos = object.yvel

        object.yvel -= object.ypos
        object.yvel >>= 16
        object.yvel <<= 4
        object.yvel /= 90
        object.yvel <<= 12

        if helper1.bossHealth == 2
            helper2.maxOscillation = 0x1180
        else
            helper2.maxOscillation = 0xC80
        end if
    end if
end function

end function

public function Helper1Controller_FourWings
    switch helper1.controllerState
    case 0
        object.state = EggMobile_StopMovement
        helper2.stopTimer = 120
        helper2.bossInvincible = true
        break

    case 1
        object.state = EggMobile_MoveToRight
        object.oscillationValue = -64
        object.xvel = 0x16000
        helper2.bossInvincible = true
        break

    case 2
        object.state = EggMobile_MoveToLeft
        break

    case 3
        object.state = EggMobile_MoveRightToCenter
        break

    case 4
        object.state = EggMobile_GroundSpinDescend
        object.yvel = 0xA000
        helper1.targetPos = object[+3].ypos
        helper1.targetPos += 0x200000
        helper2.bossInvincible = false
        break

    case 5
        object.state = EggMobile_StopMovement
        helper2.stopTimer = 20
        break

    case 6
        if player[SLOT_PLAYER1].xpos > object.xpos
            object.direction = FACING_LEFT
            object.state = EggMobile_GroundSpinRight
            helper1.targetPos = helper1.centerBound
            helper1.targetPos += 0x680000
            object.oscillationValue = 16
        else

    temp0 = 0
    temp1 = object.rotation
    while temp0 < 4
        if bossWing[arrayPos0].attached == true
            bossWing[arrayPos0].rotation = temp1

            Sin(bossWing[arrayPos0].xpos, temp1)
            bossWing[arrayPos0].xpos *= object.wingOffset
            bossWing[arrayPos0].xpos &= -0x10000 // Truncate the value
            bossWing[arrayPos0].xpos += object.xpos

            Cos(bossWing[arrayPos0].ypos, temp1)
            temp2 = object.wingOffset
            FlipSign(temp2)
            bossWing[arrayPos0].ypos *= temp2
            bossWing[arrayPos0].ypos &= -0x10000 // Truncate the value
            bossWing[arrayPos0].ypos += object.ypos
        end if

        temp1 += 128
        temp1 &= 511
        arrayPos0++
        temp0++
    loop
end function

event ObjectUpdate
    object.oscillation += object.oscillationValue
    if object.oscillation < 0
        object.oscillation += 0x1400
    end if
    object.oscillation %= 0x1400

    object.rotation = object.oscillation
    object.rotation /= 10

    if helper1.controllerDisabled == false
        CallFunction(helper1.controller)
    end if

    CallFunction(object.state)
    CallFunction(EggMobile_HandleWingRotation)

    if object.invulnerableTimer > 0
        if object.invulnerableTimer > 20
            temp0 = object.invulnerableTimer
            temp0 &= 15
            if temp0 == 0
                Rand(temp0, 48)
                temp0 -= 24
                temp0 <<= 16
                temp0 += object.xpos

                Rand(temp1, 48)
                temp1 -= 32
                temp1 <<= 16
                temp1 += object.ypos

                CreateTempObject(TypeName[Boss Spark], 0, temp0, temp1)
                object[tempObjectPos].inkEffect = INK_NONE
                object[tempObjectPos].alpha = 248

                Rand(temp0, 48)
                temp0 -= 24
                temp0 <<= 16
                temp0 += object.xpos

                Rand(temp1, 48)
                temp1 -= 32
                temp1 <<= 16
                temp1 += object.ypos

                CreateTempObject(TypeName[Boss Spark], 0, temp0, temp1)
                object[tempObjectPos].inkEffect = INK_NONE
                object[tempObjectPos].alpha = 248
            end if
        end if

        object.invulnerableTimer--
    end if

    if object.oscillationValue != 0
        switch helper2.sfxState
        case 0
            helper2.sfxTimer += 6
            if helper2.sfxTimer == 42
                PlaySfx(SfxName[Wing Rotate], false)
            end if

            if helper2.sfxTimer > 59
                helper2.sfxTimer -= 60
                PlaySfx(SfxName[Wing Rotate], false)
            end if
            break

        case 1
            helper2.sfxTimer += 3
            if helper2.sfxTimer > 59
                helper2.sfxTimer -= 60
                if helper2.sfxUsed == false
                    PlaySfx(SfxName[Wing Rotate], false)
                    helper2.sfxUsed = true
                else
                    PlaySfx(SfxName[Mech Step], false)
                    helper2.sfxUsed = false
                end if
            end if
            break

        case 2
            helper2.sfxTimer += 8
            if helper2.sfxTimer > 59
                helper2.sfxTimer -= 60
                PlaySfx(SfxName[Wing Rotate], false)
            end if
            break

        case 3
            helper2.sfxTimer += 10
            if helper2.sfxTimer > 59
                helper2.sfxTimer -= 60
                PlaySfx(SfxName[Wing Rotate], false)
            end if
            break

        case 4
            helper2.sfxTimer += 16
            if helper2.sfxTimer > 59
                helper2.sfxTimer -= 60
                PlaySfx(SfxName[Wing Rotate], false)
            end if
            break

        end switch
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if helper2.bossInvincible == false
            if object.invulnerableTimer == 0
                BoxCollisionTest(C_TOUCH, object.entityPos, -24, -24, 24, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    if player[currentPlayer].animation != ANI_JUMPING
                        CallFunction(Player_Hit)
                        if player[currentPlayer].state == Player_State_GotHit
                            object.animation = EGGANI_LAUGH
                        end if
                    else
                        object.oscillation >>= 5
                        object.oscillation <<= 5
                        FlipSign(player[currentPlayer].speed)
                        FlipSign(player[currentPlayer].xvel)
                        FlipSign(player[currentPlayer].yvel)

                        temp0 = 0
                        arrayPos0 = object.entityPos
                        arrayPos0--
                        temp1 = 0x10000
                        while temp0 < 4
                            if bossWing[arrayPos0].attached == true
                                CallFunction(EggMobile_DropDamagedWing)
                                if temp6 < temp1
                                    arrayPos1 = arrayPos0
                                    temp1 = temp6
                                end if
                            end if
                            arrayPos0--
                            temp0++
                        loop

                        if temp1 < 0x10000
                            bossWing[arrayPos1].attached = false
                            bossWing[arrayPos1].state = BOSSWING_FALL_DIAGONAL
                            bossWing[arrayPos1].priority = PRIORITY_ACTIVE

                            if bossWing[arrayPos1].xpos > object.xpos
                                bossWing[arrayPos1].xvel = 0x10000
                            else
                                bossWing[arrayPos1].xvel = -0x10000
                            end if

                            bossWing[arrayPos1].yvel = -0x30000
                        end if

                        helper1.bossHealth--
                        switch helper1.bossHealth
                        case 0
                            stage.timeEnabled = false
                            player.score += 1000
                            helper2.bossInvincible = true
                            object.animation = EGGANI_DEFEAT
                            object.timer = 0

                            if HUD.currentTimePeriod == TIME_BAD_FUTURE
                                object.state = EggMobile_Destroyed
                            else
                                // Before Final Fever was cut, this would have set the state to EggMobile_Destroyed_FinalFever instead
                                // Now there's a redundant time period check, what a shame
                                object.state = EggMobile_Destroyed
                            end if

                            object[+3].state++
                            object[+3].alpha >>= 1
                            object[+3].alpha <<= 1
                            helper2.secondaryState = 0
                            object.animationTimer = 0
                            helper1.controllerDisabled = true
                            helper2.sfxState = 5
                            break

                        case 1
                            object.invulnerableTimer = 120
                            helper1.controllerState = 0
                            helper1.controllerDisabled = true
                            object.state = EggMobile_Hit
                            object.yvel = 0x10000
                            helper1.controller = Helper1Controller_OneWing
                            object.timer = 0
                            object.animation = EGGANI_HURT
                            helper2.sfxState = 4
                            PlaySfx(SfxName[Boss Hit], false)
                            break

                        case 2
                            object.invulnerableTimer = 120
                            helper1.controllerState = 0
                            helper1.controllerDisabled = true
                            object.state = EggMobile_Hit
                            object.yvel = 0xE000
                            helper1.controller = Helper1Controller_TwoWings
                            object.timer = 0
                            object.animation = EGGANI_HURT
                            helper2.sfxState = 3

                            object.animation = EGGANI_HURT
                            helper2.sfxState = 2
                            PlaySfx(SfxName[Boss Hit], false)
                            break

                        end switch
                    end if
                end if
            end if
        end if

        temp0 = player[currentPlayer].collisionRight
        temp0 <<= 16
        temp0 += player[currentPlayer].xpos
        temp1 = helper2.cutsceneRightBound
        temp1 <<= 16
        if temp0 > temp1
            player[currentPlayer].xvel = 0
            player[currentPlayer].speed = 0
            player[currentPlayer].xpos = temp1
            temp0 = player[currentPlayer].collisionRight
            temp0 <<= 16
            player[currentPlayer].xpos -= temp0
        end if
    next
end event


event ObjectDraw
    temp0 = object.invulnerableTimer
    temp0 &= 1
    if temp0 == 0
        switch object.animation
        case EGGANI_IDLE
            DrawSprite(0)
            CallFunction(EggMobile_HandleAnimations)

            if object.oscillationValue != 0
                temp0 = oscillation
                temp0 &= 7
                temp0 >>= 2
                temp0 += 2
                DrawSprite(temp0)
            end if

            DrawSprite(6)
            DrawSprite(10)
            break

        case EGGANI_HURT
            DrawSprite(0)
            CallFunction(EggMobile_HandleAnimations)

            if object.oscillationValue != 0
                temp0 = oscillation
                temp0 &= 7
                temp0 >>= 2
                temp0 += 2
                DrawSprite(temp0)
            end if

            DrawSprite(9)
            DrawSprite(10)

            if object.animationTimer < 120
                object.animationTimer++
            else
                object.animationTimer = 0
                object.animation = 0
            end if
            break

        case EGGANI_LAUGH
            DrawSprite(0)
            CallFunction(EggMobile_HandleAnimations)

            if object.oscillationValue != 0
                temp0 = oscillation
                temp0 &= 7
                temp0 >>= 2
                temp0 += 2
                DrawSprite(temp0)
            end if

            temp0 = object.animationTimer
            temp0 &= 15
            temp0 >>= 3
            temp0 += 6
            DrawSprite(temp0)

            if object.animationTimer < 120
                object.animationTimer++
            else
                object.animationTimer = 0
                object.animation = 0
            end if

            DrawSprite(10)
            break

        case EGGANI_DEFEAT
            DrawSprite(1)
            DrawSprite(8)
            break

        end switch
    end if
end event

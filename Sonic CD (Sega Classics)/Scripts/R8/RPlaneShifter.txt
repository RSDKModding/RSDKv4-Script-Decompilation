event ObjectUpdate
    if player[SLOT_PLAYER1].interaction == true
        if object.propertyValue == GIMMICK_PARENT
            switch object.state
            case RPLANESHIFTER_PLANE_A_TO_B_R
                if object.animFrameTimer < 256
                    object.animFrameTimer += 8
                else
                    object.state = RPLANESHIFTER_IDLE

                    player[SLOT_PLAYER1].state = Player_State_Ground
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].drawOrder = 2
                    player[SLOT_PLAYER1].collisionPlane = COLLISION_PLANE_B
                end if

                temp0 = object.animFrameTimer
                temp0 >>= 6
                switch temp0
                case 0
                    player[SLOT_PLAYER1].frame = 1
                    break

                case 1
                    player[SLOT_PLAYER1].frame = 2
                    break

                case 2
                    player[SLOT_PLAYER1].frame = 3
                    break

                case 3
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].direction = FACING_LEFT
                    break

                end switch

                object.animationTimer++
                object.animationTimer %= 6
                break

            case RPLANESHIFTER_PLANE_B_TO_A_R
                if object.animFrameTimer > 0
                    object.animFrameTimer -= 8
                else
                    object.state = RPLANESHIFTER_IDLE

                    player[SLOT_PLAYER1].state = Player_State_Ground
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].drawOrder = 4
                    player[SLOT_PLAYER1].collisionPlane = COLLISION_PLANE_A
                end if

                temp0 = object.animFrameTimer
                temp0 >>= 6
                switch temp0
                case 0
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].direction = FACING_LEFT
                    break

                case 1
                    player[SLOT_PLAYER1].frame = 4
                    break

                case 2
                    player[SLOT_PLAYER1].frame = 0
                    break

                case 3
                    player[SLOT_PLAYER1].frame = 1
                    break
                end switch

                object.animationTimer++
                object.animationTimer %= 6
                break

            case RPLANESHIFTER_PLANE_A_TO_B_L
                if object.animFrameTimer > 256
                    object.animFrameTimer -= 8
                else
                    object.state = RPLANESHIFTER_IDLE

                    player[SLOT_PLAYER1].state = Player_State_Ground
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].drawOrder = 2
                    player[SLOT_PLAYER1].collisionPlane = COLLISION_PLANE_B
                end if

                temp0 = 512
                temp0 -= object.animFrameTimer
                temp0 >>= 6
                switch temp0
                case 0
                    player[SLOT_PLAYER1].frame = 1
                    break

                case 1
                    player[SLOT_PLAYER1].frame = 2
                    break

                case 2
                    player[SLOT_PLAYER1].frame = 3
                    break

                case 3
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].direction = FACING_RIGHT
                    break

                end switch

                object.animationTimer++
                object.animationTimer %= 6
                break

            case RPLANESHIFTER_PLANE_B_TO_A_L
                if object.animFrameTimer < 512
                    object.animFrameTimer += 8
                else
                    object.animFrameTimer = 0
                    object.state = RPLANESHIFTER_IDLE

                    player[SLOT_PLAYER1].state = Player_State_Ground
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].drawOrder = 4
                    player[SLOT_PLAYER1].collisionPlane = COLLISION_PLANE_A
                end if

                temp0 = 512
                temp0 -= object.animFrameTimer
                temp0 >>= 6
                switch temp0
                case 0
                    player[SLOT_PLAYER1].frame = 0
                    player[SLOT_PLAYER1].direction = FACING_RIGHT
                    break

                case 1
                    player[SLOT_PLAYER1].frame = 4
                    break

                case 2
                    player[SLOT_PLAYER1].frame = 0
                    break

                case 3
                    player[SLOT_PLAYER1].frame = 1
                    break

                end switch

                object.animationTimer++
                object.animationTimer %= 6
                break

            end switch

            // First Bar
            Sin(object[+1].xpos, object.animFrameTimer)
            object[+1].xpos *= 0xC00
            object[+1].xpos += object.xpos
            object[+1].ypos = object.ypos

            if object.animFrameTimer > 128
                if object.animFrameTimer < 384
                    object[+1].drawOrder = 1
                else
                    object[+1].drawOrder = 4
                end if
            else
                object[+1].drawOrder = 4
            end if

            // Second Bar
            temp0 = object.animFrameTimer
            temp0 += 170
            temp0 &= 511
            Sin(object[+2].xpos, temp0)
            object[+2].xpos *= 0xC00
            object[+2].xpos += object.xpos
            object[+2].ypos = object.ypos

            if temp0 > 128
                if temp0 < 384
                    object[+2].drawOrder = 1
                else
                    object[+2].drawOrder = 4
                end if
            else
                object[+2].drawOrder = 4
            end if

            // Third Bar
            temp0 = object.animFrameTimer
            temp0 += 341
            temp0 &= 511
            Sin(object[+3].xpos, temp0)
            object[+3].xpos *= 0xC00
            object[+3].xpos += object.xpos
            object[+3].ypos = object.ypos

            if temp0 > 128
                if temp0 < 384
                    object[+3].drawOrder = 1
                else
                    object[+3].drawOrder = 4
                end if
            else
                object[+3].drawOrder = 4
            end if
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.propertyValue == GIMMICK_PARENT
            BoxCollisionTest(C_PLATFORM, object.entityPos, -40, -8, 40, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            temp1 = checkResult
            BoxCollisionTest(C_SOLID, object.entityPos, -40, -184, 40, -168, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            BoxCollisionTest(C_TOUCH, object.entityPos, -4, -48, 4, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if object.stood == false
                    if temp1 == true
                        object.stood = true

                        player[currentPlayer].state = Player_State_Static
                        player[currentPlayer].animation = ANI_STOPPED
                        player[currentPlayer].xpos = object.xpos
                        player[currentPlayer].speed = 0
                        player[currentPlayer].xvel = 0
                        player[currentPlayer].yvel = 0
                        player[currentPlayer].scrollTracking = false

                        if player[currentPlayer].direction == FACING_RIGHT
                            if player[currentPlayer].collisionPlane == COLLISION_PLANE_A
                                object.state = RPLANESHIFTER_PLANE_A_TO_B_R
                                object.animFrameTimer = 0
                            else
                                object.state = RPLANESHIFTER_PLANE_B_TO_A_R
                                object.animFrameTimer = 256
                            end if
                        else
                            if player[currentPlayer].collisionPlane == COLLISION_PLANE_A
                                object.state = RPLANESHIFTER_PLANE_A_TO_B_L
                                object.animFrameTimer = 512
                            else
                                object.state = RPLANESHIFTER_PLANE_B_TO_A_L
                                object.animFrameTimer = 256
                            end if
                        end if
                    end if
                end if
            else
                object.stood = false
            end if
        end if
    next
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // type
            checkResult = object.propertyValue
            checkResult &= 1
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // type
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        end switch
    end if
end event

    if object.propertyValue == GIMMICK_PARENT
        temp0 = object.animationTimer
        temp0 /= 3
        DrawSprite(temp0)
        temp0 += 2
        DrawSprite(temp0)
    else
        arrayPos0 = object.entityPos
        temp0 = 0
        temp1 = 0
        while temp1 == false
            if temp0 == 3
                temp1 = true
            end if
            arrayPos0--
            temp0++

            CheckEqual(object[arrayPos0].type, TypeName[R Plane Shifter])
            temp1 = checkResult

            CheckNotEqual(object[arrayPos0].propertyValue, GIMMICK_CHILD)
            temp1 &= checkResult
        loop

        arrayPos1 = object.entityPos
        arrayPos1 -= object[ArrayPos0].entityPos
        switch arrayPos1
        case 0
            temp0  = 0
            break

        case 1
            temp0  = 0
            temp0 += 170
            temp0 &= 511
            break

        case 2
            temp0  = 0
            temp0 += 341
            temp0 &= 511
            break

        end switch

        Sin(temp2, temp0)
        temp2 *= 0xC00
        temp2 += object[arrayPos0].xpos
        DrawSpriteXY(4, temp2, object.ypos)
        DrawSpriteXY(5, temp2, object.ypos)
        DrawSpriteXY(6, temp2, object.ypos)
        DrawSpriteXY(7, temp2, object.ypos)
        DrawSpriteXY(8, temp2, object.ypos)
    end if

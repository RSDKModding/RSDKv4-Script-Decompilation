public table R8Setup_GreenFadePal
    0x000000, 0x002000, 0x006000, 0x00A000, 0x00E000, 0x00E000, 0x00A000, 0x006000, 0x002000, 0x000000

public table R8Setup_RedFadePal

end table

event ObjectUpdate
    object.rotation++
    object.rotation %= 144

    temp0 = object.vGearFrame
    temp0 %= 7
    if temp0 == 0
        temp0 = object.vGearFrame
        temp0 /= 7
        temp0 <<= 1
        temp0 += 600
        Copy16x16Tile(490, temp0)
        temp0++
        Copy16x16Tile(491, temp0)
    end if

    object.vGearFrame++
    object.vGearFrame %= 28

    temp0 = object.hCylinderFrame
    temp0 &= 3
    if temp0 == 0
        temp0 = object.hCylinderFrame
        temp0 >>= 2
        temp0 <<= 1
        temp0 += 608
        Copy16x16Tile(492, temp0)
        temp0++
        Copy16x16Tile(493, temp0)
    end if

    object.hCylinderFrame++
    object.hCylinderFrame &= 15

    temp0 = object.vCylinderFrame
    temp0 %= 5
    if temp0 == 0
        temp0 = object.vCylinderFrame
        temp0 /= 5
        temp0 <<= 2
        temp0 += 616
        Copy16x16Tile(494, temp0)
        temp0++
        Copy16x16Tile(495, temp0)
        temp0++
        Copy16x16Tile(496, temp0)
        temp0++
        Copy16x16Tile(497, temp0)
    end if

    object.vCylinderFrame++
    object.vCylinderFrame %= 20

    temp0 = object.lightsFrame
    temp0 &= 7
    if temp0 == 0
        temp0 = object.lightsFrame
        temp0 >>= 3
        temp0 += 632
        Copy16x16Tile(498, temp0)
    end if

    object.lightsFrame++
    object.lightsFrame %= 56

    object.paletteTimer++
    if object.paletteTimer == 5
        object.paletteTimer = 0
        object.currentPalette++
        object.currentPalette %= 6
        temp0 = object.currentPalette
        temp0 += 639
        Copy16x16Tile(499, temp0)
        RotatePalette(0, 166, 168, true)
    end if

    object.redGreenFadeTimer++
    if object.redGreenFadeTimer == 11
        object.redGreenFadeTimer = 0
        
        GetTableValue(temp0, object.redGreenFadeIndex, R8Setup_GreenFadePal)
        SetPaletteEntry(0, 177, temp0)

        GetTableValue(temp0, object.redGreenFadeIndex, R8Setup_RedFadePal)
        SetPaletteEntry(0, 178, temp0)

        object.redGreenFadeIndex++
        object.redGreenFadeIndex %= 10
    end if

    object.redYellowFlickerTimer++
    if object.redYellowFlickerTimer == 6
        object.redYellowFlickerTimer = 0

        GetTableValue(temp0, object.redYellowFlickerIndex, R8Setup_RedYellowFlickerPal)
        SetPaletteEntry(0, 179, temp0)
        object.redYellowFlickerIndex++
        object.redYellowFlickerIndex %= 3
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp1 = player[currentPlayer].xpos
        temp1 >>= 16
        temp1 += player[currentPlayer].collisionRight

        temp2 = player[currentPlayer].ypos
        temp2 >>= 16
        temp2 += player[currentPlayer].collisionBottom
        temp2 += 2
        Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_ANGLEB)
        if temp0 == 1
            Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_DIRECTION)
            if temp0 == 0
                player[currentPlayer].xpos += 0x10000
            else
                player[currentPlayer].xpos -= 0x10000
            end if
        else
            temp1 = player[currentPlayer].xpos
            temp1 >>= 16
            temp1 += player[currentPlayer].collisionLeft
            Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_ANGLEB)
            if temp0 == 1
                Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_DIRECTION)
                if temp0 == 0
                    player[currentPlayer].xpos += 0x10000
                else
                    player[currentPlayer].xpos -= 0x10000
                end if
            end if
        end if
    next
end event

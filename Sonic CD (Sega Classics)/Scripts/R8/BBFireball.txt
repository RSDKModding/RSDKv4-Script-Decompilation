// ========================

// Player Aliases
private alias object.state          : player.state
private alias object.collisionPlane : player.collisionPlane

// Collision Plane
private alias 0 : COLLISION_PLANE_A

// ========================
// Events
// ========================

event ObjectUpdate
	object.xpos += object.xvel
	object.ypos += object.yvel
	object.yvel += 0x2000

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if player[currentPlayer].collisionPlane == COLLISION_PLANE_A
			if player[currentPlayer].state != Player_State_Static
				BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					CallFunction(Player_Hit)
				end if
			end if
		end if
	next

	if object.outOfBounds == true
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.animationTimer
	temp0 >>= 1
	DrawSprite(temp0)

	object.animationTimer++
	object.animationTimer &= 3
end event


event ObjectStartup
	LoadSpriteSheet("R8/Objects2.gif")

	SpriteFrame(-8, -8, 16, 16, 109, 110) // #0 - Big Bomb Fireball yellow frame
	SpriteFrame(-8, -8, 16, 16, 126, 110) // #1 - Big Bomb Fireball red frame
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R8/Objects2.gif")
	SpriteFrame(-8, -8, 16, 16, 109, 110) // #0 - Big Bomb Fireball yellow frame

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

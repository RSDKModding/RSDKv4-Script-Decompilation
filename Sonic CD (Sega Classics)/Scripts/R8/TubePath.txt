event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -4, -4, 4, 4, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            switch object.propertyValue
            case PATH_SELECT
                if object.moving == false
                    player[currentPlayer].xpos = object.xpos
                    player[currentPlayer].ypos = object.ypos
                    player[currentPlayer].xvel = 0
                    player[currentPlayer].yvel = 0
                end if

                if player[currentPlayer].xvel == 0
                    if player[currentPlayer].yvel == 0
                        if player[currentPlayer].up == true
                            player[currentPlayer].xvel = 0
                            player[currentPlayer].yvel = -0x100000
                            PlaySfx(SfxName[Release], false)
                        end if

                        if player[currentPlayer].down == true
                            player[currentPlayer].xvel = 0
                            player[currentPlayer].yvel = 0x100000
                            PlaySfx(SfxName[Release], false)
                        end if

                        if player[currentPlayer].left == true
                            player[currentPlayer].xvel = -0x100000
                            player[currentPlayer].yvel = 0
                            PlaySfx(SfxName[Release], false)
                        end if

                        if player[currentPlayer].right == true
                            player[currentPlayer].xvel = 0x100000
                            player[currentPlayer].yvel = 0
                            PlaySfx(SfxName[Release], false)
                        end if
                    end if
                end if
                break

            case PATH_GATE
                if object.moving == false
                    player[currentPlayer].animationSpeed = 240
                    if player[currentPlayer].yvel > 0
                        player[currentPlayer].xpos = object.xpos
                        player[currentPlayer].ypos = object.ypos
                        player[currentPlayer].timer = 0
                        player[currentPlayer].state = Player_State_Static
                        player[currentPlayer].animation = ANI_JUMPING
                        player[currentPlayer].speed = 0
                        player[currentPlayer].tileCollisions = false
                        player[currentPlayer].collisionPlane = COLLISION_PLANE_B
                        player[currentPlayer].xvel = 0
                        player[currentPlayer].yvel = 0x100000
                        PlaySfx(SfxName[Release], false)
                    else
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].tileCollisions = true
                        player[currentPlayer].collisionPlane = COLLISION_PLANE_A
                        player[currentPlayer].yvel = -0x80000
                    end if
                end if
                break

            case PATH_UP_OR_RIGHT
                if object.moving == false
                    player[currentPlayer].animationSpeed = 240
                    player[currentPlayer].xpos = object.xpos
                    player[currentPlayer].ypos = object.ypos

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // condition
            checkResult = object.propertyValue
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            break

        end switch
    end if
end event

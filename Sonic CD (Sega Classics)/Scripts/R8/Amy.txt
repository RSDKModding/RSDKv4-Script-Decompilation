// Aliases
private alias object.value0 : object.timer

// Boss Curtain alias
private alias object[-1].xpos : bossCurtain.xpos

// Fade Screen Aliases
private alias object[+1].type : fadeScreen.type

// States
private alias 0 : AMY_ENTER_SCREEN
private alias 1 : AMY_JUMP_1
private alias 2 : AMY_JUMP_2
private alias 3 : AMY_TRIGGER_FADE
private alias 4 : AMY_FADE_WALK

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case AMY_ENTER_SCREEN
        object.frame = object.timer
        object.frame >>= 2

        object.timer++
        object.timer %= 24

        object.xpos += 0x14000
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)

        temp0 = bossCurtain.xpos
        temp0 -= 0x600000
        if object.xpos > temp0
            object.frame = 6

            object.xpos = temp0
            object.yvel = -0x30000

            object.state++
        end if
        break

    case AMY_JUMP_1
        object.yvel += 0x4000
        object.ypos += object.yvel
        ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
        if object.yvel > 0
            object.frame = 7
        end if

        if checkResult == true
            object.frame = 6
            object.yvel = -0x30000
            object.state++
        end if
        break

    case AMY_JUMP_2
        object.yvel += 0x4000
        object.ypos += object.yvel
        ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
        if object.yvel > 0
            object.frame = 7
        end if

        if checkResult == true
            object.yvel = 0
            object.state++
        end if
        break

    case AMY_TRIGGER_FADE
        object.frame = object.timer
        object.frame >>= 2

        object.timer++
        object.timer %= 24

        object.xpos += 0x14000
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)

        temp0 = bossCurtain.xpos
        temp0 -= 0x100000
        if object.xpos > temp0
            fadeScreen.type = TypeName[Fade Screen]
            object.state++
        end if
        break

    case AMY_FADE_WALK
        object.frame = object.timer
        object.frame >>= 2

        object.timer++
        object.timer %= 24

        object.xpos += 0x14000
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)
        break

    end switch
end event


event ObjectDraw
    if stage.playerListPos == PLAYER_SONIC_A
        DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
    end if
end event

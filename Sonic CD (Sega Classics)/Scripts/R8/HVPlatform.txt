event ObjectUpdate
    switch object.state
    case H_PLAT_BASE
    case V_PLAT_BASE
        if object.timer < 30
            object.timer++
        else
            object.timer = 0
            object.state++
        end if
        break

    case H_PLAT_EXTEND_1
    case H_PLAT_EXTEND_2
    case H_PLAT_EXTEND_3
    case V_PLAT_EXTEND_1
    case V_PLAT_EXTEND_2
    case V_PLAT_EXTEND_3
        if object.timer < 5
            object.timer++
        else
            object.timer = 0
            object.state++
        end if
        break

    case H_PLAT_FULL_EXTEND
    case V_PLAT_FULL_EXTEND
        if object.timer < 120
            object.timer++
        else
            object.timer = 0
            object.state++
        end if
        break

    case H_PLAT_RETRACT_1
    case H_PLAT_RETRACT_2
    case H_PLAT_RETRACT_3
    case V_PLAT_RETRACT_1
    case V_PLAT_RETRACT_2
    case V_PLAT_RETRACT_3
        if object.timer < 5
            object.timer++
        else
            object.timer = 0
            object.state++
            object.state &= V_PLAT_RETRACT_3
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		// Use different hitboxes for the Platform's different sizes
        switch object.state
        case H_PLAT_BASE
        case V_PLAT_BASE
            BoxCollisionTest(C_SOLID, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case H_PLAT_EXTEND_1
        case H_PLAT_RETRACT_3
            BoxCollisionTest(C_SOLID, object.entityPos, -24, -8, 24, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case H_PLAT_EXTEND_2
        case H_PLAT_RETRACT_2
            BoxCollisionTest(C_SOLID, object.entityPos, -40, -8, 40, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case H_PLAT_EXTEND_3
        case H_PLAT_FULL_EXTEND
        case H_PLAT_RETRACT_1
			// Widest version
            BoxCollisionTest(C_SOLID, object.entityPos, -56, -8, 56, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case V_PLAT_EXTEND_1
        case V_PLAT_RETRACT_3
            BoxCollisionTest(C_SOLID, object.entityPos, -8, -24, 8, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case V_PLAT_EXTEND_2
        case V_PLAT_RETRACT_2
            BoxCollisionTest(C_SOLID, object.entityPos, -8, -40, 8, 40, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        case V_PLAT_EXTEND_3
        case V_PLAT_FULL_EXTEND
        case V_PLAT_RETRACT_1
			// Tallest version
            BoxCollisionTest(C_SOLID, object.entityPos, -8, -56, 8, 56, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            break

        end switch
    next
end event


event ObjectDraw
	// Draw the Base Post Frame
    DrawSprite(0)

	// And then depending on the Platform's current state,
	// draw all the extended Platform frames as needed
    switch object.state
    case H_PLAT_EXTEND_1
    case H_PLAT_RETRACT_3
        DrawSprite(3)
        DrawSprite(4)
        break

    case H_PLAT_EXTEND_2
    case H_PLAT_RETRACT_2
        DrawSprite(2)
        DrawSprite(3)
        DrawSprite(4)
        DrawSprite(5)
        break

    case H_PLAT_EXTEND_3
    case H_PLAT_FULL_EXTEND
    case H_PLAT_RETRACT_1
		// Fully extended, horizontally
        DrawSprite(1)
        DrawSprite(2)
        DrawSprite(3)
        DrawSprite(4)
        DrawSprite(5)
        DrawSprite(6)
        break

    case V_PLAT_EXTEND_1
    case V_PLAT_RETRACT_3
        DrawSprite(9)
        DrawSprite(10)
        break

    case V_PLAT_EXTEND_2
    case V_PLAT_RETRACT_2
        DrawSprite(8)
        DrawSprite(9)
        DrawSprite(10)
        DrawSprite(11)
        break

    case V_PLAT_EXTEND_3
    case V_PLAT_FULL_EXTEND
    case V_PLAT_RETRACT_1
		// Fully extended, vertically
        DrawSprite(7)
        DrawSprite(8)
        DrawSprite(9)
        DrawSprite(10)
        DrawSprite(11)
        DrawSprite(12)
        break

    end switch
end event

    SpriteFrame(-24, -8, 16, 16, 75, 200)
    SpriteFrame(8, -8, 16, 16, 75, 200)
    SpriteFrame(24, -8, 16, 16, 75, 200)
    SpriteFrame(40, -8, 16, 16, 75, 200)
    SpriteFrame(-8, -56, 16, 16, 75, 200)
    SpriteFrame(-8, -40, 16, 16, 75, 200)
    SpriteFrame(-8, -24, 16, 16, 75, 200)
    SpriteFrame(-8, 8, 16, 16, 75, 200)
    SpriteFrame(-8, 24, 16, 16, 75, 200)
    SpriteFrame(-8, 40, 16, 16, 75, 200)

	// Find all HV Platform objects in the level
    foreach (TypeName[HV Platform], arrayPos0, ALL_ENTITIES)
		// Start the Platform at fully extended horizontally
        object[arrayPos0].state = V_PLAT_FULL_EXTEND
    next
end event

event RSDKDraw
	DrawSprite(0)

	temp0 = 7
	while temp0 <= 12
		DrawSprite(temp0)
		temp0++
	loop

	if editor.showGizmos == true
		editor.drawingOverlay = true
		object.inkEffect = INK_BLEND

		temp0 = 0
		while temp0 <= 6
			DrawSpriteFX(temp0, FX_INK, Object.XPos, Object.YPos)
			temp0++
		loop

		editor.drawingOverlay = false
	end if
end event

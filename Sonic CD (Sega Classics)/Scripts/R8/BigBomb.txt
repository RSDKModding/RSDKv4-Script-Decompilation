event ObjectUpdate
    switch object.state
    case BIGBOMB_SETUP
        object.priority = PRIORITY_ACTIVE
        ObjectTileCollision(CSIDE_FLOOR, 0, 27, 0)
        if checkResult == true
            if object.propertyValue == STATIC_BOMB
                object.state = BIGBOMB_WAIT_PLAYER
                object.direction = FACING_LEFT
                object.animation = BOMBANI_IDLE
            else
                object.state = BIGBOMB_MOVE_LEFT
                object.animation = BOMBANI_WALKING

                object.leftBound = object.xpos
                object.leftBound -= 0xC0000

                object.rightBound = object.xpos
                object.rightBound += 0xC0000
            end if
        else
            object.ypos += 0x10000
        end if
        break

    case BIGBOMB_MOVE_LEFT
        object.xpos -= 0xA000
        if object.xpos < object.leftBound
            object.direction = FACING_LEFT
            object.state = BIGBOMB_MOVE_RIGHT
        end if

        ObjectTileGrip(CSIDE_FLOOR, 0, 27, 0)
        BoxCollisionTest(C_TOUCH, object.entityPos, -96, -128, 96, 27, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.state = BIGBOMB_START_FUSE
            object.animation = BOMBANI_SPOTTED
        end if
        break

    case BIGBOMB_MOVE_RIGHT
        object.xpos += 0xA000
        if object.xpos > object.rightBound
            object.direction = FACING_RIGHT
            object.state = BIGBOMB_MOVE_LEFT
        end if

        ObjectTileGrip(CSIDE_FLOOR, 0, 27, 0)
        BoxCollisionTest(C_TOUCH, object.entityPos, -96, -128, 96, 27, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.state = BIGBOMB_START_FUSE
            object.animation = BOMBANI_SPOTTED
        end if
        break

    case BIGBOMB_WAIT_PLAYER
        BoxCollisionTest(C_TOUCH, object.entityPos, -96, -128, 96, 27, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.animation = BOMBANI_WALKING
            object.state++
        end if
        break

    case BIGBOMB_PREPARE_FUSE
        object.xpos += 0x20000
        ObjectTileGrip(CSIDE_FLOOR, 24, 27, 0)
        if checkResult == false
            object.state = BIGBOMB_START_FUSE
            object.animation = BOMBANI_SPOTTED
        end if
        break

    case BIGBOMB_START_FUSE
        if object.timer < 30
            object.timer++
        else
            object.timer = 0
            object.animation = BOMBANI_EXPLOSION
            object.animationTimer = 0
            object.savedYPos = object.ypos
            object.state++
        end if
        break

    case BIGBOMB_EXPLODE
        if object.timer < 120
            object.timer++
            object.savedYPos += 0x4000
        else
            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = -0x30000
            object[tempObjectPos].yvel = -0x62000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = -0x20000
            object[tempObjectPos].yvel = -0x52000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = -0x10000
            object[tempObjectPos].yvel = -0x42000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = 0
            object[tempObjectPos].yvel = -0x32000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = 0x10000
            object[tempObjectPos].yvel = -0x42000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = 0x20000
            object[tempObjectPos].yvel = -0x52000

            CreateTempObject(TypeName[BB Fireball], 0, object.xpos, object.ypos)
            object[tempObjectPos].xvel = 0x30000
            object[tempObjectPos].yvel = -0x62000

            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)

            PlaySfx(SfxName[Jump], false)
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if player[currentPlayer].collisionPlane == COLLISION_PLANE_A
            BoxCollisionTest(C_TOUCH, object.entityPos, -18, -18, 18, 18, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                CallFunction(Player_Hit)
            end if
        end if
    next
end event


event ObjectDraw
    switch object.animation
    case BOMBANI_IDLE
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        break

    case BOMBANI_WALKING
        temp0 = object.animationTimer
        temp0 /= 20
        temp0 *= 3
        temp0 += 3
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        temp0++
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        temp0++
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        object.animationTimer++
        object.animationTimer %= 80
        break

    case BOMBANI_SPOTTED
        temp0 = object.animationTimer
        temp0 /= 20
        temp0 *= 3
        temp0 += 3
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        temp0++
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        temp0++
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    case BOMBANI_EXPLOSION
        temp0 = object.animationTimer
        temp0 /= 3
        temp0 += 15
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.savedYPos)
        object.animationTimer++
        object.animationTimer %= 6
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        break

    end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // Type
			checkResult = object.propertyValue
			checkResult &= 1
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
		case 0 // Type
			object.propertyValue = editor.variableValue
			object.propertyValue &= 1
			break

		end switch
	end if
end event

// Aliases
private alias object.value0 : object.hitboxLeft
private alias object.value1 : object.hitboxRight
private alias object.value2 : object.collapseDelay
private alias object.value3 : object.bridgeLength

// Player Aliases
private alias object.xpos : player.xpos

// States
private alias 0 : CPLATFORM_CHECK_COLLISION
private alias 1 : CPLATFORM_CRUMBLE_RIGHT
private alias 2 : CPLATFORM_CRUMBLE_LEFT
private alias 3 : CPLATFORM_COLLAPSE_LEFT
private alias 4 : CPLATFORM_COLLAPSE_RIGHT
private alias 5 : CPLATFORM_RESTORE // Sonic Nexus leftover
private alias 6 : CPLATFORM_FALLING_DEBRIS

// Property Values
private alias 0 : PLATFORM_DEFAULT
private alias 1 : PLATFORM_DEBRIS_1
private alias 2 : PLATFORM_DEBRIS_2

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case CPLATFORM_CRUMBLE_RIGHT
        if object.collapseDelay > 0
            object.collapseDelay--
        else
            PlaySfx(SfxName[Crumble], false)
            object.state = CPLATFORM_COLLAPSE_RIGHT
        end if
        break

    case CPLATFORM_CRUMBLE_LEFT
        if object.collapseDelay > 0
            object.collapseDelay--
        else
            PlaySfx(SfxName[Crumble], false)
            object.state = CPLATFORM_COLLAPSE_LEFT
        end if
        break

    case CPLATFORM_COLLAPSE_LEFT
        object.collapseDelay++
        if object.collapseDelay == 10
            object.collapseDelay = 0
            if object.hitboxLeft < object.hitboxRight
                object.bridgeLength--
                object.hitboxLeft += 16

                temp0 = object.hitboxLeft
                temp0 -= 8
                temp0 <<= 16
                temp0 += object.xpos

                temp1 = object.ypos
                temp1 -= 0x80000

                CreateTempObject(TypeName[CPlatform], PLATFORM_DEFAULT, temp0, temp1)
                object[tempObjectPos].state = CPLATFORM_FALLING_DEBRIS
                object[tempObjectPos].yvel = -0x10000
                object[tempObjectPos].propertyValue = PLATFORM_DEBRIS_1

                temp1 += 0x100000
                CreateTempObject(TypeName[CPlatform], PLATFORM_DEFAULT, temp0, temp1)
                object[tempObjectPos].state = CPLATFORM_FALLING_DEBRIS
                object[tempObjectPos].propertyValue = PLATFORM_DEBRIS_2
            else
                ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            end if
        end if
        break

    case CPLATFORM_COLLAPSE_RIGHT
        object.collapseDelay++
        if object.collapseDelay == 10
            object.collapseDelay = 0
            if object.hitboxRight > object.hitboxLeft
                object.bridgeLength--
                object.hitboxRight -= 16

                temp0 = object.hitboxRight
                temp0 += 8
                temp0 <<= 16
                temp0 += object.xpos

                temp1 = object.ypos
                temp1 -= 0x80000

                CreateTempObject(TypeName[CPlatform], PLATFORM_DEFAULT, temp0, temp1)
                object[tempObjectPos].state = CPLATFORM_FALLING_DEBRIS
                object[tempObjectPos].yvel = -0x10000
                object[tempObjectPos].propertyValue = PLATFORM_DEBRIS_1

                temp1 += 0x100000
                CreateTempObject(TypeName[CPlatform], PLATFORM_DEFAULT, temp0, temp1)
                object[tempObjectPos].state = CPLATFORM_FALLING_DEBRIS
                object[tempObjectPos].propertyValue = PLATFORM_DEBRIS_2
            else
                ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            end if
        end if
        break

    case CPLATFORM_RESTORE
        if object.outOfBounds == true
            object.state = CPLATFORM_CHECK_COLLISION

            object.bridgeLength = object.propertyValue

            object.hitboxLeft = object.propertyValue
            object.hitboxLeft <<= 3

            object.hitboxRight = object.hitboxLeft
            FlipSign(object.hitboxLeft)

            object.priority = PRIORITY_BOUNDS
            object.collapseDelay = 0
        end if
        break

    case CPLATFORM_FALLING_DEBRIS
        object.yvel += 0x4000
        if object.yvel > 0
            object.ypos += object.yvel
        end if

        if object.outOfBounds == true
            object.type = TypeName[Blank Object]
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.state == CPLATFORM_CHECK_COLLISION
            BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, -16, object.hitboxRight, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if object.propertyValue > 1
                if checkResult == true
                    object.collapseDelay = 1
                    object.priority = PRIORITY_ACTIVE

                    if player[currentPlayer].xpos > object.xpos
                        object.state = CPLATFORM_CRUMBLE_RIGHT
                    else
                        object.state = CPLATFORM_CRUMBLE_LEFT
                    end if
                end if
            end if
        else
            BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, -16, object.hitboxRight, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        end if
    next
end event


event ObjectDraw
    switch object.state
    case CPLATFORM_CHECK_COLLISION
    case CPLATFORM_CRUMBLE_RIGHT
    case CPLATFORM_CRUMBLE_LEFT
    case CPLATFORM_COLLAPSE_RIGHT
        temp0 = 0

        temp1 = object.hitboxLeft
        temp1 <<= 16
        temp1 += 0x80000
        temp1 += object.xpos

        while temp0 < object.bridgeLength
            DrawSpriteXY(0, temp1, object.ypos)
            temp0++
            temp1 += 0x100000
        loop
        break

    case CPLATFORM_COLLAPSE_LEFT
        temp0 = 0

        temp1 = object.hitboxRight
        temp1 <<= 16
        temp1 -= 0x80000
        temp1 += object.xpos

        while temp0 < object.bridgeLength
            DrawSpriteXY(0, temp1, object.ypos)
            temp0++
            temp1 -= 0x100000
        loop
        break

    case CPLATFORM_FALLING_DEBRIS
        DrawSprite(object.propertyValue)
        break

    end switch
end event


event ObjectStartup
    LoadSpriteSheet("R8/Objects.gif")

    foreach (TypeName[CPlatform], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].bridgeLength = object[arrayPos0].propertyValue
        object[arrayPos0].hitboxLeft = object[arrayPos0].propertyValue
        object[arrayPos0].hitboxLeft <<= 3
        object[arrayPos0].hitboxRight = object[arrayPos0].hitboxLeft
        FlipSign(object[arrayPos0].hitboxLeft)
    next

    SpriteFrame(-8, -16, 16, 32, 222, 212)  // #0 - Collapsing Platform
    SpriteFrame(-8, -8, 16, 16, 222, 212)   // #1 - Collapsing Platform half 1
    SpriteFrame(-8, -8, 16, 16, 222, 228)   // #2 - Collapsing Platform half 2
end event

event RSDKDraw
	temp0 = 0

	temp1 = object.propertyValue
	temp1 <<= 3
	FlipSign(temp1)

	temp2 = temp1
	temp2 <<= 16
	temp2 += 0x80000
	temp2 += object.xpos
	
	while temp0 < object.propertyValue
		DrawSpriteXY(0, temp2, object.ypos)
		temp0++
		temp2 += 0x100000
	loop
end event

    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

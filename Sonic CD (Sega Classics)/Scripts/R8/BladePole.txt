// Aliases
private alias object.value0 : object.timer

// States
private alias 0 : BLADEPOLE_SETUP
private alias 1 : BLADEPOLE_MOVE_RIGHT
private alias 2 : BLADEPOLE_MOVE_LEFT

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case BLADEPOLE_SETUP
        ObjectTileGrip(CSIDE_FLOOR, 0, 48, 0)
        if checkResult == true
            object.state = BLADEPOLE_MOVE_RIGHT
            object.timer = -64
        else
            object.ypos += 0x10000
        end if
        break

    case BLADEPOLE_MOVE_RIGHT
        if object.timer < 96
            object.timer++
            object.xpos += 0x10000
            ObjectTileGrip(CSIDE_FLOOR, 0, 48, 0)
            if checkResult == false
                object.state = BLADEPOLE_MOVE_LEFT
                object.timer = -64
            end if
        else
            object.state = BLADEPOLE_MOVE_LEFT
            object.timer = -64
        end if
        break

    case BLADEPOLE_MOVE_LEFT
        if object.timer < 96
            object.timer++
            object.xpos -= 0x10000
            ObjectTileGrip(CSIDE_FLOOR, 0, 48, 0)
            if checkResult == false
                object.state = BLADEPOLE_MOVE_RIGHT
                object.timer = -64
            end if
        else
            object.state = BLADEPOLE_MOVE_RIGHT
            object.timer = -64
        end if
        break

    end switch

    switch object.frame
    case 0
    case 2
        if object.animationTimer < 4
            object.animationTimer++
        else
            object.animationTimer = 0
            object.frame++
        end if
        break

    case 1
        if object.animationTimer < 72
            object.animationTimer++
        else
            object.animationTimer = 0
            object.frame++
        end if
        break

    case 3
        if object.animationTimer < 72
            object.animationTimer++
        else
            object.animationTimer = 0
            object.frame = 0
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.frame
        case 1 // Upper Blade
            BoxCollisionTest(C_TOUCH, object.entityPos, -28, -48, 28, -40, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                CallFunction(Player_Hit)
            end if
            break

        case 3 // Lower Blade
            BoxCollisionTest(C_TOUCH, object.entityPos, -28, 16, 28, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                CallFunction(Player_Hit)
            end if
            break

        end switch
    next
end event


event ObjectDraw
    temp0 = object.timer
    temp0 += 64
    temp0 %= 18
    temp0 /= 9
    temp0 += 16
    DrawSprite(temp0)

    switch object.frame
    case 0
    case 2
        DrawSprite(0)
        object.direction = FLIP_Y
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        break

    case 1
        temp0 = object.animationTimer
        temp0 /= 3
        temp0 %= 6
        temp0 += 4
        DrawSprite(temp0)
        DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
        break

    case 3
        temp0 = object.animationTimer
        temp0 /= 3
        temp0 %= 6
        temp0 += 10
        DrawSprite(1)
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break

    end switch
end event

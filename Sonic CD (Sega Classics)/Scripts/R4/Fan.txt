// ========================
// Aliases
// ========================

// Button Aliases
private alias object.propertyValue : pushButton.pressed

// HUD Aliases
private alias object.propertyValue : HUD.currentTimePeriod

// States / Property Values
// These are based on Property Value, so technically they are named the same
private alias 0 : VERTICAL_FAN_OFF
private alias 1 : VERTICAL_FAN_ON
private alias 2 : HORIZONTAL_FAN_OFF
private alias 3 : HORIZONTAL_FAN_ON
private alias 4 : HORIZONTAL_FAN_LEFT_OFF
private alias 5 : HORIZONTAL_FAN_LEFT_ON

// Time Periods
private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case VERTICAL_FAN_OFF
    case HORIZONTAL_FAN_OFF
        if pushButton[-1].pressed == true
            object.propertyValue = VERTICAL_FAN_ON
            object.state++
        end if
        break

    case VERTICAL_FAN_ON
    case HORIZONTAL_FAN_ON
        // Bug Details:
        // This should be aliased as object[9], because this version of the game has the HUD reserved at that slot.
        // Instead, we are incorrectly using the RSDKv3 version's reserved slot

        // Fix:
        // Replace the currentTimePeriod alias with:
        // private alias object[9].propertyValue : HUD.currentTimePeriod
        if HUD[24].currentTimePeriod < TIME_GOOD_FUTURE
            object.frame++
            if object.frame > 9
                object.frame = 0
            end if
        else
            object.frame = 2
        end if
        break
    end switch
end event


event ObjectDraw
    switch object.state
    case VERTICAL_FAN_OFF
        DrawSprite(0)
        break

    case VERTICAL_FAN_ON
        temp0 = object.frame
        temp0 >>= 1
        DrawSprite(temp0)
        break

    case HORIZONTAL_FAN_OFF
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        break

    case HORIZONTAL_FAN_ON
        temp0 = object.frame
        temp0 >>= 1
        temp0 += 5
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        break
    end switch
end event


event ObjectStartup
    // Weird [9] check instead of the usual of baking the slot in the alias

        SpriteFrame(-16, -8, 32, 16, 94, 228)   // #1 - Future fan vertical on

        SpriteFrame(-8, -16, 16, 32, 93, 144)   // #7 - Future fan horizontal on
        SpriteFrame(-8, -16, 16, 32, 93, 144)   // #8 - Future fan horizontal on
        SpriteFrame(-8, -16, 16, 32, 93, 144)   // #9 - Future fan horizontal on
        break
    end switch

    foreach (TypeName[Fan], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].state = object[arrayPos0].propertyValue
        if object[arrayPos0].propertyValue > 3
            object[arrayPos0].state -= 2
            object[arrayPos0].direction = FACING_LEFT
        end if

        object[arrayPos0].propertyValue &= 1
    next
end event


// ========================
// Editor Events
// ========================

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // type
            checkResult = object.propertyValue
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // type
            object.propertyValue = editor.variableValue
            break
        end switch
    end if
end event


event RSDKDraw
    if object.propertyValue > 3
        object.direction = FACING_LEFT
    end if

    DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)
end event

event ObjectUpdate
    // Water streams are activated through a chain, originated from a blank object or a fan
    if prevObject[-1].active == true
        object.activeWaterStream = true
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_TOUCH, object.entityPos, object.hitboxLeft, object.hitboxTop, object.hitboxRight, object.hitboxBottom, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Hurt
                    if player[currentPlayer].state != Player_State_Death
                        if player[currentPlayer].state != Player_State_Drown
                            if player[currentPlayer].state != Player_State_Static
                                player.waterStream = true

                                player[currentPlayer].state = Player_State_WaterCurrent
                                player[currentPlayer].animation = ANI_FANROTATE

                                temp0 = player[currentPlayer].xpos
                                temp0 -= object.xpos

                                temp1 = player[currentPlayer].ypos
                                temp1 -= object.ypos

                                ATan2(temp2, temp0, temp1)
                                temp2 += object.oscillation

                                Cos256(player[currentPlayer].xvel, temp2)
                                Sin256(player[currentPlayer].yvel, temp2)
                                player[currentPlayer].xvel <<= 10
                                player[currentPlayer].yvel <<= 10
                                player[currentPlayer].speed = player[currentPlayer].xvel
                            end if
                        end if
                    end if
                end if
            else
                if player[currentPlayer].state == Player_State_WaterCurrent
                    if player.waterStream == false
                        if player[currentPlayer].left == true
                            player[currentPlayer].xvel = -0x20000
                        end if

                        if player[currentPlayer].right == true
                            player[currentPlayer].xvel = 0x20000
                        end if

                        player[currentPlayer].speed = player[currentPlayer].xvel
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
        next
    end if
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value, activeRange
            checkResult = object.propertyValue
            checkResult %= 11
            break

        case 0 // streamRange
            checkResult = object.propertyValue
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value, activeRange
            object.propertyValue = editor.variableValue
            object.propertyValue %= 11
            break

        case 0 // streamRange
            object.propertyValue = editor.variableValue
            break
        end switch
    end if
end event

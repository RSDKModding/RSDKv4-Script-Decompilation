event ObjectUpdate
    if object.state == BLOCKROWSB_MOVE_RIGHT
        object.xvel += 0x8000
        if object.xvel >= 0xC00000
            object.xvel = 0xC00000
            object.state = BLOCKROWSB_MOVE_LEFT
        end if
    else
        object.xvel -= 0x8000
        if object.xvel <= 0
            object.xvel = 0
            object.state = BLOCKROWSB_MOVE_RIGHT
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = object.xpos
        temp1 = 0
        temp2 = 0
        object.xpos -= 0x600000
        object.xpos += object.xvel
        BoxCollisionTest(C_SOLID, object.entityPos, -32, -32, 32, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            if object.state == BLOCKROWSB_MOVE_RIGHT
                player[currentPlayer].xpos += 0x8000
            else
                player[currentPlayer].xpos -= 0x8000
            end if

            if player[currentPlayer].speed == 0
                if object.stood == false
					// Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                    object.stood = true
                end if
            else
                object.stood = false
            end if
            break

        case COL_LEFT
            temp1 = true
            break

        case COL_RIGHT
            temp2 = true
            break
        end switch

        object.xpos = temp0
        if object.propertyValue > 1
            BoxCollisionTest(C_SOLID, object.entityPos, -160, -32, -128, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == COL_RIGHT
                temp2 = true
            end if
        end if

        BoxCollisionTest(C_SOLID, object.entityPos, 128, -32, 160, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_LEFT
            temp1 = true
        end if

        if temp1 == true
            if temp2 == true
                CallFunction(Player_Kill)
            end if
        end if

        if player[currentPlayer].gravity == GRAVITY_AIR
            object.stood = false
        end if
    next
end event


event ObjectDraw
    temp0 = object.xpos
    temp0 -= 0x600000
    temp0 += object.xvel
    DrawSpriteXY(0, temp0, object.ypos)
    DrawSpriteXY(1, temp0, object.ypos)
    DrawSpriteXY(2, temp0, object.ypos)
    DrawSpriteXY(3, temp0, object.ypos)
end event


event ObjectStartup
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        temp0 = 163
        temp1 = 1
        temp2 = 196
        break

    case TIME_PAST
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 157
        temp2 = 34
        break

    case TIME_GOOD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 190
        temp2 = 34
        break

    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 223
        temp2 = 34
        break
    end switch

	// Cube of blocks
    SpriteFrame(-32, -32, 32, 32, temp0, temp1)
    SpriteFrame(0, -32, 32, 32, temp0, temp1)
    SpriteFrame(-32, 0, 32, 32, temp0, temp1)
    SpriteFrame(0, 0, 32, 32, temp0, temp1)

    foreach (TypeName[Block Rows B], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].priority = PRIORITY_XBOUNDS
        switch object[arrayPos0].propertyValue
        case 1
            object[arrayPos0].xvel = 0x400000
            break

        case 2
            object[arrayPos0].xvel = 0x800000
            break

        case 3
            object[arrayPos0].xvel = 0xC00000
            object[arrayPos0].state = BLOCKROWSB_MOVE_LEFT
            break

        case 4
            object[arrayPos0].xvel = 0x800000
            object[arrayPos0].state = BLOCKROWSB_MOVE_LEFT
            break
        end switch
    next
end event

event RSDKDraw
	temp0 = object.xpos
	temp0 -= 0x600000

	switch object.propertyValue
	case 1
		temp0 += 0x400000
		break

	case 2
		temp0 += 0x800000
		break

	case 3
		temp0 += 0xC00000
		break

	case 4
		temp0 += 0x800000
		break
	end switch

	DrawSpriteXY(0, temp0, object.ypos)
	DrawSpriteXY(1, temp0, object.ypos)
	DrawSpriteXY(2, temp0, object.ypos)
	DrawSpriteXY(3, temp0, object.ypos)

	temp1 = object.xpos
	temp1 -= 0x600000
	temp1 += 0xC00000
	temp2 = object.xpos
	temp2 -= 0x600000
	if object.propertyValue < 3
		DrawArrow(temp0, object.ypos, temp1, object.ypos, 255, 128, 0)
	else
		DrawArrow(temp0, object.ypos, temp2, object.ypos, 255, 128, 0)
	end if

	if editor.showGizmos == true
		editor.drawingOverlay = true

		temp3 = object.ypos
		temp3 -= 0x200000
		temp4 = object.xpos
		temp4 -= 0x600000
		DrawRectOutline(temp4, temp3, 0xC0, 0x40, 255, 255, 255, 255)

		temp4 -= 0x200000
		DrawRectOutline(temp4, temp3, 0x100, 0x40, 255, 255, 0, 255)

		if object.propertyValue < 3
			DrawArrow(object.xpos, object.ypos, temp2, object.ypos, 255, 255, 255)
		else
			DrawArrow(object.xpos, object.ypos, temp1, object.ypos, 255, 255, 255)
		end if

		editor.drawingOverlay = false
	end if
end event

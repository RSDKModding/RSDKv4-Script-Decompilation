// Aliases
private alias object.value1 : object.minHeight

// Player Aliases
private alias object.ypos : player.ypos

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0 : FAN_ELEVATOR_CLOSED
private alias 1 : FAN_ELEVATOR_OPEN

// Time Periods
private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
    if object.state == FAN_ELEVATOR_CLOSED
        if object.yvel < 0x10000
            object.yvel += 0x2000
        end if

        if object.ypos == object.minHeight
            temp0 = player[SLOT_PLAYER1].ypos
            temp0 -= object.ypos
            if temp0 < 0xA00000
                object.yvel = 0
            end if
        else
            temp0 = object.ypos
            temp0 -= object.minHeight
            if temp0 > 0x2A00000
                object.ypos = object.minHeight
            end if
        end if
    end if

    object.ypos += object.yvel

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -28, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.yvel = -0x20000
            if object.state == FAN_ELEVATOR_CLOSED
                PlaySfx(SfxName[Switch], false)
            end if

            object.state = FAN_ELEVATOR_OPEN
        else
            if object.yvel < 0
                object.yvel = 0
            end if

            object.state = FAN_ELEVATOR_CLOSED
        end if
    next
end event


event ObjectDraw
    if object.state == FAN_ELEVATOR_CLOSED
        DrawSprite(0)
        DrawSprite(1)
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
    else
        DrawSprite(2)
        DrawSprite(3)
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects3.gif")

    // This was an < 2 check in the RSDKv3 version, < 3 here
    if HUD.currentTimePeriod < TIME_BAD_FUTURE
        SpriteFrame(-32, -32, 32, 48, 84, 46)   // #0 - Fan Elevator Body
        SpriteFrame(-32, 16, 32, 16, 110, 144)  // #1 - Fan closed looking down
        SpriteFrame(-32, -32, 32, 48, 84, 95)   // #2 - Fan Elevator Body pressed
        SpriteFrame(-32, 16, 32, 16, 110, 161)  // #3 - Fan open looking down
    else
        SpriteFrame(-32, -32, 32, 48, 84, 46)   // #0 - Fan Elevator Body
        SpriteFrame(-32, 16, 32, 16, 126, 228)  // #1 - Fan closed looking down
        SpriteFrame(-32, -32, 32, 48, 84, 95)   // #2 - Fan Elevator Body pressed
        SpriteFrame(-32, 16, 32, 16, 158, 228)  // #3 - Fan open looking down
    end if

    foreach (TypeName[Fan Elevator], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].direction = FACING_LEFT
        object[arrayPos0].priority  = PRIORITY_XBOUNDS
        object[arrayPos0].minHeight = object[arrayPos0].ypos
    next
end event


// ========================

event RSDKDraw
    DrawSprite(0)

    object.direction = FACING_LEFT
    DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
    DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
end event

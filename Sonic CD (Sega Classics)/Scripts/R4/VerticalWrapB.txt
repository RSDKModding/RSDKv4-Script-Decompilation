// Player Aliases
private alias object.ypos : player.ypos
private alias object.yvel : player.yvel

// ========================
// Events
// ========================

event ObjectUpdate
	if player[SLOT_PLAYER1].yvel >= 0
		if screen[0].cameraY >= 0x878
			temp1 = player[SLOT_PLAYER1].ypos
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				temp0 = player[currentPlayer].ypos
				temp0 -= temp1
				Abs(temp0)
				if temp0 < 0x8C0000
					player[currentPlayer].ypos -= 0x8000000
				end if
			next

			screen[0].cameraY -= 0x800
			screen.yoffset -= 0x800
		end if
	end if

	if player[SLOT_PLAYER1].yvel <= 0
		if screen[0].cameraY <= 120
			temp1 = player[SLOT_PLAYER1].ypos
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				temp0 = player[currentPlayer].ypos
				temp0 -= temp1
				Abs(temp0)
				if temp0 < 0x8C0000
					player[currentPlayer].ypos += 0x8000000
				end if
			next

			screen[0].cameraY += 0x800
			screen.yoffset += 0x800
		end if
	end if
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event

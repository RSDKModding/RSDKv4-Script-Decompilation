event ObjectUpdate
    switch object.state
    case CRUSHBLOCKS_DELAY_1
        if object.timer < 60
            object.timer++
        else
            object.timer = 0
            object.offset.x = 0x8000
            object.state++
        end if
        break

    case CRUSHBLOCKS_OPEN
        if object.xvel < 0x600000
            object.xvel += 0x8000
        else
            object.offset.x = 0
            object.state++
        end if
        break

    case CRUSHBLOCKS_DELAY_2
        if object.timer < 120
            object.timer++
        else
            object.timer = 0
            object.offset.x = -0x20000
            object.state++
        end if
        break

    case CRUSHBLOCKS_CRUSH
        if object.xvel > 0
            object.xvel -= 0x20000
        else
            object.offset.x = 0
            object.state = CRUSHBLOCKS_DELAY_1
        end if
        break
    end switch

    temp0 = object.xpos
    temp1 = 0
    temp2 = 0
    object.xpos -= object.xvel

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -80, -64, -48, -32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos -= object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_RIGHT
            temp1 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, -80, -32, -16, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos -= object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = 0
            end if
            break

        case COL_RIGHT
            temp1 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, -80, 0, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos -= object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = 0
            end if
            break

        case COL_RIGHT
            temp1 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, -80, 32, 48, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos -= object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_RIGHT
            temp1 = true
            break
        end switch

        object.xpos = temp0
        object.xpos += object.xvel

        BoxCollisionTest(C_SOLID, object.entityPos, -48, -64, 80, -32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos += object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_LEFT
            temp2 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 80, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos += object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_LEFT
            temp2 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, 16, 0, 80, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos += object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_LEFT
            temp2 = true
            break
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, 48, 32, 80, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos += object.offset.x
            if player[currentPlayer].speed == 0
                if object.stood == false
                    object.stood = true

                    // Truncate these values
                    player[currentPlayer].xpos &= -0x10000
                    object.xvel &= -0x10000
                end if
            else
                object.stood = false
            end if
            break

        case COL_LEFT
            temp2 = true
            break
        end switch

        if temp1 == true
            if temp2 == true
                CallFunction(Player_Kill)
            end if
        end if

        if player[currentPlayer].gravity == GRAVITY_AIR
            object.stood = false
        end if
    next

    object.xpos = temp0
end event

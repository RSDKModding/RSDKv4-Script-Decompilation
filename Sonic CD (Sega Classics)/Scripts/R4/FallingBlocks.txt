event ObjectUpdate
    switch object.state
    case FALLINGBLOCKS_SETUP
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            if object.state == FALLINGBLOCKS_SETUP
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -64, 16, 192, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.priority = PRIORITY_ACTIVE
                    object.yvel = 0

                    object.targetHeight = object.ypos
                    object.targetHeight += 0x200000
                    object.state++
                end if
            end if
        next
        break

    case FALLINGBLOCKS_CRUMBLE
        object.yvel += 0x2000
        object.ypos += object.yvel
        if object.ypos >= object.targetHeight
            object.ypos = object.targetHeight
            object.yvel = 0
            object.state++
        end if
        break

    case FALLINGBLOCKS_DELAY
        if object.timer < 30
            object.timer++
        else
            object.timer = 0
            object.state++
        end if
        break

    case FALLINGBLOCKS_FALL
        object.yvel += 0x2000
        object.ypos += object.yvel

        ObjectTileCollision(CSIDE_FLOOR, 0, 32, 0)
        if checkResult == true
            object.state++
        end if
        break
    end switch

    if object.outOfBounds == true
        object.ypos = object.originPos.y
        object.priority = PRIORITY_BOUNDS
        object.state = FALLINGBLOCKS_SETUP
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID2, object.entityPos, -16, -64, 16, 62, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_BOTTOM
            if player[currentPlayer].gravity == GRAVITY_AIR
                CallFunction(Player_Hit)
            else
                CallFunction(Player_Kill)
            end if
        end if
    next
end event

end event

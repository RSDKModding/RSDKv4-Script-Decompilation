event ObjectUpdate
    if prevObject[-1].active == true
        object.activeWaterStream = true
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_TOUCH, object.entityPos, -32, object.hitboxTop, 32, object.hitboxBottom, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Hurt
                    if player[currentPlayer].state != Player_State_Death
                        if player[currentPlayer].state != Player_State_Drown
                            if player[currentPlayer].state != Player_State_Static
                                player[currentPlayer].state = Player_State_WaterCurrent
                                player[currentPlayer].animation = ANI_FANROTATE

                                if player[currentPlayer].left == false
                                    if player[currentPlayer].right == false
                                        if player[currentPlayer].xpos < object.xpos
                                            player[currentPlayer].xpos += 0x10000
                                        end if
                                        if player[currentPlayer].xpos > object.xpos
                                            player[currentPlayer].xpos -= 0x10000
                                        end if
                                    end if
                                end if

                                player[currentPlayer].yvel = -0x40000

                                if player.waterStream == false
                                    player[currentPlayer].xvel = 0
                                    player[currentPlayer].speed = 0
                                    player.waterStream = true
                                end if
                            end if
                        end if
                    end if
                end if
            else
                if player[currentPlayer].state == Player_State_WaterCurrent
                    if player.waterStream == false
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
        next
    end if
end event


event ObjectStartup
    foreach (TypeName[Water Stream U], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].hitboxTop = object[arrayPos0].propertyValue
        object[arrayPos0].hitboxTop <<= 3
        object[arrayPos0].hitboxBottom = object[arrayPos0].hitboxTop
        FlipSign(object[arrayPos0].hitboxTop)
        object[arrayPos0].activeWaterStream = false
    next
end event

event RSDKDraw
    if editor.showGizmos == true
        DrawLine(object.xpos, object.ypos, object[-1].xpos, object.ypos, 0, 255, 255)
        DrawArrow(object[-1].xpos, object.ypos, object[-1].xpos, object[-1].ypos, 0, 255, 255)
    end if

    object.inkEffect = INK_ALPHA
    object.alpha = 198
    DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)

    temp0 = object.ypos
    temp0 -= 0x400000
    DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 255, 0)
end event

// ========================
// Events
// ========================

event ObjectUpdate
    object.timer++
    if object.timer > 1
        tileLayer[0].deformationOffsetW++
        tileLayer[1].deformationOffsetW++
        object.timer = 0
    end if

    if object.usePaletteCycle == true
        object.paletteTimer++
        if object.paletteTimer == 3
            object.paletteTimer = 0

            CopyPalette(3, object.paletteSrcIndex, 0, 176, 16)
            object.paletteSrcIndex += 16
            object.paletteSrcIndex %= object.paletteMaxIndex
        end if
    end if

    if Water_flashTimer > 0
        Water_flashTimer--
    end if
end event


event ObjectDraw
    temp0 = stage.waterLevel
    temp0 -= screen.yoffset

    if temp0 < 0
        temp0 = 0
    end if

    if temp0 > screen.ysize
        temp0 = screen.ysize
    end if

    SetActivePalette(0, 0, temp0)

    if Water_flashTimer > 0
        SetActivePalette(2, temp0, screen.ysize)
    else
        SetActivePalette(1, temp0, screen.ysize)
    end if

    player.waterStream = false
end event


event ObjectStartup
    object[SLOT_R4SETUP].type = TypeName[R4 Setup]
    object[SLOT_R4SETUP].priority = PRIORITY_ACTIVE
    object[SLOT_R4SETUP].drawOrder = 0

    arrayPos0 = 0
    CallFunction(R4Setup_Deform_Left)

    arrayPos0 = 128
    CallFunction(R4Setup_Deform_Left)

    arrayPos0 = 160
    CallFunction(R4Setup_Deform_Right)

    arrayPos0 = 256
    while arrayPos0 < 576
        arrayPos1 = arrayPos0
        arrayPos1 -= 256
        stage[arrayPos0].deformationData1 = stage[arrayPos1].deformationData1
        arrayPos0++
    loop

    SetLayerDeformation(DEFORM_BG_WATER, 128, 4, 0, 0, 0)

    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadPalette("R4A_WaterPal.act", 1, 0, 0, 256)
        if stage.actNum == 2
            LoadPalette("R4A_PalCycle.act", 3, 0, 0, 64)
            object[SLOT_R4SETUP].usePaletteCycle = true
            object[SLOT_R4SETUP].paletteMaxIndex = 64
        end if
        break

    case TIME_PAST
        LoadPalette("R4B_WaterPal.act", 1, 0, 0, 256)
        if stage.actNum == 2
            LoadPalette("R4B_PalCycle.act", 3, 0, 0, 48)
            object[SLOT_R4SETUP].usePaletteCycle = true
            object[SLOT_R4SETUP].paletteMaxIndex = 48
        end if
        break

    case TIME_GOOD_FUTURE
        LoadPalette("R4C_WaterPal.act", 1, 0, 0, 256)
        LoadPalette("R4C_PalCycle.act", 3, 0, 0, 64)
        object[SLOT_R4SETUP].usePaletteCycle = true
        object[SLOT_R4SETUP].paletteMaxIndex = 64
        break

    case TIME_BAD_FUTURE
        LoadPalette("R4D_WaterPal.act", 1, 0, 0, 256)
        LoadPalette("R4D_PalCycle.act", 3, 0, 0, 48)
        object[SLOT_R4SETUP].usePaletteCycle = true
        object[SLOT_R4SETUP].paletteMaxIndex = 48
        break
    end switch

    LoadPalette("ElectricFlash.act", 2, 0, 0, 256)
    Water_flashTimer = 0
end event

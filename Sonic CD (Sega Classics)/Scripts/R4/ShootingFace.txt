// ========================
// Aliases
// ========================

// Aliases
private alias object.value0 : object.timer

// Face Projectile Aliases
private alias object.xpos      : faceProjectile.xpos
private alias object.ypos      : faceProjectile.ypos
private alias object.xvel      : faceProjectile.xvel
private alias object.direction : faceProjectile.direction

// Property Values
private alias 0 : LOOKING_RIGHT
private alias 1 : LOOKING_LEFT

// ========================
// Events
// ========================

event ObjectUpdate
    object.timer++
    if object.timer == 32
        object.timer = 0

        CreateTempObject(TypeName[Face Projectile], 0, object.xpos, object.ypos)

        if object.propertyValue == LOOKING_RIGHT
            faceProjectile[tempObjectPos].xpos += 0x100000
            faceProjectile[tempObjectPos].direction = FACING_RIGHT
            faceProjectile[tempObjectPos].xvel = -0x30000
        else
            faceProjectile[tempObjectPos].xpos -= 0x100000
            faceProjectile[tempObjectPos].direction = FACING_LEFT
            faceProjectile[tempObjectPos].xvel = 0x30000
        end if

        faceProjectile[tempObjectPos].ypos += 0x40000
    end if
end event


event ObjectDraw
    DrawSprite(object.propertyValue)
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects2.gif")

    foreach (TypeName[Shooting Face], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].drawOrder = 4
    next

    SpriteFrame(-8, -16, 16, 32, 118, 151)  // #0 - Face Projectile facing left
    SpriteFrame(-8, -16, 16, 32, 135, 151)  // #1 - Face Projectile facing right
end event


// ========================
// Editor Events
// ========================

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // direction
            checkResult = object.propertyValue
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // direction
            object.propertyValue = editor.variableValue
            break

        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event


event RSDKLoad
    LoadSpriteSheet("R4/Objects2.gif")

    SpriteFrame(-8, -16, 16, 32, 118, 151)  // #0 - Face Projectile facing left
    SpriteFrame(-8, -16, 16, 32, 135, 151)  // #1 - Face Projectile facing right

    AddEditorVariable("direction")
    SetActiveVariable("direction")
    AddEnumVariable("Facing left", 0)
    AddEnumVariable("Facing right", 1)
end event

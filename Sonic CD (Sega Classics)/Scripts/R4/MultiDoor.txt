event ObjectUpdate
    switch object.state
    case MULTIDOOR_INACTIVE
        object.offset.y = object.ypos

        // Check if the previous or next button got pressed
        if pushButton[-1].pressed == true
            object.state++
        end if

        if pushButton[+1].pressed == true
            object.state++
        end if
        break

    case MULTIDOOR_OPEN
        if object.timer < 128
            object.timer++
            object.ypos += 0x10000
            object.offset.y -= 0x10000
        else
            object.state++
        end if
        break
    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -32, -65, 0, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        BoxCollisionTest(C_SOLID, object.entityPos, 32, -65, 64, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_TOP
            player[currentPlayer].ypos += 0x10000
        end if

        temp0 = object.ypos
        object.ypos = object.offset.y
        BoxCollisionTest(C_SOLID, object.entityPos, -64, -64, -32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        BoxCollisionTest(C_SOLID, object.entityPos, 0, -64, 32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        object.ypos = temp0
    next
end event

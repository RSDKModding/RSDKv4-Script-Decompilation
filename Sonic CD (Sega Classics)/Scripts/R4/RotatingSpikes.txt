public function RotatingSpikes_DebugDraw
	DrawSprite(0)

	temp0 = 0
	temp1 = 16

	temp4 = debugMode[0].currentSelection
	temp4 -= RotatingSpikes_startDebugID
	while temp0 < temp4
		Cos(temp2, object.angle)
		temp2 *= temp1
		temp2 <<= 7
		temp2 += object.xpos

		Sin(temp3, object.angle)
		temp3 *= temp1
		temp3 <<= 7
		temp3 += object.ypos

		DrawSpriteXY(1, temp2, temp3)
		temp0++
		temp1 += 16
	loop

	Cos(temp2, object.angle)
	temp2 *= temp1
	temp2 <<= 7
	temp2 += object.xpos

	Sin(temp3, object.angle)
	temp3 *= temp1
	temp3 <<= 7
	temp3 += object.ypos

	DrawSpriteXY(2, temp2, temp3)
end function


public function RotatingSpikes_DebugSpawn

	temp0++

end function

event ObjectUpdate
	object.angle += 2
	object.angle &= 511

	temp0 = 0

	temp1 = object.chainAmount
	temp1--
	temp1 <<= 4

	temp2 = object.xpos
	temp3 = object.ypos

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		Cos(object.xpos, object.angle)
		object.xpos *= temp1
		object.xpos <<= 7
		object.xpos += temp2

		Sin(object.ypos, object.angle)
		object.ypos *= temp1
		object.ypos <<= 7
		object.ypos += temp3

		BoxCollisionTest(C_TOUCH, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			CallFunction(Player_Hit)
		end if
	next

	object.xpos = temp2
	object.ypos = temp3
end event

	temp0 = 0
	RotatingSpikes_startDebugID = DebugMode_ObjCount
	RotatingSpikes_startDebugID--
	while temp0 < 4
		SetTableValue(TypeName[Rotating Spikes], DebugMode_ObjCount, DebugMode_TypesTable)

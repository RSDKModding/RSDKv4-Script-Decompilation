event ObjectUpdate
    switch object.state
    case COUNTDOWNBUBBLE_SETUP
        object.ypos += object.yvel

        object.timer++
        if object.timer > 6
            object.timer = 0
            object.frame++
        end if

        if object.frame < 5 // NOTE: This used to check against 4 for RSDKv3
            Sin(object.xpos, object.angle)
            object.xpos <<= 9
            object.xpos += object.drawPos.x

            object.angle += 4
            object.angle &= 511
        else
            temp0 = object.propertyValue
            temp0 *= 3
            object.frame += temp0

            object.drawPos.x = object.ixpos
            object.drawPos.x -= screen.xoffset

            object.drawPos.y = object.iypos
            object.drawPos.y -= screen.yoffset

            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_1
        object.timer++
        if object.timer == 6
            object.timer = 0
            object.frame++
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_2
        object.timer++
        if object.timer == 6
            object.timer = 0
            object.frame++
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_3
        object.timer++
        if object.timer == 8
            object.timer = 0
            object.frame--
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_4
        object.timer++
        if object.timer == 8
            object.timer = 0
            object.frame++
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_5
        object.timer++
        if object.timer == 8
            object.timer = 0
            object.frame--
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_6
        object.timer++
        if object.timer == 8
            object.timer = 0
            object.frame++
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_7
        object.timer++
        if object.timer == 8
            object.timer = 0
            object.frame--
            object.state++
        end if
        break

    case COUNTDOWNBUBBLE_FRAME_8
        object.timer++
        if object.timer == 9
            object.type = TypeName[Blank Object]
        end if
        break
    end switch
end event


event ObjectDraw
    if object.state == COUNTDOWNBUBBLE_SETUP
        DrawSprite(object.frame)
    else
        DrawSpriteScreenXY(object.frame, object.drawPos.x, object.drawPos.y)
    end if
end event

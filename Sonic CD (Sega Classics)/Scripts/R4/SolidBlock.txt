event ObjectUpdate
    switch object.propertyValue
    case ONE_BLOCK
    case ONE_BLOCK_ALT

            BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        next
        break

    case TWO_BLOCKS_HORIZONTAL

event ObjectStartup
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        temp0 = 163
        temp1 = 1
        temp2 = 196
        break

    case TIME_PAST
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 157
        temp2 = 34
        break

    case TIME_GOOD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 190
        temp2 = 34
        break

    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 223
        temp2 = 34
        break
    end switch

    // R4 Blocks
    SpriteFrame(-16, -16, 32, 32, temp0, temp1)
    SpriteFrame(-16, -16, 32, 32, temp2, temp1)
    SpriteFrame(-32, -16, 32, 32, temp0, temp1)
    SpriteFrame(0, -16, 32, 32, temp2, temp1)
    SpriteFrame(-16, -32, 32, 32, temp0, temp1)
    SpriteFrame(-16, 0, 32, 32, temp2, temp1)
    SpriteFrame(-32, -32, 32, 32, temp0, temp1)
    SpriteFrame(0, -32, 32, 32, temp2, temp1)
    SpriteFrame(-32, 0, 32, 32, temp2, temp1)
    SpriteFrame(0, 0, 32, 32, temp0, temp1)
    SpriteFrame(-64, -16, 32, 32, temp0, temp1)
    SpriteFrame(32, -16, 32, 32, temp2, temp1)

    foreach (TypeName[Solid Block], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].drawOrder = 4
    next

    temp0 = 0
    SolidBlock_startDebugID = DebugMode_ObjCount
    while temp0 < BLOCK_TYPE_COUNT
        SetTableValue(TypeName[Solid Block], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(SolidBlock_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(SolidBlock_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            break

        case 0 // blocks
            checkResult = object.propertyValue
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            break

        case 0 // blocks
            object.propertyValue = editor.variableValue
            break
        end switch
    end if
end event

// Aliases
private alias object.value0 : object.waterEntity

// Player Aliases
private alias object.xpos : player.xpos

// Water Aliases
private alias object.ypos : water.ypos

// Property Values
private alias 0 : BOSSWATERTRIGGER_RAISE_WATER
private alias 1 : BOSSWATERTRIGGER_SET_Y_BOUNDS

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.propertyValue
    case BOSSWATERTRIGGER_RAISE_WATER
        if player[SLOT_PLAYER1].xpos > object.xpos
			// Y - 1172 in map coords
            if stage.newWaterLevel > 0x4940000
                stage.newWaterLevel = 0x4940000

                arrayPos0 = object.waterEntity

                water[arrayPos0].ypos = screen.yoffset
                water[arrayPos0].ypos += 256
                water[arrayPos0].ypos <<= 16

                stage.newYBoundary2 = 0x5D8
                stage.curYBoundary2 = 0x5D8
            end if
        else
			// Y - 3072 in map coords
            if stage.newWaterLevel < 0xC000000
                stage.newWaterLevel = 0xC000000
                stage.newYBoundary2 = 0x900
            end if

            arrayPos0 = object.waterEntity

            temp0 = screen.yoffset
            temp0 += 256
            temp0 <<= 16
            if water[arrayPos0].ypos >= temp0
                water[arrayPos0].ypos = 0xC000000
            end if
        end if
        break

    case BOSSWATERTRIGGER_SET_Y_BOUNDS
        if player[SLOT_PLAYER1].xpos > object.xpos
            stage.newYBoundary2 = 0x5A8
        else
            stage.newYBoundary2 = 0x5D8
        end if
        break
    end switch
end event


event ObjectStartup
    foreach (TypeName[Water], arrayPos0, ALL_ENTITIES)
        arrayPos1 = arrayPos0
    next

    foreach (TypeName[Boss W Trigger], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].waterEntity = arrayPos1
    next
end event

// Editor Events

end event

end event

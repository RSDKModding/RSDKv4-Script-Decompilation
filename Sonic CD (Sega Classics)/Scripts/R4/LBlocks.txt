event ObjectUpdate
    switch object.state
    case LBLOCKS_STEP_LEFT
        if object.timer < 14
            object.timer++
        else
            object.timer = 0
            object.state = LBLOCKS_ROTATE_LEFT
            PlaySfx(SfxName[Switch], false)
        end if
        break

    case LBLOCKS_STEP_RIGHT
        if object.timer < 14
            object.timer++
        else
            object.timer = 0
            object.state = LBLOCKS_ROTATE_RIGHT
            PlaySfx(SfxName[Switch], false)
        end if
        break

    case LBLOCKS_ROTATE_LEFT
        if object.timer < 8
            object.timer++
            object.angle++
        else
            object.timer = 0
            object.state = LBLOCKS_INACTIVE
            PlaySfx(SfxName[Switch], false)
        end if
        break

    case LBLOCKS_ROTATE_RIGHT
        if object.timer < 8
            object.timer++
            object.angle--
        else
            object.timer = 0
            object.state = LBLOCKS_INACTIVE
            PlaySfx(SfxName[Switch], false)
        end if
        break
    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

		if object.state == LBLOCKS_INACTIVE
            temp1 = checkResult
        else
            temp1 = COL_TOP
        end if

		if object.state < 3
            temp0 = object.angle
            temp0 >>= 3
            switch temp0
            case 0
                BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, -16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                BoxCollisionTest(C_SOLID, object.entityPos, -48, -16, -16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == COL_TOP
                    if temp1 == COL_NONE
                        object.state = LBLOCKS_STEP_LEFT
                    end if
                end if
                break

            case 1
                BoxCollisionTest(C_SOLID, object.entityPos, -16, 16, 16, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                BoxCollisionTest(C_SOLID, object.entityPos, -48, -16, -16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == COL_TOP
                    if temp1 == COL_NONE
                        object.state = LBLOCKS_STEP_LEFT
                    end if
                end if
                break

            case 2
                BoxCollisionTest(C_SOLID, object.entityPos, -16, 16, 16, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                BoxCollisionTest(C_SOLID, object.entityPos, 16, -16, 48, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == COL_TOP
                    if temp1 == COL_NONE
                        object.state = LBLOCKS_STEP_RIGHT
                    end if
                end if
                break

            case 3
                BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, -16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                BoxCollisionTest(C_SOLID, object.entityPos, 16, -16, 48, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == COL_TOP
                    if temp1 == COL_NONE
                        object.state = LBLOCKS_STEP_RIGHT
                    end if
                end if
                break
            end switch
        end if
    next
end event

    temp0 = object.angle
    if object.state == LBLOCKS_ROTATE_RIGHT
        temp0 += 3
    end if

    temp0 >>= 2
    temp0 *= 3

    temp0++
    DrawSprite(temp0)

    temp0++
    DrawSprite(temp0)

    temp0++
    DrawSprite(temp0)

    DrawSprite(0)
end event

		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
		case 0 // block set
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
		case 0 // block set

			break

		end switch
	end if
end event

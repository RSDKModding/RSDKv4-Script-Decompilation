event ObjectUpdate
    // Extends the active range of the designated object
    arrayPos0 = object.entityPos
    arrayPos0 -= object.propertyValue
    object[arrayPos0].priority = PRIORITY_ACTIVE
end event

// Editor Events
// ========================

    DrawSprite(0)
end event

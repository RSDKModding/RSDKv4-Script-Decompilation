event ObjectUpdate
    if pushButton[+1].pressed == true
        if object.yvel > -0x10000
            object.yvel -= 512
        end if

        object.ypos += object.yvel
        if object.ypos < object.maxHeight
            object.ypos = object.maxHeight
            object.yvel = 0
        end if
    else
        if object.yvel < 0x10000
            object.yvel += 512
        end if

        object.ypos += object.yvel
        if object.ypos > object.minHeight
            object.ypos = object.minHeight
            object.yvel = 0
        end if
    end if

    pushButton[+1].xpos = object.xpos
    pushButton[+1].xpos += 0x100000

    pushButton[+1].ypos = object.ypos
    pushButton[+1].ypos -= 0x480000

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -32, -1, 0, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].ypos += 0x10000
            break

        case COL_BOTTOM
            if player[currentPlayer].gravity == GRAVITY_GROUND
                CallFunction(Player_Kill)
            end if
        end switch

        BoxCollisionTest(C_SOLID, object.entityPos, 0, -73, 32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].ypos += 0x10000
            break

        case COL_BOTTOM
            if player[currentPlayer].gravity == GRAVITY_GROUND
                CallFunction(Player_Kill)
            end if

        end switch
    next
end event

event ObjectStartup
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        temp0 = 163
        temp1 = 1
        temp2 = 196
        break

    case TIME_PAST
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 157
        temp2 = 34
        break

    case TIME_GOOD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 190
        temp2 = 34
        break

    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 223
        temp2 = 34
        break
    end switch

    // R4 Blocks
    SpriteFrame(0, -64, 32, 32, temp0, temp1)
    SpriteFrame(0, -32, 32, 32, temp0, temp1)
    SpriteFrame(-32, 0, 32, 32, temp0, temp1)
    SpriteFrame(0, 0, 32, 32, temp0, temp1)
    SpriteFrame(-32, 32, 32, 32, temp0, temp1)
    SpriteFrame(0, 32, 32, 32, temp0, temp1)

    foreach (TypeName[Button Blocks 1], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].minHeight = object[arrayPos0].ypos
        object[arrayPos0].maxHeight = object[arrayPos0].ypos
        object[arrayPos0].maxHeight -= 0x800000
    next
end event

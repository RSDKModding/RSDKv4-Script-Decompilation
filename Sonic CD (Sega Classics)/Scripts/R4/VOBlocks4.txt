// Aliases
private alias object.value0 : object.drawPos.y
private alias object.value1 : object.collisionOffset.y
private alias object.value2 : object.blockSheet
private alias object.value3 : object.spikeSheet

// Player Aliases
private alias object.gravity : player.gravity

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// Time Periods
private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
    object.angle = oscillation
    object.angle += object.propertyValue
    object.angle <<= 1

    Sin(object.collisionOffset.y, object.angle)
    object.collisionOffset.y <<= 13
    object.collisionOffset.y += object.ypos
    object.collisionOffset.y &= -0x10000
    object.collisionOffset.y -= object.drawPos.y

    temp0 = object.ypos
    object.ypos = object.drawPos.y

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -16, -64, 16, 62, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_BOTTOM
            if player[currentPlayer].gravity == GRAVITY_AIR
                CallFunction(Player_Hit)
            else
                CallFunction(Player_Kill)
            end if
        end if
    next

    object.ypos = temp0
end event


event ObjectDraw
    Sin(object.drawPos.y, object.angle)
    object.drawPos.y <<= 13
    object.drawPos.y += object.ypos
    object.drawPos.y &= -0x10000 // Truncate the value

    object.spriteSheet = object.blockSheet
    DrawSpriteXY(0, object.xpos, object.drawPos.y)
    DrawSpriteXY(1, object.xpos, object.drawPos.y)
    DrawSpriteXY(2, object.xpos, object.drawPos.y)

    object.spriteSheet = object.spikeSheet
    DrawSpriteXY(3, object.xpos, object.drawPos.y)
end event


event ObjectStartup
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        temp0 = 163
        temp1 = 1
        temp2 = 196
        break

    case TIME_PAST
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 157
        temp2 = 34
        break

    case TIME_GOOD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 190
        temp2 = 34
        break

    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 223
        temp2 = 34
        break
    end switch

    // R4 Blocks
    SpriteFrame(-16, -64, 32, 32, temp0, temp1)
    SpriteFrame(-16, -32, 32, 32, temp0, temp1)
    SpriteFrame(-16, 0, 32, 32, temp2, temp1)
    temp0 = object.spriteSheet

    LoadSpriteSheet("Global/Items3.gif")
    temp1 = object.spriteSheet

    SpriteFrame(-16, 32, 32, 32, 50, 100)  // #3 - Spikes

    foreach (TypeName[VOBlocks4], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].blockSheet = temp0
        object[arrayPos0].spikeSheet = temp1
    next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    CallFunction(EditorHelpers_FindTimePeriod)

    switch checkResult
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        break

    case TIME_PAST
    case TIME_GOOD_FUTURE
    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        break
    end switch

    temp6 = object.spriteSheet

    LoadSpriteSheet("Global/Items3.gif")
    temp7 = object.spriteSheet

    temp1 = oscillation
    temp1 += object.propertyValue
    temp1 <<= 1
    Sin(temp0, temp1)
    temp0 <<= 13
    temp0 += object.ypos
    temp0 &= 0xFFFF0000 // Truncate the value

    object.spriteSheet = temp6
    DrawSpriteXY(0, object.xpos, temp0)
    DrawSpriteXY(1, object.xpos, temp0)
    DrawSpriteXY(2, object.xpos, temp0)

    object.spriteSheet = temp7
    DrawSpriteXY(3, object.xpos, temp0)
end event

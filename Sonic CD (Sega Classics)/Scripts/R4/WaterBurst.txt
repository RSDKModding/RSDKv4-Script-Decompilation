event ObjectUpdate
    object.timer++
    if object.timer == 8
        Rand(temp0, 80)
        temp0 -= 40
        temp0 <<= 16
        temp0 += object.xpos

        Rand(temp1, 16)
        temp1 -= 8
        temp1 <<= 16
        temp1 += object.ypos
        temp1 -= 0x300000

        CreateTempObject(TypeName[Burst Splash], 0, temp0, temp1)
        Rand(burstSplash[tempObjectPos].xvel, 10)
        burstSplash[tempObjectPos].xvel -= 5
        burstSplash[tempObjectPos].xvel <<= 13
        burstSplash[tempObjectPos].yvel = -0x20000

        object.timer = 0
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -66, 48, -32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            player[currentPlayer].ypos += 0x20000
        end if

        BoxCollisionTest(C_TOUCH, object.entityPos, -128, -24, 128, 400, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            if player[currentPlayer].state != Player_State_Death
                if player[currentPlayer].state != Player_State_Drown
                    player[currentPlayer].state = Player_State_WaterCurrent
                    player[currentPlayer].animation = ANI_FANROTATE

                    player[currentPlayer].controlMode = CONTROLMODE_NONE
                    player[currentPlayer].left  = false
                    player[currentPlayer].right = false
                    player[currentPlayer].up    = false
                    player[currentPlayer].down  = false

                    player[currentPlayer].yvel = -0x100000
                    player[currentPlayer].timer = 0
                end if
            end if

            BoxCollisionTest(C_TOUCH, object.entityPos, -128, 48, 128, 332, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                temp0 = object.xpos
                temp0 -= player[currentPlayer].xpos
                temp0 >>= 2
                if temp0 > 0x100000
                    temp0 = 0x100000
                end if

                if temp0 < -0x100000
                    temp0 = -0x100000
                end if

                player[currentPlayer].xpos += temp0
            end if
        else
            if player[currentPlayer].state == Player_State_WaterCurrent
                player[currentPlayer].state = Player_State_Air
                player[currentPlayer].controlMode = currentPlayer
                player[currentPlayer].animation = ANI_BOUNCING

                player[currentPlayer].xvel = 0
                player[currentPlayer].speed = 0
            end if
        end if
    next
end event


event ObjectDraw
    object.ypos = stage.newWaterLevel
    object.ypos -= 0x300000

    temp0 = oscillation
    temp0 <<= 2
    Sin(temp1, temp0)
    temp1 <<= 9
    object.ypos += temp1

    temp0 = object.frameTimer
    temp0 /= 3
    temp0 += 8
    DrawSprite(temp0)

    temp0 += 4
    DrawSprite(temp0)

    temp0 += 4
    DrawSprite(temp0)

    temp0 += 4
    DrawSprite(temp0)

    object.frameTimer++
    object.frameTimer %= 12
    DrawSprite(object.frame)

    temp0 = object.frame
    temp0 += 4
    DrawSprite(temp0)

    object.frame++
    object.frame &= 3
end event

event ObjectUpdate
	object.xvel += 0x8000
	object.xvel %= 0x900000

	object.priority = PRIORITY_BOUNDS

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = object.xpos

		object.xpos -= 0xA00000
		object.xpos -= object.xvel

		temp1 = 0
		temp2 = 0
		temp3 = 0
		while temp1 < 4
			BoxCollisionTest(C_SOLID, object.entityPos, -32, -64, 32, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			switch checkResult
			case COL_TOP
				player[currentPlayer].xpos -= 0x8000
				if player[currentPlayer].speed == 0
					if object.stood == false
						object.stood = true

						// Truncate these values
						player[currentPlayer].xpos &= -0x10000
						object.xvel &= -0x10000
					end if
				else
					object.stood = false
				end if
				break

			case COL_LEFT
				temp2 = true
				break

			case COL_RIGHT
				temp3 = true
				break
			end switch

			object.xpos += 0x900000
			temp1++
		loop

		object.xpos = temp0
		object.xpos -= 0x1300000
		object.xpos += object.xvel

		temp1 = 0
		while temp1 < 4
			BoxCollisionTest(C_SOLID, object.entityPos, -32, 0, 32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			switch checkResult
			case COL_TOP
				player[currentPlayer].xpos += 0x8000
				if player[currentPlayer].speed == 0
					if object.stood == false
						object.stood = true

						// Truncate these values
						player[currentPlayer].xpos &= -0x10000
						object.xvel &= -0x10000
					end if
				else
					object.stood = false
				end if
				break

			case COL_LEFT
				temp2 = true
				break

			case COL_RIGHT
				temp3 = true
				break
			end switch

			object.xpos += 0x900000
			temp1++
		loop

		if player[currentPlayer].gravity == GRAVITY_AIR
			object.stood = false
		end if

		object.xpos = temp0
		BoxCollisionTest(C_SOLID, object.entityPos, -224, -128, -192, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		switch checkResult
		case COL_LEFT
			temp2 = true
			break

		case COL_RIGHT
			temp3 = true
			break
		end switch

		BoxCollisionTest(C_SOLID, object.entityPos, 192, -128, 224, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		switch checkResult
		case COL_LEFT
			temp2 = true
			break

		case COL_RIGHT
			temp3 = true
			break
		end switch

		if temp2 == true
			if temp3 == true
				CallFunction(Player_Kill)
			end if
		end if
	next
end event


event ObjectDraw
	temp0 = object.xpos
	temp0 -= 0xA00000
	temp0 -= object.xvel

	temp1 = object.ypos
	temp1 -= 0x200000
	temp2 = 0
	while temp2 < 4
		DrawSpriteXY(0, temp0, temp1)
		DrawSpriteXY(1, temp0, temp1)
		DrawSpriteXY(2, temp0, temp1)
		DrawSpriteXY(3, temp0, temp1)
		temp0 += 0x900000
		temp2++
	loop

	temp0 = object.xpos
	temp0 -= 0x1300000
	temp0 += object.xvel

	temp1 = object.ypos
	temp1 += 0x200000
	temp2 = 0
	while temp2 < 4
		DrawSpriteXY(0, temp0, temp1)
		DrawSpriteXY(1, temp0, temp1)
		DrawSpriteXY(2, temp0, temp1)
		DrawSpriteXY(3, temp0, temp1)
		temp0 += 0x900000
		temp2++
	loop
end event

		temp2 = 196

	case TIME_PAST
		LoadSpriteSheet("R4/Objects2.gif")
		temp0 = 1
		temp1 = 157
		temp2 = 34
		break

	case TIME_GOOD_FUTURE
		LoadSpriteSheet("R4/Objects2.gif")

	case TIME_BAD_FUTURE

end event

event RSDKDraw
	temp0 = object.xpos
	temp0 -= 0xA00000

	temp1 = object.ypos
	temp1 -= 0x200000
	temp2 = 0
	while temp2 < 4
		DrawSpriteXY(0, temp0, temp1)
		DrawSpriteXY(1, temp0, temp1)
		DrawSpriteXY(2, temp0, temp1)
		DrawSpriteXY(3, temp0, temp1)
		temp0 += 0x900000
		temp2++
	loop

	temp3 = temp0
	temp3 -= 0x900000

	temp0 = object.xpos
	temp0 -= 0x1300000
	DrawArrow(object.xpos, temp1, temp0, temp1, 255, 128, 0)

	temp1 = object.ypos
	temp1 += 0x200000
	temp2 = 0
	while temp2 < 4
		DrawSpriteXY(0, temp0, temp1)
		DrawSpriteXY(1, temp0, temp1)
		DrawSpriteXY(2, temp0, temp1)
		DrawSpriteXY(3, temp0, temp1)
		temp0 += 0x900000
		temp2++
	loop

	DrawArrow(object.xpos, temp1, temp0, temp1, 255, 128, 0)

	if editor.showGizmos == true
		editor.drawingOverlay = true

		temp2 = object.ypos
		temp2 -= 0x200000
		DrawArrow(temp3, temp2, object.xpos, temp2, 255, 255, 255)

		temp2 = object.xpos
		temp2 -= 0x1300000
		DrawArrow(temp2, temp1, object.xpos, temp1, 255, 255, 255)

		temp0 = object.xpos
		temp0 -= 0x1300000

		temp1 = object.ypos
		temp1 -= 0x400000
		DrawRectOutline(temp0, temp1, 0x240, 0x80, 255, 255, 0, 255)

		editor.drawingOverlay = false
	end if
end event

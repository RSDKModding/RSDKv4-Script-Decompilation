// Aliases
private alias object.value1 : object.drawPos.x
private alias object.value2 : object.collisionOffset.x
private alias object.value3 : object.againstBlock

// Player Aliases
private alias object.xpos : player.xpos

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// Time Periods
private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
    object.angle = oscillation
    object.angle += object.propertyValue
    object.angle <<= 1

    Cos(object.collisionOffset.x, object.angle)
    object.collisionOffset.x <<= 12
    object.collisionOffset.x += object.xpos
    object.collisionOffset.x &= -0x10000 // Truncate the value
    object.collisionOffset.x -= object.drawPos.x

    object.againstBlock = false

    temp0 = object.xpos
    object.xpos = object.drawPos.x

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_SOLID, object.entityPos, -48, -16, 48, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        switch checkResult
        case COL_TOP
            player[currentPlayer].xpos += object.collisionOffset.x
            break

        case COL_LEFT
            if object[-1].againstBlock == true
                CallFunction(Player_Kill)
            end if
            break

        case COL_RIGHT
            object.againstBlock = true
            break
        end switch
    next

    object.xpos = temp0
end event


event ObjectDraw
    Cos(object.drawPos.x, object.angle)
    object.drawPos.x <<= 12
    object.drawPos.x += object.xpos
    object.drawPos.x &= -0x10000 // Truncate the value
    DrawSpriteXY(0, object.drawPos.x, object.ypos)
    DrawSpriteXY(1, object.drawPos.x, object.ypos)
    DrawSpriteXY(2, object.drawPos.x, object.ypos)
end event


event ObjectStartup
    switch HUD.currentTimePeriod
    case TIME_PRESENT
        LoadSpriteSheet("R4/Objects.gif")
        temp0 = 163

        temp2 = 196
        break

    case TIME_PAST
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1
        temp1 = 157
        temp2 = 34
        break

    case TIME_GOOD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")
        temp0 = 1

        temp2 = 34
        break

    case TIME_BAD_FUTURE
        LoadSpriteSheet("R4/Objects2.gif")

        temp1 = 223
        temp2 = 34
        break
    end switch

    // R4 Blocks
    SpriteFrame(-48, -16, 32, 32, temp0, temp1)
    SpriteFrame(-16, -16, 32, 32, temp0, temp1)
    SpriteFrame(16, -16, 32, 32, temp0, temp1)
end event

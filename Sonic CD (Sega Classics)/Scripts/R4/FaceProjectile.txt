// ========================
// Aliases
// ========================

// Player Aliases
private alias object.yvel  : player.yvel
private alias object.speed : player.speed
private alias object.left  : player.left
private alias object.right : player.right

// ========================
// Events
// ========================

event ObjectUpdate
    object.xpos += object.xvel
    if object.outOfBounds == true
        object.type = TypeName[Blank Object]
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if player[currentPlayer].yvel > -0xE0000
            BoxCollisionTest(C_TOUCH, object.entityPos, -13, -8, 13, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if object.direction == FACING_RIGHT
                    if player[currentPlayer].left == true
                        if player[currentPlayer].speed > -0x30000
                            player[currentPlayer].speed = -0x30000
                        end if
                    else
                        if player[currentPlayer].right == true
                            player[currentPlayer].speed = -0x20000
                        else
                            player[currentPlayer].speed = -0x30000
                        end if
                    end if
                else
                    if player[currentPlayer].right == true
                        if player[currentPlayer].speed < 0x30000
                            player[currentPlayer].speed = 0x30000
                        end if
                    else
                        if player[currentPlayer].left == true
                            player[currentPlayer].speed = 0x20000
                        else
                            player[currentPlayer].speed = 0x30000
                        end if
                    end if
                end if
            end if
        end if
    next

    object.animationTimer++
    object.animationTimer %= 20
end event


event ObjectDraw
    if object.animationTimer > 9
        object.direction += 2
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        object.direction -= 2
    else
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects2.gif")
    SpriteFrame(-16, -8, 32, 16, 152, 159)  // #0 - Face Projectile
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R4/Objects2.gif")
    SpriteFrame(-16, -8, 32, 16, 152, 159)  // #0 - Face Projectile
    
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

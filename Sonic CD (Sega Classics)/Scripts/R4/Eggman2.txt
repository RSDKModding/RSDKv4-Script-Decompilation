public function EggBubble_HandleBubbleMovement
    arrayPos0 = 1
    temp0 = object.angle
    while arrayPos0 <= object.maxBubble
        if bubble2[+arrayPos0].state < BOSSBUBBLE2_CONSUMED
            bubble2[+arrayPos0].drawOrder = 3

            // Orbit-like movement
            Cos(temp1, temp0)
            temp1 *= object.bubbleRadius
            temp1 >>= 1

            // Horizontal movement
            Cos(bubble2[+arrayPos0].xpos, object.rotation)
            bubble2[+arrayPos0].xpos *= temp1
            bubble2[+arrayPos0].xpos >>= 9
            bubble2[+arrayPos0].xpos += object.xpos

            // Vertical movement
            Sin(bubble2[+arrayPos0].ypos, temp0)
            bubble2[+arrayPos0].ypos *= object.bubbleRadius
            bubble2[+arrayPos0].ypos >>= 1
            bubble2[+arrayPos0].ypos += object.ypos
        end if

        temp0 += object.bubbleOffset
        arrayPos0++
    loop
end function


public function EggBubble_HandleBubbleTurn
    arrayPos0 = 1
    temp0 = object.angle
    while arrayPos0 <= object.maxBubble
        if bubble2[+arrayPos0].state < BOSSBUBBLE2_CONSUMED
            if temp0 < 128
                bubble2[+arrayPos0].drawOrder = 3
            else
                if temp0 > 383
                    bubble2[+arrayPos0].drawOrder = 3
                else
                    bubble2[+arrayPos0].drawOrder = 2
                end if
            end if

            Cos(temp1, temp0)
            temp1 *= object.bubbleRadius
            temp1 >>= 1

            Cos(bubble2[+arrayPos0].xpos, object.rotation)
            bubble2[+arrayPos0].xpos *= temp1
            bubble2[+arrayPos0].xpos >>= 9
            bubble2[+arrayPos0].xpos += object.xpos

            Sin(bubble2[+arrayPos0].ypos, temp0)
            bubble2[+arrayPos0].ypos *= object.bubbleRadius
            bubble2[+arrayPos0].ypos >>= 1
            bubble2[+arrayPos0].ypos += object.ypos
        end if

        temp0 += object.bubbleOffset
        temp0 &= 511
        arrayPos0++
    loop
end function


public function EggBubble_CheckShieldCondition
    arrayPos0 = 1
    temp0 = 0
    while arrayPos0 <= object.maxBubble
        if bubble2[+arrayPos0].type == TypeName[Boss Bubble 2]
            temp0++
        end if
        arrayPos0++
    loop

    checkResult = false
    switch temp0
    case 3
        if object.maxBubble > 4
            checkResult = true
        end if
        break
    case 4
    case 5
    case 6
    case 7
        if temp0 < object.maxBubble
            checkResult = true
        end if
        break
    case 8
    case 9
    case 10
    case 11
    case 12
    case 13
    case 14
        temp1 = object.maxBubble
        temp1 -= temp0
        if temp1 > 1
            checkResult = true
        end if
        break
    end switch
end function


public function EggBubble_CheckBubbleAmount
    arrayPos0 = 1
    temp0 = 0
    while arrayPos0 <= object.maxBubble
        if bubble2[+arrayPos0].type == TypeName[Boss Bubble 2]
            temp0++
        end if
        arrayPos0++
    loop

    arrayPos0 = 1
    while arrayPos0 <= object.maxBubble
        if arrayPos0 <= temp0
            bubble2[+arrayPos0].type = TypeName[Boss Bubble 2]
            bubble2[+arrayPos0].state = BOSSBUBBLE2_BREATHABLE
            bubble2[+arrayPos0].timer = 0
        else
            bubble2[+arrayPos0].type = TypeName[Blank Object]
        end if
        arrayPos0++
    loop

    object.maxBubble = temp0
    object.bubbleOffset = 512
    object.bubbleOffset /= object.maxBubble
end function


public function EggBubble_DestroyBubbles
    arrayPos0 = 1
    while arrayPos0 <= object.maxBubble
        if bubble2[+arrayPos0].type == TypeName[Boss Bubble 2]
            bubble2[+arrayPos0].state = BOSSBUBBLE2_CONSUMED
            bubble2[+arrayPos0].timer = 0
        end if
        arrayPos0++
    loop
end function

event ObjectUpdate
    switch object.state
    case R4_EGG_BUBBLE_PLAYER_APPROACH
        stage.newXBoundary1 = object.ixpos
        stage.newXBoundary1 -= 96
        stage.newXBoundary1 -= screen.xcenter

        stage.newXBoundary2 = stage.newXBoundary1
        stage.newXBoundary2 += screen.xcenter
        stage.newXBoundary2 += screen.xcenter

        BoxCollisionTest(C_TOUCH, object.entityPos, -80, -256, 64, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.state = R4_EGG_BUBBLE_INTRO

            helper.xpos = object.xpos
            helper.xpos -= 0x600000
            helper.ypos = object.ypos

            CreateTempObject(TypeName[Fade Music], FADEMUSIC_TO_BOSS, object.xpos, object.ypos)
        end if
        break

    case R4_EGG_BUBBLE_INTRO
        if object.timer < 60
            object.timer++
        else
            object.timer = 0
            object.bubbleSelector = 0
            object.state = R4_EGG_BUBBLE_SPAWN_BUBBLES
        end if
        break

    case R4_EGG_BUBBLE_SPAWN_BUBBLES
        if object.timer < 320
            temp0 = object.timer
            temp0 %= 20
            if temp0 == 0
                PlaySfx(SfxName[Drowning], false)
            end if

            if object.bubbleSpawnTimer > 0
                object.bubbleSpawnTimer--
            else
                object.bubbleSpawnTimer = 10

                CreateTempObject(TypeName[Boss Bubble 1], 0, object.xpos, object.ypos)
                Rand(bubble1[tempObjectPos].xpos, 28)
                bubble1[tempObjectPos].xpos -= 14
                bubble1[tempObjectPos].xpos <<= 16
                bubble1[tempObjectPos].xpos += object.xpos
                bubble1[tempObjectPos].offset.x = bubble1[tempObjectPos].xpos
                bubble1[tempObjectPos].ypos += 0x9C0000

                Rand(bubble1[tempObjectPos].shift.x, 4)
                bubble1[tempObjectPos].shift.x += 9

                bubble1[tempObjectPos].bossEntity = object.entityPos

                if object.bubbleSelector == 1
                    Rand(bubble1[tempObjectPos].radiusSize, 32)
                    bubble1[tempObjectPos].radiusSize -= 16
                    bubble1[tempObjectPos].radiusSize <<= 16
                else
                    bubble1[tempObjectPos].radiusSize = 0
                end if

                object.bubbleSelector++
                object.bubbleSelector &= 1
            end if
            object.timer++
        else
            object.state = R4_EGG_BUBBLE_DELAY
            object.timer = 0
        end if
        break

    case R4_EGG_BUBBLE_DELAY
        if object.timer < 180
            object.timer++
        else
            object.timer     = 0
            object.maxBubble = 16
            object.angle     = 0

            object.bubbleOffset = 32
            object.bubbleRadius = 0x1000

            object.rotation = 256
            object.state    = R4_EGG_BUBBLE_EXPAND_SHIELD

            arrayPos0 = 1
            while arrayPos0 < 17
                bubble2[+arrayPos0].type = TypeName[Boss Bubble 2]
                arrayPos0++
            loop

            CallFunction(EggBubble_HandleBubbleMovement)
        end if
        break

    case R4_EGG_BUBBLE_EXPAND_SHIELD
        object.angle += 8
        object.angle &= 511
        if object.bubbleRadius < 0x3000
            object.bubbleRadius += 96
        else
            object.state = R4_EGG_BUBBLE_START_BATTLE
        end if

        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_START_BATTLE
        object.angle += 2
        object.angle &= 511

        if object.timer < 20
            object.timer++
        else
            object.state = R4_EGG_BUBBLE_ATTACK_R_TO_L
            helper.angle = 0
        end if

        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_ATTACK_R_TO_L
        Cos(object.xpos, helper.angle)
        object.xpos *= 0x3000
        object.xpos += helper.xpos

        Sin(object.ypos, helper.angle)
        object.ypos *= 0x2000
        object.ypos += helper.ypos

        if helper.angle == 12
            CallFunction(EggBubble_FireBullet)
        end if

        if helper.angle < 256
            helper.angle += 2
        else
            object.state = R4_EGG_BUBBLE_TURN_TO_RIGHT
        end if

        object.angle += 2
        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_TURN_TO_RIGHT
        object.angle += 8
        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleTurn)

        if object.rotation == 128
            object.direction = FACING_LEFT
        end if

        if object.rotation > 0
            object.rotation -= 2
        else
            CallFunction(EggBubble_CheckShieldCondition)
            if checkResult == true
                object.state = R4_EGG_BUBBLE_REBUILD_SHIELD_L
                object.timer = 0
            else
                object.state = R4_EGG_BUBBLE_ATTACK_L_TO_R
            end if
        end if
        break

    case R4_EGG_BUBBLE_ATTACK_L_TO_R
        Cos(object.xpos, helper.angle)
        object.xpos *= 0x3000
        object.xpos += helper.xpos

        Sin(object.ypos, helper.angle)
        object.ypos *= 0x2000
        object.ypos += helper.ypos

        if helper.angle == 244
            CallFunction(EggBubble_FireBullet)
        end if

        if helper.angle > 0
            helper.angle -= 2
        else
            object.state = R4_EGG_BUBBLE_TURN_TO_LEFT
        end if

        object.angle += 2
        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_TURN_TO_LEFT
        object.angle += 8
        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleTurn)

        if object.rotation == 128
            object.direction = FACING_RIGHT
        end if
        if object.rotation < 256
            object.rotation += 2
        else
            CallFunction(EggBubble_CheckShieldCondition)
            if checkResult == true
                object.state = R4_EGG_BUBBLE_REBUILD_SHIELD_R
                object.timer = 0
            else
                object.state = R4_EGG_BUBBLE_ATTACK_R_TO_L
            end if
        end if
        break

    case R4_EGG_BUBBLE_REBUILD_SHIELD_L
        object.xpos += 0x10000
        if object.timer < 20
            object.timer++
            object.angle += 2
            object.angle &= 511
            CallFunction(EggBubble_HandleBubbleMovement)

            if object.timer == 20
                object.animation = EGGANI_LAUGH
            end if
        else
            if object.bubbleRadius > 0xC00
                object.bubbleRadius -= 256
            else
                object.state = R4_EGG_BUBBLE_MOVE_TO_R
                CallFunction(EggBubble_CheckBubbleAmount)
                CallFunction(EggBubble_FireBullet)
            end if

            object.angle += 8
            object.angle &= 511
            CallFunction(EggBubble_HandleBubbleMovement)
        end if
        break

    case R4_EGG_BUBBLE_MOVE_TO_R
        object.xpos += 0x10000
        if object.bubbleRadius < 0x3000
            object.bubbleRadius += 96
            object.angle += 8
        else
            object.angle += 2
        end if

        temp0 = helper.xpos
        temp0 += 0x600000
        if object.xpos > temp0
            object.xpos = temp0
            helper.angle = 0
            object.state = R4_EGG_BUBBLE_TURN_TO_LEFT
        end if

        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_REBUILD_SHIELD_R
        object.xpos -= 0x10000
        if object.timer < 20
            object.timer++
            object.angle += 2
            object.angle &= 511
            CallFunction(EggBubble_HandleBubbleMovement)

            if object.timer == 20
                object.animation = EGGANI_LAUGH
            end if
        else
            if object.bubbleRadius > 0xC00
                object.bubbleRadius -= 256
            else
                object.state = R4_EGG_BUBBLE_MOVE_TO_L
                CallFunction(EggBubble_CheckBubbleAmount)
                CallFunction(EggBubble_FireBullet)
            end if

            object.angle += 8
            object.angle &= 511
            CallFunction(EggBubble_HandleBubbleMovement)
        end if
        break

    case R4_EGG_BUBBLE_MOVE_TO_L
        object.xpos -= 0x10000
        if object.bubbleRadius < 0x3000
            object.bubbleRadius += 96
            object.angle += 8
        else
            object.angle += 2
        end if

        temp0 = helper.xpos
        temp0 -= 0x600000
        if object.xpos < temp0
            object.xpos = temp0
            helper.angle = 256
            object.state = R4_EGG_BUBBLE_TURN_TO_RIGHT
        end if

        object.angle &= 511
        CallFunction(EggBubble_HandleBubbleMovement)
        break

    case R4_EGG_BUBBLE_EXPLODE
        if object.timer < 70
            temp0 = object.timer
            temp0 &= 3
            if temp0 == 0
                Rand(temp0, 80)
                temp0 -= 40
                temp0 <<= 16
                temp0 += object.xpos

                Rand(temp1, 56)
                temp1 -= 28
                temp1 <<= 16
                temp1 += object.ypos

                CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
                PlaySfx(SfxName[Explosion], false)
            end if

            object.timer++
        else
            object.state     = R4_EGG_BUBBLE_DEFEAT
            object.animation = EGGANI_DEFEAT
            object.timer     = 0
            object.direction = FACING_LEFT
            player.score += 1000
            CallFunction(EggBubble_DestroyBubbles)
        end if
        break

    case R4_EGG_BUBBLE_DEFEAT
        if object.timer < 70
            object.timer++
        else
            object.timer    = 0
            object.priority = PRIORITY_ACTIVE

            object.state      = R4_EGG_BUBBLE_FLEE
            object.animation  = EGGANI_ESCAPE
            object.frameTimer = 0

            stage.newWaterLevel = 0x8000000 // Y - 2048 in map coords
            CreateTempObject(TypeName[Fade Music], FADEMUSIC_TO_LEVEL, object.xpos, object.ypos)
        end if
        break

    case R4_EGG_BUBBLE_FLEE
        object.xpos += 0x1C000
        if object.outOfBounds == true
            object.state     = R4_EGG_BUBBLE_RESTORE_BOUNDS
            object.animation = EGGANI_NONE
        end if
        break

    case R4_EGG_BUBBLE_RESTORE_BOUNDS
        if object.timer < 60
            stage.curXBoundary2 += 4
            object.timer++
        else
            ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
            stage.newXBoundary2 = 0xF00
        end if
        break
    end switch

    if object.state > R4_EGG_BUBBLE_EXPAND_SHIELD
        if object.state < R4_EGG_BUBBLE_EXPLODE
            foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
                BoxCollisionTest(C_TOUCH, object.entityPos, -24, -16, 24, 16, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
                if checkResult == true
                    CheckEqual(player[currentPlayer].animation, ANI_JUMPING)
                    temp0 = checkResult

                    CheckEqual(player[currentPlayer].animation, ANI_SPINDASH)
                    temp0 |= checkResult

                    arrayPos0 = currentPlayer
                    arrayPos0 += playerCount
                    if object[arrayPos0].type == TypeName[Invincibility]
                        temp0 = true
                    end if

                    if temp0 == true
                        FlipSign(player[currentPlayer].xvel)
                        FlipSign(player[currentPlayer].speed)
                        player[currentPlayer].xvel >>= 1
                        player[currentPlayer].speed >>= 1

                        player[currentPlayer].yvel = player[currentPlayer].ypos
                        player[currentPlayer].yvel -= object.ypos
                        player[currentPlayer].yvel >>= 3

                        object.state     = R4_EGG_BUBBLE_EXPLODE
                        object.animation = EGGANI_HURT
                        object.timer     = 0
                    else
                        if player[currentPlayer].invincibleTimer == 0
                            player[currentPlayer].state = Player_State_GotHit
                            if player[currentPlayer].xpos > object.xpos
                                player[currentPlayer].speed = 0x20000
                            else
                                player[currentPlayer].speed = -0x20000
                            end if
                        end if

                        object.animation = EGGANI_LAUGH
                    end if
                end if
            next
        end if
    end if

    if object.state < R4_EGG_BUBBLE_RESTORE_BOUNDS
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            temp0 = player[currentPlayer].collisionRight
            temp0 <<= 16
            temp0 += player[currentPlayer].xpos
            temp1 = stage.curXBoundary2
            temp1 <<= 16
            if temp0 > temp1
                player[currentPlayer].xvel = 0
                player[currentPlayer].speed = 0
                player[currentPlayer].xpos = temp1
                temp0 = player[currentPlayer].collisionRight
                temp0 <<= 16
                player[currentPlayer].xpos -= temp0
            end if
        next
    end if
end event


event ObjectDraw
    switch object.animation
    case EGGANI_IDLE
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        break

    case EGGANI_LAUGH
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)

        temp0 = object.frameTimer
        temp0 &= 15
        temp0 >>= 3
        temp0 += 3
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

        if object.frameTimer < 112
            object.frameTimer++
        else
            object.frameTimer = 0
            object.animation = EGGANI_IDLE
        end if
        break

    case EGGANI_HURT
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        break

    case EGGANI_DEFEAT
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)
        break

    case EGGANI_ESCAPE
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)

        temp0 = object.frameTimer
        temp0 >>= 2
        temp0 += 7
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

        object.frameTimer++
        object.frameTimer &= 15
        break
    end switch
end event

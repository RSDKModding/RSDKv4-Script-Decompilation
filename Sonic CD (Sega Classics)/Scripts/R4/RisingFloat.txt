// Aliases
private alias object.value1 : object.originPos.x
private alias object.value2 : object.originPos.y

// Player Aliases
private alias object.xpos  : player.xpos
private alias object.ypos  : player.ypos
private alias object.xvel  : player.xvel
private alias object.speed : player.speed

// States
private alias 0 : RISINGFLOAT_INACTIVE
private alias 1 : RISINGFLOAT_RISE

// ========================
// Events
// ========================

event ObjectUpdate
    object.ypos = stage.waterLevel
    object.ypos <<= 16

    if object.state == RISINGFLOAT_RISE
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            object.xpos = player[currentPlayer].xpos
            if object.xpos < 0x10100000
                object.xpos = 0x10100000
            end if

            if object.xpos > 0x10F20000
                object.xpos = 0x10F20000
            else
                if player[currentPlayer].ypos > 0x5BE0000
                    if player[currentPlayer].ypos < object.ypos
                        ObjectTileCollision(CSIDE_LWALL, 16, 6, 0)
                        if checkResult == true
                            if player[currentPlayer].speed > 0x10000
                                player[currentPlayer].speed = 0x10000
                                player[currentPlayer].xvel = 0x10000
                            end if
                        end if

                        ObjectTileCollision(CSIDE_RWALL, -16, 6, 0)
                        if checkResult == true
                            if player[currentPlayer].speed < -0x10000
                                player[currentPlayer].speed = -0x10000
                                player[currentPlayer].xvel = -0x10000
                            end if
                        end if

                        player[currentPlayer].xpos = object.xpos
                    end if
                else
                    ObjectTileCollision(CSIDE_LWALL, 16, 0, 0)
                    ObjectTileCollision(CSIDE_RWALL, -16, 0, 0)
                end if
            end if
        next
    end if

    if object.outOfBounds == true
        temp0 = object.xpos
        temp1 = object.ypos

        object.xpos = object.originPos.x
        object.ypos = object.originPos.y

        if object.outOfBounds == true
            object.state    = RISINGFLOAT_INACTIVE
            object.priority = PRIORITY_BOUNDS
        else
            object.xpos = temp0
            object.ypos = temp1
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -10, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            player[currentPlayer].ypos += 0x20000
            if object.state == RISINGFLOAT_INACTIVE
                player[currentPlayer].speed = 0
                player[currentPlayer].xvel  = 0

                stage.newWaterLevel = 0x5E40000 // Y - 1508 in map coords

                object.state    = RISINGFLOAT_RISE
                object.priority = PRIORITY_ACTIVE
            end if
        end if
    next
end event


event ObjectDraw
    DrawSprite(0)
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects2.gif")
    SpriteFrame(-16, -8, 32, 16, 67, 235)  // #0 - Wood Log

    foreach (TypeName[Rising Float], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].originPos.x = object[arrayPos0].xpos
        object[arrayPos0].originPos.y = object[arrayPos0].ypos
    next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R4/Objects2.gif")
    SpriteFrame(-16, -8, 32, 16, 67, 235)  // #0 - Wood Log
    
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

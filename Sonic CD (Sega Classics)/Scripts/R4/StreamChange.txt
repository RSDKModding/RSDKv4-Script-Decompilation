// Aliases
private alias object.value1 : object.currentType

// Water Stream Alias
private alias object.value1 : object.oscillation

// Push Button alias
private alias object.propertyValue : pushButton.pressed

// Property Values
private alias 0 : SEARCH_NEXT_OBJ
private alias 1 : SEARCH_PREV_OBJ

// ========================
// Events
// ========================

event ObjectUpdate
    if pushButton[-2].pressed == true
        if object.propertyValue != object.currentType
            arrayPos0 = object.entityPos
            object[arrayPos0].currentType = object.propertyValue
            if object.propertyValue == SEARCH_NEXT_OBJ
                arrayPos0++
                while object[arrayPos0].type != TypeName[Stream Change]
                    if object[arrayPos0].type == TypeName[Water Stream L]
                        object[arrayPos0].type = TypeName[Water Stream R]
                    else
                        if object[arrayPos0].type == TypeName[Water Stream R]
                            object[arrayPos0].type = TypeName[Water Stream L]
                        end if
                    end if

                    if object[arrayPos0].type == TypeName[Water Stream U]
                        object[arrayPos0].type = TypeName[Water Stream D]
                    else
                        if object[arrayPos0].type == TypeName[Water Stream D]
                            object[arrayPos0].type = TypeName[Water Stream U]
                        end if
                    end if

                    if object[arrayPos0].type == TypeName[Water Stream C]
                        if object[arrayPos0].oscillation == 64
                            object[arrayPos0].oscillation = 192
                        else
                            object[arrayPos0].oscillation = 64
                        end if
                    end if

                    arrayPos0++
                loop
            else
                arrayPos0--
                while object[arrayPos0].type != TypeName[Stream Change]
                    if object[arrayPos0].type == TypeName[Water Stream L]
                        object[arrayPos0].type = TypeName[Water Stream R]
                    else
                        if object[arrayPos0].type == TypeName[Water Stream R]
                            object[arrayPos0].type = TypeName[Water Stream L]
                        end if
                    end if

                    if object[arrayPos0].type == TypeName[Water Stream U]
                        object[arrayPos0].type = TypeName[Water Stream D]
                    else
                        if object[arrayPos0].type == TypeName[Water Stream D]
                            object[arrayPos0].type = TypeName[Water Stream U]
                        end if
                    end if

                    if object[arrayPos0].type == TypeName[Water Stream C]
                        if object[arrayPos0].oscillation == 64
                            object[arrayPos0].oscillation = 192
                        else
                            object[arrayPos0].oscillation = 64
                        end if
                    end if

                    arrayPos0--
                loop
            end if

            object[arrayPos0].currentType = object.propertyValue
        end if
    end if
end event

event RSDKDraw
    temp0 = object.xpos
    temp1 = temp0
    temp0 -= 0x100000
    temp1 += 0x100000
    DrawSpriteXY(0, temp0, object.ypos)
    DrawSpriteXY(0, temp1, object.ypos)

    temp2 = object.ypos
    temp3 = temp2
    temp2 -= 0x80000
    temp3 += 0x80000
    DrawArrow(temp0, temp2, temp1, temp2, 0, 255, 0)
    DrawArrow(temp1, temp3, temp0, temp3, 255, 0, 0)
end event

    CreateTempObject(TypeName[Vertical Door], VDOOR_BUTTON_HOLD, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
    switch object.propertyValue
    case VDOOR_BUTTON_HOLD
        if pushButton[-1].pressed == false
            if object.timer > 0
                object.timer -= 0x80000
                object.ypos += 0x80000
            end if
        else
            if object.timer < 0x800000
                object.timer += 0x80000
                object.ypos -= 0x80000
            end if
        end if
        break

    case VDOOR_CLOSE_AFTER_PLAYER_ROOF
        if pushButton[-1].pressed == true
            object.state = VERTICALDOOR_ACTIVE
        end if

        temp0 = object.xpos
        temp0 += 0x180000
        if player[SLOT_PLAYER1].xpos > temp0
            object.state = VERTICALDOOR_INACTIVE
        end if

        if object.state == VERTICALDOOR_INACTIVE
            if object.timer > 0
                object.timer -= 0x80000
                object.ypos += 0x80000
            end if
        else
            if object.timer < 0x800000
                object.timer += 0x80000
                object.ypos -= 0x80000
            end if
        end if
        break

    case VDOOR_CLOSE_AFTER_PLAYER_FLOOR
        if pushButton[-1].pressed == true
            object.state = VERTICALDOOR_ACTIVE
        end if

        temp0 = object.xpos
        temp0 += 0x180000
        if player[SLOT_PLAYER1].xpos > temp0
            object.state = VERTICALDOOR_INACTIVE
        end if

        if object.state == VERTICALDOOR_INACTIVE
            if object.timer > 0
                object.timer -= 0x80000
                object.ypos -= 0x80000
            end if
        else
            if object.timer < 0x800000
                object.timer += 0x80000
                object.ypos += 0x80000
            end if
        end if
        break
    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        temp0 = player[currentPlayer].ypos
        BoxCollisionTest(C_SOLID2, object.entityPos, -8, -64, 8, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == COL_BOTTOM
            player[currentPlayer].ypos = temp0
        end if
    next
end event


event ObjectDraw
    DrawSprite(0)
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects.gif")
    SpriteFrame(-8, -64, 16, 128, 1, 69)  // #0 - Vertical Door

    SetTableValue(TypeName[Vertical Door], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(VerticalDoor_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(VerticalDoor_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
    DebugMode_ObjCount++
end event

event ObjectUpdate
    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.propertyValue
        case TRIGGER_FROM_FAN
            // Triggers from a fan
            if Fan[-1].active == true
                BoxCollisionTest(C_TOUCH, object.entityPos, -112, -64, 0, 160, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    if player[currentPlayer].state != Player_State_Death
                        if player[currentPlayer].state != Player_State_Drown
                            player[currentPlayer].state = Player_State_WaterCurrent
                            player[currentPlayer].animation = ANI_FANROTATE

                            player[currentPlayer].controlMode = CONTROLMODE_NONE
                            player[currentPlayer].left  = false
                            player[currentPlayer].right = false
                            player[currentPlayer].up    = false
                            player[currentPlayer].down  = false

                            if player[currentPlayer].ypos < object.ypos
                                player[currentPlayer].xvel = -0x40000
                                player[currentPlayer].speed = -0x40000
                            else
                                player[currentPlayer].xvel = 0
                                player[currentPlayer].speed = 0
                            end if

                            player[currentPlayer].yvel = -0x40000
                        end if
                    end if
                else
                    if player[currentPlayer].state == Player_State_WaterCurrent
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
            break

        case STREAM_RIGHT_XBOUNDS
            BoxCollisionTest(C_TOUCH, object.entityPos, -128, 0, 112, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Death
                    if player[currentPlayer].state != Player_State_Drown
                        player[currentPlayer].state = Player_State_WaterCurrent
                        player[currentPlayer].animation = ANI_FANROTATE

                        player[currentPlayer].controlMode = CONTROLMODE_NONE
                        player[currentPlayer].left  = false
                        player[currentPlayer].right = false
                        player[currentPlayer].up    = false
                        player[currentPlayer].down  = false

                        player[currentPlayer].xvel  = 0x60000
                        player[currentPlayer].speed = 0x60000
                        player[currentPlayer].yvel  = 0x20000
                    end if
                end if
            else
                if player[currentPlayer].state == Player_State_WaterCurrent
                    temp0 = object.xpos
                    temp0 += 0x840000
                    if player[currentPlayer].xpos > temp0
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
            break

        case STREAM_LEFT_XBOUNDS
            BoxCollisionTest(C_TOUCH, object.entityPos, -128, 0, 112, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Death
                    if player[currentPlayer].state != Player_State_Drown
                        player[currentPlayer].state = Player_State_WaterCurrent
                        player[currentPlayer].animation = ANI_FANROTATE

                        player[currentPlayer].controlMode = CONTROLMODE_NONE
                        player[currentPlayer].left  = false
                        player[currentPlayer].right = false
                        player[currentPlayer].up    = false
                        player[currentPlayer].down  = false

                        player[currentPlayer].xvel  = -0x60000
                        player[currentPlayer].speed = -0x60000
                        player[currentPlayer].yvel  = 0x20000
                    end if
                end if
            else
                if player[currentPlayer].state == Player_State_WaterCurrent
                    temp0 = object.xpos
                    temp0 += 0x840000
                    if player[currentPlayer].xpos > temp0
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
            break

        case STREAM_LEFT_YBOUNDS
            BoxCollisionTest(C_TOUCH, object.entityPos, -128, 0, 112, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Death
                    if player[currentPlayer].state != Player_State_Drown
                        player[currentPlayer].state = Player_State_WaterCurrent
                        player[currentPlayer].animation = ANI_FANROTATE

                        player[currentPlayer].controlMode = CONTROLMODE_NONE
                        player[currentPlayer].left  = false
                        player[currentPlayer].right = false
                        player[currentPlayer].up    = false
                        player[currentPlayer].down  = false

                        player[currentPlayer].xvel  = -0x60000
                        player[currentPlayer].speed = -0x60000
                        player[currentPlayer].yvel  = 0x20000
                    end if
                end if
            else
                if player[currentPlayer].ypos > object.ypos
                    if player[currentPlayer].state == Player_State_WaterCurrent
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
            break

        case STREAM_RIGHT
            BoxCollisionTest(C_TOUCH, object.entityPos, 0, -64, 256, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Death
                    if player[currentPlayer].state != Player_State_Drown
                        player[currentPlayer].state = Player_State_WaterCurrent
                        player[currentPlayer].animation = ANI_FANROTATE

                        player[currentPlayer].xvel  = 0x40000
                        player[currentPlayer].speed = 0x40000
                        player[currentPlayer].yvel  = 0
                    end if
                end if
            end if
            break

        case STREAM_PUSH_DOWN
            BoxCollisionTest(C_TOUCH, object.entityPos, -64, -160, 64, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                if player[currentPlayer].state != Player_State_Death
                    if player[currentPlayer].state != Player_State_Drown
                        player[currentPlayer].state = Player_State_WaterCurrent
                        player[currentPlayer].animation = ANI_FANROTATE

                        player[currentPlayer].controlMode = CONTROLMODE_NONE
                        player[currentPlayer].left  = false
                        player[currentPlayer].right = false
                        player[currentPlayer].up    = false
                        player[currentPlayer].down  = false

                        if player[currentPlayer].xpos < object.xpos
                            player[currentPlayer].xvel = 0x40000
                            player[currentPlayer].speed = 0x40000
                        else
                            player[currentPlayer].xvel = 0
                            player[currentPlayer].speed = 0
                        end if

                        player[currentPlayer].yvel = 0x40000
                    end if
                end if
            else
                if player[currentPlayer].ypos > object.ypos
                    if player[currentPlayer].state == Player_State_WaterCurrent
                        player[currentPlayer].state = Player_State_Air
                        player[currentPlayer].angle = 0
                        player[currentPlayer].controlMode = currentPlayer
                        player[currentPlayer].animation = ANI_WALKING
                    end if
                end if
            end if
            break
        end switch
    next
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 5
            break

        case 0 // type
            checkResult = object.propertyValue
            checkResult &= 5
            break

        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 5
            break

        case 0 // type
            object.propertyValue = editor.variableValue
            object.propertyValue &= 5
            break

        end switch
    end if
end event


event RSDKDraw
    object.inkEffect = INK_ALPHA
    object.alpha = 198

    LoadSpriteSheet("Global/Display.gif")
    temp5 = object.spriteSheet

    LoadSpriteSheet("Players/Sonic3.gif")
    temp6 = object.spriteSheet

    DrawSpriteFX(3, FX_INK, object.xpos, object.ypos)

    switch object.propertyValue
    case TRIGGER_FROM_FAN
        if Editor.ShowGizmos == true
            if object[-1].Type == TypeName[Fan]
                DrawLine(object.xpos, object.ypos, object[-1].xpos, object.ypos, 0, 255, 255)
                DrawLine(object[-1].xpos, object.ypos, object[-1].xpos, object[-1].ypos, 0, 255, 255)
            end if
        end if

        temp0 = object.xpos
        temp0 -= 0x400000
        temp1 = object.ypos
        temp1 -= 0x400000
        DrawArrow(object.xpos, object.ypos, temp0, temp1, 255, 255, 0)
        break

    case STREAM_RIGHT_XBOUNDS
    case STREAM_LEFT_YBOUNDS
        temp0 = object.xpos
        temp0 += 0x600000
        temp1 = object.ypos
        temp1 += 0x200000
        DrawArrow(object.xpos, object.ypos, temp0, temp1, 255, 255, 0)
        break

    case STREAM_LEFT_XBOUNDS
        temp0 = object.xpos
        temp0 -= 0x600000
        temp1 = object.ypos
        temp1 += 0x200000
        DrawArrow(object.xpos, object.ypos, temp0, temp1, 255, 255, 0)
        break

    case STREAM_RIGHT
        temp0 = object.xpos
        temp0 += 0x400000
        DrawArrow(object.xpos, object.ypos, temp0, object.ypos, 255, 255, 0)
        break

    case STREAM_PUSH_DOWN
        temp0 = object.xpos
        temp0 += 0x400000
        temp1 = object.ypos
        temp1 += 0x400000
        DrawArrow(object.xpos, object.ypos, temp0, temp1, 255, 255, 0)
        break
    end switch
end event

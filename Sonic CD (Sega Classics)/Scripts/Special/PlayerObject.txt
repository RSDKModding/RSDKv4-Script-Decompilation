public function Player_HandleMovement
	// Update Sonic's tilt based on what directions the player is currently holding
	// Left takes priority over right, similarly to the maingame
	if player.left == true
		player.tilt--

		// Min tilt of -8
		if player.tilt < -8
			player.tilt = -8
		end if
	else
		if player.right == true
			player.tilt++

			// Max tilt of 8
			if player.tilt > 8
				player.tilt = 8
			end if
		else
			// Neither left nor right are held, restore Sonic's tilt to neutral position
			if player.tilt > 0
				player.tilt--
			end if

			if player.tilt < 0
				player.tilt++
			end if
		end if
	end if

	if player.left == true
		player.angle += 2
	end if

	if player.right == true
		player.angle -= 2
	end if

	if player.angle < 0
		// It's worth noting, Angle uses a 512-based value, which is why this is "allowed"
		player.angle += 512
	end if

	player.angle &= 511

	// Update movement based on the player's angle
	Sin(temp0, player.angle)
	temp0 *= player.speed
	temp0 >>= 9
	player.xpos += temp0

	Cos(temp0, player.angle)
	temp0 *= player.speed
	temp0 >>= 9
	player.zpos += temp0
end function


public function Player_ProcessAnimation
	// This function is used for animating the object, here's a short overview of the values it uses
	// (Frame values are in accordance with SpriteFrames set in ObjectStartup)
	// -> player.animationSpeed is the speed at which Sonic should animate, it's added to his timer every frame
	// -> player.frameTimer is the timer used for animating Sonic, think of it akin to object.animationTimer
	//     - The next frame is triggered whenever the value is 240 or above, the speed is controlled by player.animationSpeed
	//     - This doesn't get reset whenever the frame transitions, so keep that in mind too
	// -> player.frameEnd is the final frame of the animation
	// -> player.frameLoop is the loop point for the animation to go back to after reaching its end
	// -> player.frame is the frame to be displayed when drawing the object

	// For each animation's corresponding values, check out the ANI_* aliases up above
	player.frameTimer += player.animationSpeed
	if player.frameTimer > 239
		player.frameTimer -= 240

		player.frame++
		if player.frame > player.frameEnd
			player.frame = player.frameLoop
		end if
	end if
end function


public function Player_BumperInteraction
	// Preconditions:
	// - temp0 is player's truncated xpos, and
	// - temp1 is truncated zpos

	// Reset temp2, it's gonna get used as a bitfield for where bumper collision has been sensed,
	// see below for which bits correspond to what "sensors"
	temp2 = 0

	// Check upper-left tile, uses the first bit
	temp0 -= 8
	temp1 -= 8
	Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)
	if checkResult == 3
		SetBit(temp2, 0, true)
	end if

	// Check upper-right tile, uses the second bit
	temp0 += 16
	Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)
	if checkResult == 3
		SetBit(temp2, 1, true)
	end if

	// Check bottom left tile, uses the third bit
	temp0 -= 16
	temp1 += 16
	Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)
	if checkResult == 3
		SetBit(temp2, 2, true)
	end if

	// Check bottom right tile, uses the fourth bit
	temp0 += 16
	Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)
	if checkResult == 3
		SetBit(temp2, 3, true)
	end if

	// Was any collision sensed?
	if temp2 > 0
		temp3 = player.speed
		temp3 += 0x10000

		if player.timer != 32
			PlaySfx(SfxName[Bumper 2], false)
		end if

		player.timer = 32

		// Jump to the corresponding collision match, each match is labelled in accordance with what sensors are triggered
		switch temp2
		case 1 // Upper left tile only
			player.screenXPos = temp3
			player.screenYPos = temp3
			break

		case 2 // Upper right tile only
			player.screenXPos = temp3
			player.screenYPos = temp3
			FlipSign(player.screenXPos)
			break

		case 3 // Upper left tile and upper right
			player.screenXPos = 0
			player.screenYPos = temp3
			break

		case 4 // Bottom left tile only
			player.screenXPos = temp3
			player.screenYPos = temp3
			FlipSign(player.screenYPos)
			break

		case 5 // Upper left tile and bottom left tile
			player.screenXPos = temp3
			player.screenYPos = 0
			break

		case 6 // Upper right tile and bottom left tile
		case 7 // Upper left tile, upper right tile, and bottom left tile
			player.screenXPos = temp3
			player.screenYPos = temp3
			break

		case 8  // Bottom right tile only
		case 14 // Upper right tile, bottom left tile, and bottom right tile
		case 15 // Upper left tile, upper right tile, bottom left tile, and bottom right tile
			player.screenXPos = temp3
			player.screenYPos = temp3
			FlipSign(player.screenXPos)
			FlipSign(player.screenYPos)
			break

		case 9  // Upper left tile and bottom right tile
		case 11 // Upper left tile, upper right tile, and bottom right tile
			player.screenXPos = temp3
			player.screenYPos = temp3
			FlipSign(player.screenXPos)
			break

		case 10 // Upper right tile and bottom right tile
			player.screenXPos = temp3
			player.screenYPos = 0
			FlipSign(player.screenXPos)
			break

		case 12 // Bottom left tile and bottom right tile
			player.screenXPos = 0
			player.screenYPos = temp3
			FlipSign(player.screenYPos)
			break

		case 13 // Upper left tile, bottom left tile, and bottom right tile
			player.screenXPos = temp3
			player.screenYPos = temp3
			FlipSign(player.screenYPos)
			break

		end switch
	end if
end function


public function Player_TileInteraction
	// Get the player's truncated xpos and zpos for tile collision purposes
	temp0 = player.xpos
	temp0 >>= 16
	temp1 = player.zpos
	temp1 >>= 16

	// Pass the results over to the bumper function for bumping interations
	CallFunction(Player_BumperInteraction)

	// Get the player's truncated xpos and zpos for tile collision purposes (again)
	temp0 = player.xpos
	temp0 >>= 16
	temp1 = player.zpos
	temp1 >>= 16

	// Find what tile type the player is currently standing on
	Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)
	switch checkResult
	case 1
		// Offroad tile, slow down Sonic and kick up some dust
		object[3].type = TypeName[Dust Puff]

		// Note that we're giving it a Draw Order of 4, not only does this make it draw above Sonic but
		// more importantly this makes it so that it won't be treated as a 3D object
		object[3].drawOrder = 4

		if HUD.speedShoes == 0
			if player.speed > 0x28000
				// While on dust without speed shoes, the maximum speed is 2.5 px per frame
				player.speed = 0x28000
			end if
		else
			if player.speed > 0x50000
				// While on dust *with* speed shoes, the maximum speed is now bumped up to 5 px per frame
				player.speed = 0x50000
			end if
		end if
		break

	case 2
		// Water tile, slow down the player and make a splash
		object[3].type = TypeName[Water Splash]
		object[3].drawOrder = 4

		// Give the water splash a Z Pos of just a tad bit lower than Sonic, in order to make it draw behind him
		object[3].screenDepth = 0x57FE

		if HUD.speedShoes > 0
			if player.speed > 0x50000
				// If Sonic has speed shoes, then enforce a max speed of 5px per frame
				player.speed = 0x50000
			end if
		end if
		break

	// 3, the value used by the bumper tiles, is skipped here
	// It's handled in Player_BumperInteraction instead, called above

	case 4
		// Ouch, a Cruncher!
		// Make Sonic fall

		player.state = SSSONIC_TRIPPED
		player.timer = 136

		player.frame     = ANI_TRIP_START
		player.frameLoop = ANI_TRIP_LOOP
		player.frameEnd  = ANI_TRIP_END

		player.animationSpeed = 40
		player.frameTimer = 0

		// Go to a speed of 1px per frame
		player.speed = 0x10000

		if HUD.rings > 0
			PlaySfx(SfxName[Lose Rings], false)

			// Cut the player's rings in half
			temp0 = HUD.rings
			HUD.rings >>= 1

			// Find how many rings were lost
			temp0 -= HUD.rings
			if temp0 > 8
				temp0 = 8
			end if

			// If Sonic has 0 rings now, then reset the UFO streak
			if HUD.rings == 0
				HUD.lastUFOType = -1
			end if

			// Create all the dropped rings
			while temp0 > 0
				CreateTempObject(TypeName[Ring], 0, player.xpos, 0)
				object[tempObjectPos].priority = PRIORITY_ACTIVE

				// Move the Ring to Sonic
				// (XPos is matched already via object spawning function)
				Ring[tempObjectPos].zpos = player.zpos

				// Randomise the X Velocity
				Rand(temp1, 64)
				temp1 -= 32
				temp1 <<= 10
				Ring[tempObjectPos].xvel = temp1

				// Make the Rings fall in the general direction of Sonic, though
				Sin(temp1, player.angle)
				temp1 *= 96
				Ring[tempObjectPos].xvel += temp1

				// Randomise the Y Velocity as well
				Rand(temp1, 64)
				temp1 += 32
				temp1 <<= 12
				Ring[tempObjectPos].yvel = temp1

				// No further stuff needed for Y Velocity since there's really only one way for them to go - up!

				// Randomise the Ring's Z Velocity too
				Rand(temp1, 64)
				temp1 -= 32
				temp1 <<= 10
				Ring[tempObjectPos].zvel = temp1

				// And make it somewhat match Sonic's direction
				Cos(temp1, player.angle)
				temp1 *= 96
				Ring[tempObjectPos].zvel += temp1

				temp0--
			loop
		end if
		break

	case 5
		// Spring, send the player up, up, and away into the skies
		player.state = SSSONIC_SPRING
		player.timer = 0

		player.frame     = ANI_JUMPING_START
		player.frameLoop = ANI_JUMPING_LOOP
		player.frameEnd  = ANI_JUMPING_END

		player.animationSpeed = 80
		player.frameTimer = 0

		// Ascending at a rate of 8.75 pixels per frame
		player.yvel = 0x8C000

		PlaySfx(SfxName[Spring], false)
		break

	case 6
		// Fan, starting gliding
		player.state = SSSONIC_FAN
		player.timer = 0

		player.frameLoop = ANI_FAN_LOOP
		player.frameEnd  = ANI_FAN_END

		player.animationSpeed = 24

	case 7
		// Large arrow booster pad, facing left
		player.timer = 40

		// Get the tile's flip directions

// ========================
// Events
// ========================

event ObjectUpdate
	if HUD.speedShoes > 0
		HUD.speedShoes--
	end if

	switch player.state
	case SSSONIC_INTROPOSE
		CallFunction(Player_ProcessAnimation)
		player.timer++
		if player.timer == 120
			// Stop posing and start turning around
			player.timer = 0

			player.state = SSSONIC_INTROTURNAROUND

			player.frame     = ANI_FACINGAHEAD_START
			player.frameLoop = ANI_FACINGAHEAD_LOOP
			player.frameEnd  = ANI_FACINGAHEAD_END

			player.frameTimer = 0
			player.animationSpeed = 20
		end if
		break

	case SSSONIC_INTROTURNAROUND
		CallFunction(Player_ProcessAnimation)

		player.timer++
		if player.timer == 140
			// Fully turned around now and ready to go!
			player.timer = 0

			// Start advancing ahead
			player.state = SSSONIC_WALKING

			player.frame     = ANI_WALKING_START
			player.frameLoop = ANI_WALKING_LOOP
			player.frameEnd  = ANI_WALKING_END

			player.speed = 0
		end if
		break

	case SSSONIC_WALKING
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x50000
			player.speed += 0x800
		else
			player.speed = 0x50000
		end if

		if keyDown[0].down == true
			player.speed -= 0xC00

			if player.speed < 0x1E000
				player.speed = 0x1E000
			end if
		end if

		// Sonic's animation speed is dependant on his actual speed
		player.animationSpeed = player.speed
		player.animationSpeed *= 15
		player.animationSpeed /= 0x14000
		player.animationSpeed += 20

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Update Sonic's gimmick interaction timer
		if player.timer > 0
			player.timer--
			player.xpos += player.screenXPos
			player.zpos += player.screenYPos
		end if

		CallFunction(Player_TileInteraction)

		// Check for jumping
		if player.jumpPress == true
			player.state = SSSONIC_JUMPING
			player.timer = 0

			player.frame     = ANI_JUMPING_START
			player.frameLoop = ANI_JUMPING_LOOP
			player.frameEnd  = ANI_JUMPING_END

			player.frameTimer = 0
			player.animationSpeed = 80

			// Note: We've got a slightly stronger strength here compared to the original's 0x46000
			player.yvel = 0x58000
			PlaySfx(SfxName[Jump], false)
		end if
		break

	case SSSONIC_JUMPING
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x50000
			player.speed += 0x800
		end if

		if keyDown[0].down == true
			player.speed -= 0xC00

			if player.speed < 0x1E000
				player.speed = 0x1E000
			end if
		end if

		if player.jumpHold == false
			if player.yvel > 0x30000
				player.yvel = 0x30000
			end if
		end if

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Gravity of 0.125 per frame
		player.yvel -= 0x2800
		player.ypos += player.yvel

		// Touched the ground?
		if player.ypos < 0
			player.ypos = 0

			if HUD.ufosCount > 0
				if HUD.speedShoes == 0
					player.state     = SSSONIC_WALKING

					player.frame     = ANI_WALKING_START
					player.frameLoop = ANI_WALKING_LOOP
					player.frameEnd  = ANI_WALKING_END

					player.frameTimer = 0
				else
					player.state     = SSSONIC_SPEEDSHOESRUN

					player.frame     = ANI_RUN_START
					player.frameLoop = ANI_RUN_LOOP
					player.frameEnd  = ANI_RUN_END

					player.animationSpeed = 80
					player.frameTimer = 0
				end if
			else
				player.controlMode = CONTROLMODE_NONE
				player.state = SSSONIC_FINISHSTAND
				stage.timeEnabled = false
			end if
		end if
		break

	case SSSONIC_SPEEDBOOSTER
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x50000
			player.speed += 0x800
		end if

		if keyDown[0].down == true
			player.speed -= 0xC00

			if player.speed < 0x1E000
				player.speed = 0x1E000
			end if
		end if

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Update the booster timer
		if player.timer > 0
			player.timer--
			player.xpos += player.screenXPos
			player.zpos += player.screenYPos
		else
			// Booster's over, restore the player to normal
			if HUD.speedShoes == 0
				player.state     = SSSONIC_WALKING

				player.frame     = ANI_WALKING_START
				player.frameLoop = ANI_WALKING_LOOP
				player.frameEnd  = ANI_WALKING_END

				player.frameTimer = 0
			else
				player.state     = SSSONIC_SPEEDSHOESRUN

				player.frame     = ANI_RUN_START
				player.frameLoop = ANI_RUN_LOOP
				player.frameEnd  = ANI_RUN_END

				player.animationSpeed = 80
				player.frameTimer = 0
			end if
		end if

		CallFunction(Player_TileInteraction)

		// See if the player wants to jump out of the boost
		if player.jumpPress == true
			player.state = SSSONIC_JUMPING
			player.timer = 0

			player.frame     = ANI_JUMPING_START
			player.frameLoop = ANI_JUMPING_LOOP
			player.frameEnd  = ANI_JUMPING_END

			// Note: We've got a slightly stronger strength here compared to the original's 0x46000
			player.animationSpeed = 80
			player.yvel = 0x58000

			PlaySfx(SfxName[Jump], false)
		end if
		break

	case SSSONIC_FAN
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x50000
			player.speed += 0x800
		end if

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Gravity of 0.03125 pixels per frame, the fan provides some air resistance to lessen the normal gravity from 0.125 pixels per frame
		player.yvel -= 0x800
		player.ypos += player.yvel

		// Touched the ground?
		if player.ypos < 0
			player.ypos = 0

			if HUD.ufosCount > 0
				if HUD.speedShoes == 0
					player.state     = SSSONIC_WALKING

					player.frame     = ANI_WALKING_START
					player.frameLoop = ANI_WALKING_LOOP
					player.frameEnd  = ANI_WALKING_END

					player.frameTimer = 0
				else
					player.state     = SSSONIC_SPEEDSHOESRUN

					player.frame     = ANI_RUN_START
					player.frameLoop = ANI_RUN_LOOP
					player.frameEnd  = ANI_RUN_END

					player.animationSpeed = 80
					player.frameTimer = 0
				end if
			else
				player.state = SSSONIC_FINISHSTAND
				player.controlMode = CONTROLMODE_NONE
				stage.timeEnabled = false
			end if
		end if
		break

	case SSSONIC_TRIPPED
		CallFunction(Player_ProcessPlayer)

		// Lock movement
		player.left = false
		player.right = false

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Get the player's truncated xpos and zpos
		temp0 = player.xpos
		temp0 >>= 16

		temp1 = player.zpos
		temp1 >>= 16

		Get16x16TileInfo(checkResult, temp0, temp1, TILEINFO_ANGLEA)

		// If the player's hit a bumper tile, then get up and start walking again
		if checkResult == 3
			player.state     = SSSONIC_WALKING

			player.frame     = ANI_WALKING_START
			player.frameLoop = ANI_WALKING_LOOP
			player.frameEnd  = ANI_WALKING_END

			player.frameTimer = 0
		end if

		if player.timer > 0
			player.timer--
		else
			if HUD.speedShoes == 0
				player.state     = SSSONIC_WALKING

				player.frame     = ANI_WALKING_START
				player.frameLoop = ANI_WALKING_LOOP
				player.frameEnd  = ANI_WALKING_END

				player.frameTimer = 0
			else
				player.state     = SSSONIC_SPEEDSHOESRUN

				player.frame     = ANI_RUN_START
				player.frameLoop = ANI_RUN_LOOP
				player.frameEnd  = ANI_RUN_END

				player.animationSpeed = 80
				player.frameTimer = 0
			end if
		end if
		break

	case SSSONIC_SPEEDSHOESRUN
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x70000
			player.speed += 0x1000
		end if

		if keyDown[0].down == true
			player.speed -= 0xC00

			if player.speed < 0x1E000
				player.speed = 0x1E000
			end if
		end if

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		if player.timer > 0
			player.timer--
			player.xpos += player.screenXPos
			player.zpos += player.screenYPos
		end if

		CallFunction(Player_TileInteraction)

		if player.jumpPress == true
			player.state = SSSONIC_JUMPING
			player.timer = 0

			player.frame     = ANI_JUMPING_START
			player.frameLoop = ANI_JUMPING_LOOP
			player.frameEnd  = ANI_JUMPING_END

			player.frameTimer = 0
			player.animationSpeed = 80
			player.yvel = 0x58000

			PlaySfx(SfxName[Jump], false)
		end if

		if HUD.speedShoes == 0
			player.state     = SSSONIC_WALKING

			player.frame     = ANI_WALKING_START
			player.frameLoop = ANI_WALKING_LOOP
			player.frameEnd  = ANI_WALKING_END

			player.frameTimer = 0
		end if
		break

	case SSSONIC_FINISHSTAND
		player.frame = ANI_STANDING_START
		player.timer = 0
		player.speed = 0
		break

	case SSSONIC_CAMERAPAN
		// This state is given to Sonic from the Time Stone object, not from himself

		if player.timer < 128
			player.timer++
			player.angle -= 2
			if player.angle < 0
				player.angle += 512
			end if
		else
			if object[3].type == TypeName[Blank Object]
				// Spawn the Time Stone and place it 24 pixels above the screen
				ResetObjectEntity(3, TypeName[Time Stone], 0, 0, -0x180000)
				object[3].ixpos = screen.xcenter
				object[3].priority = PRIORITY_ACTIVE
			end if
		end if

		// Make Sonic's rotation based on how far into the pan we are
		// -> 81 is the starting Sprite Frame ID of the rotation frames
		player.frame = player.timer
		player.frame >>= 4
		player.frame += 81
		break

	case SSSONIC_STONEGRABBED
		if player.timer == 308
			// Spawn the Stage Results, as the "TIME STONES" variant
			object[30].type = TypeName[Stage Finish]
			object[30].resultsType = STAGEFINISH_T_STONEOBTAINED
		else
			player.timer++
		end if
		break

	case SSSONIC_SPRING
		CallFunction(Player_ProcessPlayer)

		if player.speed < 0x50000
			player.speed += 0x800
		end if

		if keyDown[0].down == true
			player.speed -= 0xC00

			if player.speed < 0x1E000
				player.speed = 0x1E000
			end if
		end if

		CallFunction(Player_HandleMovement)
		CallFunction(Player_ProcessAnimation)

		// Update gravity with an eight of a pixel per frame as a gravity value
		player.yvel -= 0x2000
		player.ypos += player.yvel

		// Touched the ground?
		if player.ypos < 0
			player.ypos = 0
			if HUD.ufosCount > 0
				if HUD.speedShoes == 0
					player.state    = SSSONIC_WALKING

					player.frame     = ANI_WALKING_START
					player.frameLoop = ANI_WALKING_LOOP
					player.frameEnd  = ANI_WALKING_END

					player.frameTimer = 0
				else
					player.state     = SSSONIC_SPEEDSHOESRUN

					player.frame     = ANI_RUN_START
					player.frameLoop = ANI_RUN_LOOP
					player.frameEnd  = ANI_RUN_END

					player.animationSpeed = 80
					player.frameTimer = 0
				end if
			else
				player.controlMode = CONTROLMODE_NONE
				player.state = SSSONIC_FINISHSTAND
			end if
		end if
		break

	end switch

	// Enforce stage bounds
	// These stage size values are set by the stage's BGEffects object on Startup
	if player.xpos > stage.curXBoundary2
		player.xpos = stage.curXBoundary2
	end if

	if player.xpos < stage.curXBoundary1
		player.xpos = stage.curXBoundary1
	end if

	// The Stage Bounds are setup for a 2D plane, which is why zpos is being paired with the "Y" Bounds here
	if player.zpos > stage.curYBoundary2
		player.zpos = stage.curYBoundary2
	end if

	if player.zpos < stage.curYBoundary1
		player.zpos = stage.curYBoundary1
	end if

	// So we're kind of not actually moving around a 3D stage - instead, we're moving the entire world around Sonic
	// So do the calculations for that
	tileLayer[0].angle = player.angle

	// X/Z movement
	Sin(tileLayer[0].xpos, tileLayer[0].angle)
	Cos(tileLayer[0].zpos, tileLayer[0].angle)
	tileLayer[0].xpos *= -0x2C00
	tileLayer[0].zpos *= -0x2C00
	tileLayer[0].xpos += player.xpos
	tileLayer[0].zpos += player.zpos

	// Y movement is much easier, no complex calcuations needed here
	// Just take Sonic's Y Position, smooth it out a bit, and offset it by 88 pixels
	tileLayer[0].ypos = player.ypos
	tileLayer[0].ypos /= 3
	tileLayer[0].ypos += 0x580000
end event


event ObjectDraw
	// First draw Sonic's shadow
	// Find his the ground position to draw the shadow on
	temp0 = tileLayer[0].ypos
	temp0 >>= 8
	temp0 *= 96
	temp0 /= 0x5800
	temp0 += 128

	// And now draw the shadow at that given spot
	DrawSpriteScreenXY(0, screen.xcenter, temp0)

	// And now draw Sonic himself
	// Get the Y position to draw him at
	temp0 = tileLayer[0].ypos
	temp0 -= player.ypos
	temp0 >>= 8
	temp0 *= 96
	temp0 /= 0x5800
	temp0 += 128

	// If in the walking animation, jump to its corresponding special drawing code,
	// otherwise just do the standard drawing sprite routine
	switch player.state
	default
		// Just draw Sonic's sprite, nothing special needed here
		DrawSpriteScreenXY(player.frame, screen.xcenter, temp0)
		break

	case SSSONIC_WALKING
		// Bump Sonic's sprite based on how "tilted" he is

		temp1 = player.frame

		temp2 = player.tilt
		temp2 >>= 2
		temp2 += 2

		// And now jump to the result
		// Alternate SpriteFrames are used, as well as sprite flipping too
		switch temp2
		case 0
			temp1 += 6
			player.direction = FACING_LEFT
			break

		case 1
			temp1 += 12
			player.direction = FACING_LEFT
			break

		case 2
			player.direction = FACING_RIGHT
			break

		case 3
			temp1 += 18
			player.direction = FACING_RIGHT
			break

		case 4
			temp1 += 24
			player.direction = FACING_RIGHT
			break

		end switch

		// And now draw the result sprite from that
		DrawSpriteScreenFX(temp1, FX_FLIP, screen.xcenter, temp0)
		break

	end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // StartDir
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // StartDir
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event

// ========================

// Aliases
private alias object.value0 : object.timer

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0	: BLUEBIRD_FLYRIGHT
private alias 1 : BLUEBIRD_FLYLEFT

// Time Period
private alias 2 : TIME_GOOD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
	// This Object is always running & updating, regardless of the Stage's current state
	// However, it will only draw itself under the specific conditions, though it is being
	// updated all the same

	switch object.state
	case BLUEBIRD_FLYRIGHT
		if object.angle > 0
			object.angle -= 2
		else
			// Apex hit, turn around
			object.state     = BLUEBIRD_FLYLEFT
			object.direction = FACING_LEFT
		end if
		break

	case BLUEBIRD_FLYLEFT
		if object.angle < 256
			object.angle += 2
		else
			// Other apex hit now, return to the other direction
			object.state     = BLUEBIRD_FLYRIGHT
			object.direction = FACING_RIGHT
		end if
		break

	end switch

	// Animate the Blue Bird
	object.timer++
	if object.timer > 19
		object.timer = 0
		object.frame++

		// The Bird bounces between 2 frames total
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == true
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if

	if HUD.currentTimePeriod == TIME_GOOD_FUTURE
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Global/Items3.gif")

	SpriteFrame(-8, -8, 16, 16, 240, 199)
	SpriteFrame(-8, -8, 16, 16, 240, 216)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Items3.gif")
	SpriteFrame(-8, -8, 16, 16, 240, 199)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

// ========================

// Aliases
private alias object.value0 : object.timer

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0	: FLYINGBIRD_FLYRIGHT
private alias 1 : FLYINGBIRD_FLYLEFT

// Time Period
private alias 2 : TIME_GOOD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
	// Note about the Object:
	// It's always running and animating, regardless of the stage's current state
	switch object.state
	case FLYINGBIRD_FLYRIGHT
		if object.angle > 0
			object.angle -= 2
		else
			// Hit the peak, turn around
			object.state     = FLYINGBIRD_FLYLEFT
			object.direction = FACING_LEFT
		end if
		break

	case FLYINGBIRD_FLYLEFT
		if object.angle < 256
			object.angle += 2
		else
			// Hit the other side of the peak, turn right around again
			object.state     = FLYINGBIRD_FLYRIGHT
			object.direction = FACING_RIGHT
		end if
		break

	end switch

	// Animate the object
	object.timer++
	if object.timer > 19
		object.timer = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == true
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if

	if HUD.currentTimePeriod == TIME_GOOD_FUTURE
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects.gif")

	// Bird Frames
	SpriteFrame(-8, -8, 16, 16, 34, 186)
	SpriteFrame(-8, -8, 16, 16, 34, 203)

	// Find all Flying Bird Objects in the stage
	foreach (TypeName[Flying Bird], arrayPos0, ALL_ENTITIES)
		// Set the Object's Draw Order to be above most other objects and the stage
		object[arrayPos0].drawOrder = 5
	next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R7/Objects.gif")
	SpriteFrame(-8, -8, 16, 16, 34, 186)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

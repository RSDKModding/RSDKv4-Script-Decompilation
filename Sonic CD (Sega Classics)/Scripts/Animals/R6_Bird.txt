// ========================
// Aliases

// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.angle

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0	: FLYINGBIRD_FLYRIGHT
private alias 1 : FLYINGBIRD_FLYLEFT

// Time Period
private alias 2 : TIME_GOOD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case FLYINGBIRD_FLYRIGHT
		if object.angle > 0
			object.angle -= 2
		else
			// Apex reached, turn around
			object.state     = FLYINGBIRD_FLYLEFT
			object.direction = FACING_LEFT
		end if
		break

	case FLYINGBIRD_FLYLEFT
		if object.angle < 256
			object.angle += 2
		else
			// Crest of the path reached, turn right back around again
			object.state     = FLYINGBIRD_FLYRIGHT
			object.direction = FACING_RIGHT
		end if
		break
	end switch

	// Animate the Bird
	object.timer++
	if object.timer > 19
		object.timer = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == true
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	else
		if HUD.currentTimePeriod == TIME_GOOD_FUTURE
			Cos(temp0, object.angle)
			temp0 <<= 14
			temp0 += object.xpos

			Sin(temp1, object.angle)
			temp1 <<= 14
			temp1 += object.ypos

			DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R6/Objects.gif")

	SpriteFrame(-12, -8, 24, 16, 197, 34)
	SpriteFrame(-12, -8, 24, 16, 197, 51)

	// Find and setup all Flying Bird Objects in the Level
	foreach (TypeName[Flying Bird], arrayPos0, ALL_ENTITIES)
		// Make all Flying Birds draw ontop of other objects & level tiles
		object[arrayPos0].drawOrder = 5
	next
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R6/Objects.gif")
	SpriteFrame(-12, -8, 24, 16, 197, 34)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

// ========================
// Aliases

// Aliases
private alias object.value0 : object.timer

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0	: BIRD_FLYRIGHT
private alias 1 : BIRD_FLYLEFT

// Time Period
private alias 2 : TIME_GOOD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case BIRD_FLYRIGHT
		if object.angle > 0
			object.angle -= 2
		else
			object.state     = BIRD_FLYLEFT
			object.direction = FACING_LEFT
		end if
		break

	case BIRD_FLYLEFT
		if object.angle < 256
			object.angle += 2
		else
			object.state     = BIRD_FLYRIGHT
			object.direction = FACING_RIGHT
		end if
		break

	end switch

	// Animate the Bird
	object.timer++
	if object.timer > 3
		object.timer = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == true
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos

		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	else
		if HUD.currentTimePeriod == TIME_GOOD_FUTURE
			Cos(temp0, object.angle)
			temp0 <<= 14
			temp0 += object.xpos

			Sin(temp1, object.angle)
			temp1 <<= 14
			temp1 += object.ypos

			DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects3.gif")

	SpriteFrame(-8, -8, 16, 16, 132, 1)
	SpriteFrame(-8, -8, 16, 16, 132, 18)

	// Setup all Flying Bird Objects in the level
	// Interestingly, we're iterating through them manually instead
	// of using RSDKv4's foreach functionality
	arrayPos0 = 32
	while arrayPos0 < 0x420
		if object[arrayPos0].type == TypeName[Flying Bird]
			// Give all Bird a high Draw Order so that they're above most other Objects
			// & stage tiles
			object[arrayPos0].drawOrder = 5
		end if

		arrayPos0++
	loop
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R3/Objects3.gif")
	SpriteFrame(-8, -8, 16, 16, 132, 1)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

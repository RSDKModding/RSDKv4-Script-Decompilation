// ========================
// Aliases

// Aliases
private alias object.value0 : object.timer

// HUD Aliases
private alias object[9].propertyValue : HUD.currentTimePeriod

// States
private alias 0	: YELLOWFISH_SWIMRIGHT
private alias 1 : YELLOWFISH_SWIMLEFT

// Time Period
private alias 2 : TIME_GOOD_FUTURE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case YELLOWFISH_SWIMRIGHT
		if object.angle < 512
			object.angle += 2

			// Move a pixel right
			object.xpos += 0x10000
		else
			// Apex reached, turn around
			object.state     = YELLOWFISH_SWIMLEFT
			object.direction = FACING_LEFT
		end if
		break

	case YELLOWFISH_SWIMLEFT
		if object.angle > 0
			object.angle -= 2

			// Just keep swimming, move a pixel left now
			object.xpos -= 0x10000
		else
			// Other side's apex reached, turn right back around again
			object.state     = YELLOWFISH_SWIMRIGHT
			object.direction = FACING_RIGHT
		end if
		break

	end switch

	object.timer++
	if object.timer > 19
		object.timer = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	// With this current system set up, that does indeed mean the sprite gets drawn twice if
	// in the Good Future and the Metal Sonic Projector has been destroyed
	if metalSonic.destroyed == true
		Sin(temp1, object.angle)
		temp1 <<= 12
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, object.xpos, temp1)
	end if

	if HUD.currentTimePeriod == TIME_GOOD_FUTURE
		Sin(temp1, object.angle)
		temp1 <<= 12
		temp1 += object.ypos

		DrawSpriteFX(object.frame, FX_FLIP, object.xpos, temp1)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R4/Objects3.gif")

	// Fishy Frames
	SpriteFrame(-8, -12, 16, 24, 1, 92)
	SpriteFrame(-8, -12, 16, 24, 18, 92)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R4/Objects3.gif")
	SpriteFrame(-8, -12, 16, 24, 1, 92)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

private alias 0 : FLOWERPOD_IDLE
private alias 1 : FLOWERPOD_EXPLOSION

private alias 4 : FLOWERPOD_CONTROL_PLAYER

// PodSeed Property Values
private alias 1 : TRIGGER_RESULTS

event ObjectUpdate
	switch object.state
	case FLOWERPOD_IDLE
		temp0 = object.ypos
		temp0 += 0x40005A
		if player[SLOT_PLAYER1].ypos < temp0
			stage.newXBoundary1 = object.xpos
			stage.newXBoundary1 >>= 16
			stage.newXBoundary1 -= screen.xcenter

			stage.newXBoundary2 = object.xpos
			stage.newXBoundary2 >>= 16
			stage.newXBoundary2 += screen.xcenter
		end if

		if object.timer < 60
			object.frame = 1
		else
			temp0 = object.timer
			temp0 -= 60
			temp0 >>= 1
			switch temp0
			case 0
				object.frame = 2
				break

			case 1
			case 2
				object.frame = 1
				break

			case 3
				object.frame = 2
				break

			case 4
			case 5
				object.frame = 1
				break

			case 6
				object.frame = 2
				break

			end switch
		end if

		object.timer++
		if object.timer == 74
			object.timer = 0
		end if
		break

	case FLOWERPOD_EXPLOSION
		if object.timer < 120
			object.timer++
			temp0 = object.timer
			temp0 &= 3
			if temp0 == 3
				Rand(temp0, 64)
				temp0 -= 32
				temp0 <<= 16
				temp0 += object.xpos

				Rand(temp1, 40)
				temp1 -= 40
				temp1 <<= 16
				temp1 += object.ypos

				CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
				PlaySfx(SfxName[Explosion], false)
			end if
		else
			object.state = FLOWERPOD_VANISH
			object.timer = 256
			object.inkEffect = INK_ALPHA
			object.alpha = 255

			temp0 = object.xpos
			temp0 -= 0x30000
			temp1 = object.ypos
			temp1 -= 0x180000
			CreateTempObject(TypeName[Pod Seed], TRIGGER_RESULTS, temp0, temp1)
			object[tempObjectPos].xvel = -0x20000
			object[tempObjectPos].yvel = -0x60000
			object[tempObjectPos].parentPod = object.entityPos

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = -0x15554
			object[tempObjectPos].yvel = -0x60000

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = -0xAAAA
			object[tempObjectPos].yvel = -0x60000

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 0
			object[tempObjectPos].yvel = -0x60000

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 0xAAAA
			object[tempObjectPos].yvel = -0x60000

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 0x15554
			object[tempObjectPos].yvel = -0x60000

			temp0 += 0x10000
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 0x20000
			object[tempObjectPos].yvel = -0x60000
		end if
		break

	case FLOWERPOD_VANISH
		if object.timer > 0
			object.timer -= 16
			object.alpha = object.timer
		end if
		break

	case FLOWERPOD_TRIGGER_ACTFINISH
		if player[SLOT_PLAYER1].controlMode == CONTROLMODE_NONE
			player[SLOT_PLAYER1].left     = false
			player[SLOT_PLAYER1].right    = true
			player[SLOT_PLAYER1].up       = false
			player[SLOT_PLAYER1].down     = false
			player[SLOT_PLAYER1].jumpHold = false
		end if

		temp0 = object.xpos
		temp0 += 0xA00000

		if player[SLOT_PLAYER1].xpos > temp0
			PlayMusic(1)
			object.state = FLOWERPOD_CONTROL_PLAYER
			object[30].type = TypeName[Act Finish]
			object[30].drawOrder = 6
			object[30].priority  = PRIORITY_ACTIVE
		end if
		break

	case FLOWERPOD_CONTROL_PLAYER
		if player[SLOT_PLAYER1].controlMode == CONTROLMODE_NONE
			player[SLOT_PLAYER1].right = true
		end if
		break

	end switch

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == FLOWERPOD_IDLE
			if player[currentPlayer].animation == ANI_JUMPING
				BoxCollisionTest(C_TOUCH, object.entityPos, -28, -46, 28, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					stage.timeEnabled = false
					object.state = FLOWERPOD_EXPLOSION
					object.frame = 3
					object.timer = 0

					player[currentPlayer].yvel >>= 1
					FlipSign(player[currentPlayer].yvel)
				end if
			end if
		end if

		temp0 = screen.xcenter
		temp0 += 40
		temp0 <<= 16
		temp1 = object.xpos
		temp1 += temp0

		if player[currentPlayer].xpos > temp1
			player[currentPlayer].xpos  = temp1
			player[currentPlayer].speed = 0
			player[currentPlayer].xvel  = 0
			if player[currentPlayer].gravity == GRAVITY_GROUND
				player[currentPlayer].yvel = 0
			end if
		end if
	next
end event


event ObjectDraw
	if object.state < FLOWERPOD_VANISH
		DrawSprite(0)
		DrawSprite(object.frame)
	else
		DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
		DrawSpriteFX(object.frame, FX_INK, object.xpos, object.ypos)
	end if
end event

	SpriteFrame(-32, 6, 64, 42, 1, 55)		// #0 - FlowerPod Stem
	SpriteFrame(-32, -48, 64, 54, 1, 1)		// #1 - FlowerPod Bulb
	SpriteFrame(-32, -48, 64, 54, 1, 98)	// #2 - FlowerPod Bulb Flash
	SpriteFrame(-32, -10, 64, 16, 1, 153)	// #3 - FlowerPod Bulb Destroyed

	foreach (TypeName[Flower Pod], arrayPos0, ALL_ENTITIES)
		temp0 = screen.xcenter
		temp0 -= 160
		temp0 <<= 16
		object[arrayPos0].xpos -= temp0
		object[arrayPos0].priority = PRIORITY_XBOUNDS
	next
end event

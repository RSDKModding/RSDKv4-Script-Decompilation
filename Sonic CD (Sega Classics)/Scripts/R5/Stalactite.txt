end function


public function Stalactite_DebugSpawn
	CreateTempObject(TypeName[Stalactite], 0, object.xpos, object.ypos)
	object[tempObjectPos].yOriginPos = object[tempObjectPos].ypos
end function


// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case STALACTITE_IDLE
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// First check if the player's within falling trigger range
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, 0, 4, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				// Make the Stalactite start falling

				if object.priority != PRIORITY_XBOUNDS_DESTROY
					// Make it always active, since it'll end up falling out of bounds
					object.priority = PRIORITY_ACTIVE
				end if

				// Reset its falling properties
				object.yvel  = 0
				object.state = STALACTITE_FALLING
			end if
		next
		break

	case STALACTITE_FALLING
		// Update Gravity, maximum velocity of 8 pixels
		if object.yvel < 0x80000
			// Gravity rate of 0.25 pixels per frame
			object.yvel += 0x4000
		end if

		object.ypos += object.yvel

		if object.outOfBounds == true
			// If this object goes offscreen, then put it back at its starting position and into the temp buffer state
			object.state = STALACTITE_OFFSCREEN
			object.ypos  = object.yOriginPos
		end if

		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Just act as a plain 'ol hitbox too, in case the player jumps at it for whatever reason
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -24, 8, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

			if checkResult == true
				CallFunction(Player_Hit)
			end if
		next
		break

	case STALACTITE_OFFSCREEN
		// This state is here just in case the Stalactite fell off screen, but the Player is still near it too
		// In this case, we need the Object to stay away just a bit longer so that it doesn't suddenly snap back to position
		if object.outOfBounds == true
			// It's safe to go back now, restore
			if object.priority != PRIORITY_XBOUNDS_DESTROY
				object.priority = PRIORITY_BOUNDS
			end if

			object.state = STALACTITE_IDLE
		end if
		break

	end switch
end event


event ObjectDraw
	switch object.state
	case STALACTITE_IDLE
	case STALACTITE_FALLING
		DrawSprite(0)
		break

	end switch
end event


event ObjectStartup
	LoadSpriteSheet("R5/Objects.gif")

	switch HUD.currentTimePeriod
	case TIME_PRESENT
		SpriteFrame(-8, -24, 16, 48, 172, 207)
		break

	case TIME_PAST
		SpriteFrame(-8, -24, 16, 48, 189, 207)
		break

	case TIME_GOOD_FUTURE
		SpriteFrame(-8, -24, 16, 48, 172, 207)
		break

	case TIME_BAD_FUTURE
		SpriteFrame(-8, -24, 16, 48, 155, 207)
		break

	end switch

	foreach (TypeName[Stalactite], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].yOriginPos = object[arrayPos0].ypos
	next

	SetTableValue(TypeName[Stalactite], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(Stalactite_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(Stalactite_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

		SpriteFrame(-8, -24, 16, 48, 189, 207)
		break

	case TIME_GOOD_FUTURE
		SpriteFrame(-8, -24, 16, 48, 172, 207)
		break

	case TIME_BAD_FUTURE
		SpriteFrame(-8, -24, 16, 48, 155, 207)
		break

	end switch

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

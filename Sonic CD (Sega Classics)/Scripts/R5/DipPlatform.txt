event ObjectUpdate
	if object.stood == true
		if object.drawPos.y < 16
			object.drawPos.y++

			temp0 = object.drawPos.y
			temp0 &= 3

			if temp0 == 3
				// The Platform will be moved 1 pixel down
				object.collisionOffset.y = 0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
		end if
	else
		if object.drawPos.y > 0
			object.drawPos.y--

			temp0 = object.drawPos.y
			temp0 &= 3

			if temp0 == 3
				// The Platform should be moved 1 pixel up
				object.collisionOffset.y = -0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
		end if
	end if

	object.stood = false

	switch object.propertyValue
	case 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Small Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -16, 16, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true

				// Move the Player along with the Platform
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
		next
		break

	case 1
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Medium-sized Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true

				// Move the Player in accordance with the Platform
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
		next
		break

	case 2
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Large Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true
				
				// As needed, nudge the Player so that the follow the Platform
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
		next
		break

	case 3
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Small Conveyor Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -16, 16, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true

				// Update the Player's Y Position to follow the Platform
				player[currentPlayer].ypos += object.collisionOffset.y

				// Also, update their X Position as needed with the Conveyor Belt too
				if conveyorBelt.flag == 0
					player[currentPlayer].xpos += conveyorBelt.speed
				else
					player[currentPlayer].xpos -= conveyorBelt.speed
				end if
			end if
		next
		break

	case 4
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Medium-sized Conveyor Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true

				// Make the Player follow the Platform
				player[currentPlayer].ypos += object.collisionOffset.y

				// And make them follow the Conveyor Belt as well
				if conveyorBelt.flag == 0
					player[currentPlayer].xpos += conveyorBelt.speed
				else
					player[currentPlayer].xpos -= conveyorBelt.speed
				end if
			end if
		next
		break

	case 5
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Large Conveyor Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true

				// The Platform moves vertically, so should the Player!
				player[currentPlayer].ypos += object.collisionOffset.y

				// And don't forget about the Conveyor Belt, too!
				if conveyorBelt.flag == 0
					player[currentPlayer].xpos += conveyorBelt.speed
				else
					player[currentPlayer].xpos -= conveyorBelt.speed
				end if
			end if
		next
		break
	end switch

	object.ypos += object.collisionOffset.y
end event


event ObjectDraw
	switch object.propertyValue
	case 0
	case 1
	case 2
		// Standard Platform
		DrawSprite(object.propertyValue)
		break
		
	// These below 3 are Conveyor Platforms, so they need the Conveyor overlay Sprite drawn ontop of them too
	case 3
		DrawSprite(0)
		DrawSprite(3)
		break

	case 4
		DrawSprite(1)
		DrawSprite(4)
		break

	case 5
		DrawSprite(2)
		DrawSprite(5)
		break

	end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Type
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // Type
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event


event RSDKDraw
	if object.propertyValue >= 3
		temp0 = -3
		temp0 += object.propertyValue
		DrawSprite(temp0)
	end if
	
	DrawSprite(object.propertyValue)
end event

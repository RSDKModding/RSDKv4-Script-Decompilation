// ========================
// Aliases

// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.loops

// ========================
// Events
// ========================

event ObjectDraw
	DrawSprite(object.frame)

	// Rise by 3 pixels
	object.ypos -= 0x30000
	object.timer++

	// Animate the Explosion
	if object.timer > 3
		object.timer = 0
		object.frame++
		if object.frame > 4
			object.frame = 0
			object.loops++
			if object.loops == 2
				// Upon reaching the third loop, unload
				object.type = TypeName[Blank Object]
			end if
		end if
	end if
end event


event ObjectStartup
	if stage.playerListPos == PLAYER_SONIC_A
		LoadSpriteSheet("Global/Items2.gif")
	else
		LoadSpriteSheet("Global/Items2_t.gif")
	end if

	// Explosion Frames
	SpriteFrame(-16, -8, 32, 16, 1, 1)
	SpriteFrame(-16, -16, 32, 32, 1, 84)
	SpriteFrame(-16, -16, 32, 32, 1, 117)
	SpriteFrame(-24, -24, 48, 48, 34, 1)
	SpriteFrame(-24, -24, 48, 48, 34, 50)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Global/Items2.gif")
	SpriteFrame(-16, -8, 32, 16, 1, 1)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

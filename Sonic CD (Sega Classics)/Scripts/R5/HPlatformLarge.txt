public function HPlatformLarge_DebugDraw
	temp0 = debugMode[0].currentSelection
	temp0 -= HPlatformLarge_startDebugID

	switch temp0
	case 0 // Right Platform
		DrawSprite(0)
		break

	case 1 // Right Platform (Conveyor)
		DrawSprite(0)
		DrawSprite(1)
		break

	end switch
end function


public function HPlatformLarge_DebugSpawn
	temp0 = debugMode[0].currentSelection
	temp0 -= HPlatformLarge_startDebugID

	switch temp0
	case 0 // Right Platform
		temp1 = START_MOVING_RIGHT
		break

	case 1 // Right Platform (Conveyor)
		temp1 = START_MOVING_RIGHT_W/CONV
		break

	end switch

	// Spawn the platform
	CreateTempObject(TypeName[HPlatform Large], temp1, object.xpos, object.ypos)

	// Truncate the position
	object[tempObjectPos].xpos &= -0x10000
	object[tempObjectPos].ypos &= -0x10000
end function

event ObjectUpdate
	// Update this Object's Oscillation Value
	// 312 may seem like a rather random value, but you'll see why it's used in a moment...
	object.oscillation++
	if object.oscillation == 312
		object.oscillation = 0
	end if

	// From this Oscillation value, get the object's Angle
	// This Angle will range from 0-510
	object.angle = object.oscillation
	object.angle <<= 6
	object.angle /= 39

	// And then from this Angle, get the base X Change Value
	Sin(object.collisionOffset.x, object.angle)

	switch object.propertyValue
	case START_MOVING_RIGHT
	case START_MOVING_RIGHT_W/CONV
		// Right-moving Platform
		object.collisionOffset.x *= 0x1800
		object.collisionOffset.x += 0x300000
		object.collisionOffset.x += object.xpos
		object.collisionOffset.x &= -0x10000 // Truncate the value
		object.collisionOffset.x -= object.drawPos.x
		break

	case START_MOVING_LEFT
	case START_MOVING_LEFT_W/CONV
		// Left-moving Platform
		object.collisionOffset.x *= -0x1800
		object.collisionOffset.x -= 0x300000
		object.collisionOffset.x += object.xpos
		object.collisionOffset.x &= -0x10000 // Truncate the value
		object.collisionOffset.x -= object.drawPos.x
		break

	case START_MOVING_RIGHT_W/SPRING
		// Right-moving Carrier Platform
		object.collisionOffset.x *= 0x1800
		object.collisionOffset.x += 0x300000
		object.collisionOffset.x += object.xpos
		object.collisionOffset.x &= -0x10000 // Truncate the value

		// Move the first following Object

		// Make it 32 pixels to the left of this Object
		object[+1].xpos = object.collisionOffset.x
		object[+1].xpos -= 0x200000

		// And 24 pixels above
		object[+1].ypos = object.ypos
		object[+1].ypos -= 0x180000

		// And then comes the second following Object

		// This one will be in the center of the Platform
		object[+2].xpos = object.collisionOffset.x

		// And also 24 pixels above
		object[+2].ypos = object.ypos
		object[+2].ypos -= 0x180000

		// And then the last, third riding Object

		// 32 pixels to the right
		object[+3].xpos = object.collisionOffset.x
		object[+3].xpos += 0x200000

		// And, of course, 24 pixels above again
		object[+3].ypos = object.ypos
		object[+3].ypos -= 0x180000

		object.collisionOffset.x -= object.drawPos.x
		break

	case START_MOVING_LEFT_W/SPRING
		// Left-moving Carrier Platform
		object.collisionOffset.x *= -0x1800
		object.collisionOffset.x -= 0x300000
		object.collisionOffset.x += object.xpos
		object.collisionOffset.x &= -0x10000 // Truncate the value

		// Move the first Following Object

		// 32 pixels to the left
		object[+1].xpos = object.collisionOffset.x
		object[+1].xpos -= 0x200000

		// 24 pixels above
		object[+1].ypos = object.ypos
		object[+1].ypos -= 0x180000

		// Next Object!

		// Same X as the Platform
		object[+2].xpos = object.collisionOffset.x

		// 24 pixels above
		object[+2].ypos = object.ypos
		object[+2].ypos -= 0x180000

		// Final Rider Object!

		// 32 pixels right
		object[+3].xpos = object.collisionOffset.x
		object[+3].xpos += 0x200000

		// And 24 pixels above
		object[+3].ypos = object.ypos
		object[+3].ypos -= 0x180000

		object.collisionOffset.x -= object.drawPos.x
		break

	end switch

	if object.stood == true
		if object.drawPos.y < 16
			object.drawPos.y++
			temp0 = object.drawPos.y
			temp0 &= 3
			if temp0 == 3
				// Dip a pixel down
				object.collisionOffset.y = 0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
		end if
	else
		if object.drawPos.y > 0
			object.drawPos.y--
			temp0 = object.drawPos.y
			temp0 &= 3
			if temp0 == 3
				// Go back up a pixel
				object.collisionOffset.y = -0x10000
			else
				object.collisionOffset.y = 0
			end if
		else
			object.collisionOffset.y = 0
		end if
	end if

	object.stood = false

	// Before doing anything else, back up the Object's base position
	temp0 = object.xpos

	// Then, move the Object to where its Platform is
	object.xpos = object.drawPos.x

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		// And now, jump to the handling type needed
		switch object.propertyValue
		case START_MOVING_RIGHT
		case START_MOVING_LEFT
			// Standard Platform
			BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				// Since the Player's on the Platform, let the Platform Object know so it can dip
				object.stood = true

				// And carry the Player along with the Platform
				player[currentPlayer].xpos += object.collisionOffset.x
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
			break

		case START_MOVING_RIGHT_W/CONV
		case START_MOVING_LEFT_W/CONV
			// Conveyor Platform
			if object.cooldown == 0
				BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					// Now that the Player's on the Platform, let the Platform Object know
					object.stood = true

					// Move the Player along with the Platform to keep them on
					player[currentPlayer].xpos += object.collisionOffset.x
					player[currentPlayer].ypos += object.collisionOffset.y

					if conveyorBelt.flag == 0
						player[currentPlayer].xpos += conveyorBelt.speed
					else
						player[currentPlayer].xpos -= conveyorBelt.speed
					end if

					// On the note of keeping the Player on the Platform...

					// Activate a small cooldown if the Player is seen to be on the edge of this Platform
					// This is to prevent an issue where the Player will repeatedly get pushed off and regrabbed by the Platform

					if player[currentPlayer].animation == ANI_FLAILING1
						object.cooldown = 30
					end if

					if player[currentPlayer].animation == ANI_FLAILING2
						object.cooldown = 30
					end if
				end if
			else
				// Hold...
				object.cooldown--
			end if
			break
		end switch
	next

	// And now restore the Object's Position
	object.xpos = temp0
end event


event ObjectDraw
	// Move the Object to match where it should be
	object.ypos += object.collisionOffset.y

	switch object.propertyValue
	case START_MOVING_RIGHT
	case START_MOVING_RIGHT_W/SPRING
		// Right-moving Platform
		Sin(object.drawPos.x, object.angle)
		object.drawPos.x *= 0x1800
		object.drawPos.x += 0x300000
		object.drawPos.x += object.xpos
		object.drawPos.x &= -0x10000 // Truncate the value
		DrawSpriteXY(0, object.drawPos.x, object.ypos)
		break

	case START_MOVING_RIGHT_W/CONV
		// Right-moving Conveyor Platform
		Sin(object.drawPos.x, object.angle)
		object.drawPos.x *= 0x1800
		object.drawPos.x += 0x300000
		object.drawPos.x += object.xpos
		object.drawPos.x &= -0x10000 // Truncate the value
		DrawSpriteXY(0, object.drawPos.x, object.ypos)
		DrawSpriteXY(1, object.drawPos.x, object.ypos)
		break

	case START_MOVING_LEFT
	case START_MOVING_LEFT_W/SPRING
		// Left-moving Platform
		Sin(object.drawPos.x, object.angle)
		object.drawPos.x *= -0x1800
		object.drawPos.x -= 0x300000
		object.drawPos.x += object.xpos
		object.drawPos.x &= -0x10000 // Truncate the value
		DrawSpriteXY(0, object.drawPos.x, object.ypos)
		break

	case START_MOVING_LEFT_W/CONV
		// Left-moving Conveyor Platform
		Sin(object.drawPos.x, object.angle)
		object.drawPos.x *= -0x1800
		object.drawPos.x -= 0x300000
		object.drawPos.x += object.xpos
		object.drawPos.x &= -0x10000 // Truncate the value
		DrawSpriteXY(0, object.drawPos.x, object.ypos)
		DrawSpriteXY(1, object.drawPos.x, object.ypos)
		break

	end switch
end event


event ObjectStartup
	// Use different graphics for the different Time Periods
	switch HUD.currentTimePeriod
	case TIME_PRESENT
		LoadSpriteSheet("R5/Objects.gif")
		SpriteFrame(-48, -16, 96, 32, 1, 84)
		SpriteFrame(-48, -16, 96, 16, 1, 191)
		break

	case TIME_PAST
		LoadSpriteSheet("R5/Objects3.gif")
		SpriteFrame(-48, -16, 96, 32, 1, 203)
		SpriteFrame(-48, -16, 96, 16, 159, 131)
		break

	case TIME_GOOD_FUTURE
		LoadSpriteSheet("R5/Objects3.gif")
		SpriteFrame(-48, -16, 96, 32, 1, 203)
		SpriteFrame(-48, -16, 96, 16, 159, 165)
		break

	case TIME_BAD_FUTURE
		LoadSpriteSheet("R5/Objects3.gif")
		SpriteFrame(-48, -16, 96, 32, 1, 203)
		SpriteFrame(-48, -16, 96, 16, 159, 199)
		break

	end switch

	temp0 = 0
	HPlatformLarge_startDebugID = DebugMode_ObjCount
	while temp0 < 2
		SetTableValue(TypeName[HPlatform Large], DebugMode_ObjCount, DebugMode_TypesTable)
		SetTableValue(HPlatformLarge_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
		SetTableValue(HPlatformLarge_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
		DebugMode_ObjCount++
		temp0++
	loop
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Direction
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // Direction
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event


event RSDKDraw
	temp2 = false
	
	switch object.propertyValue
	case START_MOVING_RIGHT_W/CONV
		temp2 = true
		// [Fallthrough]

	case START_MOVING_RIGHT
	case START_MOVING_RIGHT_W/SPRING
		temp0 = 0x1800
		temp1 = 0x300000
		break
	
	case START_MOVING_LEFT_W/CONV
		temp2 = true
		// [Fallthrough]

	case START_MOVING_LEFT
	case START_MOVING_LEFT_W/SPRING
		temp0 = -0x1800
		temp1 = -0x300000
		break
		
	end switch
	
	temp3 = object.xpos
	temp3 &= 0xFFFF0000 // Truncate the value
	DrawSpriteXY(0, temp3, object.ypos)
	DrawSpriteXY(temp2, temp3, object.ypos)
	
	if editor.showGizmos == true
		editor.drawingOverlay = true
		
		// Show the Platform's path
		
		Sin(temp4, 128)
		temp4 *= temp0
		temp4 += temp1
		temp4 += object.xpos
		temp4 &= 0xFFFF0000 // Truncate the value
		DrawLine(temp3, object.ypos, temp4, object.ypos, 255, 255, 255)
		
		editor.drawingOverlay = false
	end if
end event

private alias 1 : BOSSSPIKES_AWAITARENAENTER

// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case BOSSSPIKES_AWAITPLAYER
        if player[SLOT_PLAYER1].xpos > object.xpos
            // Initalise the arena

            // Backup the Stage's normal right bounds into the Boss Top's Values
            bossTop.oldBoundsR = stage.curXBoundary2

            // Set left bounds
            stage.newXBoundary1 = object.ixpos
            stage.newXBoundary1 += 328
            stage.newXBoundary1 -= screen.xcenter

            // And right bounds
            stage.newXBoundary2 = stage.newXBoundary1
            stage.newXBoundary2 += screen.xcenter

            // Set the Boss Top's
            bossTop.boundsR = stage.newXBoundary2
            bossTop.boundsR += 160
            bossTop.boundsR <<= 16

            stage.newXBoundary2 += screen.xcenter
            stage.newYBoundary2 = object.iypos

            object.state = BOSSSPIKES_AWAITARENAENTER
            object.priority = PRIORITY_ACTIVE

			// Disable the Conveyor Belt controller
            conveyorBelt.active = false
        end if
        break

    case BOSSSPIKES_AWAITARENAENTER
        temp0 = object.xpos
        temp0 += 0x300000
        if player[SLOT_PLAYER1].xpos > temp0
            object.state = BOSSSPIKES_RAISE
        end if
        break

    case BOSSSPIKES_RAISE
        if object.timer < 64
            object.timer += 2
            object.ypos -= 0x20000
        else
            object.timer = 0
            object.state = BOSSSPIKES_ADVANCE
        end if
        break

    case BOSSSPIKES_ADVANCE
        if object.timer < 192
            object.timer += 2
            object.xpos += 0x20000
        else
            object.timer = 0
            object.state = BOSSSPIKES_STAND
            object.priority = PRIORITY_BOUNDS
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.state < BOSSSPIKES_RAISE
			// Just act as a box for the player
            BoxCollisionTest(C_SOLID, object.entityPos, -24, -32, 0, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        else
			// See if the player is within range of behind the Spike Wall
            BoxCollisionTest(C_TOUCH, object.entityPos, -64, -32, 0, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

			if checkResult == true
				// If they are, then move them to be in front instead
                player[currentPlayer].xpos = player[currentPlayer].collisionRight
                player[currentPlayer].xpos <<= 16
                player[currentPlayer].xpos += object.xpos
            end if
        end if

		// Check collision for the Spikes side of things
        BoxCollisionTest(C_TOUCH, object.entityPos, 0, -32, 24, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

		if checkResult == true
            CallFunction(Player_Hit)

            if player[currentPlayer].state == Player_State_GotHit
				// If Sonic's been hurt, then set the Boss Platform's speed
                bossPlatform.beltSpeed = 0xC000
            end if
        end if
    next
end event

end event

    else
        SpriteFrame(-24, -32, 48, 64, 76, 191)
    end if
end event

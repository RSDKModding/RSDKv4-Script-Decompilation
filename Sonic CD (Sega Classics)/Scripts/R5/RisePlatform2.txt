// Aliases
private alias object.value0 : object.stood
private alias object.value1 : object.drawPos.y
private alias object.value2 : object.collisionOffset.y
private alias object.value3 : object.yvel
private alias object.value4 : object.originPos.y
private alias object.value5 : object.timer

// States
private alias 0 : RISEPLATFORM2_AWAITPLAYER
private alias 1 : RISEPLATFORM2_PRELAUNCH
private alias 2 : RISEPLATFORM2_LAUNCH
private alias 3 : RISEPLATFORM2_RISE
private alias 4 : RISEPLATFORM2_HALT
private alias 5 : RISEPLATFORM2_ARRIVED

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case RISEPLATFORM2_AWAITPLAYER
		// Waiting for the Player to step on it to Launch
		object.collisionOffset.y = 0

		if object.stood == true
			// If the Player's stepped onto the Platform, then get ready to launch
			object.state    = RISEPLATFORM2_PRELAUNCH
			object.priority = PRIORITY_ACTIVE
			object.yvel     = 0
			object.timer    = 0
		end if
		break

	case RISEPLATFORM2_PRELAUNCH
		// Pausing for a moment before Launching
		object.collisionOffset.y = object.yvel

		if object.timer < 8
			object.timer++
		else
			object.timer = 0
			object.state++
		end if
		break

	case RISEPLATFORM2_LAUNCH
		// If the Object hasn't yet reached a speed of 4px per frame, then make it go faster by 0.0625 more per frame
		if object.yvel > -0x40000
			object.yvel -= 0x1000
		else
			// Sufficient speed reached, go to the next timer-based State
			object.state++
		end if

		object.collisionOffset.y = object.yvel
		break

	case RISEPLATFORM2_RISE
		// Just rise for a brief moment
		if object.timer < 14
			object.timer++
		else
			// Time's up, start slowing down...
			object.timer = 0
			object.state++
		end if

		object.collisionOffset.y = object.yvel
		break

	case RISEPLATFORM2_HALT
		// Until the Object is fully stopped, decelerate it at a speed of 0.0625 pixels per frame
		if object.yvel < 0
			object.yvel += 0x1000
		else
			// Ding! - And now we've arrived!
			object.state++
		end if

		object.collisionOffset.y = object.yvel
		break

	case RISEPLATFORM2_ARRIVED
		// Now, the Object just acts as a standard Platform
		if object.stood == true
			if object.drawPos.y < 16
				object.drawPos.y++
				temp0 = object.drawPos.y
				temp0 &= 3
				if temp0 == 3
					object.collisionOffset.y = 0x10000
				else
					object.collisionOffset.y = 0
				end if
			else
				object.collisionOffset.y = 0
			end if
		else
			if object.drawPos.y > 0
				object.drawPos.y--
				temp0 = object.drawPos.y
				temp0 &= 3
				if temp0 == 3
					object.collisionOffset.y = -0x10000
				else
					object.collisionOffset.y = 0
				end if
			else
				object.collisionOffset.y = 0
			end if
		end if
		break

	end switch

	// If the Object is Out Of Bounds, check to see if it's origin Position is OOB too before moving it back
	if object.outOfBounds == true
		temp0 = object.ypos
		object.ypos = object.originPos.y

		if object.outOfBounds == true
			object.drawPos.y = 0
			object.state = RISEPLATFORM2_AWAITPLAYER
		else
			object.ypos = temp0
		end if
	end if

	object.stood = false
end event


event ObjectDraw
	object.ypos += object.collisionOffset.y
	DrawSprite(0)
end event


event ObjectStartup
	LoadSpriteSheet("R5/Objects.gif")

	// Large Platform Frame
	SpriteFrame(-48, -16, 96, 32, 1, 84)

	// Cycle through every Rise Platform 2 Objects in the Scene
	arrayPos0 = 32
	while arrayPos0 < 1056
		if object[arrayPos0].type == TypeName[Rise Platform 2]
			// Store the Object's Starting Position
			object[arrayPos0].originPos.y = object[arrayPos0].ypos
			object[arrayPos0].priority = PRIORITY_XBOUNDS
		end if

		arrayPos0++
	loop
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)

	if editor.showGizmos == true
		// Draw the Platform's destination
		temp0 = object.ypos
		temp0 -= 0x1400000

		editor.drawingOverlay = true
		DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 255, 255)
		editor.drawingOverlay = false
	end if
end event

	LoadSpriteSheet("R5/Objects.gif")
	SpriteFrame(-48, -16, 96, 32, 1, 84)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

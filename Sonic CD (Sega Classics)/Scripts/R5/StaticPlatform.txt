event ObjectUpdate
	switch object.propertyValue
	case PLATFORM_SMALL
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -16, 16, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break

	case PLATFORM_MEDIUM
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break

	case PLATFORM_LARGE
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break

	case PLATFORM_SMALL_CONVEYOR
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// (with the Conveyor code removed, functionally identical to the normal Small Platform)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -16, 16, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break

	case PLATFORM_MEDIUM_CONVEYOR
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// (Also with no Conveyor code associated with it... pretty much the same as the normal Medium Platform type)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break

	case PLATFORM_LARGE_CONVEYOR
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// (You may notice some corresponding code is missing, that's because the took the Conveyor out of the name!)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -48, -16, 48, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		next
		break
	end switch
end event


event ObjectDraw
	switch object.propertyValue
	case PLATFORM_SMALL
	case PLATFORM_MEDIUM
	case PLATFORM_LARGE
		// Standard Platform
		DrawSprite(object.propertyValue)
		break

	// These next 3 types are for Conveyor Belt types, where the Conveyor Belt Frame is overlayed atop the normal Platform Sprite

	case PLATFORM_SMALL_CONVEYOR
		DrawSprite(0)
		DrawSprite(3)
		break

	case PLATFORM_MEDIUM_CONVEYOR
		DrawSprite(1)
		DrawSprite(4)
		break

	case PLATFORM_LARGE_CONVEYOR
		DrawSprite(2)
		DrawSprite(5)
		break

	end switch
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // Type
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // Type
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event


event RSDKDraw
	if object.propertyValue >= 3
		temp0 = -3
		temp0 += object.propertyValue
		DrawSprite(temp0)
	end if

	DrawSprite(object.propertyValue)
end event

		player[SLOT_PLAYER1].ypos += object.ypos

			player[SLOT_PLAYER1].speed -= player[SLOT_PLAYER1].xpos
			FlipSign(player[SLOT_PLAYER1].speed)
		end if
	end if

	if object.propertyValue == 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			if player[currentPlayer].state != Player_State_Static
				BoxCollisionTest(C_TOUCH, object.entityPos, -8, -16, 8, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

					player[currentPlayer].state       = Player_State_Static
					player[currentPlayer].interaction = false

					object.timer    = 0
					object.priority = PRIORITY_ACTIVE

				end if
			end if
		next
	end if
end event

// Aliases
private alias object.value0 : object.originPosY

// States
private alias 0 : DROPROCK_IDLE
private alias 1 : DROPROCK_FALLING
private alias 2 : DROPROCK_WAITFORRESPAWN
private alias 3 : DROPROCK_FRAGMENT

// Property Values
private alias 1 : ROCK_DEBRIS_1
private alias 2 : ROCK_DEBRIS_2
private alias 3 : ROCK_DEBRIS_3
private alias 4 : ROCK_DEBRIS_4

// ========================
// Function Declarations
// ========================

reserve function DropRock_DebugDraw
reserve function DropRock_DebugSpawn

// ========================
// Function Definitions
// ========================

public function DropRock_DebugDraw
    DrawSprite(0)
end function


public function DropRock_DebugSpawn
    CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_1, object.xpos, object.ypos)
    object[tempObjectPos].originPosY = object[tempObjectPos].ypos
end function


// ========================
// Events
// ========================

event ObjectUpdate
    switch object.state
    case DROPROCK_FALLING
		// If not yet falling at a speed of 8 px/frame, accelerate a rate of 0.25 per frame
        if object.yvel < 0x80000
            object.yvel += 0x4000
        end if

        object.ypos += object.yvel

		// Only check for Collision if the Object is falling at a rate faster than 2px per frame
        if object.yvel > 0x20000
            ObjectTileCollision(CSIDE_FLOOR, 0, 12, 0)
            if checkResult == true
                PlaySfx(SfxName[Explosion], false)

				// Make the Rock invisible and make it wait for respawning
                object.state = DROPROCK_WAITFORRESPAWN

				// Create all the Rock Fragments now
				// The appropriate info for each Fragment is listed as needed, in the format of
				// x: X offset from the Rock's base position
				// y: Y offset from the Rock's base position
				// xvel: Starting X Velocity of the Fragment
				// yvel: Starting Y Velocity of the Fragment

				// Top left Fragment!
				// - x: 8 pixels left
				// - y: 8 pixels up
				// - xvel: 1 pixel per frame left
				// - yvel: 4 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_1, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos -= 0x80000
                object[tempObjectPos].ypos -= 0x80000
                object[tempObjectPos].xvel = -0x10000
                object[tempObjectPos].yvel = -0x40000
                object[tempObjectPos].drawOrder = 5

				// Top right Fragment:
				// - x: 8 pixels right
				// - y: 8 pixels up
				// - xvel: 1 pixel per frame right
				// - yvel: 4 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_2, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos += 0x80000
                object[tempObjectPos].ypos -= 0x80000
                object[tempObjectPos].xvel = 0x10000
                object[tempObjectPos].yvel = -0x40000
                object[tempObjectPos].drawOrder = 5

				// Top right Fragment:
				// - x: 8 pixels right
				// - y: 8 pixels up
				// - xvel: 1 pixel per frame right
				// - yvel: 4 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_3, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos -= 0x80000
                object[tempObjectPos].ypos += 0x80000
                object[tempObjectPos].xvel = -0x20000
                object[tempObjectPos].yvel = -0x20000
                object[tempObjectPos].drawOrder = 5

				// Bottom right Fragment:
				// - x: 8 pixels right
				// - y: 8 pixels down
				// - xvel: 2 pixels per frame right
				// - yvel: 2 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_4, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos += 0x80000
                object[tempObjectPos].ypos += 0x80000
                object[tempObjectPos].xvel = 0x20000
                object[tempObjectPos].yvel = -0x20000
                object[tempObjectPos].drawOrder = 5

				// And then carrying on from the Explosion SFX from a while ago, create an Explosion effect too now
                CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
                object[tempObjectPos].drawOrder = 5
            end if
        end if
        break

    case DROPROCK_WAITFORRESPAWN
		// This state is given to the Rock after it explodes,
		// in order to make sure it doesn't warp back to its
		// starting position right in front of the Player
        if object.outOfBounds == true
            object.ypos = object.originPosY

            if object.priority != PRIORITY_XBOUNDS_DESTROY
                object.priority = PRIORITY_BOUNDS
            end if

            object.state = DROPROCK_IDLE
        end if
        break

    case DROPROCK_FRAGMENT
        object.xpos += object.xvel // Update horizontal movements
        object.yvel += 0x4000      // Make the Fragment accelerate vertically at a rate of 0.25px/frame
        object.ypos += object.yvel // And then update vertical movements

		// Unload when no longer needed
        if object.outOfBounds == true
            object.type = TypeName[Blank Object]
        end if
        break

    end switch

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.state
        case DROPROCK_IDLE
			// First, act as a Box for the Player
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

			// Then, see if the Player has walked below the Rock
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, 0, 4, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

			if checkResult == true
				// If they have, then start Falling
                object.originPosY = object.ypos

                if object.priority != PRIORITY_XBOUNDS_DESTROY
                    object.priority = PRIORITY_ACTIVE
                end if

                object.yvel  = 0
                object.state = DROPROCK_FALLING
            end if
			// [Fallthrough, to check if the Player has actually touched the rock]

        case DROPROCK_FALLING
			// Check Collision with Player
            BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 12, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

			// Did the Player hit their head on the bottom of this rock?
			if checkResult == COL_BOTTOM
                PlaySfx(SfxName[Explosion], false)

                object.state = DROPROCK_WAITFORRESPAWN

                CallFunction(Player_Hit)

				// Create all the Rock Fragments, since the Rock's been exploded

				// The initial info for each Fragment is listed as needed, in the following format:
				// x: X offset from the Rock's base position
				// y: Y offset from the Rock's base position
				// xvel: Starting X Velocity of the Fragment
				// yvel: Starting Y Velocity of the Fragment

				// Top left Fragment:
				// - x: 8 pixels left
				// - y: 8 pixels up
				// - xvel: 1 pixel per frame right
				// - yvel: 4 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_1, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos -= 0x80000
                object[tempObjectPos].ypos -= 0x80000
                object[tempObjectPos].xvel = -0x10000
                object[tempObjectPos].yvel = -0x40000
                object[tempObjectPos].drawOrder = 5

				// Top right Fragment:
				// - x: 8 pixels right
				// - y: 8 pixels up
				// - xvel: 1 pixel per frame right
				// - yvel: 4 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_2, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos += 0x80000
                object[tempObjectPos].ypos -= 0x80000
                object[tempObjectPos].xvel = 0x10000
                object[tempObjectPos].yvel = -0x40000
                object[tempObjectPos].drawOrder = 5

				// Bottom left Fragment:
				// - x: 8 pixels left
				// - y: 8 pixels down
				// - xvel: 2 pixels per frame left
				// - yvel: 3 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_3, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos -= 0x80000
                object[tempObjectPos].ypos += 0x80000
                object[tempObjectPos].xvel = -0x20000
                object[tempObjectPos].yvel = -0x30000
                object[tempObjectPos].drawOrder = 5

				// Bottom right Fragment:
				// - x: 8 pixels right
				// - y: 8 pixels down
				// - xvel: 2 pixels per frame right
				// - yvel: 3 pixels per frame up
                CreateTempObject(TypeName[Drop Rock], ROCK_DEBRIS_4, object.xpos, object.ypos)
                object[tempObjectPos].state = DROPROCK_FRAGMENT
                object[tempObjectPos].xpos += 0x80000
                object[tempObjectPos].ypos += 0x80000
                object[tempObjectPos].xvel = 0x20000
                object[tempObjectPos].yvel = -0x30000
                object[tempObjectPos].drawOrder = 5

                CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
                object[tempObjectPos].drawOrder = 5
            end if
            break

        end switch
    next
end event


event ObjectDraw
    switch object.state
    case DROPROCK_IDLE
    case DROPROCK_FALLING
        // Draw the Rock Sprite
        DrawSprite(0)
        break

    case DROPROCK_FRAGMENT
        // Draw the Fragment's given Frame
        DrawSprite(object.propertyValue)
        break

    end switch
end event


event ObjectStartup
    LoadSpriteSheet("R5/Objects.gif")

    SpriteFrame(-16, -16, 32, 32, 92, 18) // #0 - Main Rock Frame
    SpriteFrame(-8, -8, 16, 16, 92, 18)   // #1 - Fragment Top Left Frame
    SpriteFrame(-8, -8, 16, 16, 108, 18)  // #2 - Fragment Top Right Frame
    SpriteFrame(-8, -8, 16, 16, 92, 34)   // #3 - Fragment Bottom Left Frame
    SpriteFrame(-8, -8, 16, 16, 108, 34)  // #4 - Fragment Bottom Right Frame

    SetTableValue(TypeName[Drop Rock], DebugMode_ObjCount, DebugMode_TypesTable)
    SetTableValue(DropRock_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
    SetTableValue(DropRock_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
    DebugMode_ObjCount++
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    if editor.showGizmos == true
        // Draw a downwards arrow to show that this Rock is of the falling variety
        temp0 = object.ypos
        temp0 += 0x300000

        editor.drawingOverlay = true
        DrawArrow(object.xpos, object.ypos, object.xpos, temp0, 255, 255, 255)
        editor.drawingOverlay = false
    end if

    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R5/Objects.gif")
    SpriteFrame(-16, -16, 32, 32, 92, 18)

    // Although used by the Object, it's not to be set from the Editor
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

public function R5Setup_PresentStageUpdate
	// Only animate when the currently active BG is the Cave
	if stage[0].activeLayer == LAYER_INSIDE

		// Get the Palette Animation Frame to use
		temp0 = object.timer
		temp0 /= 6

		switch temp0
		case 0
			// Cases 0/3 are all the same and could probably be combined,
			// akin to 4/8 or 5/7, but I suppose they just didn't feel like doing that...
			CopyPalette(1, 0, 0, 176, 16)
			break

		case 1
			CopyPalette(1, 0, 0, 176, 16)
			break

		case 2
			CopyPalette(1, 0, 0, 176, 16)
			break

		case 3
			CopyPalette(1, 0, 0, 176, 16)
			break

		case 4
		case 8
			CopyPalette(1, 16, 0, 176, 16)
			break

		case 5
		case 7
			CopyPalette(1, 32, 0, 176, 16)
			break

		case 6
			CopyPalette(1, 48, 0, 176, 16)
			break

		end switch

		// Animate as needed
		object.timer++
		object.timer %= 54
	end if

	CopyPalette(3, temp0, 0, 192, 16)
end function

event ObjectUpdate
	if conveyorBelt.active == true
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			// Only affect the Player if they're on the Ground
			if player[currentPlayer].gravity == GRAVITY_GROUND
				// The Tunnel state makes Players intangible to Conveyor Belts
				if player[currentPlayer].state != Player_State_TubeRoll
					// Get the X Position of the tile on the Player's right side
					object.beltStoodTileX = player[currentPlayer].xpos
					object.beltStoodTileX >>= 16
					object.beltStoodTileX += player[currentPlayer].collisionRight

					// Get the Y Position of the tile right underneath tthe Player
					object.beltStoodTileY = player[currentPlayer].ypos
					object.beltStoodTileY >>= 16
					object.beltStoodTileY += player[currentPlayer].collisionBottom
					object.beltStoodTileY += 2

					// Get the B Angle/Flags value of the Tile at the specified positions
					Get16x16TileInfo(temp0, object.beltStoodTileX, object.beltStoodTileY, TILEINFO_ANGLEB)

					// See if it's a Conveyor Belt tile
					if temp0 == TILE_CONVEYORBELT
						// If it is, then move the Player in the corresponding direction and at the corresponding rate
						if conveyorBelt.flag == FACING_RIGHT
							player[currentPlayer].xpos += conveyorBelt.speed
						else
							player[currentPlayer].xpos -= conveyorBelt.speed
						end if
					else
						// If the right tile did't have anything, then check the one to the Player's left
						object.beltStoodTileX = player[currentPlayer].xpos
						object.beltStoodTileX >>= 16
						object.beltStoodTileX += player[currentPlayer].collisionLeft

						// Get the B Angle/Flags value of the Tile at the specified positions
						Get16x16TileInfo(temp0, object.beltStoodTileX, object.beltStoodTileY, TILEINFO_ANGLEB)

						// See if it's a Conveyor Belt tile
						if temp0 == TILE_CONVEYORBELT
							// If this one is a Conveyer tile, then convey the player
							if conveyorBelt.flag == FACING_RIGHT
								player[currentPlayer].xpos += conveyorBelt.speed
							else
								player[currentPlayer].xpos -= conveyorBelt.speed
							end if
						end if
					end if
				end if
			end if
		next

		// Some palette cycling to make the Conveyer Belts look like they're moving
		object.beltTimer++
		if object.beltTimer == 5
			object.beltTimer = 0
			object.beltFrame++
			object.beltFrame %= 3
		end if
	end if

	temp0 = object.beltFrame
	temp0 += conveyorBelt.flag

	temp0 <<= 4
	CopyPalette(2, temp0, 0, 160, 16)

	CallFunction(object.stateStageUpdate)
end event


event ObjectStartup
	object[SLOT_R5SETUP].type     = TypeName[R5 Setup]
	object[SLOT_R5SETUP].priority = PRIORITY_ACTIVE

	conveyorBelt.active = true
	conveyorBelt.flag   = 0
	conveyorBelt.frame  = 0

	switch HUD.currentTimePeriod
	case TIME_PRESENT
		// Give the Conveyer Belts a speed of 1 pixel per frame
		conveyorBelt.speed = 0x10000

		temp0 = 0
		temp1 = 0
		temp2 = 16

		// Setup the palette cycle for conveyors, using the R5A bank
		while temp0 < 96
			LoadPalette("R5A_PalCycle.act", 2, temp0, temp1, temp2)
			temp1 += 16
			temp2 += 16

			LoadPalette("R5A_PalCycle.act", 3, temp0, temp1, temp2)
			temp1 += 16
			temp2 += 16
			temp0 += 16
		loop

		// Setup the palette cycle for the stage
		object[SLOT_R5SETUP].stateStageUpdate = R5Setup_PresentStageUpdate
		LoadPalette("R5A_GlowEffect.act", 1, 0, 0, 80)
		break

	case TIME_PAST
		// In the Past, technology wasn't as developed, so give the Conveyer Belts a slower speed of 0.5 pixels per frame
		conveyorBelt.speed = 0x8000

		// Setup the palette cycle for the stage
		LoadPalette("R5B_PalCycle.act", 2, 0, 0, 96)
		object[SLOT_R5SETUP].stateStageUpdate = R5Setup_PastStageUpdate
		break

	case TIME_GOOD_FUTURE
		// Technology is pretty advanced in the future, the Conveyer Belts have grown so much to have a speed of 2 pixels per frame now!
		conveyorBelt.speed = 0x20000

		// Setup the palette cycle for the stage
		LoadPalette("R5C_PalCycle.act", 2, 0, 0, 96)
		object[SLOT_R5SETUP].stateStageUpdate = R5Setup_GoodFutureStageUpdate
		break

	case TIME_BAD_FUTURE
		// Even after taking over the world, Eggman was kind enough to let Conveyer Belt innovations still continue!
		// What a kind man, Conveyer Belts still go at a speed of 2 pixels per frame now!
		conveyorBelt.speed = 0x20000

		// Setup the palette cycle for the stage
		LoadPalette("R5D_PalCycle.act", 2, 0, 0, 96)
		object[SLOT_R5SETUP].stateStageUpdate = R5Setup_BadFutureStageUpdate
		break

	end switch
end event

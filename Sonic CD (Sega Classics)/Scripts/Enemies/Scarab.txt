private function Scarab_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= Scarab_startDebugID
    if temp0 == 0
        DrawSprite(0)
    else
        DrawSprite(2)
    end if
end function


private function Scarab_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= Scarab_startDebugID
    CreateTempObject(TypeName[Scarab], temp0, object.xpos, object.ypos)
    object[tempObjectPos].originPos.x = object.xpos
    object[tempObjectPos].originPos.y = object.ypos
end function

event ObjectUpdate
    switch object.state
    case SCARAB_MOVE
        object.priority = PRIORITY_ACTIVE
        if object.timer < 60
            object.timer++
            object.xpos += object.xvel
        else
            object.timer = 0
            object.state = SCARAB_STOP
        end if

        ObjectTileGrip(CSIDE_FLOOR, 0, 12, 0)
        if object.xvel > 0
            temp0 = object.xpos
            temp0 -= object.originPos.x
            if temp0 > 0x400000
                FlipSign(object.xvel)
            end if
        else
            temp0 = object.originPos.x
            temp0 -= object.xpos
            if temp0 > 0x400000
                FlipSign(object.xvel)
            end if
        end if

        object.animationTimer += object.animationSpeed
        object.animationTimer %= 40
        break

    case SCARAB_STOP
        if object.timer < 30
            object.timer++
        else
            object.timer = 0
            object.state = SCARAB_MOVE
        end if
        break
    end switch

    if object.launchDelay > 0
        if player[SLOT_PLAYER1].state == Player_State_Death
            object.launchDelay = 0
        else
            if object.direction == FACING_RIGHT
                player[SLOT_PLAYER1].xpos = object.xpos
                player[SLOT_PLAYER1].xpos -= 0x180000

                player[SLOT_PLAYER1].ypos = object.ypos
                player[SLOT_PLAYER1].ypos -= 0x30000
            else
                player[SLOT_PLAYER1].xpos = object.xpos
                player[SLOT_PLAYER1].xpos += 0x180000

                player[SLOT_PLAYER1].ypos = object.ypos
                player[SLOT_PLAYER1].ypos -= 0x30000
            end if

            object.launchDelay--
            if object.launchDelay == 0
                player[SLOT_PLAYER1].state = Player_State_Roll
                player[SLOT_PLAYER1].speed = object.launchSpeed
                player[SLOT_PLAYER1].xvel = player[SLOT_PLAYER1].speed
            end if
        end if
    end if

    if object.caryingMonitor == true
        if object[+1].type == TypeName[Monitor]
            object[+1].xpos = object.xpos
            object[+1].xpos -= 0x180000

            object[+1].ypos = object.ypos
            object[+1].ypos -= 0x40000
        else
            object[+1].type = TypeName[Blank Object]
            object.caryingMonitor = false
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        // This is always true
        if object.state < 2
            if object.launchDelay == 0
                if object.direction == FACING_RIGHT
                    BoxCollisionTest(C_TOUCH, object.entityPos, -32, -12, -12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                    if checkResult == true
                        if object.stood == false
                            object.launchDelay = 120

                            player[currentPlayer].animation = ANI_JUMPING

                            player[currentPlayer].state = Player_State_Static
                            player[currentPlayer].speed = 0

                            player[currentPlayer].xvel = 0
                            player[currentPlayer].yvel = 0

                            player[currentPlayer].xpos = object.xpos
                            player[currentPlayer].xpos -= 0x180000

                            player[currentPlayer].ypos = object.ypos
                            player[currentPlayer].ypos -= 0x30000

                            player[currentPlayer].direction = FACING_RIGHT

                            if warp.destination > WARPDEST_NONE
                                if warp.timer > 99
                                    // They (still) forgot to fix the warp timer here
                                    if warp.timer < 220
                                        warp.destination = WARPDEST_NONE
                                    end if
                                end if

                                warp.timer = 0
                            end if

                            object.stood = true
                        end if
                    else
                        object.stood = false
                        BoxCollisionTest(C_TOUCH, object.entityPos, -12, -12, 20, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                        if checkResult == true
                            CallFunction(Player_BadnikBreak)
                        end if
                    end if
                else
                    BoxCollisionTest(C_TOUCH, object.entityPos, 12, -12, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                    if checkResult == true
                        if object.stood == false
                            object.launchDelay = 120

                            player[currentPlayer].animation = ANI_JUMPING

                            player[currentPlayer].state = Player_State_Static
                            player[currentPlayer].speed = 0

                            player[currentPlayer].xvel = 0
                            player[currentPlayer].yvel = 0

                            player[currentPlayer].xpos = object.xpos
                            player[currentPlayer].xpos += 0x180000

                            player[currentPlayer].ypos = object.ypos
                            player[currentPlayer].ypos -= 0x30000

                            player[currentPlayer].direction = FACING_LEFT

                            if warp.destination > WARPDEST_NONE
                                if warp.timer > 99
                                    // They (still) forgot to fix the warp timer here
                                    if warp.timer < 220
                                        warp.destination = WARPDEST_NONE
                                    end if
                                end if

                                warp.timer = 0
                            end if

                            object.stood = true
                        end if
                    else
                        object.stood = false
                        BoxCollisionTest(C_TOUCH, object.entityPos, -20, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                        if checkResult == true
                            CallFunction(Player_BadnikBreak)
                        end if
                    end if
                end if
            end if
        end if
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)

    if object.outOfBounds == true
        temp0 = object.xpos
        temp1 = object.ypos

        object.xpos = object.originPos.x
        object.ypos = object.originPos.y

        // If starting position was out of bounds too, then we can safely go back
        if object.outOfBounds == true
            object.priority = PRIORITY_BOUNDS

            object.state = SCARAB_MOVE
            object.timer = 0

            object.animationTimer = 0
            if object.xvel < 0
                FlipSign(object.xvel)
            end if
        else
            object.xpos = temp0
            object.ypos = temp1
        end if
    end if
end event


event ObjectDraw
    // This is always true
    if object.state < 2
        if object.drawOrder == 3
            object.frame = object.animationTimer
            object.frame /= 20
            object.frame += object[arrayPos0].propertyValue
            DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)

            object.frame += 2
            SetDrawListEntityRef(object.entityPos, 4, screen[4].drawListSize)
            screen[4].drawListSize++
            object.drawOrder = 4
        else
            DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
            object.drawOrder = 3
        end if
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R8/Objects.gif")

    SpriteFrame(-24, -12, 24, 24, 107, 206) // #0 - Scarab Arms frame 0
    SpriteFrame(-32, -12, 32, 24, 132, 206) // #1 - Scarab Arms frame 1

    SpriteFrame(-32, -12, 56, 24, 165, 206) // #2 - Scarab frame 0
    SpriteFrame(-32, -12, 56, 24, 165, 231) // #3 - Scarab frame 1

    SpriteFrame(-24, -12, 24, 24, 107, 206) // #4 - Scarab Arms frame 0
    SpriteFrame(-32, -12, 32, 16, 67, 240)  // #5 - Scarab Arms frame 1

    SpriteFrame(-32, -12, 56, 24, 165, 206) // #6 - Scarab frame 0
    SpriteFrame(-40, -12, 64, 24, 100, 231) // #7 - Scarab frame 1

    foreach (TypeName[Scarab], arrayPos0, ALL_ENTITIES)
        object[arrayPos0].originPos.x = object[arrayPos0].xpos
        object[arrayPos0].originPos.y = object[arrayPos0].ypos

        switch object[arrayPos0].propertyValue
        case DIR_FACING_RIGHT
            // Set the animation to use un-streched arms
            object[arrayPos0].propertyValue = DIR_FACING_RIGHT

            object[arrayPos0].xvel = 0x6000
            object[arrayPos0].animationSpeed = 2
            object[arrayPos0].direction = FACING_RIGHT
            object[arrayPos0].launchSpeed = -0xA0000
            break

        case DIR_FACING_LEFT
            // Set the animation to use un-streched arms
            object[arrayPos0].propertyValue = DIR_FACING_RIGHT

            object[arrayPos0].xvel = 0x6000
            object[arrayPos0].animationSpeed = 2
            object[arrayPos0].direction = FACING_LEFT
            object[arrayPos0].launchSpeed = 0xA0000
            break

        case DIR_FACING_RIGHT_S
            // Set the animation to use streched arms
            object[arrayPos0].propertyValue = DIR_FACING_RIGHT_W_OBJ

            object[arrayPos0].xvel = 0x3000
            object[arrayPos0].animationSpeed = 1
            object[arrayPos0].direction = FACING_RIGHT
            object[arrayPos0].launchSpeed = -0x50000
            break

        case DIR_FACING_LEFT_S
            // Set the animation to use streched arms
            object[arrayPos0].propertyValue = DIR_FACING_RIGHT_W_OBJ

            object[arrayPos0].xvel = 0x3000
            object[arrayPos0].animationSpeed = 1
            object[arrayPos0].direction = FACING_LEFT
            object[arrayPos0].launchSpeed = 0x50000
            break

        case DIR_FACING_RIGHT_W_OBJ
            // Set the animation to use un-streched arms
            object[arrayPos0].propertyValue = DIR_FACING_RIGHT

            object[arrayPos0].xvel = 0x6000
            object[arrayPos0].animationSpeed = 2
            object[arrayPos0].direction = FACING_RIGHT
            object[arrayPos0].launchSpeed = -0xA0000
            object[arrayPos0].caryingMonitor = true
            break
        end switch
    next

    temp0 = 0
    Scarab_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Scarab], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Scarab_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Scarab_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult %= 5
            break

        case 0 // type
            checkResult = object.propertyValue
            checkResult %= 5
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue %= 5
            break

        case 0 // type
            object.propertyValue = editor.variableValue
            object.propertyValue %= 5
            break
        end switch
    end if
end event


event RSDKDraw
    object.direction = FACING_RIGHT

    switch object.propertyValue
    case DIR_FACING_LEFT
        object.direction = FACING_LEFT
        // [Fallthrough]

    case DIR_FACING_RIGHT
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        break

    case DIR_FACING_LEFT_S
        object.direction = FACING_LEFT
        // [Fallthrough]

    case DIR_FACING_RIGHT_S
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)
        break

    case DIR_FACING_RIGHT_W_OBJ
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)

        if editor.showGizmos == true
            editor.drawingOverlay = true

            if object[+1].Type == TypeName[Monitor]
                DrawLine(object.xpos, object.ypos, object[+1].xpos, object[+1].ypos, 0, 0, 255)
            end if

            editor.drawingOverlay = false
        end if
        break
    end switch
end event

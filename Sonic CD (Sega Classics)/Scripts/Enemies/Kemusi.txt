            end if
        end if

            object.ypos = object.bodyPosition1.y

            end if
        end if

        object.ypos = temp5

        temp4 += temp2

event ObjectDraw
    if object.quality == QUALITY_GOOD
        GetBit(object.direction, object.pieceDirection, 3)
        DrawSpriteFX(2, FX_FLIP, object.bodyPosition3.x, object.bodyPosition3.y)

        GetBit(object.direction, object.pieceDirection, 2)
        DrawSpriteFX(2, FX_FLIP, object.bodyPosition2.x, object.bodyPosition2.y)

        GetBit(object.direction, object.pieceDirection, 1)
        DrawSpriteFX(2, FX_FLIP, object.bodyPosition1.x, object.bodyPosition1.y)
    else
        GetBit(object.direction, object.pieceDirection, 3)
        DrawSpriteFX(3, FX_FLIP, object.bodyPosition3.x, object.bodyPosition3.y)

        GetBit(object.direction, object.pieceDirection, 2)
        DrawSpriteFX(3, FX_FLIP, object.bodyPosition2.x, object.bodyPosition2.y)

        GetBit(object.direction, object.pieceDirection, 1)
        DrawSpriteFX(3, FX_FLIP, object.bodyPosition1.x, object.bodyPosition1.y)
    end if

    // Get the direction of the head
    GetBit(object.direction, object.pieceDirection, 0)

    switch object.state
    case KEMUSI_INIT
    case KEMUSI_FINDGROUND
    case KEMUSI_SPIKED_RETRACT
    case KEMUSI_NORMAL_RETRACT
        // Closed mouth frame
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        break

    case KEMUSI_SPIKED_ADVANCE
    case KEMUSI_NORMAL_ADVANCE
        // Open mouth frame
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        break
    end switch
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    temp0 = object.xpos
    temp0 += 0x240000

    DrawSpriteXY(object.propertyValue, temp0, object.ypos)

    temp0 -= 0xC0000
    DrawSpriteXY(object.propertyValue, temp0, object.ypos)

    temp0 -= 0xC0000
    DrawSpriteXY(object.propertyValue, temp0, object.ypos)

    temp0 -= 0xC0000
    DrawSpriteXY(2, temp0, object.ypos) // smile :)
end event

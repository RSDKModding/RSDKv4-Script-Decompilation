// ========================

private function Sasuri_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= Sasuri_startDebugID
    if temp0 == 0
        // Good quality
        DrawSprite(2)
        DrawSprite(0)
        DrawSprite(3)
        DrawSprite(10)
    else
        // Bad quality
        DrawSprite(6)
        DrawSprite(1)
        DrawSprite(7)
        DrawSprite(12)
    end if
end function


private function Sasuri_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= Sasuri_startDebugID
    CreateTempObject(TypeName[Sasuri], temp0, object.xpos, object.ypos)
end function

event ObjectUpdate
    object.frame++
    object.frame %= 60

    if object.quality == QUALITY_GOOD
        switch object.state
        case SASURI_FINDGROUND
            ObjectTileCollision(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == true
                object.state = SASURI_CRAWLLEFT
            else
                object.ypos += 0x10000
            end if
            break

        case SASURI_CRAWLLEFT
            object.xpos -= 0x10000
            object.moveTimer--
            if object.moveTimer <= -80
                object.state = SASURI_CRAWLRIGHT
                object.direction = FACING_LEFT
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == false
                object.state = SASURI_CRAWLRIGHT
                object.moveTimer = 0
                object.direction = FACING_LEFT
            end if
            break

        case SASURI_CRAWLRIGHT
            object.xpos += 0x10000
            object.moveTimer++
            if object.moveTimer >= 80
                object.state = SASURI_CRAWLLEFT
                object.direction = FACING_RIGHT
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == false
                object.state = SASURI_CRAWLLEFT
                object.moveTimer = 0
                object.direction = FACING_RIGHT
            end if
            break

        case SASURI_SPOTPLAYER
            if object.cannonTimer < 30
                object.cannonTimer++
            else
                object.state = SASURI_SHOOT
                object.cannonTimer = 0
            end if
            break

        case SASURI_SHOOT
            if object.cannonTimer < 40
                if object.cannonTimer == 10
                    PlaySfx(SfxName[Shot], false)

                    CreateTempObject(TypeName[Sasuri Shot], 0, object.xpos, object.ypos)
                    object[tempObjectPos].ypos -= 0x180000

                    if object.direction == FACING_RIGHT
                        object[tempObjectPos].xpos -= 0x70000
                        object[tempObjectPos].xvel = -0x30000
                    else
                        object[tempObjectPos].xpos += 0x70000
                        object[tempObjectPos].xvel = 0x30000
                    end if

                    object.drawOrder = 4
                end if

                object.cannonTimer++
            else
                object.state = SASURI_RETRACTCANNON
                object.cannonTimer = 0
                object.drawOrder = 3
            end if
            break

        case SASURI_RETRACTCANNON
            if object.cannonTimer < 30
                object.cannonTimer++
            else
                if object.direction == FACING_RIGHT
                    object.state = SASURI_CRAWLLEFT
                else
                    object.state = SASURI_CRAWLRIGHT
                end if
            end if
            break
        end switch
    else
        switch object.state
        case SASURI_FINDGROUND
            ObjectTileGrip(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == true
                object.state = SASURI_CRAWLLEFT
            else
                object.ypos += 0x10000
            end if
            break

        case SASURI_CRAWLLEFT
            object.xpos -= 0x8000
            object.moveTimer--
            if object.moveTimer <= -160
                object.state = SASURI_CRAWLRIGHT
                object.direction = FACING_LEFT
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == false
                object.state = SASURI_CRAWLRIGHT
                object.moveTimer = 0
                object.direction = FACING_LEFT
            end if
            break

        case SASURI_CRAWLRIGHT
            object.xpos += 0x8000
            object.moveTimer++
            if object.moveTimer >= 160
                object.state = SASURI_CRAWLLEFT
                object.direction = FACING_RIGHT
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 13, 0)
            if checkResult == false
                object.state = SASURI_CRAWLLEFT
                object.moveTimer = 0
                object.direction = FACING_RIGHT
            end if
            break

        case SASURI_SPOTPLAYER
            if object.cannonTimer < 30
                object.cannonTimer++
            else
                object.state = SASURI_SHOOT
                object.cannonTimer = 0
            end if
            break

        case SASURI_SHOOT
            if object.cannonTimer < 40
                object.cannonTimer++
            else
                object.state = SASURI_RETRACTCANNON
                object.cannonTimer = 0
            end if
            break

        case SASURI_RETRACTCANNON
            if object.cannonTimer < 30
                object.cannonTimer++
            else
                if object.direction == FACING_RIGHT
                    object.state = SASURI_CRAWLLEFT
                else
                    object.state = SASURI_CRAWLRIGHT
                end if
            end if
            break
        end switch
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -22, -14, 22, 14, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if

        switch object.state
        case SASURI_CRAWLLEFT
        case SASURI_CRAWLRIGHT
            if object.cannonTimer == 0
                BoxCollisionTest(C_TOUCH, object.entityPos, -64, -24, 64, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state = SASURI_SPOTPLAYER

                    if player[currentPlayer].xpos < object.xpos
                        object.direction = FACING_RIGHT
                    else
                        object.direction = FACING_LEFT
                    end if
                end if
            else
                object.cannonTimer = 0
            end if
            break
        end switch
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event


event ObjectDraw
    if object.quality == QUALITY_GOOD
        if object.frame < 30
            DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
        else
            DrawSpriteFX(4, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        end if

        switch object.state
        case SASURI_FINDGROUND
        case SASURI_CRAWLLEFT
        case SASURI_CRAWLRIGHT
        case SASURI_SPOTPLAYER
        case SASURI_RETRACTCANNON
            DrawSpriteFX(10, FX_FLIP, object.xpos, object.ypos)
            break

        case SASURI_SHOOT
            DrawSpriteFX(11, FX_FLIP, object.xpos, object.ypos)
            break
        end switch
    else
        if object.frame < 30
            DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(7, FX_FLIP, object.xpos, object.ypos)
        else
            DrawSpriteFX(8, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
            DrawSpriteFX(9, FX_FLIP, object.xpos, object.ypos)
        end if

        switch object.state
        case SASURI_FINDGROUND
        case SASURI_CRAWLLEFT
        case SASURI_CRAWLRIGHT
        case SASURI_SPOTPLAYER
        case SASURI_RETRACTCANNON
            DrawSpriteFX(12, FX_FLIP, object.xpos, object.ypos)
            break

        case SASURI_SHOOT
            DrawSpriteFX(13, FX_FLIP, object.xpos, object.ypos)
            break
        end switch
    end if
end event

    SpriteFrame(-1, -24, 16, 16, 133, 110)
    SpriteFrame(3, -29, 16, 16, 133, 110)

    temp0 = 0
    Sasuri_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Sasuri], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Sasuri_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Sasuri_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    if object.propertyValue == QUALITY_GOOD
        DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(6, FX_FLIP, object.xpos, object.ypos)
    else
        DrawSpriteFX(4, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
        DrawSpriteFX(7, FX_FLIP, object.xpos, object.ypos)
    end if
end event

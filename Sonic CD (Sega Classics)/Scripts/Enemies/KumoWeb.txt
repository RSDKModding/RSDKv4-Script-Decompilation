// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.attachedPlayer

// Player Aliases
private alias object.xpos  : player.xpos
private alias object.ypos  : player.ypos
private alias object.speed : player.speed

// States
private alias 0 : KUMOKUMOWEB_THROWN
private alias 1 : KUMOKUMOWEB_ATTACHED

// ========================
// Events
// ========================

event ObjectUpdate
    if object.state == KUMOKUMOWEB_THROWN
        // Fly through the air
        object.xpos += object.xvel
        if object.outOfBounds == true
            // Unload if out of bounds
            object.type = TypeName[Blank Object]
        else
            temp0 = screen.yoffset
            temp0 -= 8
            temp0 <<= 16
            if object.ypos < temp0
                // If far vertically too, then unload too
                object.type = TypeName[Blank Object]
            end if
        end if
    else
        // Attached to a Player
        if object.timer < 120
            object.timer++

            // Slow the Player down to a maximum speed of 0.25 pixels a second
            currentPlayer = object.attachedPlayer

            // Right-wards limit
            if player[currentPlayer].speed > 0x4000
                player[currentPlayer].speed = 0x4000
            end if

            // Left-wards limit
            if player[currentPlayer].speed < -0x4000
                player[currentPlayer].speed = -0x4000
            end if
        else
            // The Web's sticky time is over, unload now
            object.type = TypeName[Blank Object]
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        // Check to see if the Web should attach to Sonic
        // (Although it doesn't matter too much anyway, this check continues to run even if the Web is already in its ATTACHED state)
        BoxCollisionTest(C_TOUCH, object.entityPos, -6, -6, 6, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.state = KUMOKUMOWEB_ATTACHED
            object.attachedPlayer = currentPlayer
        end if
    next
end event


event ObjectDraw
    if object.state == KUMOKUMOWEB_ATTACHED
        // Attached Webs should stick to the Player
        currentPlayer = object.attachedPlayer
        object.xpos = player[currentPlayer].xpos
        object.ypos = player[currentPlayer].ypos
    end if

    temp0 = object.frame
    temp0 /= 3
    DrawSprite(temp0)

    object.frame++
    object.frame %= 6
end event


event ObjectStartup
    LoadSpriteSheet("R5/Objects.gif")

    // Kumo Kumo Web Frames
    SpriteFrame(-12, -8, 24, 16, 69, 1)
    SpriteFrame(-12, -8, 24, 16, 94, 1)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R5/Objects.gif")
    SpriteFrame(-12, -8, 24, 16, 69, 1)
    
    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

// Aliases
private alias object.value0        : object.timer
private alias object.propertyValue : object.quality

// Player Aliases
private alias object.value16 : debugMode.currentSelection

// States
private alias 0 : ANTON_FINDGROUND
private alias 1 : ANTON_MOVING_LEFT
private alias 2 : ANTON_MOVING_RIGHT

// Badnik Quality / Property Values
private alias 0 : QUALITY_GOOD
private alias 1 : QUALITY_BAD

// ========================
// Function Declarations
// ========================

reserve function Anton_DebugDraw
reserve function Anton_DebugSpawn

// ========================
// Static Values
// ========================

private value Anton_startDebugID = 0

// ========================
// Function Definitions
// ========================

private function Anton_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= Anton_startDebugID
    DrawSprite(4)
    if temp0 == 0
        DrawSprite(0) // Good quality
    else
        DrawSprite(2) // Bad quality
    end if
end function


private function Anton_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= Anton_startDebugID
    CreateTempObject(TypeName[Anton], temp0, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
    if object.quality == QUALITY_GOOD
        object.frame += 2

        switch object.state
        case ANTON_FINDGROUND
            ObjectTileCollision(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == true
                object.state = ANTON_MOVING_LEFT
            else
                object.ypos += 0x20000
            end if
            break

        case ANTON_MOVING_LEFT
            object.xpos -= 0x10000
            object.timer--
            if object.timer == -128
                object.state = ANTON_MOVING_RIGHT
                object.direction = FLIP_X
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == false
                object.state = ANTON_MOVING_RIGHT
                object.timer = -128
                object.direction = FLIP_X
            end if
            break

        case ANTON_MOVING_RIGHT
            object.xpos += 0x10000
            object.timer++
            if object.timer == 128
                object.state = ANTON_MOVING_LEFT
                object.direction = FLIP_NONE
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == false
                object.state = ANTON_MOVING_LEFT
                object.timer = 128
                object.direction = FLIP_NONE
            end if
            break
        end switch
    else
        object.frame++

        switch object.state
        case ANTON_FINDGROUND
            ObjectTileCollision(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == true
                object.state = ANTON_MOVING_LEFT
            else
                object.ypos += 0x20000
            end if
            break

        case ANTON_MOVING_LEFT
            object.xpos -= 0x8000
            object.timer--
            if object.timer == -256
                object.state = ANTON_MOVING_RIGHT
                object.direction = FLIP_X
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == false
                object.state = ANTON_MOVING_RIGHT
                object.timer = -256
                object.direction = FLIP_X
            end if
            break

        case ANTON_MOVING_RIGHT
            object.xpos += 0x8000
            object.timer++
            if object.timer == 256
                object.state = ANTON_MOVING_LEFT
                object.direction = FLIP_NONE
            end if

            ObjectTileGrip(CSIDE_FLOOR, 0, 20, 0)
            if checkResult == false
                object.state = ANTON_MOVING_LEFT
                object.timer = 256
                object.direction = FLIP_NONE
            end if
            break
        end switch
    end if

    object.frame &= 15

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -12, -18, 12, 20, currentPlayer, object[currentPlayer].value41, object[currentPlayer].value39, object[currentPlayer].value42, object[currentPlayer].value40)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if
    next
end event


event ObjectDraw
    temp0 = object.frame
    temp0 >>= 3

    if object.quality == QUALITY_GOOD
        temp0 += 4
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Wheel

        temp0 -= 4
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Anton

        temp0 += 6
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Smoke Puff
    else
        temp0 += 4

        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Wheel

        temp0 -= 2
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Anton

        temp0 += 4
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos) // Smoke Puff
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R1/Objects.gif")

    SpriteFrame(-12, -19, 24, 32, 1, 1)   // #0 - Anton Frame 0
    SpriteFrame(-12, -20, 24, 32, 1, 1)   // #1 - Anton Frame 1
    SpriteFrame(-12, -19, 24, 32, 26, 1)  // #2 - Old Anton Frame 0
    SpriteFrame(-12, -20, 24, 32, 26, 1)  // #3 - Old Anton Frame 1

    SpriteFrame(-8, 4, 16, 16, 1, 34)     // #4 - Anton Wheel Frame 0
    SpriteFrame(-8, 4, 16, 16, 18, 34)    // #5 - Anton Wheel Frame 1

    SpriteFrame(14, -4, 8, 8, 35, 34)     // #6 - Anton Smoke Puff Frame 0
    SpriteFrame(18, -5, 8, 8, 35, 34)     // #7 - Anton Smoke Puff Frame 1

    temp0 = 0
    Anton_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Anton], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Anton_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Anton_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
        case 0 // condition
            GetBit(checkResult, object.quality, 0)
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
        case 0 // condition
            object.quality = editor.variableValue
            object.quality &= QUALITY_BAD
            break
        end switch
    end if
end event


event RSDKDraw
    // Wheel
    object.frame = object.quality
    object.frame += 2
    DrawSprite(object.frame)

    // Anton
    DrawSprite(object.quality)
end event


event RSDKLoad
    LoadSpriteSheet("R1/Objects.gif")

    SpriteFrame(-12, -19, 24, 32, 1, 1)     // #0 - Anton
    SpriteFrame(-12, -19, 24, 32, 26, 1)    // #1 - Old Anton

    SpriteFrame(-8, 4, 16, 16, 1, 34)       // #2 - Anton Wheel
    SpriteFrame(-8, 4, 16, 16, 18, 34)      // #3 - Old Anton Wheel

    AddEditorVariable("condition")
    SetActiveVariable("condition")
    AddEnumVariable("Good Quality", QUALITY_GOOD)
    AddEnumVariable("Bad Quality",  QUALITY_BAD)
end event

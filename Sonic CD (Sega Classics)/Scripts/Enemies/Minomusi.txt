event ObjectUpdate
    if object.outOfBounds == true
        object.state = MINOMUSI_DETECT_PLAYER
        object.timer = 0
        object.spikeTimer = 0
        object.ypos = object.originPos.y
        object.priority = PRIORITY_BOUNDS
    end if

    if object.quality == QUALITY_GOOD
        switch object.state
        case MINOMUSI_DETECT_PLAYER
            object.frame = 0
            if object.timer > 0
                object.timer--
            end if
            break

        case MINOMUSI_DROP
            if object.timer < 12
                object.ypos += 0x80000
                object.timer++
            else
                object.timer = 0
                object.state++
            end if
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_1
            if object.spikeTimer < 30
                object.spikeTimer++
            else
                object.spikeTimer = 0
                object.state++
                object.frame = 3
            end if
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_2
            object.state++
            object.frame = 4
            PlaySfx(SfxName[Spikes], false)
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_3
            if object.spikeTimer < 3
                object.spikeTimer++
            else
                object.spikeTimer = 0
                object.state++
                object.frame = 3
            end if
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_4
            object.state++
            object.frame = 0
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_5
            object.state++
            object.frame = 3
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_6
            object.state++
            object.frame = 4
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_7
            if object.spikeTimer < 30
                object.spikeTimer++
            else
                object.spikeTimer = 0
                object.state++
                object.frame = 3
            end if
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_8
            object.frame = 0
            if object.timer < 2
                object.state = MINOMUSI_GOOD_SPIKE_ANIM_1
                object.timer++
            else
                object.state++
                object.timer = 0
            end if
            break

        case MINOMUSI_GOOD_SPIKE_ANIM_9
            if object.timer < 20
                object.timer++
            else
                object.timer = 0
                object.state++
            end if
            break

        case MINOMUSI_GOOD_RETIRE
            if object.timer < 12
                object.ypos -= 0x80000
                object.timer++
            else
                object.timer = 120
                object.state = MINOMUSI_DETECT_PLAYER
            end if
            break
        end switch
    else
        switch object.state
        case MINOMUSI_DETECT_PLAYER
            object.frame = 1
            if object.timer > 0
                object.timer--
            end if
            break

        case MINOMUSI_DROP
            object.frame = 1
            if object.timer < 12
                object.ypos += 0x80000
                object.timer++
            else
                object.timer = 0
                object.state++
            end if
            break

        case MINOMUSI_BAD_FLIP
            if object.spikeTimer < 10
                object.spikeTimer++
            else
                object.spikeTimer = 0
                object.state++
                object.frame = 2
            end if
            break

        case MINOMUSI_BAD_FLIP_2
            if object.spikeTimer < 10
                object.spikeTimer++
            else
                object.spikeTimer = 0
                object.frame = 1
                object.timer++
                if object.timer == 3
                    object.timer = 0
                    object.state++
                else
                    object.state = MINOMUSI_BAD_FLIP
                end if
            end if
            break

        case MINOMUSI_BAD_RETIRE
            if object.timer < 48
                object.ypos -= 0x20000
                object.timer++
            else
                object.timer = 120
                object.state = MINOMUSI_DETECT_PLAYER
            end if
            break
        end switch
    end if

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -8, -6, 8, 22, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if

        if object.quality == QUALITY_GOOD
            if object.type == TypeName[Minomusi]
                if object.frame > 2
                    BoxCollisionTest(C_TOUCH, object.entityPos, -17, -4, 17, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                    if checkResult == true
                        CallFunction(Player_Hit)
                    end if
                end if
            end if
        end if

        if object.state == MINOMUSI_DETECT_PLAYER
            if object.timer == 0
                BoxCollisionTest(C_TOUCH, object.entityPos, -64, 0, 64, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.state = MINOMUSI_DROP
                    object.priority = PRIORITY_ACTIVE
                end if
            end if
        end if
    next
end event


event ObjectDraw
    temp0 = object.ypos
    temp0 -= object.originPos.y
    temp0 >>= 19
    temp0 += 5
    DrawSpriteXY(temp0, object.xpos, object.originPos.y)

    DrawSprite(object.frame)
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event

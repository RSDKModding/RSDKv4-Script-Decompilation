// ========================

private function Ga_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= Ga_startDebugID
    if temp0 == 0
        DrawSprite(3) // Good quality
    else
        DrawSprite(7) // Bad quality
    end if
end function


private function Ga_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= Ga_startDebugID
    CreateTempObject(TypeName[Ga], temp0, object.xpos, object.ypos)
end function

// Events

event ObjectUpdate
    switch object.state
    case GA_IDLE
        BoxCollisionTest(C_TOUCH, object.entityPos, -96, -256, 96, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.drawOrder = 5

            object.state = GA_REVEAL

            object.frameTimer = 0
            object.frame = 0

            object.priority = PRIORITY_ACTIVE

            object.originPos.x = object.xpos
            object.originPos.y = object.ypos
        end if
        break

    case GA_RESPAWN
        if object.outOfBounds == true
            object.state = GA_IDLE
            if object.priority != PRIORITY_XBOUNDS_DESTROY
                object.priority = PRIORITY_BOUNDS
            end if
        end if
        break

    case GA_REVEAL
        if object.timer < 110
            object.timer++
            if object.timer == 30
                object.frame++
            end if

            if object.timer == 60
                object.frame++
            end if
        else
            if object.xpos > object.xpos
                object.xvel = -0x8000
                object.direction = FACING_RIGHT
            else
                object.xvel = 0x8000
                object.direction = FACING_LEFT
            end if

            object.state = GA_FLY
            object.timer = 0

            if object.quality > QUALITY_GOOD
                object.xvel >>= 1
            end if

            object.yvel = 0x8000
        end if

        if object.outOfBounds == true
            object.state = GA_RESPAWN
            object.timer = 0

            object.xpos = object.originPos.x
            object.ypos = object.originPos.y
        end if
        break

    case GA_FLY
        object.xpos += object.xvel
        object.ypos += object.yvel

        if object.timer < 65
            object.timer++
        else
            object.timer = 0
            FlipSign(object.yvel)

            // Only good quality Ga entities drop sparkles
            if object.quality == QUALITY_GOOD
                CreateTempObject(TypeName[Ga Sparkle], 0, object.xpos, object.ypos)
                object[tempObjectPos].drawOrder = 5
            end if
        end if

        object.frameTimer++
        object.frameTimer %= 20

        object.frame = object.frameTimer
        object.frame /= 5

        if object.quality == QUALITY_GOOD
            object.frame += 3
        else
            object.frame += 7
        end if

        if object.outOfBounds == true
            object.state = GA_RESPAWN
            object.timer = 0

            object.xpos = object.originPos.x
            object.ypos = object.originPos.y
        end if
        break
    end switch

    if object.state > GA_REVEAL
        foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
            BoxCollisionTest(C_TOUCH, object.entityPos, -16, -8, 16, 8, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
            if checkResult == true
                CallFunction(Player_BadnikBreak)
            end if
        next
    end if

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event


event ObjectDraw
    if object.state > GA_RESPAWN
        DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
    end if
end event

    SpriteFrame(-12, -8, 32, 16, 165, 183)  // #4  - Ga Fly Frame 1
    SpriteFrame(-12, -8, 32, 16, 166, 233)  // #5  - Ga Fly Frame 2
    SpriteFrame(-12, -8, 32, 16, 165, 183)  // #6  - Ga Fly Frame 3

    // Old Ga
    SpriteFrame(-12, -12, 32, 24, 1, 157)   // #7  - Ga Fly Frame 0
    SpriteFrame(-12, -8, 32, 16, 1, 182)    // #8  - Ga Fly Frame 1
    SpriteFrame(-12, -8, 32, 16, 1, 199)    // #9  - Ga Fly Frame 2
    SpriteFrame(-12, -8, 32, 16, 1, 182)    // #10 - Ga Fly Frame 3

    temp0 = 0
    Ga_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Ga], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Ga_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Ga_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

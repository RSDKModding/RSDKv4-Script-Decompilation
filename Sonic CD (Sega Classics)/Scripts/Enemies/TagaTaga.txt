private function TagaTaga_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= TagaTaga_startDebugID
    if temp0 == 0
        DrawSprite(0) // Good quality
    else
        DrawSprite(6) // Bad quality
    end if
end function


private function TagaTaga_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= TagaTaga_startDebugID
    CreateTempObject(TypeName[Taga Taga], temp0, object.xpos, object.ypos)
    object[tempObjectPos].originPos.y = object[tempObjectPos].ypos
end function

event ObjectUpdate
    if object.quality == QUALITY_GOOD
        switch object.state
        case TAGATAGA_SET_LAUNCH_SPEED
            object.yvel = -0x40000
            object.timer = 1
            object.state = TAGATAGA_EMERGE
            break

        case TAGATAGA_EMERGE
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 24
                object.state = TAGATAGA_MOVE_PINCH_1
            end if
            break

        case TAGATAGA_MOVE_PINCH_1
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 48
                object.state = TAGATAGA_TURN_1
            end if
            break

        case TAGATAGA_TURN_1
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 56
                object.state = TAGATAGA_TURN_2
            end if
            break

        case TAGATAGA_TURN_2
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 64
                object.state = TAGATAGA_TURN_3
            end if
            break

        case TAGATAGA_TURN_3
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 96
                object.state = TAGATAGA_MOVE_PINCH_2
            end if
            break

        case TAGATAGA_MOVE_PINCH_2
            object.yvel += 0x1000
            object.ypos += object.yvel
            object.timer++
            if object.timer > 128
                object.state = TAGATAGA_SUBMERGE
                PlaySfx(SfxName[Water Splash], false)
            end if
            break

        case TAGATAGA_SUBMERGE
            object.timer++
            if object.timer > 204
                object.state = TAGATAGA_SET_LAUNCH_SPEED
                object.timer = 0
                object.ypos = object.originPos.y
                PlaySfx(SfxName[Water Splash], false)
            end if
            break
        end switch
    else
        switch object.state
        case TAGATAGA_SET_LAUNCH_SPEED
            object.yvel = -0x30000
            object.timer = 1
            object.state = TAGATAGA_EMERGE
            break

        case TAGATAGA_EMERGE
            object.yvel += 0xEC4
            object.ypos += object.yvel
            object.timer++
            if object.timer > 40
                object.state = TAGATAGA_MOVE_PINCH_1
            end if
            break

        case TAGATAGA_MOVE_PINCH_1
            object.yvel += 0xEC4
            object.ypos += object.yvel
            object.timer++
            if object.timer > 50
                object.state = TAGATAGA_TURN_1
            end if
            break

        case TAGATAGA_TURN_1
            object.yvel += 0xEC4
            object.ypos += object.yvel
            object.timer++
            if object.timer > 60
                object.state = TAGATAGA_TURN_2
            end if
            break

        case TAGATAGA_TURN_2
            object.yvel += 0xEC4
            object.ypos += object.yvel
            object.timer++
            if object.timer > 104
                object.state = TAGATAGA_TURN_3
                PlaySfx(SfxName[Water Splash], false)
            end if
            break

        case TAGATAGA_TURN_3
            object.timer++
            if object.timer > 188
                object.state = TAGATAGA_SET_LAUNCH_SPEED
                object.timer = 0
                object.ypos = object.originPos.y
                PlaySfx(SfxName[Water Splash], false)
            end if
            break
        end switch
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        if object.quality == QUALITY_GOOD
            if object.state < TAGATAGA_SUBMERGE
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
                if checkResult == true
                    CallFunction(Player_BadnikBreak)
                end if
            end if
        else
            if object.state < TAGATAGA_TURN_3
                BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
                if checkResult == true
                    CallFunction(Player_BadnikBreak)
                end if
            end if
        end if
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event


event ObjectDraw
    if object.quality == QUALITY_GOOD
        switch object.state
        case TAGATAGA_SET_LAUNCH_SPEED
        case TAGATAGA_EMERGE
            DrawSprite(0)
            break

        case TAGATAGA_MOVE_PINCH_1
            temp0 = object.timer
            temp0 &= 15
            if temp0 < 8
                DrawSprite(0)
            else
                DrawSprite(1)
            end if
            break

        case TAGATAGA_TURN_1
            DrawSprite(2)
            break

        case TAGATAGA_TURN_2
            DrawSprite(3)
            break

        case TAGATAGA_TURN_3
            DrawSprite(4)
            break

        case TAGATAGA_MOVE_PINCH_2
            temp0 = object.timer
            temp0 &= 15
            if temp0 < 8
                DrawSprite(4)
            else
                DrawSprite(5)
            end if
            break
        end switch
    else
        switch object.state
        case TAGATAGA_SET_LAUNCH_SPEED
        case TAGATAGA_EMERGE
            DrawSprite(6)
            break

        case TAGATAGA_MOVE_PINCH_1
            DrawSprite(7)
            break

        case TAGATAGA_TURN_1
            DrawSprite(8)
            break

        case TAGATAGA_TURN_2
            DrawSprite(9)
            break
        end switch
    end if
end event

    TagaTaga_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Taga Taga], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(TagaTaga_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(TagaTaga_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event

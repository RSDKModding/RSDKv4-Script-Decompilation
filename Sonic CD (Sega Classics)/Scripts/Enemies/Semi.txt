event ObjectUpdate
    if object.outOfBounds == true
        object.state = SEMI_RESET
        object.timer = 0

        object.xpos = object.originPos.x
        object.ypos = object.originPos.y
    end if

    switch object.state
    case SEMI_START_FLY
        object.priority = PRIORITY_ACTIVE

        object.xpos += object.xvel
        object.ypos += object.yvel

        if object.timer < 96
            object.timer++
        else
            object.timer = 0
            object.state++
        end if
        break

    case SEMI_BOMB_THROW
        object.xpos += object.xvel
        if object.quality == QUALITY_GOOD
            if object.timer == 0
                object.timer = 60
                CreateTempObject(TypeName[Semi Bomb], 0, object.xpos, object.ypos)
                object[tempObjectPos].ypos += 0xC0000
            else
                object.timer--
            end if
        end if
        break

    case SEMI_RESET
        if object.outOfBounds == true
            object.state = SEMI_IDLE
            object.priority = PRIORITY_BOUNDS
        end if
        break
    end switch

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        switch object.state
        case SEMI_IDLE
            BoxCollisionTest(C_TOUCH, object.entityPos, -96, -128, 96, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                object.state = SEMI_START_FLY
                if object.xpos > player[currentPlayer].xpos
                    if object.quality == QUALITY_GOOD
                        object.xvel = 0x10000
                        object.yvel = -0x8000
                    else
                        object.xvel = 0xC000
                        object.yvel = 0x6000
                    end if
                else
                    if object.quality == QUALITY_GOOD
                        object.xvel = -0x10000
                        object.yvel = -0x8000
                    else
                        object.xvel = -0xC000
                        object.yvel = 0x6000
                    end if
                end if
            end if
            break

        case SEMI_START_FLY
        case SEMI_BOMB_THROW
            BoxCollisionTest(C_TOUCH, object.entityPos, -14, -12, 14, 12, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
            if checkResult == true
                CallFunction(Player_BadnikBreak)
            end if
            break
        end switch
    next
end event


event ObjectDraw
    switch object.state
    case SEMI_IDLE
        DrawSprite(object.quality)
        break

    case SEMI_START_FLY
    case SEMI_BOMB_THROW
        temp0 = object.frame
        temp0 >>= 2

        if object.quality == QUALITY_GOOD
            temp0 += 2
        else
            temp0 += 6
        end if

        if object.frame > 7
            object.direction = FACING_LEFT
        else
            object.direction = FACING_RIGHT
        end if

        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
        object.frame++
        object.frame &= 15
        break
    end switch
end event

        object[arrayPos0].originPos.x = object[arrayPos0].xpos
        object[arrayPos0].originPos.y = object[arrayPos0].ypos
        object[arrayPos0].drawOrder = 5
    next

    temp0 = 0
    Semi_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Semi], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Semi_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Semi_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event

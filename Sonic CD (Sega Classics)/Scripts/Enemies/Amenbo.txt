private function Amenbo_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= Amenbo_startDebugID
    if temp0 == 0
        DrawSprite(0) // Good quality
    else
        DrawSprite(2) // Bad quality
    end if
end function


private function Amenbo_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= Amenbo_startDebugID
    CreateTempObject(TypeName[Amenbo], temp0, object.xpos, object.ypos)
end function

event ObjectUpdate
    if object.quality == QUALITY_GOOD
        switch object.state
        case WATERSKIMMER_DASH_LEFT
            if object.xvel > -0x18000
                object.xvel -= 0x400
                object.xpos += object.xvel
            else
                object.state++
            end if
            object.frame++
            object.frame %= 22
            break

        case WATERSKIMMER_FLOAT_LEFT
            if object.xvel < 0
                object.xvel += 280
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.state++
                object.timer = 60
            end if
            break

        case WATERSKIMMER_TURN_TO_RIGHT
            if object.timer > 0
                object.timer--
            else
                object.state++
                object.direction = FACING_LEFT
                object.timer = 60
            end if
            break

        case WATERSKIMMER_WAIT
            if object.timer > 0
                object.timer--
            else
                object.state++
            end if
            break

        case WATERSKIMMER_DASH_RIGHT
            if object.xvel < 0x18000
                object.xvel += 0x400
                object.xpos += object.xvel
            else
                object.state++
            end if
            object.frame++
            object.frame %= 22
            break

        case WATERSKIMMER_FLOAT_RIGHT
            if object.xvel > 0
                object.xvel -= 280
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.state++
                object.timer = 60
            end if
            break

        case WATERSKIMMER_TURN_TO_LEFT
            if object.timer > 0
                object.timer--
            else
                object.state++
                object.direction = FACING_RIGHT
                object.timer = 60
            end if
            break

        case WATERSKIMMER_CHARGE_LEFT
            if object.timer > 0
                object.timer--
            else
                object.state = WATERSKIMMER_DASH_LEFT
            end if
            break

        end switch

        if object.bulletTimer > 0
            object.bulletTimer--
            if object.bulletTimer == 0
                PlaySfx(SfxName[Shot], false)
                CreateTempObject(TypeName[Amenbo Shot], 0, object.xpos, object.ypos)
                if object.direction == 0
                    object[tempObjectPos].xvel = -0x20000
                else
                    object[tempObjectPos].xvel = 0x20000
                end if
            end if
        end if
    else
        switch object.state
        case WATERSKIMMER_DASH_LEFT
            if object.xvel > -0x10000
                object.xvel -= 0x400
                object.xpos += object.xvel
            else
                object.state++
            end if
            object.frame++
            object.frame %= 22
            break

        case WATERSKIMMER_FLOAT_LEFT
            if object.xvel < 0
                object.xvel += 280
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.state++
                object.timer = 60
            end if
            break

        case WATERSKIMMER_TURN_TO_RIGHT
            if object.timer > 0
                object.timer--
            else
                object.state++
                object.direction = FACING_LEFT
                object.timer = 60
            end if
            break

        case WATERSKIMMER_CHARGE_RIGHT
            if object.timer > 0
                object.timer--
            else
                object.state++
            end if
            break

        case WATERSKIMMER_DASH_RIGHT
            if object.xvel < 0x10000
                object.xvel += 0x400
                object.xpos += object.xvel
            else
                object.state++
            end if
            object.frame++
            object.frame %= 22
            break

        case WATERSKIMMER_FLOAT_RIGHT
            if object.xvel > 0
                object.xvel -= 280
                object.xpos += object.xvel
            else
                object.xvel = 0
                object.state++
                object.timer = 60
            end if
            break

        case WATERSKIMMER_TURN_TO_LEFT
            if object.timer > 0
                object.timer--
            else
                object.state++
                object.direction = FACING_RIGHT
                object.timer = 60
            end if
            break

        case WATERSKIMMER_CHARGE_LEFT
            if object.timer > 0
                object.timer--
            else
                object.state = WATERSKIMMER_DASH_LEFT
            end if
            break

        end switch
    end if

    object.ypos = stage.waterLevel
    object.ypos -= 10
    object.ypos <<= 16

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -12, -8, 12, 8, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if
        if object.bulletTimer == 0
            if object.direction == FACING_RIGHT
                BoxCollisionTest(C_TOUCH, object.entityPos, -96, -16, 0, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.bulletTimer = 120
                end if
            else
                BoxCollisionTest(C_TOUCH, object.entityPos, 0, -16, 96, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
                if checkResult == true
                    object.bulletTimer = 120
                end if
            end if
        end if
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event


event ObjectDraw
    if object.quality == QUALITY_GOOD
        temp0 = object.frame
        temp0 /= 11
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
    else
        temp0 = object.frame
        temp0 /= 11
        temp0 += 2
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects.gif")

    SpriteFrame(-20, -12, 32, 24, 67, 231)  // #0 - Water Skimmer frame 0
    SpriteFrame(-20, -12, 32, 24, 100, 231) // #1 - Water Skimmer frame 1

    SpriteFrame(-20, -12, 32, 24, 1, 231)   // #2 - Broken Water Skimmer frame 0
    SpriteFrame(-20, -12, 32, 24, 34, 231)  // #3 - Broken Water Skimmer frame 1

    temp0 = 0
    Amenbo_startDebugID = DebugMode_ObjCount
    while temp0 < 2
        SetTableValue(TypeName[Amenbo], DebugMode_ObjCount, DebugMode_TypesTable)
        SetTableValue(Amenbo_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
        SetTableValue(Amenbo_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
        DebugMode_ObjCount++
        temp0++
    loop
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(object.propertyValue)
end event

    AddEnumVariable("Bad Quality", QUALITY_BAD)
end event

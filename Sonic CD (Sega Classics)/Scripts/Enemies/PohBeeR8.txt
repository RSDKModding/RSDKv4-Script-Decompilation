event ObjectUpdate
    if object.outOfBounds == true
        object.xpos = object.originPos.x
        object.timer = 0
        object.state = POHBEE_MOVING_RIGHT
    end if

    object.rotation += 9
    object.rotation &= 0x7FF

    if object.quality == QUALITY_GOOD
        spikeBallN2.rotation += 10
        spikeBallN2.rotation &= 0x7FF

        switch object.state
        case POHBEE_MOVING_RIGHT
            object.direction = FACING_LEFT
            if object.timer < 128
                object.timer++
                object.xpos += 0x10000
            else
                object.state = POHBEE_MOVING_LEFT
                object.timer = 0
            end if
            break

        case POHBEE_MOVING_LEFT
            object.direction = FACING_RIGHT
            if object.timer < 128
                object.timer++
                object.xpos -= 0x10000
            else
                object.state = POHBEE_MOVING_RIGHT
                object.timer = 0
            end if
            break
        end switch

        object.pohBeeOscillation += 8
        object.pohBeeOscillation &= 511
        Sin(object.ypos, object.pohBeeOscillation)
        object.ypos <<= 12
        object.ypos += object.originPos.y

        temp0 = object.rotation
        temp0 >>= 2
        Sin(spikeBallN1.oscillation, temp0)
        spikeBallN1.oscillation >>= 2
        spikeBallN1.oscillation += 136
        spikeBallN1.oscillation *= 120
        spikeBallN1.oscillation >>= 7

        temp0 = spikeBallN2.rotation
        temp0 >>= 2
        Sin(spikeBallN2.oscillation, temp0)
        spikeBallN2.oscillation >>= 2
        spikeBallN2.oscillation += 144
        spikeBallN2.oscillation *= 112
        spikeBallN2.oscillation >>= 7
    else
        switch object.state
        case POHBEE_MOVING_RIGHT
            object.direction = FACING_LEFT
            if object.timer < 256
                object.timer++
                object.xpos += 0x8000
            else
                object.state = POHBEE_MOVING_LEFT
                object.timer = 0
            end if
            break

        case POHBEE_MOVING_LEFT
            object.direction = FACING_RIGHT
            if object.timer < 256
                object.timer++
                object.xpos -= 0x8000
            else
                object.state = POHBEE_MOVING_RIGHT
                object.timer = 0
            end if
            break
        end switch

        object.pohBeeOscillation += 6
        object.pohBeeOscillation &= 511
        Sin(object.ypos, object.pohBeeOscillation)
        object.ypos <<= 11
        object.ypos += object.originPos.y

        temp0 = object.rotation
        temp0 >>= 2
        Sin(spikeBallN1.oscillation, temp0)
        spikeBallN1.oscillation >>= 2
        spikeBallN1.oscillation += 136
        spikeBallN1.oscillation *= 120
        spikeBallN1.oscillation >>= 7
    end if

    object.sballDistance = object.ypos
    object.sballDistance += 0xD0000

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -20, -16, 20, 14, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if

        temp0 = object.xpos
        temp1 = object.ypos

        Cos(object.xpos, spikeBallN1.oscillation)
        object.xpos *= 0x2A00
        object.xpos += temp0

        Sin(object.ypos, spikeBallN1.oscillation)
        object.ypos *= 0x2A00
        object.ypos += object.sballDistance

        BoxCollisionTest(C_TOUCH, object.entityPos, -9, -9, 9, 9, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            CallFunction(Player_Hit)
        end if

        if object.quality == QUALITY_GOOD
            Cos(object.xpos, spikeBallN2.oscillation)
            object.xpos *= 0x1200
            object.xpos += temp0

            Sin(object.ypos, spikeBallN2.oscillation)
            object.ypos *= 0x1200
            object.ypos += object.sballDistance

            BoxCollisionTest(C_TOUCH, object.entityPos, -9, -9, 9, 9, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
            if checkResult == true
                CallFunction(Player_Hit)
            end if
        end if

        object.xpos = temp0
        object.ypos = temp1
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event

private function KamaKama_DebugDraw
    temp0 = debugMode[0].currentSelection
    temp0 -= KamaKama_startDebugID
    DrawSprite(0)

    if temp0 == 0
        DrawSprite(4) // Body, Good quality
        DrawSprite(6) // Head, Good quality
    else
        DrawSprite(10) // Body, Bad quality
        DrawSprite(12) // Head, Bad quality
    end if
end function


private function KamaKama_DebugSpawn
    temp0 = debugMode[0].currentSelection
    temp0 -= KamaKama_startDebugID
    CreateTempObject(TypeName[Kama Kama], temp0, object.xpos, object.ypos)
    object[tempObjectPos].originPos.x = object.xpos
    object[tempObjectPos].originPos.y = object.ypos
end function

event ObjectUpdate
    object.animationTimer++
    object.animationTimer &= 15

    switch object.state
    case KAMAKAMA_FINDGROUND
        ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
        if checkResult == true
            object.state = KAMAKAMA_GUARD
        else
            object.ypos += 0x20000
        end if
        break

    case KAMAKAMA_GUARD
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0)
        if object.xpos > player[SLOT_PLAYER1].xpos
            object.direction = FACING_RIGHT
        else
            object.direction = FACING_LEFT
        end if

        BoxCollisionTest(C_TOUCH, object.entityPos, -96, -48, 96, 48, 0, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            object.state = KAMAKAMA_LAUNCH_BLADE
            object.priority = PRIORITY_ACTIVE

            // Only badniks in a good state still have blades to throw
            if object.quality == QUALITY_GOOD
                CreateTempObject(TypeName[Kama Kama Blade], 24, object.xpos, object.ypos)
                object[tempObjectPos].direction = object.direction
                object[tempObjectPos].iypos -= 6
                object[tempObjectPos].kamakamaEntityPos = object.entityPos

                if object.direction == FACING_RIGHT
                    object[tempObjectPos].ixpos -= 8
                else
                    object[tempObjectPos].ixpos += 8
                end if

                CreateTempObject(TypeName[Kama Kama Blade], 56, object.xpos, object.ypos)
                object[tempObjectPos].direction = object.direction
                object[tempObjectPos].iypos -= 4
                object[tempObjectPos].kamakamaEntityPos = object.entityPos

                if object.direction == FACING_RIGHT
                    object[tempObjectPos].ixpos -= 16
                else
                    object[tempObjectPos].ixpos += 16
                end if
            end if
        end if
        break

    case KAMAKAMA_LAUNCH_BLADE
        if object.timer < 110
            object.timer++
        else
            object.state = KAMAKAMA_PREPARE_CHARGE
            object.timer = 0
        end if
        break

    case KAMAKAMA_PREPARE_CHARGE
        if object.timer < 48
            object.timer++
        else
            object.state = KAMAKAMA_CHARGE
            object.timer = 0
            object.frame = 0

            if object.direction == FACING_RIGHT
                object.xvel = -0x10000
            else
                object.xvel = 0x10000
            end if

            if object.quality != QUALITY_GOOD
                object.xvel >>= 1
            end if
        end if
        break

    case KAMAKAMA_CHARGE
        object.frame++
        object.frame &= 31

        object.xpos += object.xvel
        ObjectTileGrip(CSIDE_FLOOR, 0, 19, 0) // Check if it's reaching a ledge
        if checkResult == false
            object.state = KAMAKAMA_SWITCH_DIRECTION
            object.timer = 60
        end if
        break

    case KAMAKAMA_SWITCH_DIRECTION
        object.frame++
        object.frame &= 31
        if object.timer > 0
            object.timer--
        else
            object.state = KAMAKAMA_CHARGE
            FlipSign(object.xvel)
            if object.direction == FACING_RIGHT
                object.direction = FACING_LEFT
            else
                object.direction = FACING_RIGHT
            end if
        end if
        break
    end switch

    // Reset object when out of bounds
    if object.outOfBounds == true
        object.priority = PRIORITY_BOUNDS

        object.state = KAMAKAMA_GUARD
        object.timer = 0

        object.animationTimer = 0
        object.frame = 0

        object.xpos = object.originPos.x
        object.ypos = object.originPos.y
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 20, currentPlayer, player[currentPlayer].hitboxLeft, player[currentPlayer].hitboxTop, player[currentPlayer].hitboxRight, player[currentPlayer].hitboxBottom)
        if checkResult == true
            CallFunction(Player_BadnikBreak)
        end if
    next

    // Check if this badnik should be a flower
    CallFunction(StageSetup_CheckGoodFuture)
end event

    case KAMAKAMA_GUARD

        temp0 >>= 3
        DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

        if object.quality == QUALITY_GOOD
            temp1 = temp0
            temp1 += 4
            DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)

            temp1 += 12
            DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)

            temp1 -= 10
            DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)

            DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)

            temp1 += 2

    case KAMAKAMA_LAUNCH_BLADE

        if object.quality == QUALITY_GOOD

        DebugMode_ObjCount++
        temp0++
    loop

event RSDKEdit
    if editor.returnVariable == true
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            checkResult = object.propertyValue
            checkResult &= 1
            break

        case 0 // condition
            checkResult = object.propertyValue
            checkResult &= 1
            break
        end switch
    else
        switch editor.variableID
        case EDIT_VAR_PROPVAL // Property Value
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break

        case 0 // condition
            object.propertyValue = editor.variableValue
            object.propertyValue &= 1
            break
        end switch
    end if
end event


event RSDKDraw
    DrawSprite(0)

    switch object.quality
    case QUALITY_GOOD
        DrawSprite(1)
        DrawSprite(2)
        DrawSprite(3)
        break

    case QUALITY_BAD
        DrawSprite(4)
        DrawSprite(5)
        break
    end switch
end event

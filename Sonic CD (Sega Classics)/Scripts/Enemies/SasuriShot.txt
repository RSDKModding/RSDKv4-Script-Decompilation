event ObjectUpdate
	object.xpos += object.xvel

	if object.outOfBounds == true
		// If Out Of Bounds, then Unload
		object.type = TypeName[Blank Object]
	else
		// Another condition for Unloading - be far enough from the screen on the Y Axis too
		temp0 = screen.yoffset
		temp0 -= 8
		temp0 <<= 16
		if object.ypos < temp0
			object.type = TypeName[Blank Object]
		end if
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -6, -6, 6, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			CallFunction(Player_Hit)
		end if
	next
end event


event ObjectDraw
	temp0 = object.frame
	temp0 >>= 1
	DrawSprite(temp0)

	object.frame++
	object.frame &= 3
end event


event ObjectStartup
	LoadSpriteSheet("R5/Objects.gif")

	// Shot Frames
	SpriteFrame(-8, -8, 16, 16, 67, 27)
	SpriteFrame(-8, -8, 16, 16, 133, 127)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("R5/Objects.gif")
	SpriteFrame(-8, -8, 16, 16, 67, 27)

	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

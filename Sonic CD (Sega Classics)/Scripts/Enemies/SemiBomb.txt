// ========================
// Aliases
// ========================

// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.yvel

// States
private alias 0 : SEMIBOMB_FINDGROUND
private alias 1 : SEMIBOMB_TICKING

// ========================
// Events
// ========================

event ObjectUpdate
    if object.outOfBounds == true
        object.type = TypeName[Blank Object]
    end if

    if object.state == SEMIBOMB_FINDGROUND
        object.ypos += object.yvel
        object.yvel += 0x2000

        ObjectTileCollision(CSIDE_FLOOR, 0, 8, 0)
        if checkResult == true
            object.state = SEMIBOMB_TICKING
        end if
    else
        if object.timer < 240
            object.timer++
        else
            object.type = TypeName[Blank Object]
            CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
            PlaySfx(SfxName[Explosion], false)
        end if

        if object.timer > 120
            object.frame = object.timer
            object.frame &= 3
            object.frame >>= 1
        end if
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            CallFunction(Player_Hit)
        end if
    next
end event


event ObjectDraw
    DrawSprite(object.frame)
end event


event ObjectStartup
    LoadSpriteSheet("R6/Objects.gif")

    SpriteFrame(-8, -8, 16, 16, 43, 67)  // #0 - Bomb
    SpriteFrame(-8, -8, 16, 16, 60, 67)  // #1 - Flashing Bomb
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R6/Objects.gif")
    SpriteFrame(-8, -8, 16, 16, 43, 67)  // #0 - Bomb

    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

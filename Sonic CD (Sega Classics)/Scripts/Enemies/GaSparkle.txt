// Aliases
private alias object.value0 : object.timer

// ========================
// Events
// ========================

event ObjectUpdate
    object.timer++
    object.ypos += 0x10000
    if object.timer == 6
        object.timer = 0
        object.frame++
        object.frame &= 3
    end if

    if object.outOfBounds == true
        object.type = TypeName[Blank Object]
    end if
end event


event ObjectDraw
    DrawSprite(object.frame)
end event


event ObjectStartup
    LoadSpriteSheet("Global/Items.gif")

    SpriteFrame(-8, -8, 16, 16, 1, 202)   // #0 - Ga Sparkle Frame 0
    SpriteFrame(-8, -8, 16, 16, 18, 202)  // #1 - Ga Sparkle Frame 1
    SpriteFrame(-8, -8, 16, 16, 1, 219)   // #2 - Ga Sparkle Frame 2
    SpriteFrame(-8, -8, 16, 16, 18, 219)  // #3 - Ga Sparkle Frame 3
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad

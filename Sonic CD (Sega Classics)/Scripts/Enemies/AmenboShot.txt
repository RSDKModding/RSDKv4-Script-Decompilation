// Aliases
private alias object.value0 : object.frameTimer

// ========================
// Events
// ========================

event ObjectUpdate
    object.xpos += object.xvel

    if object.outOfBounds == true
        object.type = TypeName[Blank Object]
    end if

    foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
        BoxCollisionTest(C_TOUCH, object.entityPos, -4, -4, 4, 4, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
        if checkResult == true
            CallFunction(Player_Hit)
        end if
    next
end event


event ObjectDraw
    DrawSprite(object.frame)

    object.frameTimer++
    if object.frameTimer > 5
        object.frameTimer = 0
        object.frame++
        object.frame &= 1
    end if
end event


event ObjectStartup
    LoadSpriteSheet("R4/Objects.gif")
    SpriteFrame(-4, -4, 8, 8, 197, 52)
    SpriteFrame(-4, -4, 8, 8, 197, 60)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
    DrawSprite(0)
end event


event RSDKLoad
    LoadSpriteSheet("R4/Objects.gif")

    SpriteFrame(-4, -4, 8, 8, 197, 52)  // #0 - Skimmer Bullet

    SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

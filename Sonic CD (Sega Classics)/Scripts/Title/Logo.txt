// Aliases
private alias object.value0 : object.fade

// States
private alias 0	:	LOGO_WHITE_FADEIN
private alias 1	:	LOGO_SONIC_ANIMATE
private alias 2	:	LOGO_SONIC_RESET

// ========================
// Events
// ========================

event ObjectDraw
	temp0 = screen.xcenter
	temp0 <<= 1
	DrawSpriteScreenXY(16, 0, 100)
	DrawSpriteScreenXY(17, temp0, 100)

	switch object.state
	case LOGO_WHITE_FADEIN
		SetScreenFade(255, 255, 255, object.fade)

		// Draw boths halves of the Emblem
		object.direction = FACING_RIGHT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		object.direction = FACING_LEFT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		// Draw Sonic
		DrawSpriteScreenXY(2, screen.xcenter, 120)

		// Draw the Ribbon
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)

		// And draw his initial Arm frame too
		temp0 += 121
		DrawSpriteScreenXY(4, temp0, 110)

		if object.fade > 0
			object.fade -= 8
		else
			object.state = LOGO_SONIC_ANIMATE
			object.frame = 4
		end if
		break

	case LOGO_SONIC_ANIMATE
		// Draw both ends of the Emblem
		object.direction = FACING_RIGHT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		object.direction = FACING_LEFT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		// Draw the main Sonic sprite
		DrawSpriteScreenXY(2, screen.xcenter, 120)

		// If needed, draw his blinking frame
		if object.frame > 7
			if object.frame < 13
				temp0 = screen.xcenter
				temp0 -= 43
				DrawSpriteScreenXY(3, temp0, 48)
			end if
		end if

		// Draw the ribbon ontop of that
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)

		// And afterwards, draw his arm
		temp0 += 121
		DrawSpriteScreenXY(object.frame, temp0, 110)

		object.fade++
		if object.fade > 1
			object.fade = 0
			object.frame++
			if object.frame > 15
				object.state = LOGO_SONIC_RESET
				object[+1].type = TypeName[Start]
				object[+1].ixpos = screen.xcenter
				object[+1].iypos = 200
			end if
		end if
		break
	case LOGO_SONIC_RESET
		// Draw both ends of the Emblem
		object.direction = FACING_RIGHT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		object.direction = FACING_LEFT
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)

		// Draw the main Sonic sprite
		DrawSpriteScreenXY(2, screen.xcenter, 120)

		// Draw the ribbon
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)

		// And afterwards, draw his arm
		temp0 += 121
		DrawSpriteScreenXY(4, temp0, 110)
		break
	end switch
end event

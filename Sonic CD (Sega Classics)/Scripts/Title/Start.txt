public function TouchStart_HandleCheatCodes
    temp0 = CHEATINPUT_NONE

    if keyPress[0].up == true
        temp0 = CHEATINPUT_UP
    end if

    if keyPress[0].down == true
        temp0 = CHEATINPUT_DOWN
    end if

    if keyPress[0].left == true
        temp0 = CHEATINPUT_LEFT
    end if

    end if

        temp0 = CHEATINPUT_BUTTONA
    end if

        temp0 = CHEATINPUT_BUTTONB
    end if

    if keyPress[0].buttonC == true
        temp0 = CHEATINPUT_BUTTONC
    end if

    // Stage Select Cheat
    // up, down, down, left, right, B
    switch object.stageSelectCheatPos
    case 0
        if temp0 == CHEATINPUT_UP
            object.stageSelectCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.stageSelectCheatPos = 0
            end if
        end if
        break

    case 1
        if temp0 == CHEATINPUT_DOWN
            object.stageSelectCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.stageSelectCheatPos = 0
            end if
        end if
        break

    case 2
        if temp0 == CHEATINPUT_DOWN
            object.stageSelectCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.stageSelectCheatPos = 0
            end if
        end if
        break

    case 3
        if temp0 == CHEATINPUT_LEFT
            object.stageSelectCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.stageSelectCheatPos = 0
            end if
        end if
        break

    case 4
        if temp0 == CHEATINPUT_RIGHT
            object.stageSelectCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.stageSelectCheatPos = 0
            end if
        end if
        break

    case 5
        if temp0 == CHEATINPUT_BUTTONB
            object.stageSelectCheatPos = 0

            // Go to the Stage Select
            object.state = START_GOTO_STAGESELECT
            StopMusic()
            PlaySfx(SfxName[Ring L], false)

        end if

            end if
        end if
        break

            object.soundTestCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.soundTestCheatPos = 0
            end if
        end if
        break

    case 2
        if temp0 == CHEATINPUT_DOWN
            object.soundTestCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.soundTestCheatPos = 0
            end if
        end if
        break

    case 3
        if temp0 == CHEATINPUT_LEFT
            object.soundTestCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.soundTestCheatPos = 0
            end if
        end if
        break

    case 4
        if temp0 == CHEATINPUT_RIGHT
            object.soundTestCheatPos++
        else
            if temp0 != CHEATINPUT_NONE
                object.soundTestCheatPos = 0
            end if
        end if
        break

    case 5
        if temp0 == CHEATINPUT_BUTTONA
            object.soundTestCheatPos = 0

            // Go to the Sound Test
            object.state = START_GOTO_SOUNDTEST
            StopMusic()

        end if
        break

                object.daGardenCheatPos = 0
            end if

event ObjectUpdate
    if object.state == START_SETUP
        if options.gameMode == MODE_NOSAVE
            CallFunction(TouchStart_HandleCheatCodes)
        end if
    end if

    switch object.state
    case START_SETUP
        CheckTouchRect(0, 0, screen.xsize, 240)

        if keyPress[0].start == true
            checkResult = false
        end if

        if keyPress[0].buttonA == true
            checkResult = false
        end if

        if keyPress[0].buttonB == true
            checkResult = false
        end if

        if keyPress[0].buttonC == true
            checkResult = false
        end if

        if checkResult > -1
            object.state = START_GOTO_GAME
            StopMusic()
            PlaySfx(SfxName[Select], false)
        else
            object.demoTimer++
            if object.demoTimer == 1550
                object.state = START_GOTO_DEMO
            end if
        end if
        break

    case START_GOTO_GAME
        object.fade += 4
        if object.fade == 384
            if options.soundtrack == OST_JP
                CallNativeFunction2(PlayVideo, "Opening", 0)
            else
                CallNativeFunction2(PlayVideo, "OpeningUS", 0)
            end if

            stage.activeList = REGULAR_STAGE
            stage.listPos = 0
            LoadStage()
        end if

        SetScreenFade(0, 0, 0, object.fade)
        break

    case START_GOTO_STAGESELECT
        object.fade += 4
        if object.fade == 384
            options.gameMode = MODE_NOSAVE
            stage.listPos = STAGE_P_STAGESELECT
            LoadStage()
        end if

        SetScreenFade(0, 0, 0, object.fade)
        break

    case START_GOTO_SOUNDTEST
        object.fade += 4
        if object.fade == 384
            options.gameMode = MODE_NOSAVE
            stage.listPos = STAGE_P_SOUNDTEST
            LoadStage()
        end if

        SetScreenFade(0, 0, 0, object.fade)
        break

    case START_GOTO_DAGARDEN // Oh?? a new secret emerges...
        object.fade += 4
        if object.fade == 384
            options.gameMode = MODE_NOSAVE
            stage.listPos = STAGE_P_DAGARDEN
            LoadStage()
        end if

        SetScreenFade(0, 0, 0, object.fade)
        break

    case START_GOTO_DEMO
        object.fade += 4
        if object.fade == 384
            fadeColor = 0

            player.score = 0
            player.lives = 3
            player.scoreBonus = 50000

            generator.destroyed  = false
            metalSonic.destroyed = false

            goodFuture.enabled = false
            goodFuture.list    = 0
            goodFuture.count   = 0

            options.gameMode    = MODE_NOSAVE
            options.attractMode = true

            switch options.demoNumber
            case DEMO_R11A
                stage.activeList = REGULAR_STAGE
                stage.listPos = 0
                break

            case DEMO_SS1
                stage.activeList = SPECIAL_STAGE
                stage.listPos = 0
                break

            case DEMO_R43C
                stage.activeList = REGULAR_STAGE
                stage.listPos = 28
                break

            case DEMO_SS6
                stage.activeList = SPECIAL_STAGE
                stage.listPos = 5
                break

            case DEMO_R82A
                stage.activeList = REGULAR_STAGE
                stage.listPos = 64
                break
            end switch

            options.demoNumber++
            options.demoNumber %= DEMO_MAX
            LoadStage()
        end if

        SetScreenFade(0, 0, 0, object.fade)
        break
    end switch
end event


event ObjectDraw
    object.animationTimer++
    if object.animationTimer > 59
        object.animationTimer = 0
    end if

    if object.animationTimer > 29
        if options.touchControls == false
            DrawSprite(0)
        else
            DrawSprite(1)
        end if
    end if
end event


event ObjectStartup
    LoadSpriteSheet("Title/Title.gif")

    switch engine.language
    case LANG_ENGLISH
    case LANG_JAPANESE
    case LANG_PORTUGUESE
    case LANG_RUSSIAN
        SpriteFrame(-43, 0, 87, 11, 106, 103)    // #0 - Press Start
        SpriteFrame(-56, 0, 117, 11, 106, 91)    // #1 - Touch to Start!
        break

    case LANG_FRENCH
        SpriteFrame(-60, 0, 120, 11, 391, 387)   // #0 - Appui sur Start
        SpriteFrame(-94, 0, 189, 11, 322, 399)   // #1 - Touche pour Commencer !

    case LANG_ITALIAN
        SpriteFrame(-44, 0, 88, 11, 270, 411)    // #0 - Premi Start
        SpriteFrame(-76, 0, 152, 11, 359, 411)   // #1 - Tocca Per Iniziare!
        break

    case LANG_DEUTSCH
        SpriteFrame(-52, -2, 106, 13, 405, 373)  // #0 - Press to Start
        SpriteFrame(-87, 0, 175, 11, 336, 423)   // #1 - Touch to Start
        break

    case LANG_SPANISH
        SpriteFrame(-43, 0, 87, 11, 274, 435)    // #0 - Pulsa Start
        SpriteFrame(-74, 0, 149, 11, 362, 435)   // #1 - ¡Toca para Empezar!
        break
    end switch

    options.touchControls = false
end event

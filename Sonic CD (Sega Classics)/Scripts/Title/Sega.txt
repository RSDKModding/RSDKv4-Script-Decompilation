// ========================
// Aliases
// ========================

private alias object.value0	:	object.timer
private alias object.value3	:	object.unused

// States
private alias 0	:	SEGA_SETUP
private alias 1	:	SEGA_FADEIN
private alias 2	:	SEGA_JINGLE
private alias 3	:	SEGA_FADEOUT
private alias 4	:	SEGA_END

// Ink Effect Aliases
private alias 2 : INK_ALPHA  // Uses transparency value of Object.Alpha

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case SEGA_SETUP

		// The game doesn't read the soundtrack here anymore...

		object.state = SEGA_FADEIN
		object.timer = 480
		object.alpha = 256
		object.inkEffect = INK_ALPHA

		// Start the screen off all black
		SetScreenFade(0, 0, 0, 255)
		object.ixpos = screen.xcenter
		break
	case SEGA_FADEIN
		if object.timer > 0
			// Fade in, 8 by 8
			object.timer -= 8
		else
			// Jingle! Looks like there's no Engine.OnlineActive here now!
			object.state = SEGA_JINGLE
		end if
		SetScreenFade(0, 0, 0, object.timer)
		break

	case SEGA_JINGLE
		object.timer++

		// Hold for a moment, to let the player bask in this wonderful screen

		if object.timer == 160
			object.timer = 0
			object.state = SEGA_FADEOUT
		end if

		if object.timer == 2
			// Play the SFX almost immediately after entering this state
			PlaySfx(SfxName[Sega Chant], 0)
		end if
		break

	case SEGA_FADEOUT
		// Do note - this state is kinda messed up, they forget a break at the end so
		// this state doesn't reallly fade out
		if object.alpha > 0
			object.alpha -= 8
		else
			if object.timer == 16
				ResetObjectEntity(object.entityPos, TypeName[Sonic], 0, object.xpos, object.ypos)
				object.inkEffect = INK_ALPHA
				object.alpha = 0
				object.timer = 0
				object.value3 = 96
			else
				object.timer++
			end if
		end if
		// No break was used here, so it falls through.
		// break
	end switch
end event


event ObjectDraw
	// First, draw the background
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 96, 255)

	// Then, draw the pixel sega logo.
	if object.alpha > 8
		DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Title/Title.gif")
	SpriteFrame(-49, -16, 98, 31, 106, 1)
end event


// ========================
// Editor Events
// ========================

event RSDKDraw
	DrawSprite(0)
end event


event RSDKLoad
	LoadSpriteSheet("Title/Title.gif")
	SpriteFrame(-49, -16, 98, 31, 106, 1)
	
	SetVariableAlias(ALIAS_VAR_PROPVAL, "unused")
end event

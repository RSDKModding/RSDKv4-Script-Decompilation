// ========================
// Events
// ========================

public function TailsObject_ProcessTailSprite
    switch player.tailAnim
    case 0 // ANI_STOPPED
        if player.frame == 0
            player.tailFrame++
            player.tailFrame %= 40
        end if
        break

    case 1 // ANI_WAITING
    case 3 // ANI_LOOKINGUP
    case 4 // ANI_LOOKINGDOWN
        player.tailFrame++
        player.tailFrame %= 40
        break

    case 7  // ANI_SKIDDING
    case 9  // ANI_SPINDASH
    case 10 // ANI_JUMPING
    case 43 // Unknown, this was probably "SSRoll" in Sonic 1 before it got moved to its own ani file in that game
        player.tailFrame++
        player.tailFrame %= 12
        break

    case 17 // ANI_PUSHING
        player.tailFrame++
        player.tailFrame %= 30
        break
    end switch
end function

event ObjectUpdate
    if stage.debugMode == true
        CallFunction(Player_ProcessUpdate)

        CheckEqual(options.attractMode, false)
        temp0 = checkResult

        CheckEqual(keyPress[0].buttonB, true)
        temp0 &= checkResult

        if temp0 == true
            player.type = TypeName[Debug Mode]
            player.yvel = 0
            player.state = Player_State_Static
            player.frame = 0
            player.rotation = 0
            player.interaction = false
            player.drawOrder = 4
            player.priority = PRIORITY_ACTIVE
            player.blinkTimer = 0
            player.visible = true
            player.abilityTimer = 0
            player.drownTimer = 0
            player.drownLevel = 0
            player.frame = debugMode.currentSelection

            screen[0].cameraEnabled = true
            screen[0].cameraStyle = options.originalControls

            StopSfx(SfxName[Flying])
            StopSfx(SfxName[Jump])

            if stage.state == STAGE_FROZEN
                stage.state = STAGE_RUNNING
            end if

            if object[SLOT_PLAYER2].type == TypeName[Blank Object]
                object[SLOT_PLAYER2].priority = PRIORITY_ACTIVE
            end if

            if object[+playerCount].propertyValue == 3
                object[+playerCount].type = TypeName[Invincibility]
                object[+playerCount].propertyValue = 0
            end if
        else
            if player.gravity == GRAVITY_GROUND
                player.jumpAbilityState = 0
            end if

            CallFunction(player.state)
            ProcessAnimation()
            CallFunction(TailsObject_ProcessTailSprite)

            if player.animation == ANI_JUMPING
                screen[0].adjustCameraY = player.jumpOffset
            else
                if screen[0].adjustCameraY == player.jumpOffset
                    screen[0].adjustCameraY = 0
                    player.iypos += player.jumpOffset
                end if
            end if

            if player.collisionDisabled == false
                player.prevGravity = player.gravity
                player.prevGravity ^= GRAVITY_AIR
                ProcessObjectMovement()
                checkResult |= player.prevGravity
                player.prevGravity ^= GRAVITY_AIR

                if checkResult == GRAVITY_AIR
                    player.badnikBonus = 0
                    if player.animation == ANI_JUMPING
                        if player.down == false
                            if player.state != Player_State_Roll
                                if player.state != Player_State_TubeRoll
                                    player.animation = ANI_WALKING
                                    screen[0].adjustCameraY = 0
                                    player.iypos += player.jumpOffset
                                end if
                            end if
                        end if
                    end if
                end if
            else
                player.collisionDisabled = false
            end if
        end if
    else
        CallFunction(Player_ProcessUpdate)

        if player.gravity == GRAVITY_GROUND
            player.jumpAbilityState = 0
        end if

        CallFunction(player.state)
        ProcessAnimation()
        CallFunction(TailsObject_ProcessTailSprite)

        if player.animation == ANI_JUMPING
            screen[0].adjustCameraY = player.jumpOffset
        else
            if screen[0].adjustCameraY == player.jumpOffset
                screen[0].adjustCameraY = 0
                player.iypos += player.jumpOffset
            end if
        end if

        if player.collisionDisabled == false
            temp0 = player.prevGravity
            player.prevGravity = player.gravity
            ProcessObjectMovement()
            player.prevGravity ^= GRAVITY_AIR
            CheckEqual(player.gravity, GRAVITY_GROUND)
            player.prevGravity |= checkResult
            player.prevGravity ^= GRAVITY_AIR

            if temp0 == GRAVITY_AIR
                if player.prevGravity == GRAVITY_GROUND
                    player.badnikBonus = 0
                    if player.animation == ANI_JUMPING
                        if player.down == false
                            if player.state != Player_State_Roll
                                if player.state != Player_State_TubeRoll
                                    player.animation = ANI_WALKING
                                    screen[0].adjustCameraY = 0

                                    player.iypos += player.jumpOffset
                                end if
                            end if
                        end if
                    end if
                end if
            end if
        else
            player.collisionDisabled = false
        end if
    end if
end event


event ObjectDraw
    if player.animation != player.prevAnimation
        player.prevAnimation = player.animation
        player.frame = 0
        player.animationTimer = 0
        player.animationSpeed = 0
    end if

    if player.tailAnim != player.animation
        if player.animation > 4 // Animation 4 is ANI_LOOKINGDOWN, it's notable they used a direct constant rather than the normal ANI_* variables
            player.tailFrame = 0
        end if

        if player.tailAnim > 4
            player.tailFrame = 0
        end if

        player.tailAnim = player.animation
    end if

	// Draw the Tails part of Tails
    switch player.tailAnim
    case 0 // ANI_STOPPED
        if player.frame == 0
            temp0 = player.tailFrame
            temp0 >>= 3
            if player.visible == true
                DrawSpriteFX(temp0, FX_FLIP, player.xpos, player.ypos)
            end if
        end if
        break

    case 1  // ANI_WAITING
    case 3  // ANI_LOOKINGUP
    case 4  // ANI_LOOKINGDOWN
    case 37 // ANI_CONTINUE_UP
        temp0 = player.tailFrame
        temp0 >>= 3
        if player.visible == true
            DrawSpriteFX(temp0, FX_FLIP, player.xpos, player.ypos)
        end if
        break

    case 7 // ANI_SKIDDING
    case 9 // ANI_SPINDASH
        temp0 = player.tailFrame
        temp0 >>= 2
        temp0 += 11
        if player.visible == true
            DrawSpriteFX(temp0, FX_FLIP, player.xpos, player.ypos)
        end if
        break

    case 10 // ANI_JUMPING
    case 43 // Unknown, this was probably "SSRoll" before it got moved to its own ani file
        temp0 = player.tailFrame
        temp0 >>= 2
        CheckEqual(player.xvel, 0)
        temp1 = checkResult
        CheckEqual(player.yvel, 0)
        temp1 &= checkResult
        temp2 = player.rotation
        if temp1 == 0
            ATan2(player.rotation, player.xvel, player.yvel)
            player.rotation += 16
            player.rotation &= 255
            player.rotation >>= 5
            switch player.rotation
            case 0
            case 8
                temp0 += 5
                player.rotation = 0x00
                break

            case 1
                temp0 += 8
                if player.direction == FACING_RIGHT
                    player.rotation = 0x40
                else
                    player.rotation = 0x00
                end if
                break

            case 2
                temp0 += 5
                player.rotation = 0x40
                break

            case 3
                temp0 += 8
                if player.direction == FACING_RIGHT
                    player.rotation = 0x80
                else
                    player.rotation = 0x40
                end if
                break

            case 4
                temp0 += 5
                player.rotation = 0x80
                break

            case 5
                temp0 += 8
                if player.direction == FACING_RIGHT
                    player.rotation = 0xC0
                else
                    player.rotation = 0x80
                end if
                break

            case 6
                temp0 += 5
                player.rotation = 0xC0
                break

            case 7
                temp0 += 8
                if player.direction == FACING_RIGHT
                    player.rotation = 0x00
                else
                    player.rotation = 0xC0
                end if
                break
            end switch

            if player.direction == FACING_LEFT
                player.rotation += 0x80
            end if
        else
            temp0 += 5
            player.rotation = 0x00
        end if

        player.rotation <<= 1
        if player.visible == true
            DrawSpriteFX(temp0, FX_ROTATE, player.xpos, player.ypos)
        end if
        player.rotation = temp2
        break

    case 17 // ANI_PUSHING
        temp0 = player.tailFrame
        temp0 /= 10
        temp0 += 11
        if player.visible == true
            DrawSpriteFX(temp0, FX_FLIP, player.xpos, player.ypos)
        end if
        break
    end switch

	// Draw Tails himself
	// (His separate tails are drawn in the process above, notably done before this function call since Tails is supposed to be ontop of his Tails)
    DrawObjectAnimation()
end event

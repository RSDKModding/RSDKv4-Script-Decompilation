// Aliases
private alias object.value0 : object.timer
private alias object.value1 : object.offset.y
private alias object.value3 : object.offset.x

// States
private alias 0 : METALSONIC_SETUP
private alias 1 : METALSONIC_GOTO_CENTER
private alias 2 : METALSONIC_SET_DIRECTION
private alias 3 : METALSONIC_START_CHARGE
private alias 4 : METALSONIC_LEAVE

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case METALSONIC_SETUP
		object.offset.x = screen.xcenter
		object.offset.x <<= 16

		if object.xpos < object.offset.x
			object.direction = FACING_RIGHT
		else
			object.direction = FACING_LEFT
		end if

		object.state++
		break

	case METALSONIC_GOTO_CENTER
		temp0 = object.offset.x
		temp0 -= object.xpos
		temp0 >>= 5

		temp1 = 0x780000
		temp1 -= object.ypos
		temp1 >>= 5

		object.xpos += temp0
		object.ypos += temp1

		object.timer++
		if object.timer == 120
			object.timer = 0
			object.offset.y = object.ypos
			object.state++
		end if

		object.animationTimer++
		object.animationTimer &= 7
		break

	case METALSONIC_SET_DIRECTION
		object.angle += 8
		object.angle &= 511
		Sin(object.ypos, object.angle)
		object.ypos <<= 9
		object.ypos += object.offset.y

		object.timer++
		if object.timer == 100
			object.timer = 0
			object.offset.y = object.ypos
			object.direction++
			object.direction &= FACING_LEFT
			object.state++
		end if

		object.animationTimer++
		object.animationTimer &= 7
		break

	case METALSONIC_START_CHARGE
		object.angle += 64
		object.angle &= 511
		Sin(object.ypos, object.angle)
		object.ypos <<= 7
		object.ypos += object.offset.y

		if object.direction == FACING_LEFT
			object.xpos += 0x8000
		else
			object.xpos -= 0x8000
		end if

		object.timer++
		if object.timer == 90
			object.timer = 0
			object.xvel = 0x10000
			object.state++
		end if

		object.animationTimer++
		object.animationTimer &= 7
		break

	case METALSONIC_LEAVE
		if object.direction == FACING_RIGHT
			object.xpos += object.xvel
		else
			object.xpos -= object.xvel
		end if

		object.xvel += 0x3000
		object.animationTimer += 2
		object.animationTimer %= 5
		break

	end switch

	if object.outOfBounds == true
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	// Jet Flame
	temp0 = object.animationTimer
	temp0 >>= 1
	temp0++
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

	// Metal
	DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
end event

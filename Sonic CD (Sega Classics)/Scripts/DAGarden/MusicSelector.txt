// States
private alias 0 : MUSICSELECTOR_BLANK
private alias 1 : MUSICSELECTOR_STOP_MOVE_L
private alias 2 : MUSICSELECTOR_MOVE_L
private alias 3 : MUSICSELECTOR_STOP_MOVE_R
private alias 4 : MUSICSELECTOR_MOVE_R

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case MUSICSELECTOR_STOP_MOVE_L
		object.xpos -= 0xC0000
		temp0 = screen.xcenter
		temp0 <<= 16
		if object.xpos <= temp0
			object.xpos = temp0
			object.state = MUSICSELECTOR_BLANK
		end if
		break

	case MUSICSELECTOR_MOVE_L
		object.xpos -= 0xC0000
		if object.xpos < -0x600000
			temp0 = screen.xsize
			temp0 += 192
			temp0 <<= 16
			object.xpos += temp0
			object.state = MUSICSELECTOR_STOP_MOVE_L
			if object.frame < 30
				object.frame++
			else
				object.frame = MUSICSELECTOR_BLANK
			end if
		end if
		break

	case MUSICSELECTOR_STOP_MOVE_R
		object.xpos += 0xC0000
		temp0 = screen.xcenter
		temp0 <<= 16
		if object.xpos >= temp0
			object.xpos = temp0
			object.state = MUSICSELECTOR_BLANK
		end if
		break

	case MUSICSELECTOR_MOVE_R
		object.xpos += 0xC0000
		temp0 = screen.xsize
		temp0 += 96
		temp0 <<= 16
		if object.xpos > temp0
			object.xpos -= temp0
			object.xpos -= 0x600000
			object.state = MUSICSELECTOR_STOP_MOVE_R
			if object.frame > 0
				object.frame--
			else
				object.frame = 30
			end if
		end if
		break

	end switch
end event


event ObjectDraw
	DrawSprite(object.frame)
end event

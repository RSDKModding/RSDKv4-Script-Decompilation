event ObjectUpdate
	// Update Little Planet control
	if menuControl.state == MENUCONTROL_IDLE
		// Planet Movement
		if keyDown[0].left == true
			if object.xpos > -0x400
				object.xpos--
			end if
		end if

		if keyDown[0].right == true
			if object.xpos < 0x400
				object.xpos++
			end if
		end if

		if keyDown[0].up == true
			object.ypos--
		end if

		if keyDown[0].down == true
			object.ypos++
		end if

		// Increase rotation speed
		if keyDown[0].start == true
			if object.rotationDirection == FACING_LEFT
				if object.rotationSpeed > -256
					object.rotationSpeed -= 2
				end if
			else
				if object.rotationSpeed < 256
					object.rotationSpeed += 2
				end if
			end if
		end if

		// Flip Rotation
		if keyPress[0].start == true
			object.rotationDirection++
			object.rotationDirection &= 1
			object.rotationSpeed = 0
		end if

		// Switch Zoom In / Zoom Out
		if keyPress[0].buttonC == true
			object.zoomIn++
			object.zoomIn &= 1
		end if

		// Zoom In/Out
		if keyDown[0].buttonC == true
			if object.zoomIn == false
				if object.zoomDepth > 48
					object.zoomDepth -= 8
				end if
			else
				if object.zoomDepth < 0x4000
					object.zoomDepth += 8
				end if
			end if
		end if

		// No touch input :(
	end if

	object.rotationLimit += object.rotationSpeed
	if object.rotationLimit < 0
		object.rotationLimit += 0x2000
	end if

	object.rotationLimit &= 0x1FFF
	object.rotation = object.rotationLimit
	object.rotation >>= 4
end event


event ObjectDraw
	if object.frame == 22
		DrawRect(0, 32, screen.xsize, 176, 64, 64, 32, 176)
	end if

	if object.zoomDepth > 0
		temp0 = object.xpos
		temp0 <<= 11
		temp0 /= object.zoomDepth
		temp0 += screen.xcenter

		temp1 = object.ypos
		temp1 <<= 11
		temp1 /= object.zoomDepth
		temp1 += 120

		object.scale = 0x20000
		object.scale /= object.zoomDepth
		DrawSpriteScreenFX(object.frame, FX_ROTOZOOM, temp0, temp1)

		temp2 = object.animationTimer
		temp2 >>= 3
		temp3 = object.frame
		switch temp2
		case 0
			temp3++
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 1
			temp3 += 2
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 2
			temp3 += 3
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 3
			temp3 += 4
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 4
			temp3 += 5
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 24
			temp3 += 6
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 25
			temp3 += 7
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 26
			temp3 += 8
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 27
			temp3 += 9
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		case 28
			temp3 += 10
			DrawSpriteScreenFX(temp3, FX_ROTOZOOM, temp0, temp1)
			break

		end switch

		object.animationTimer++
		object.animationTimer &= 511
	end if
end event

// ========================

// Aliases
private alias object.propertyValue : object.spawnedStar // Correspond to Star Sprite Frames

// States
private alias 0 : SHOOTINGSTAR_SETUP
private alias 1 : SHOOTINGSTAR_MOVEMENT

// Property Values
private alias 0 : WHITE_STAR
private alias 6 : GREEN_STAR

// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case SHOOTINGSTAR_SETUP
		object.xvel = -0x14000
		object.yvel = 0x20000
		object.state++
		break

	case GREEN_STAR
		object.xpos += object.xvel
		object.ypos += object.yvel
		break

	end switch

	if object.outOfBounds == true
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.animationSpeed
	temp0 >>= 2
	temp0 += object.spawnedStar
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)

	object.animationSpeed++
	object.animationSpeed %= 24
end event


event ObjectStartup
	LoadSpriteSheet("DAGarden/Objects.gif")

	SpriteFrame(-4, -12, 9, 24, 387, 854)		// #0  - White Star frame 0

	SpriteFrame(-12, -12, 24, 24, 442, 879)		// #9  - Green Star Frame 3
	SpriteFrame(-12, -12, 24, 24, 467, 879)		// #10 - Green Star Frame 4
	SpriteFrame(-10, -12, 19, 24, 492, 879)		// #11 - Green Star Frame 5
end event

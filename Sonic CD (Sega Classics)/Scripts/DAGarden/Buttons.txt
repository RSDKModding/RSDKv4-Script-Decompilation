// States
private alias 1 : MENUCONTROL_FADE_IN
private alias 3 : MENUCONTROL_MUSIC_CONTROLS

private alias 6 : MENUCONTROL_EXIT

private alias 1 : PRIORITY_ACTIVE

event ObjectUpdate
	if menuControl.state > MENUCONTROL_FADE_IN
		if menuControl.state < MENUCONTROL_SET_MUSIC
			temp0 = object.xpos
			temp0 >>= 16
			temp1 = temp0

			temp0 -= 32
			temp0 -= screen.xoffset

			temp1 += 32
			temp1 -= screen.xoffset

			CheckTouchRect(temp0, 0, temp1, 32)
			if checkResult > -1
				object.buttonPressed = object.buttonID
				object.buttonPressed++

				switch object.buttonID
				case 0
					if littlePlanet.zoomDepth < 0x4000
						littlePlanet.zoomDepth += 8
					end if
					break

				case 1
					if littlePlanet.zoomDepth > 48
						littlePlanet.zoomDepth -= 8
					end if
					break

				case 2
					if littlePlanet.rotationSpeed < 256
						littlePlanet.rotationSpeed += 2
					end if
					break

				case 3
					if littlePlanet.rotationSpeed > -256
						littlePlanet.rotationSpeed -= 2
					end if
					break

				end switch
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					switch object.buttonID
					// Zooming
					case 0
						if object.buttonPressed == 1
							object.buttonPressed = 0
							object.button1Value = 1
						end if
						break

					case 1
						if object.buttonPressed == 2
							object.buttonPressed = 0
							object.button1Value = 0
						end if
						break

					// Rotation
					case 2
						if object.buttonPressed == 3
							object.buttonPressed = 0
							object.button2Value = 3
						end if
						break

					case 3
						if object.buttonPressed == 4
							object.buttonPressed = 0
							object.button2Value = 2
						end if
						break

					// Music Selector
					case 4
						if object.buttonPressed == 5
							object.buttonPressed = 0
							PlaySfx(SfxName[Menu Button], false)
							musicSelector.state = MUSICSELECTOR_MOVE_R
						end if
						break

					case 5
						if object.buttonPressed == 6
							object.buttonPressed = 0
							menuControl.state = MENUCONTROL_SET_MUSIC
							StopMusic()
						end if
						break

					case 6
						if object.buttonPressed == 7
							object.buttonPressed = 0
							PlaySfx(SfxName[Menu Button], false)
							musicSelector.state = MUSICSELECTOR_MOVE_L
						end if
						break

					case 7
						if object.buttonPressed == 8
							object.buttonPressed = 0

							PlaySfx(SfxName[Select], false)

							menuControl.state = MENUCONTROL_MUSIC_CONTROLS
							musicSelector.xpos = screen.xsize
							musicSelector.xpos += 96
							musicSelector.xpos <<= 16

							musicSelector.state = MUSICSELECTOR_STOP_MOVE_L
							musicSelector.drawOrder = 5
							musicSelector.priority = PRIORITY_ACTIVE

							object.button1Value = 4
							object.button2Value = 5
							object.button3Value = 6
						end if
						break

					// Exit
					case 8
						if object.buttonPressed == 9
							object.buttonPressed = 0
							StopMusic()
							PlaySfx(SfxName[Select], false)
							menuControl.state = MENUCONTROL_EXIT
						end if
						break

					end switch
				else
					// Reset buttons
					switch object.buttonID
					case 0
						if object.buttonPressed == 1
							object.buttonPressed = 0
						end if
						break

					case 1
						if object.buttonPressed == 2
							object.buttonPressed = 0
						end if
						break

					case 2
						if object.buttonPressed == 3
							object.buttonPressed = 0
						end if
						break

					case 3
						if object.buttonPressed == 4
							object.buttonPressed = 0
						end if
						break

					case 4
						if object.buttonPressed == 5
							object.buttonPressed = 0
						end if
						break

					case 5
						if object.buttonPressed == 6
							object.buttonPressed = 0
						end if
						break

					case 6
						if object.buttonPressed == 7
							object.buttonPressed = 0
						end if
						break

					case 7
						if object.buttonPressed == 8
							object.buttonPressed = 0
						end if
						break

					case 8
						if object.buttonPressed == 9
							object.buttonPressed = 0
						end if
						break

					end switch
				end if
			end if
		end if
	end if
end event


event ObjectDraw
	temp0 = object.buttonID
	temp0 <<= 1

	temp1 = object.buttonID
	temp1++
	if object.buttonPressed == temp1
		temp0++
	end if
	DrawSprite(temp0)
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			checkResult = object.propertyValue
			break

		case 0 // ButtonID
			checkResult = object.propertyValue
			break

		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // Property Value
			object.propertyValue = editor.variableValue
			break

		case 0 // ButtonID
			object.propertyValue = editor.variableValue
			break

		end switch
	end if
end event


event RSDKDraw
	temp0 = object.buttonID
	temp0 <<= 1
	DrawSprite(temp0)
end event

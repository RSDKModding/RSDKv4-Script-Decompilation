// Aliases
private alias object.value0     : object.timer
private alias object[+7].value2 : characterSpawn.type

// States
private alias 0  : BGANIMATION_DAY
private alias 1  : BGANIMATION_DAY_TO_EVENING
private alias 2  : BGANIMATION_EVENING
private alias 3  : BGANIMATION_EVENING_TO_DUSK
private alias 4  : BGANIMATION_DUSK
private alias 5  : BGANIMATION_DUSK_TO_NIGHT
private alias 6  : BGANIMATION_NIGHT
private alias 7  : BGANIMATION_NIGHT_TO_MIDNIGHT
private alias 8  : BGANIMATION_MIDNIGHT
private alias 9  : BGANIMATION_MIDNIGHT_TO_MORNING
private alias 10 : BGANIMATION_MORNING
private alias 11 : BGANIMATION_MORNING_TO_DAY

// Ink Effect
private alias 2 : INK_ALPHA

// ========================
// Events
// ========================

event ObjectDraw
	if object.drawOrder == 3
		// Pretty straightforward process, when the day time is set, a timer increases.
		// Once the timer reachs the limit, start transitioning, then repeat process.
		switch object.state
		case BGANIMATION_DAY
			DrawSprite(0)
			DrawSprite(1)
			object.timer++
			if object.timer > 480
				object.timer = 0
				object.alpha = 0
				object.inkEffect = INK_ALPHA
				object.state++
			end if
			break

		case BGANIMATION_DAY_TO_EVENING
			DrawSprite(0)
			DrawSprite(1)
			if object.alpha < 248
				object.alpha += 4
			else
				object.alpha = 255
				object.state++
			end if
			DrawSpriteFX(2, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(3, FX_INK, object.xpos, object.ypos)
			break

		case BGANIMATION_EVENING
			DrawSprite(2)
			DrawSprite(3)
			object.timer++
			if object.timer > 480
				object.timer = 0
				object.alpha = 0
				object.state++
			end if
			break

		case BGANIMATION_EVENING_TO_DUSK
			DrawSprite(2)
			DrawSprite(3)
			if object.alpha < 248
				object.alpha += 4
			else
				object.alpha = 255
				characterSpawn.type = 1
				object.state++
			end if
			DrawSpriteFX(4, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(5, FX_INK, object.xpos, object.ypos)
			break

		case BGANIMATION_DUSK
			DrawSprite(4)
			DrawSprite(5)
			object.timer++
			if object.timer > 480
				object.timer = 0
				object.alpha = 0
				object.state++
			end if
			break

		case BGANIMATION_DUSK_TO_NIGHT
			DrawSprite(4)
			DrawSprite(5)
			if object.alpha < 248
				object.alpha += 4
			else
				object.alpha = 255
				object.state++
			end if
			DrawSpriteFX(6, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(7, FX_INK, object.xpos, object.ypos)
			break

		case BGANIMATION_NIGHT
			DrawSprite(6)
			DrawSprite(7)
			object.timer++
			if object.timer > 480
				object.timer = 0
				object.alpha = 0
				object.state++
			end if
			break

		case BGANIMATION_NIGHT_TO_MIDNIGHT
			DrawSprite(6)
			DrawSprite(7)
			if object.alpha < 248
				object.alpha += 4
			else
				object.alpha = 255
				object.state++
			end if
			DrawSpriteFX(8, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(9, FX_INK, object.xpos, object.ypos)
			break

		case BGANIMATION_MIDNIGHT
			DrawSprite(8)
			DrawSprite(9)
			object.timer++
			if object.timer > 512
				object.timer = 0
				object.alpha = 0
				characterSpawn.type = 0
				object.state++
			end if
			break

		case BGANIMATION_MIDNIGHT_TO_MORNING
			DrawSprite(8)
			DrawSprite(9)
			if object.alpha < 248
				object.alpha += 2
			else
				object.alpha = 255
				object.state++
			end if
			DrawSpriteFX(10, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(11, FX_INK, object.xpos, object.ypos)
			break

		case BGANIMATION_MORNING
			DrawSprite(10)
			DrawSprite(11)
			object.timer++
			if object.timer > 480
				object.timer = 0
				object.alpha = 0
				object.state++
			end if
			break

		case BGANIMATION_MORNING_TO_DAY
			DrawSprite(10)
			DrawSprite(11)
			if object.alpha < 248
				object.alpha += 4
			else
				object.alpha = 255
				object.state = BGANIMATION_DAY
			end if
			DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
			DrawSpriteFX(1, FX_INK, object.xpos, object.ypos)
			break

		end switch

		SetDrawListEntityRef(object.entityPos, 4, screen[4].drawListSize)
		screen[4].drawListSize++
		object.drawOrder = 4
	else
		// Draw borders
		DrawRect(0, 0, screen.xsize, 32, 0, 0, 0, 255)
		DrawRect(0, 208, screen.xsize, 32, 0, 0, 0, 255)
		object.drawOrder = 3
	end if
end event

	SpriteFrame(0, 0, 256, 192, 0, 192)			// #4  - Sky, Dusk Half 1
	SpriteFrame(256, 0, 256, 192, 0, 192)		// #5  - Sky, Dusk Half 2
	SpriteFrame(0, 0, 256, 192, 256, 192)		// #6  - Sky, Night Half 1
	SpriteFrame(256, 0, 256, 192, 256, 192)		// #7  - Sky, Night Half 2
	SpriteFrame(0, 0, 256, 192, 0, 384)			// #8  - Sky, Midnight Half 1
	SpriteFrame(256, 0, 256, 192, 0, 384)		// #9  - Sky, Midnight Half 2
	SpriteFrame(0, 0, 256, 192, 256, 384)		// #10 - Sky, Morning Half 1

	object[31].type = TypeName[BG Animation]

	object[31].ypos = 0x180000
end event
